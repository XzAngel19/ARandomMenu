local function resolvePath(path)
	for _, root in ipairs({ envGlobal, _G }) do
		if type(root) == "table" then
			local cur = root
			for key in string.gmatch(path, "[^%.]+") do
				local ok, nxt = pcall(function()
					return cur[key]
				end)
				cur = ok and nxt or nil
				if cur == nil then
					break
				end
			end
			if type(cur) == "function" then
				return cur
			end
		end
	end
	return nil
end

function Zenith.probe()
	local caps = { status = {}, via = {}, fn = {} }
	for _, entry in ipairs(KernelMATRIX) do
		local name, aliases = entry[1], entry[2]
		local found, usedAlias
		for _, alias in ipairs(aliases) do
			local f = resolvePath(alias)
			if f then
				found, usedAlias = f, alias
				break
			end
		end
		caps.status[name] = found and "ok" or "missing"
		caps.via[name] = usedAlias or "-"
		caps.fn[name] = found
	end
	Zenith.capabilities = caps
	return caps
end

function Zenith.cap(name)
	local c = Zenith.capabilities
	return c ~= nil and c.status[name] ~= nil and c.status[name] ~= "missing"
end

function Zenith.capFn(name)
	local c = Zenith.capabilities
	return c and c.fn[name] or nil
end

local modules, order = {}, {}

function Zenith.register(name, spec)
	if modules[name] then
		return
	end
	modules[name] = {
		requires = spec.requires or {},
		optional = spec.optional or {},
		init = spec.init,
		stop = spec.stop,
		state = "pending",
		owned = {},
	}
	table.insert(order, name)
end

function Zenith.track(conn, bucket)
	if not conn then
		return conn
	end
	table.insert(Zenith.connections, conn)
	if type(bucket) == "table" then
		table.insert(bucket, conn)
	end
	return conn
end

function Zenith.startAll()
	local caps = Zenith.capabilities or Zenith.probe()
	for _, name in ipairs(order) do
		local m = modules[name]
		if m and m.state ~= "loaded" and m.init then
			local ok, err = pcall(m.init, caps, {
				name = name,
				track = function(conn)
					return Zenith.track(conn, m.owned)
				end,
			})
			m.state = ok and "loaded" or "failed"
			if not ok then
				Zenith.log("Error", name .. " init fallo: " .. tostring(err))
			end
		end
	end
end

function Zenith.stopAll()
	for i = #order, 1, -1 do
		local name = order[i]
		local m = modules[name]
		if m and m.state == "loaded" then
			if m.stop then
				pcall(m.stop)
			end
			for j = #m.owned, 1, -1 do
				local c = m.owned[j]
				pcall(function()
					if typeof(c) == "RBXScriptConnection" then
						c:Disconnect()
					elseif type(c) == "table" and type(c.Disconnect) == "function" then
						c:Disconnect()
					end
				end)
			end
			table.clear(m.owned)
			m.state = "pending"
		end
	end
end

function Zenith.instancePath(inst)
	if typeof(inst) ~= "Instance" then
		return tostring(inst)
	end
	if inst == game then
		return "game"
	end
	local chain = {}
	local cur = inst
	local guard = 0
	while cur and cur ~= game and guard < 64 do
		guard += 1
		local okP, par = pcall(function()
			return cur.Parent
		end)
		if not okP then
			break
		end
		if par == game then
			local okS, cls = pcall(function()
				return cur.ClassName
			end)
			if okS and cls and cls ~= "" then
				table.insert(chain, 1, string.format("game:GetService(%q)", cls))
				return table.concat(chain, "")
			end
		end
		local okN, nm = pcall(function()
			return cur.Name
		end)
		local segName = okN and tostring(nm) or "Instance"
		if string.match(segName, "^[%a_][%w_]*$") then
			table.insert(chain, 1, "." .. segName)
		else
			table.insert(chain, 1, string.format("[%q]", segName))
		end
		cur = par
	end
	if #chain == 0 then
		return "nil"
	end
	local head = chain[1]
	if string.sub(head, 1, 1) == "." then
		chain[1] = string.sub(head, 2)
	end
	return "nil --[[" .. table.concat(chain, "") .. "]]"
end

function Zenith.decompileInstance(inst)
	if typeof(inst) ~= "Instance" then
		return nil
	end
	local decFn = Zenith.capFn("decompile")
	if not decFn then
		Zenith.log("Warn", "Decompile no disponible para: " .. Zenith.instancePath(inst))
		return nil
	end
	local ok, src = pcall(decFn, inst)
	if ok and type(src) == "string" then
		local out = Zenith.outputs and Zenith.outputs["Decompiler Output"]
		if out and type(out.setText) == "function" then
			out.setText(src)
		end
		Zenith.log("Info", "Decompiled: " .. Zenith.instancePath(inst))
		return src
	end
	return nil
end

		local idx = 1
		for i, m in ipairs(PARENT_MODES) do
			if m == st.mode then idx = i end
		end
		st.mode = PARENT_MODES[(idx % #PARENT_MODES) + 1]
		U.zGui = findZenithGui()
		if not attach(U.gui, st.mode) then
			toast("No se pudo re-parentar")
		else
			toast("Parent: " .. st.mode .. " -> " .. st.parentLabel)
		end
	end

	local function renderCards(g)
		for _, c in ipairs(U.cards) do c:Destroy() end
		U.cards = {}
		if not g then
			U.pageChip.Text = "- / -"
			return
		end
		local total = #g.calls
		local pages = math.max(1, math.ceil(total / PER_PAGE))
		if st.page > pages then st.page = pages end
		if st.page < 1 then st.page = 1 end
		U.pageChip.Text = string.format("%d / %d", st.page, pages)
		local first = total - (st.page - 1) * PER_PAGE
		local last = math.max(1, first - PER_PAGE + 1)
		local order = 0
		for i = first, last, -1 do
			order += 1
			buildCard(g.calls[i], order)
		end
	end

	local function computeGroups(S)
		local mode = TABS[st.tab].dir
		local needle = string.lower(st.search)
		local byKey, order, counts = {}, {}, { C2S = 0, S2C = 0, EV = 0 }
		if not S then return order, byKey, counts end
		local ign = S.ignoredNames or {}
		for _, e in ipairs(S.entries) do
			local d = e.dir
			local cat = (d == "INT" or d == "ATTR") and "EV" or d
			if counts[cat] ~= nil then
				local ln = lowerCache[e]
				if not ln then
					ln = string.lower(tostring(e.name))
					lowerCache[e] = ln
				end
				if not ign[ln] then
					counts[cat] += 1
					if cat == mode and (needle == "" or string.find(ln, needle, 1, true)) then
						local key = tostring(e.className) .. "|" .. tostring(e.path) .. "|" .. tostring(e.method)
						local g = byKey[key]
						if not g then
							g = { key = key, name = e.name, className = e.className, method = e.method, path = e.path, calls = {}, count = 0 }
							byKey[key] = g
							order[#order + 1] = g
						end
						g.count += 1
						g.calls[g.count] = e
						g.last = e
					end
				end
			end
		end
		return order, byKey, counts
	end

	end

	local function collectHookRows(S)
		local rows = {}
		local function H(id, text) rows[#rows + 1] = { id = "h:" .. id, text = text } end
		local function R(id, text, state, detail, btnText, onClick)
			rows[#rows + 1] = { id = id, text = text, state = state, detail = detail, btnText = btnText, onClick = onClick }
		end
		local function recState(r) if not r then return "off" end; return r.active and "on" or "warn" end
		if not S then
			H("none", "RSPY")
			R("nospy", "Zenith.RSpy", "bad", "modulo no cargado")
		else
			H("hooks", "HOOKS")
			local nc = S.namecallRec
			R("nc", "__namecall", recState(nc), nc and (nc.active and ("activo (" .. tostring(S.namecallVia or "hook") .. ")") or "pass-through") or "no instalado")
			local dh = S.directHooks or {}
			local act = 0
			for _, r in ipairs(dh) do if r.active then act += 1 end end
			R("dh", "hookfunction / oth", #dh == 0 and "off" or (act > 0 and "on" or "warn"), string.format("%d/%d activos", act, #dh))
			for i, r in ipairs(dh) do
				local okN, nm = pcall(debug.info, r.target, "n")
				R("dh" .. i, "   " .. ((okN and type(nm) == "string" and nm ~= "") and nm or ("fn#" .. i)), recState(r), r.usedOth and "oth" or "hookfunction")
			end
			R("ni", "__newindex (callbacks)", recState(S.newindexRec), S.newindexRec and (S.newindexRec.active and "activo" or "pass-through") or "no instalado")
			R("sg", "Signal hook (__index)", recState(S.signalHook), S.signalHook and (S.signalHook.active and "activo" or "pass-through") or "no instalado")
			R("ac", "Actors", S.actorChannel and "on" or "off", string.format("%d conexiones", #(S.actorConns or {})))
			local s2c = #(S.s2cConns or {})
			R("s2c", "S2C connections", s2c > 0 and "on" or "off", tostring(s2c))
			local wc = 0
			for _ in pairs(S.wrappedCallbacks or {}) do wc += 1 end
			R("wc", "Callbacks wrapped", wc > 0 and "on" or "off", tostring(wc))
			R("int", "Interactions", #(S.interactionConns or {}) > 0 and "on" or "off", tostring(#(S.interactionConns or {})))
			R("att", "Attributes", #(S.attrConns or {}) > 0 and "on" or "off", tostring(#(S.attrConns or {})))
			H("pipe", "PIPELINE")
			local on = getFlag(RN.enabled)
			R("spy", "Remote Spy", on and "on" or "off", on and "capturando" or "pausado", on and "Pausar" or "Activar", function() setFlag(RN.enabled, not getFlag(RN.enabled)) end)
			local q = #(S.queue or {})
			R("q", "Cola", q > 1500 and "warn" or "on", string.format("%d en cola", q))
			R("log", "Entradas", "on", string.format("%d / %d  (seq %d)", #S.entries, tonumber(Zenith.values[RN.maxLogs]) or 300, S.seq or 0))
		end
		H("par", "OVERLAY PARENT")
		R("par", "Overlay", U.gui and U.gui.Parent and "on" or "bad", (U.gui and U.gui.Parent and pathOfInst(U.gui.Parent) or "?") .. " [" .. st.parentLabel .. "]", "Cambiar", cycleParent)
		R("zui", "Zenith GUI", U.zGui and "on" or "off", U.zGui and pathOfInst(U.zGui) or "no detectado", "Detectar", function() U.zGui = findZenithGui(); toast(U.zGui and "Zenith GUI detectado" or "No detectado") end)
		R("ord", "DisplayOrder", "on", tostring(U.gui and U.gui.DisplayOrder or 0))
		if S then
			local bl, n = {}, 0
			for inst, v in pairs(S.blocked or {}) do
				if v == true and typeof(inst) == "Instance" then
					n += 1
					if n <= 30 then bl[#bl + 1] = inst end
				end
			end
			H("blk", "BLOCKED REMOTES (" .. n .. ")")
			for i, inst in ipairs(bl) do
				R("b" .. i, clip(pathOfInst(inst), 40), "bad", inst.ClassName, "Unblock", function() S.blocked[inst] = nil; st.dirty = true end)
			end
		end
		H("cap", "CAPABILITIES")
		local have, total, missing = 0, 0, {}
		for name, statusVal in pairs(Zenith.capabilities and Zenith.capabilities.status or {}) do
			total += 1
			if statusVal ~= nil and statusVal ~= "missing" then
				have += 1
			else
				missing[#missing + 1] = name
			end
		end
		table.sort(missing)
		R("capn", "Disponibles", have == total and "on" or "warn", string.format("%d / %d", have, total))
		if #missing > 0 then R("capm", "Faltan", "warn", table.concat(missing, ", ")) end
		return rows
	end
	local function updateHooks()
		local rows = collectHookRows(Zenith.RSpy)
		U.hookPool.begin()
		for i, d in ipairs(rows) do
			local r = U.hookPool.get(d.id, i)
			r.name.Text = d.text
			if r.dot then
				r.dot.BackgroundColor3 = d.state == "on" and T.good or d.state == "warn" and T.warn or d.state == "bad" and T.bad or T.mute
				r.detail.Text = d.detail or ""
				if d.btnText then
					r.btn.Visible, r.btn.Text, r.onClick = true, d.btnText, d.onClick
					r.detail.Size = UDim2.new(0.58, -78, 1, 0)
				else
					r.btn.Visible, r.onClick = false, nil
					r.detail.Size = UDim2.new(0.58, -10, 1, 0)
				end
			end
		end
		U.hookPool.finish()
	end

	function refresh(force)
		if not st.alive or not U.gui then return end
		local S = Zenith.RSpy
		updateStatus(S)
		local tab = TABS[st.tab]
		if tab.page == "hooks" then return updateHooks() end
		if tab.page == "settings" then
			for _, fn in ipairs(U.refreshers) do fn() end
			return
		end
		local g0 = curGroup()
		if g0 and g0.last then
			U.blockBtn.Text = (S and S.blocked[g0.last.remote] == true) and "Unblock" or "Block"
		end
		if st.paused and not force then return end
		local sig = table.concat({ S and S.seq or -1, S and #S.entries or -1, (S and S.entries[1] and S.entries[1].id) or 0, S and S.ignoreGen or 0, getFlag(RN.enabled) and 1 or 0, st.tab, st.search }, "|")
		if sig == st.sig and not st.dirty and not force then return end
		st.sig, st.dirty = sig, false
		local order, byKey, counts = computeGroups(S)
		st.byKey, st.nGroups = byKey, #order
		U.tabBtns[1].Text = "Outgoing  " .. counts.C2S
		U.tabBtns[2].Text = "Incoming  " .. counts.S2C
		U.tabBtns[3].Text = "Events  " .. counts.EV
		if not byKey[st.selKey] then
			st.selKey, st.page = order[1] and order[1].key or nil, 1
		end
		U.listPool.begin()
		for i = 1, math.min(#order, MAX_GROUPS) do
			local g = order[i]
			local r = U.listPool.get(g.key, i)
			local blocked = S and g.last and S.blocked[g.last.remote] == true
			local ic = ICON[g.className] or { "A", T.good }
			r.icon.Text, r.icon.TextColor3 = ic[1], ic[2]
			r.name.Text = g.name
			r.name.TextColor3 = blocked and T.bad or T.text
			r.count.Text = "x" .. g.count
			r.root.BackgroundColor3 = (st.selKey == g.key) and T.sel or T.panel
		end
		U.listPool.finish()
		local g = byKey[st.selKey]
		if g then
			U.info.Text = g.path .. "  :" .. tostring(g.method)
			U.empty.Visible = false
			local cs = table.concat({ g.key, g.count, st.page, g.last.id }, "|")
			if cs ~= st.cardSig or force then
				st.cardSig = cs
				renderCards(g)
			end
		else
			U.info.Text = ""
			st.cardSig = ""
			renderCards(nil)
			U.empty.Text = getFlag(RN.enabled) and "Sin llamadas todavia..." or "Remote Spy esta apagado. Activalo en Settings o en la pestana Hooks."
			U.empty.Visible = true
		end
	end

-- RSpyPro (Sub-arg block/ignore, arg mutators, return spoofers, S2C replay, diff & fuzzer).
do
	local Pro = {
		subBlocked = {},
		subIgnored = {},
		argMutators = setmetatable({}, { __mode = "k" }),
		retSpoofers = setmetatable({}, { __mode = "k" }),
	}

	local function quoteBytes(str)
		local out = {}
		for i = 1, #str do
			local b = string.byte(str, i)
			if b == 34 then
				out[#out + 1] = '\\"'
			elseif b == 92 then
				out[#out + 1] = "\\\\"
			elseif b == 10 then
				out[#out + 1] = "\\n"
			elseif b == 13 then
				out[#out + 1] = "\\r"
			elseif b == 9 then
				out[#out + 1] = "\\t"
			elseif b >= 32 and b <= 126 then
				out[#out + 1] = string.char(b)
			else
				out[#out + 1] = string.format("\\%03d", b)
			end
		end
		return '"' .. table.concat(out) .. '"'
	end

	function Pro.serializeValue(v, depth, activeStack)
		depth = depth or 0
		activeStack = activeStack or {}
		local t = typeof(v)
		if t == "nil" then
			return "nil"
		elseif t == "boolean" or t == "number" then
			return tostring(v)
		elseif t == "string" then
			return quoteBytes(v)
		elseif t == "Instance" then
			return Zenith.instancePath and Zenith.instancePath(v) or tostring(v)
		elseif t == "buffer" then
			local len = buffer.len(v)
			local previewLen = math.min(len, 512)
			local raw = buffer.readstring(v, 0, previewLen)
			if len > previewLen then
				return string.format("buffer.fromstring(%s) --[[ +%d bytes ]]", quoteBytes(raw), len - previewLen)
			end
			return string.format("buffer.fromstring(%s)", quoteBytes(raw))
		elseif t == "Axes" then
			return string.format("Axes.new(%s)", tostring(v))
		elseif t == "Faces" then
			return string.format("Faces.new(%s)", tostring(v))
		elseif t == "Vector2int16" then
			return string.format("Vector2int16.new(%d, %d)", v.X, v.Y)
		elseif t == "table" then
			if activeStack[v] then
				return "{ --[[ cyclic ]] }"
			end
			if depth >= 8 then
				return "{ --[[ max depth ]] }"
			end
			activeStack[v] = true
			local parts = {}
			for k, val in pairs(v) do
				local ks = (type(k) == "string" and string.match(k, "^[%a_][%w_]*$")) and k or ("[" .. Pro.serializeValue(k, depth + 1, activeStack) .. "]")
				parts[#parts + 1] = ks .. " = " .. Pro.serializeValue(val, depth + 1, activeStack)
			end
			activeStack[v] = nil
			return "{ " .. table.concat(parts, ", ") .. " }"
		end
		return tostring(v)
	end

	local function subKey(remote, firstArg)
		return tostring(remote) .. "::" .. typeof(firstArg) .. ":" .. tostring(firstArg)
	end

	function Pro.setSubRule(kind, remote, firstArg, enabled)
		local tbl = (kind == "block") and Pro.subBlocked or Pro.subIgnored
		tbl[subKey(remote, firstArg)] = enabled and true or nil
	end

	function Pro.shouldBlock(remote, args)
		local first = type(args) == "table" and args[1] or nil
		return Pro.subBlocked[subKey(remote, first)] == true
	end

	function Pro.shouldIgnore(remote, args)
		local first = type(args) == "table" and args[1] or nil
		return Pro.subIgnored[subKey(remote, first)] == true
	end

	function Pro.setArgMutator(remote, mutatorFn)
		Pro.argMutators[remote] = mutatorFn
	end

	function Pro.setReturnSpoofer(remote, spooferFn)
		Pro.retSpoofers[remote] = spooferFn
	end

	function Pro.applyOutboundIntercept(remote, packedArgs)
		local mut = Pro.argMutators[remote]
		if type(mut) ~= "function" then
			return packedArgs
		end
		local ok, replaced = pcall(mut, packedArgs)
		if ok and type(replaced) == "table" then
			return replaced
		end
		return packedArgs
	end

	function Pro.applyReturnSpoof(remote, packedResults)
		local sp = Pro.retSpoofers[remote]
		if type(sp) ~= "function" then
			return packedResults
		end
		local ok, replaced = pcall(sp, packedResults)
		if ok and type(replaced) == "table" then
			return replaced
		end
		return packedResults
	end

	function Pro.replayServerToClient(remote, ...)
		if typeof(remote) ~= "Instance" or not remote:IsA("RemoteEvent") then
			return false, "Solo RemoteEvent soporta replay S2C"
		end
		local fs = Zenith.capFn("firesignal")
		if fs then
			local ok, err = pcall(fs, remote.OnClientEvent, ...)
			return ok, err
		end
		local getConns = Zenith.capFn("getconnections")
		if getConns then
			local okC, conns = pcall(getConns, remote.OnClientEvent)
			if okC and type(conns) == "table" then
				local firedCount = 0
				for _, conn in ipairs(conns) do
					if conn.Enabled ~= false and type(conn.Fire) == "function" then
						pcall(conn.Fire, conn, ...)
						firedCount += 1
					end
				end
				return firedCount > 0, firedCount == 0 and "Sin conexiones activas" or nil
			end
		end
		return false, "firesignal y getconnections no disponibles"
	end

	function Pro.diffEntries(entryA, entryB)
		if not entryA or not entryB then
			return "-- Selecciona dos llamadas validas para comparar"
		end
		local lines = {
			string.format("-- Diff: #%d vs #%d (%s)", entryA.id or 0, entryB.id or 0, tostring(entryA.name or entryB.name)),
		}
		local maxN = math.max(entryA.args and (entryA.args.n or #entryA.args) or 0, entryB.args and (entryB.args.n or #entryB.args) or 0)
		for i = 1, maxN do
			local va = Pro.serializeValue(entryA.args and entryA.args[i])
			local vb = Pro.serializeValue(entryB.args and entryB.args[i])
			if va == vb then
				lines[#lines + 1] = string.format("  [arg %d] == %s", i, va)
			else
				lines[#lines + 1] = string.format("- [arg %d] %s", i, va)
				lines[#lines + 1] = string.format("+ [arg %d] %s", i, vb)
			end
		end
		return table.concat(lines, "\n")
	end

	function Pro.fuzzRemote(remote, method, argIndex, baseArgs)
		if typeof(remote) ~= "Instance" then
			return 0
		end
		local fn = remote[method]
		if type(fn) ~= "function" then
			return 0
		end
		local probes = { 0, -1, 1e9, 0 / 0, "", "__ZENITH_FUZZ__", false, true, {}, nil }
		local sent = 0
		for _, payload in ipairs(probes) do
			local copyArgs = table.clone(baseArgs or {})
			copyArgs[argIndex or 1] = payload
			local n = math.max(copyArgs.n or #copyArgs, argIndex or 1)
			local ok = pcall(fn, remote, table.unpack(copyArgs, 1, n))
			if ok then
				sent += 1
			end
		end
		return sent
	end

	local function getActiveEntry()
		local S = Zenith.RSpy
		if not S then
			return nil
		end
		if S.selectedLabel and S.byLabel and S.byLabel[S.selectedLabel] then
			return S.byLabel[S.selectedLabel]
		end
		return S.entries and S.entries[#S.entries] or nil
	end

	Zenith.RSpyPro = Pro
	Zenith.register("RSpyPro", {
		requires = {},
		optional = { "firesignal", "getconnections" },
		init = function()
			Zenith.RSpyPro = Pro

			Zenith.actions["Sub-Block Arg1"] = function()
				local e = getActiveEntry()
				if not e or not e.remote then
					return Zenith.log("Warn", "RSpyPro: selecciona una llamada primero")
				end
				local first = e.args and e.args[1]
				local k = subKey(e.remote, first)
				local nextOn = not (Pro.subBlocked[k] == true)
				Pro.setSubRule("block", e.remote, first, nextOn)
				Zenith.log("Info", string.format("RSpyPro: Sub-Block (%s, %s) = %s", tostring(e.name), tostring(first), tostring(nextOn)))
			end

			Zenith.actions["Sub-Ignore Arg1"] = function()
				local e = getActiveEntry()
				if not e or not e.remote then
					return Zenith.log("Warn", "RSpyPro: selecciona una llamada primero")
				end
				local first = e.args and e.args[1]
				local k = subKey(e.remote, first)
				local nextOn = not (Pro.subIgnored[k] == true)
				Pro.setSubRule("ignore", e.remote, first, nextOn)
				Zenith.log("Info", string.format("RSpyPro: Sub-Ignore (%s, %s) = %s", tostring(e.name), tostring(first), tostring(nextOn)))
			end

			Zenith.actions["Replay S2C"] = function()
				local e = getActiveEntry()
				if not e or not e.remote then
					return Zenith.log("Warn", "RSpyPro: selecciona un RemoteEvent S2C primero")
				end
				local args = e.args or { n = 0 }
				local ok, err = Pro.replayServerToClient(e.remote, table.unpack(args, 1, args.n or #args))
				Zenith.log(ok and "Info" or "Warn", ok and ("RSpyPro: S2C replayed -> " .. tostring(e.name)) or ("RSpyPro S2C: " .. tostring(err)))
			end

			Zenith.actions["Fuzz Remote"] = function()
				local e = getActiveEntry()
				if not e or not e.remote or e.dir == "S2C" or e.dir == "INT" or e.dir == "ATTR" then
					return Zenith.log("Warn", "RSpyPro: selecciona una llamada C2S para fuzzear")
				end
				local sent = Pro.fuzzRemote(e.remote, e.method, 1, e.args)
				Zenith.log("Info", string.format("RSpyPro: Fuzz '%s' completo (%d probes enviadas)", tostring(e.name), sent))
			end

			Zenith.actions["Diff Last 2"] = function()
				local S = Zenith.RSpy
				local e = getActiveEntry()
				if not S or not e then
					return Zenith.log("Warn", "RSpyPro: se necesitan al menos 2 llamadas del mismo remote")
				end
				local matches = {}
				for i = #S.entries, 1, -1 do
					local cur = S.entries[i]
					if cur.remote == e.remote and cur.method == e.method then
						matches[#matches + 1] = cur
						if #matches == 2 then
							break
						end
					end
				end
				if #matches < 2 then
					return Zenith.log("Warn", "RSpyPro: captura al menos 2 llamadas de " .. tostring(e.name))
				end
				local txt = Pro.diffEntries(matches[2], matches[1])
				local out = Zenith.outputs and Zenith.outputs["Spy Log"]
				if out and type(out.setText) == "function" then
					out.setText(txt)
				end
				Zenith.log("Info", "RSpyPro: Diff generado en Spy Log")
			end
		end,
		stop = function()
			table.clear(Pro.subBlocked)
			table.clear(Pro.subIgnored)
			table.clear(Pro.argMutators)
			table.clear(Pro.retSpoofers)
			for _, k in ipairs({ "Sub-Block Arg1", "Sub-Ignore Arg1", "Replay S2C", "Fuzz Remote", "Diff Last 2" }) do
				Zenith.actions[k] = nil
			end
		end,
	})
end

-- Explorer & Inspector de conexiones (DataModel + Nil Instances).
do
	local N = {
		root = "Explorer Root",
		search = "Explorer Search",
		list = "Explorer List",
		output = "Explorer Inspector",
		connList = "Explorer Connections",
		openParent = "Explorer Parent",
		openSelected = "Explorer Open",
		refresh = "Explorer Refresh",
		copyPath = "Explorer Copy Path",
		decompileInst = "Explorer Decompile",
		triggerInst = "Explorer Trigger",
		destroyInst = "Explorer Destroy",
		toggleConn = "Explorer Toggle Conn",
		fireConn = "Explorer Fire Conn",
	}

	local DEFAULT_ROOTS = {
		"Workspace", "Players", "ReplicatedStorage", "ReplicatedFirst",
		"StarterGui", "StarterPack", "Lighting", "SoundService", "CoreGui", "Nil Instances",
	}

	local CLASS_TAGS = {
		LocalScript = "[LS]", ModuleScript = "[MS]", Script = "[SS]",
		RemoteEvent = "[RE]", RemoteFunction = "[RF]", UnreliableRemoteEvent = "[URE]",
		BindableEvent = "[BE]", BindableFunction = "[BF]",
		Folder = "[F]", Model = "[M]", Part = "[P]", MeshPart = "[MP]",
		ProximityPrompt = "[PP]", ClickDetector = "[CD]",
	}

	local INSPECT_SIGNALS = {
		"Changed", "AttributeChanged", "ChildAdded", "ChildRemoved", "DescendantAdded",
		"Activated", "MouseButton1Click", "MouseButton1Down", "Triggered",
		"MouseClick", "Touched", "OnClientEvent", "Event", "StateChanged",
	}

	local state = {
		currentParent = nil,
		currentInst = nil,
		byLabel = {},
		shownLabels = {},
		connRecords = {},
		connByLabel = {},
		selectedConnLabel = nil,
	}

	local function getRootTarget(rootName)
		if rootName == "Nil Instances" then
			return "NIL_ROOT"
		end
		local ok, svc = pcall(game.GetService, game, rootName)
		return (ok and svc) or nil
	end

	local function safeProp(inst, prop)
		local ok, val = pcall(function()
			return inst[prop]
		end)
		if ok then
			return true, val
		end
		return false, nil
	end

	local function instPath(inst)
		if Zenith.instancePath then
			return Zenith.instancePath(inst)
		end
		local ok, s = pcall(inst.GetFullName, inst)
		return ok and tostring(s) or "?"
	end

	local function collectChildren(parentTarget)
		if parentTarget == "NIL_ROOT" then
			local gni = Zenith.capFn("getnilinstances")
			if not gni then
				return {}
			end
			local ok, list = pcall(gni)
			return (ok and type(list) == "table") and list or {}
		end
		if typeof(parentTarget) ~= "Instance" then
			local roots = {}
			for _, rName in ipairs(DEFAULT_ROOTS) do
				local tgt = getRootTarget(rName)
				if tgt and tgt ~= "NIL_ROOT" then
					table.insert(roots, tgt)
				end
			end
			return roots
		end
		local ok, children = pcall(parentTarget.GetChildren, parentTarget)
		return (ok and type(children) == "table") and children or {}
	end

	local function refreshTree()
		local q = string.lower(string.match(tostring(Zenith.values[N.search] or ""), "^%s*(.-)%s*$"))
		table.clear(state.byLabel)
		table.clear(state.shownLabels)

		if state.currentParent and state.currentParent ~= "NIL_ROOT" then
			local upLabel = ".. [Parent]"
			state.byLabel[upLabel] = "GO_UP"
			table.insert(state.shownLabels, upLabel)
		elseif state.currentParent == "NIL_ROOT" then
			local upLabel = ".. [Roots]"
			state.byLabel[upLabel] = "GO_ROOTS"
			table.insert(state.shownLabels, upLabel)
		end

		local children = collectChildren(state.currentParent)
		local seenCount = {}
		for _, child in ipairs(children) do
			if typeof(child) == "Instance" then
				local okN, nm = safeProp(child, "Name")
				local okC, cls = safeProp(child, "ClassName")
				local nameStr = tostring(okN and nm or "Instance")
				local clsStr = tostring(okC and cls or "Instance")
				if q == "" or string.find(string.lower(nameStr), q, 1, true) or string.find(string.lower(clsStr), q, 1, true) then
					local tag = CLASS_TAGS[clsStr] or ("[" .. clsStr .. "]")
					local base = string.format("%s %s", tag, nameStr)
					seenCount[base] = (seenCount[base] or 0) + 1
					local lbl = seenCount[base] > 1 and string.format("%s #%d", base, seenCount[base]) or base
					state.byLabel[lbl] = child
					table.insert(state.shownLabels, lbl)
				end
			end
		end
		Zenith.setItems(N.list, state.shownLabels)
	end

	local function scanConnections(inst)
		table.clear(state.connRecords)
		table.clear(state.connByLabel)
		local labels = {}
		local getConns = Zenith.capFn("getconnections")
		if not getConns or typeof(inst) ~= "Instance" then
			Zenith.setItems(N.connList, labels)
			return
		end

		local isExec = Zenith.capFn("isexecutorclosure")
		local getInfo = Zenith.capFn("getinfo")
		local idx = 0
		for _, sigName in ipairs(INSPECT_SIGNALS) do
			local okS, sig = safeProp(inst, sigName)
			if okS and typeof(sig) == "RBXScriptSignal" then
				local okC, conns = pcall(getConns, sig)
				if okC and type(conns) == "table" then
					for _, conn in ipairs(conns) do
						idx += 1
						local fn = conn.Function
						local isLua = type(fn) == "function"
						local execFlag = isLua and isExec and select(2, pcall(isExec, fn)) == true
						local src = "-"
						if isLua and getInfo then
							local okI, info = pcall(getInfo, fn)
							if okI and type(info) == "table" then
								src = tostring(info.short_src or info.source or "-")
								if info.currentline and info.currentline > 0 then
									src = src .. ":" .. tostring(info.currentline)
								end
							end
						end
						local enabled = (conn.Enabled ~= false)
						local lbl = string.format("#%d [%s] %s (%s) %s", idx, enabled and "ON" or "OFF", sigName, execFlag and "Exec" or (isLua and "Lua" or "C"), src)
						local rec = { signalName = sigName, signal = sig, conn = conn, func = fn, enabled = enabled, source = src }
						table.insert(state.connRecords, rec)
						state.connByLabel[lbl] = rec
						table.insert(labels, lbl)
					end
				end
			end
		end
		Zenith.setItems(N.connList, labels)
	end

	local function inspectInstance(inst)
		state.currentInst = inst
		if typeof(inst) ~= "Instance" then
			return
		end
		local okN, nm = safeProp(inst, "Name")
		local okC, cls = safeProp(inst, "ClassName")
		local lines = {
			"-- Zenith Instance Inspector",
			"-- Name      : " .. tostring(okN and nm or "?"),
			"-- ClassName : " .. tostring(okC and cls or "?"),
			"-- Path      : " .. instPath(inst),
		}

		local props = { "Archivable", "Value", "Text", "Visible", "Enabled", "Disabled", "Anchored", "CanCollide", "Transparency", "Size", "Position", "CFrame", "Color", "SoundId", "AnimationId", "MeshId", "ActionText", "ObjectText", "MaxActivationDistance" }
		lines[#lines + 1] = "-- Properties:"
		for _, p in ipairs(props) do
			local okP, v = safeProp(inst, p)
			if okP and v ~= nil then
				lines[#lines + 1] = string.format("  %s = %s", p, tostring(v))
			end
		end

		local okA, attrs = pcall(inst.GetAttributes, inst)
		if okA and type(attrs) == "table" and next(attrs) ~= nil then
			lines[#lines + 1] = "-- Attributes:"
			for k, v in pairs(attrs) do
				lines[#lines + 1] = string.format("  [%s] = %s (%s)", tostring(k), tostring(v), typeof(v))
			end
		end

		scanConnections(inst)
		lines[#lines + 1] = "-- Connections found: " .. tostring(#state.connRecords)

		local out = Zenith.outputs and Zenith.outputs[N.output]
		if out and type(out.setText) == "function" then
			out.setText(table.concat(lines, "\n"))
		end
	end

	Zenith.Explorer = state
	Zenith.register("Explorer", {
		requires = {},
		optional = { "getnilinstances", "getconnections", "firesignal", "fireproximityprompt", "fireclickdetector", "getinfo", "isexecutorclosure" },
		init = function()
			Zenith.Explorer = state
			Zenith.hooks[N.root] = function(rootName)
				state.currentParent = getRootTarget(tostring(rootName or "Workspace"))
				refreshTree()
			end
			Zenith.hooks[N.search] = function()
				refreshTree()
			end
			Zenith.hooks[N.list] = function(lbl)
				local target = state.byLabel[lbl]
				if target == "GO_UP" and typeof(state.currentParent) == "Instance" then
					local okP, par = safeProp(state.currentParent, "Parent")
					state.currentParent = (okP and par ~= game) and par or nil
					refreshTree()
				elseif target == "GO_ROOTS" then
					state.currentParent = nil
					refreshTree()
				elseif typeof(target) == "Instance" then
					inspectInstance(target)
				end
			end
			Zenith.hooks[N.connList] = function(lbl)
				state.selectedConnLabel = lbl
			end
			Zenith.actions[N.openSelected] = function()
				if typeof(state.currentInst) == "Instance" then
					state.currentParent = state.currentInst
					refreshTree()
				end
			end
			Zenith.actions[N.openParent] = function()
				if typeof(state.currentParent) == "Instance" then
					local okP, par = safeProp(state.currentParent, "Parent")
					state.currentParent = (okP and par ~= game) and par or nil
					refreshTree()
				end
			end
			Zenith.actions[N.refresh] = refreshTree
			Zenith.actions[N.copyPath] = function()
				if typeof(state.currentInst) ~= "Instance" then
					return Zenith.log("Warn", "Explorer: selecciona una instancia primero")
				end
				Zenith.copy(instPath(state.currentInst))
				Zenith.log("Info", "Explorer: ruta copiada")
			end
			Zenith.actions[N.decompileInst] = function()
				local inst = state.currentInst
				if typeof(inst) == "Instance" and inst:IsA("LuaSourceContainer") and Zenith.decompileInstance then
					Zenith.decompileInstance(inst)
					if Zenith.switchTab then
						Zenith.switchTab("Decompiler")
					end
				else
					Zenith.log("Warn", "Explorer: la instancia seleccionada no es un script")
				end
			end
			Zenith.actions[N.triggerInst] = function()
				local inst = state.currentInst
				if typeof(inst) ~= "Instance" then
					return
				end
				if inst:IsA("ProximityPrompt") and Zenith.cap("fireproximityprompt") then
					pcall(Zenith.capFn("fireproximityprompt"), inst)
				elseif inst:IsA("ClickDetector") and Zenith.cap("fireclickdetector") then
					pcall(Zenith.capFn("fireclickdetector"), inst)
				elseif inst:IsA("GuiButton") and Zenith.cap("firesignal") then
					pcall(Zenith.capFn("firesignal"), inst.MouseButton1Click)
				else
					Zenith.log("Warn", "Explorer: instancia no disparable")
				end
			end
			Zenith.actions[N.destroyInst] = function()
				if typeof(state.currentInst) == "Instance" then
					pcall(state.currentInst.Destroy, state.currentInst)
					state.currentInst = nil
					refreshTree()
				end
			end
			Zenith.actions[N.toggleConn] = function()
				local rec = state.connByLabel[state.selectedConnLabel]
				if not rec or not rec.conn then
					return Zenith.log("Warn", "Explorer: selecciona una conexion primero")
				end
				if rec.enabled and type(rec.conn.Disable) == "function" then
					pcall(rec.conn.Disable, rec.conn)
				elseif not rec.enabled and type(rec.conn.Enable) == "function" then
					pcall(rec.conn.Enable, rec.conn)
				end
				scanConnections(state.currentInst)
			end
			Zenith.actions[N.fireConn] = function()
				local rec = state.connByLabel[state.selectedConnLabel]
				if not rec then
					return Zenith.log("Warn", "Explorer: selecciona una conexion primero")
				end
				local fs = Zenith.capFn("firesignal")
				if fs and rec.signal then
					pcall(fs, rec.signal)
				elseif rec.conn and type(rec.conn.Fire) == "function" then
					pcall(rec.conn.Fire, rec.conn)
				end
			end
			refreshTree()
		end,
		stop = function()
			for _, k in ipairs({ N.root, N.search, N.list, N.connList }) do
				Zenith.hooks[k] = nil
			end
			for _, k in ipairs({ N.openParent, N.openSelected, N.refresh, N.copyPath, N.decompileInst, N.triggerInst, N.destroyInst, N.toggleConn, N.fireConn }) do
				Zenith.actions[k] = nil
			end
			table.clear(state.byLabel)
			table.clear(state.shownLabels)
			table.clear(state.connRecords)
			table.clear(state.connByLabel)
			state.currentParent, state.currentInst = nil, nil
			if Zenith.Explorer == state then
				Zenith.Explorer = nil
			end
		end,
	})
end

Zenith.startAll()
