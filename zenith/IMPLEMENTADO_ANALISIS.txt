local function resolveCap(path)
	for _, root in ipairs({ envGlobal, _G }) do
		if type(root) == "table" then
			local cur = root
			for k in string.gmatch(path, "[^%.]+") do local ok, v = pcall(function() return cur[k] end); cur = ok and v or nil; if cur == nil then break end end
			if type(cur) == "function" then return cur end
		end
	end
	return nil
end
function Zenith.probe()
	local caps = { status = {}, via = {}, fn = {} }
	for _, e in ipairs(KernelMATRIX) do

	end
	Zenith.capabilities = caps; return caps
end

local _mods, _order = {}, {}
function Zenith.register(name, spec)
	if _mods[name] then return end
	_mods[name] = { init = spec.init, stop = spec.stop, state = "pending", owned = {} }
	table.insert(_order, name)
end
function Zenith.startAll()
	local caps = Zenith.capabilities or Zenith.probe()
	for _, name in ipairs(_order) do
		local m = _mods[name]
		if m and m.state ~= "loaded" and m.init then
			local ok, err = pcall(m.init, caps, { name = name, track = function(c) table.insert(m.owned, c); return c end })
			m.state = ok and "loaded" or "failed"
			if not ok then Zenith.log("Error", name .. " init fallo: " .. tostring(err)) end
		end
	end
end
function Zenith.stopAll()
	for i = #_order, 1, -1 do
		local m = _mods[_order[i]]
		if m and m.state == "loaded" then
			if m.stop then pcall(m.stop) end
			for j = #m.owned, 1, -1 do pcall(function() m.owned[j]:Disconnect() end) end
			table.clear(m.owned); m.state = "pending"
		end
	end
end

	local function serializeRaw(v, depth, seen)

			end

		elseif t == "Instance" then return instancePath(v) elseif t == "Vector3" then return string.format("Vector3.new(%s, %s, %s)", fmtNum(v.X), fmtNum(v.Y), fmtNum(v.Z)) elseif t == "Vector2" then return string.format("Vector2.new(%s, %s)", fmtNum(v.X), fmtNum(v.Y)) elseif t == "Vector3int16" then return string.format("Vector3int16.new(%d, %d, %d)", v.X, v.Y, v.Z)

		elseif t == "BrickColor" then return string.format("BrickColor.new(%s)", quoteStr(tostring(v.Name))) elseif t == "EnumItem" or t == "Enum" then return tostring(v)
		elseif t == "NumberRange" then return string.format("NumberRange.new(%s, %s)", fmtNum(v.Min), fmtNum(v.Max)) elseif t == "TweenInfo" then return string.format("TweenInfo.new(%s, %s, %s, %d, %s, %s)", fmtNum(v.Time), tostring(v.EasingStyle), tostring(v.EasingDirection), v.RepeatCount or 0, tostring(v.Reverses == true), fmtNum(v.DelayTime or 0))
		elseif t == "Ray" then return string.format("Ray.new(%s, %s)", serializeValue(v.Origin, depth + 1, seen), serializeValue(v.Direction, depth + 1, seen)) elseif t == "Region3" then local half = v.Size * 0.5
			return string.format("Region3.new(%s, %s)", serializeValue(v.CFrame.Position - half, depth + 1, seen), serializeValue(v.CFrame.Position + half, depth + 1, seen))
		elseif t == "NumberSequence" then local kps = {}
			for _, kp in ipairs(v.Keypoints or {}) do kps[#kps + 1] = string.format("NumberSequenceKeypoint.new(%s, %s, %s)", fmtNum(kp.Time), fmtNum(kp.Value), fmtNum(kp.Envelope or 0)) end
			return "NumberSequence.new({" .. table.concat(kps, ", ") .. "})"
		elseif t == "ColorSequence" then local kps = {}
			for _, kp in ipairs(v.Keypoints or {}) do kps[#kps + 1] = string.format("ColorSequenceKeypoint.new(%s, %s)", fmtNum(kp.Time), serializeValue(kp.Value, depth + 1, seen)) end
			return "ColorSequence.new({" .. table.concat(kps, ", ") .. "})"
		elseif t == "DateTime" then return string.format("DateTime.fromUnixTimestampMillis(%d)", v.UnixTimestampMillis or 0) elseif t == "buffer" then return string.format("buffer.create(%d)", buffer.len(v))
		elseif t == "function" then
			return "function() end"
		elseif t == "table" then
			if seen[v] then
				return "{ --[[ cyclic ]] }"
			end

			end

					table.insert(parts, pad .. "--[[ ... more keys ]]")
					break
				end
				local keyStr
				if type(k) == "number" then
					keyStr = "[" .. fmtNum(k) .. "]"
				elseif isIdent(k) then keyStr = k
				else
					keyStr = "[" .. serializeValue(k, depth + 1, seen) .. "]"
				end
				table.insert(parts, string.format("%s%s = %s,", pad, keyStr, serializeValue(val, depth + 1, seen)))
			end
			seen[v] = nil; if #parts == 0 then return "{}" end
			return "{\n" .. table.concat(parts, "\n") .. "\n" .. closePad .. "}"
		end
		local desc = string.gsub(tostring(v), "%]%]", "] ]")
		return string.format("nil --[[ %s: %s ]]", t, desc)
	end

	end

			end)

		end

				end
			end

		end

		end
		table.clear(state.directHooks); state.installed = false
	end
