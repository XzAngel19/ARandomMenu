function Zenith.log(lvl, msg)
	local line = "[Zenith/" .. tostring(lvl) .. "] " .. tostring(msg)
	if lvl == "Warn" or lvl == "Error" then warn(line) else print(line) end
end
function Zenith.copy(txt)
	local f = Zenith.capFn("setclipboard")
	if not f then return false end
	return (pcall(f, tostring(txt)))

function Zenith.parseCsvSet(str)
	local set = {}
	for part in string.gmatch(tostring(str or ""), "[^,]+") do
		local trimmed = string.lower((string.gsub(part, "^%s*(.-)%s*$", "%1")))
		if trimmed ~= "" then set[trimmed] = true end
	end
	return set
end
function Zenith.register(name, mod) Zenith.modules[name] = mod end
function Zenith.startAll()
	for name, mod in pairs(Zenith.modules) do
		if type(mod.init) == "function" then
			local ok, err = pcall(mod.init)
			if not ok then Zenith.log("Error", name .. " init: " .. tostring(err)) end
		end
	end
end
function Zenith.stopAll()
	for name, mod in pairs(Zenith.modules) do
		if type(mod.stop) == "function" then
			local ok, err = pcall(mod.stop)
			if not ok then Zenith.log("Error", name .. " stop: " .. tostring(err)) end
		end
	end
end

	function Dec.render()
		local cur = Dec.current
		if not cur then return end
		if cur.static then
			publish(cur.static, cur.staticExt or "txt")
			return
		end
		local mode = string.lower(tostring(Zenith.values[N.mode] or "bytecode"))
		if not cur.bc then
			local fb = tryNativeDecompile(cur.inst, "sin bytecode")
			if fb then
				publish(fb, "lua")
			else
				publish({ "-- Sin bytecode y decompile() no disponible o fallo" })
			end
			return
		end
		local builder = BUILD[mode]
		if not builder then
			publish({ "-- Mode '" .. mode .. "' no esta disponible todavia (Fase 2). Cambia Decompile Mode a Bytecode, Function, Constants, Upvalues o Protos." })
			return
		end
		local ok, res = pcall(builder, cur, viewOpts())
		if ok and type(res) == "table" then
			publish(res, mode == "source" and "lua" or "txt")
			return
		end
		local reason = ok and "resultado invalido" or res
		if mode == "source" then
			local fb = tryNativeDecompile(cur.inst, reason)
			if fb then
				publish(fb, "lua")
				return
			end
		end
		publish({ "-- Error al generar la vista: " .. tostring(reason) })
	end
	local function fetch(inst)
		local f = Zenith.capFn("getscriptbytecode")

		return bc
	end
	local function xmlClean(s)
		return (string.gsub(tostring(s or ""), "%c", function(c)
			if c == "\t" or c == "\n" or c == "\r" then return c end
			return ""
		end))
	end
	local function xmlEscape(s)
		return (string.gsub(xmlClean(s), "[&<>\"']", {
			["&"] = "&amp;",
			["<"] = "&lt;",
			[">"] = "&gt;",
			['"'] = "&quot;",
			["'"] = "&apos;",
		}))
	end
	local function cdataSafe(s)
		return (string.gsub(xmlClean(s), "]]>", "]]]]><![CDATA[>"))
	end
	local exportBusy = false

	local function buildPlaceXml()
		local timeoutSec = math.max(1, num(N.decompileTimeout, 10))
		local saveBc = flag(N.saveBytecode, false)
		local anon = flag(N.anonymizeDump, false)
		local ignoreInternals = flag(N.ignoreInternals, true)
		local svcNames = { "Workspace", "ReplicatedStorage", "Lighting", "StarterGui", "StarterPack", "StarterPlayer" }
		local xml = { '<roblox xmlns:xmime="http://www.w3.org/2005/05/xmlmime" xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance" xsi:noNamespaceSchemaLocation="http://www.roblox.com/roblox.xsd" version="4">' }
		local referent, instCount, scriptCount, truncated = 0, 0, 0, false
		local VALUE_CLASSES = { StringValue = "string", IntValue = "int", NumberValue = "double", BoolValue = "bool" }
		local function emitInst(inst, indent, isRootSvc)
			if typeof(inst) ~= "Instance" then return end
			if instCount >= MAX_EXPORT_INSTANCES then
				truncated = true
				return
			end
			if not isRootSvc then
				local okA, arch = pcall(function() return inst.Archivable end)
				if okA and arch == false then return end
			end
			local okN, iName = pcall(function() return inst.Name end)
			local okC, iClass = pcall(function() return inst.ClassName end)
			iName = tostring(okN and iName or "Instance")
			iClass = tostring(okC and iClass or "Folder")
			if ignoreInternals and (iClass == "Camera" or iClass == "Terrain" or iClass == "Player") then return end
			referent += 1
			instCount += 1
			if instCount % 250 == 0 then
				task.wait()
			end
			local pad = string.rep("\t", indent)
			xml[#xml + 1] = string.format('%s<Item class="%s" referent="RBX%d">', pad, xmlEscape(iClass), referent)
			xml[#xml + 1] = pad .. "\t<Properties>"
			xml[#xml + 1] = string.format('%s\t\t<string name="Name">%s</string>', pad, xmlEscape(iName))
			local isScrOk, isScr = pcall(inst.IsA, inst, "LuaSourceContainer")
			if isScrOk and isScr then
				scriptCount += 1
				local lbl = Index.labelOf[inst] or iName
				local src = decompileEntryToText(inst, lbl, timeoutSec, saveBc, anon)
				xml[#xml + 1] = string.format('%s\t\t<ProtectedString name="Source"><![CDATA[%s]]></ProtectedString>', pad, cdataSafe(src))
			elseif VALUE_CLASSES[iClass] then
				local okV, val = pcall(function() return inst.Value end)
				if okV and val ~= nil then
					local tag = VALUE_CLASSES[iClass]
					xml[#xml + 1] = string.format('%s\t\t<%s name="Value">%s</%s>', pad, tag, xmlEscape(tostring(val)), tag)
				end
			end
			xml[#xml + 1] = pad .. "\t</Properties>"
			local okCh, ch = pcall(inst.GetChildren, inst)
			if okCh and type(ch) == "table" then
				for _, child in ipairs(ch) do
					emitInst(child, indent + 1, false)
				end
			end
			xml[#xml + 1] = pad .. "</Item>"
		end
		for _, svcName in ipairs(svcNames) do
			local okS, svc = pcall(game.GetService, game, svcName)
			if okS and svc then
				emitInst(svc, 1, true)
			end
		end
		xml[#xml + 1] = "</roblox>"
		return table.concat(xml, "\n"), instCount, scriptCount, truncated
	end

	local XREF_CACHE_MAX = 500
	local xrefCache, xrefCount = {}, 0
	local function buildXrefIndex(bytes)
		local bc = parseCached(bytes)
		if not bc then return nil end
		link(bc)
		local imports, strings = {}, {}
		for _, p in ipairs(bc.protos) do
			local fnTag = "#" .. tostring(p.index) .. (p.name and (" (" .. p.name .. ")") or "")
			for _, k in ipairs(p.k) do
				if k.t == "import" then
					local nm = kstr(p, k, true)
					imports[#imports + 1] = { val = nm, low = string.lower(nm), fnTag = fnTag }
				elseif k.t == "string" and type(k.v) == "string" and k.v ~= "" then
					strings[#strings + 1] = { val = k.v, low = string.lower(k.v), fnTag = fnTag }
				end
			end
		end
		return { imports = imports, strings = strings }
	end
	local function getXrefIndex(bytes)
		local hit = xrefCache[bytes]
		if hit then return hit end
		local ok, res = pcall(buildXrefIndex, bytes)
		if not ok or not res then return nil end
		if xrefCount >= XREF_CACHE_MAX then
			xrefCache, xrefCount = {}, 0
		end
		xrefCache[bytes] = res
		xrefCount += 1
		return res
	end
	local PREVIEW_ESC = { ["\n"] = "\\n", ["\r"] = "\\r", ["\t"] = "\\t", ['"'] = '\\"', ["\\"] = "\\\\" }
	local function quoteShort(s)
		return '"' .. (string.gsub(s, '[%c"\\]', function(c)
			return PREVIEW_ESC[c] or string.format("\\%03d", string.byte(c))
		end)) .. '"'
	end
	local searchBusy = false
	function Dec.globalSearch(rawQuery)
		local query = string.match(tostring(rawQuery or Zenith.values[N.globalSearch] or ""), "^%s*(.-)%s*$")
		if query == "" then
			Zenith.log("Warn", "Global Search: escribe un texto o nombre de Remote")
			return 0
		end
		if searchBusy then
			Zenith.log("Warn", "Global Search: ya hay una busqueda en curso")
			return 0
		end
		local entries = table.clone(getExportTargets())
		if #entries == 0 then
			publish({ "-- Global Search: no hay scripts tras filtros" }, "txt")
			return 0
		end
		searchBusy = true
		local needle = string.lower(query)
		local out = {
			"-- Zenith Global Bytecode XRef Search",
			"-- Query: " .. query .. " | Scripts escaneados: " .. tostring(#entries),
			"",
		}
		local matchedLabels, totalHits, skippedNoBc = {}, 0, 0
		local okRun, errRun = pcall(function()
			for i, e in ipairs(entries) do
				local bytes = fetch(e.inst)
				if not bytes then
					skippedNoBc += 1
				else
					local idx = getXrefIndex(bytes)
					if idx then
						local scriptHits = {}
						for _, imp in ipairs(idx.imports) do
							if string.find(imp.low, needle, 1, true) then
								scriptHits[#scriptHits + 1] = string.format("  [IMPORT %s] %s", imp.fnTag, imp.val)
							end
						end
						for _, st in ipairs(idx.strings) do
							if string.find(st.low, needle, 1, true) then
								local cut = #st.val > 140
								local preview = quoteShort(string.sub(st.val, 1, 140)) .. (cut and "..." or "")
								scriptHits[#scriptHits + 1] = string.format("  [STRING %s] %s", st.fnTag, preview)
							end
						end
						if #scriptHits > 0 then
							matchedLabels[#matchedLabels + 1] = e.label
							totalHits += #scriptHits
							if #out < 450 then
								out[#out + 1] = string.format("== %s (%d hits)", e.label, #scriptHits)
								for k = 1, math.min(#scriptHits, 12) do
									out[#out + 1] = scriptHits[k]
								end
								if #scriptHits > 12 then
									out[#out + 1] = string.format("  ... (+%d hits mas en este script)", #scriptHits - 12)
								end
								out[#out + 1] = ""
							end
						end
					end
				end
				if i % 40 == 0 then
					task.wait()
				end
			end
		end)
		searchBusy = false
		if not okRun then
			Zenith.log("Error", "Global Search fallo: " .. tostring(errRun))
			return 0
		end
		out[3] = string.format("-- Coincidencias: %d hits en %d scripts (%d sin bytecode omitidos)", totalHits, #matchedLabels, skippedNoBc)
		if #matchedLabels == 0 then
			out[#out + 1] = "-- Sin coincidencias para: " .. query
		else
			if #out >= 450 then
				out[#out + 1] = "-- (salida truncada: usa un termino mas especifico)"
			end
			table.clear(Index.shown)
			for idx = 1, math.min(#matchedLabels, MAX_SHOWN) do
				Index.shown[idx] = matchedLabels[idx]
			end
			set(N.scriptsFound, #matchedLabels)
			if Zenith.setItems then pcall(Zenith.setItems, N.list, table.clone(Index.shown)) end
		end
		Dec.current = { label = "GlobalSearch_" .. sanitize(query), inst = nil, bc = nil, size = 0, class = "XRef", static = out, staticExt = "txt" }
		publish(out, "txt")
		Zenith.log("Info", string.format("Global Search '%s': %d hits en %d scripts", query, totalHits, #matchedLabels))
		return #matchedLabels, totalHits
	end
	local gcBusy = false
	function Dec.gcSearch(rawQuery)
		local query = string.match(tostring(rawQuery or Zenith.values[N.globalSearch] or ""), "^%s*(.-)%s*$")
		if query == "" then
			Zenith.log("Warn", "GC Search: escribe un texto en Global Search")
			return 0
		end
		if gcBusy then
			Zenith.log("Warn", "GC Search: ya hay un escaneo en curso")
			return 0
		end
		local getGC = Zenith.capFn("getgc")
		if not getGC then
			Zenith.log("Warn", "GC Search requiere getgc() en el executor")
			publish({ "-- GC Closure Search requiere getgc() en el executor" }, "txt")
			return 0
		end
		gcBusy = true
		local needle = string.lower(query)
		local isExecFn = Zenith.capFn("isexecutorclosure")
		local isLFn = Zenith.capFn("islclosure")
		local getKs = Zenith.capFn("getconstants") or (type(debug) == "table" and debug.getconstants)
		local getUps = Zenith.capFn("getupvalues") or (type(debug) == "table" and debug.getupvalues)
		local okG, gcList = pcall(getGC, true)
		if not okG or type(gcList) ~= "table" then
			gcBusy = false
			Zenith.log("Error", "getgc fallo: " .. tostring(gcList))
			return 0
		end
		local function pred(fn, obj)
			local ok, r = pcall(fn, obj)
			if ok then return r end
			return nil
		end
		local ownTables = { [Zenith]=true,[Zenith.values]=true,[Zenith.defaults]=true,[Zenith.actions]=true,[Zenith.hooks]=true,[Zenith.setters]=true,[Zenith.outputs]=true,[Zenith.lists]=true,[Dec]=true,[Index]=true,[Index.entries]=true,[Index.filteredEntries]=true,[Index.shown]=true,[Index.byLabel]=true,[Index.labelOf]=true,[xrefCache]=true,[gcList]=true }
		local out = {
			"-- Zenith Live GC Search (Closures + Tables)",
			"-- Query: " .. query .. " | GC objects: " .. tostring(#gcList),
			"",
		}
		local matchedFn, matchedTbl, scannedFn, scannedTbl = 0, 0, 0, 0
		local okScan, errScan = pcall(function()
			for i, obj in ipairs(gcList) do
				local tObj = type(obj)
				if tObj == "function" then
					local skip = false
					if isExecFn and pred(isExecFn, obj) == true then
						skip = true
					elseif isLFn and pred(isLFn, obj) == false then
						skip = true
					end
					if not skip then
						scannedFn += 1
						local src, line, fname = "?", "?", ""
						if type(debug) == "table" and type(debug.info) == "function" then
							local okI, s, l, n = pcall(debug.info, obj, "sln")
							if okI then
								src = tostring(s or "?")
								line = tostring(l or "?")
								fname = tostring(n or "")
							end
						end
						if src ~= "[C]" then
							local hits = {}
							if fname ~= "" and string.find(string.lower(fname), needle, 1, true) then
								hits[#hits + 1] = "name=" .. fname
							end
							if string.find(string.lower(src), needle, 1, true) then
								hits[#hits + 1] = "src=" .. src
							end
							if getKs then
								local okK, ks = pcall(getKs, obj)
								if okK and type(ks) == "table" then
									for ki, kv in pairs(ks) do
										if type(kv) == "string" and string.find(string.lower(kv), needle, 1, true) then
											hits[#hits + 1] = "K[" .. tostring(ki) .. "]=" .. quoteShort(string.sub(kv, 1, 80))
										end
									end
								end
							end
							if getUps then
								local okU, ups = pcall(getUps, obj)
								if okU and type(ups) == "table" then
									for ui, uv in pairs(ups) do
										if type(uv) == "string" and string.find(string.lower(uv), needle, 1, true) then
											hits[#hits + 1] = "U[" .. tostring(ui) .. "]=" .. quoteShort(string.sub(uv, 1, 80))
										end
									end
								end
							end
							if #hits > 0 then
								matchedFn += 1
								if #out < 450 then
									out[#out + 1] = string.format("[GC FN #%d] %s:%s fn=%s", matchedFn, src, line, fname ~= "" and fname or "anon")
									for k = 1, math.min(#hits, 8) do
										out[#out + 1] = "  " .. hits[k]
									end
									out[#out + 1] = ""
								end
							end
						end
					end
				elseif tObj == "table" and not ownTables[obj] and getmetatable(obj) == nil then
					scannedTbl += 1
					local hits = {}
					local seenKeys = 0
					for tk, tv in pairs(obj) do
						seenKeys += 1
						if seenKeys > 64 then break end
						if type(tk) == "string" and string.find(string.lower(tk), needle, 1, true) then
							hits[#hits + 1] = string.format("[%s] = %s", quoteShort(string.sub(tk, 1, 40)), tostring(tv))
						elseif type(tv) == "string" and string.find(string.lower(tv), needle, 1, true) then
							hits[#hits + 1] = string.format("[%s] = %s", tostring(tk), quoteShort(string.sub(tv, 1, 60)))
						end
						if #hits >= 6 then break end
					end
					if #hits > 0 then
						matchedTbl += 1
						if #out < 450 then
							out[#out + 1] = string.format("[GC TBL #%d]", matchedTbl)

		elseif t == "UDim" then
			return string.format("UDim.new(%s, %s)", fmtNum(v.Scale), fmtNum(v.Offset))
		elseif t == "Rect" then
			return string.format("Rect.new(%s, %s, %s, %s)", fmtNum(v.Min.X), fmtNum(v.Min.Y), fmtNum(v.Max.X), fmtNum(v.Max.Y))
		elseif t == "BrickColor" then

		elseif t == "EnumItem" or t == "Enum" then

		elseif t == "TweenInfo" then

		elseif t == "Region3" then
			local half = v.Size * 0.5
			return string.format("Region3.new(%s, %s)", serializeValue(v.CFrame.Position - half, depth + 1, seen), serializeValue(v.CFrame.Position + half, depth + 1, seen))
		elseif t == "NumberSequence" then
			local kps = {}
			for _, kp in ipairs(v.Keypoints or {}) do
				kps[#kps + 1] = string.format("NumberSequenceKeypoint.new(%s, %s, %s)", fmtNum(kp.Time), fmtNum(kp.Value), fmtNum(kp.Envelope or 0))
			end
			return "NumberSequence.new({" .. table.concat(kps, ", ") .. "})"
		elseif t == "ColorSequence" then
			local kps = {}
			for _, kp in ipairs(v.Keypoints or {}) do
				kps[#kps + 1] = string.format("ColorSequenceKeypoint.new(%s, %s)", fmtNum(kp.Time), serializeValue(kp.Value, depth + 1, seen))
			end
			return "ColorSequence.new({" .. table.concat(kps, ", ") .. "})"
		elseif t == "DateTime" then
			return string.format("DateTime.fromUnixTimestampMillis(%d)", v.UnixTimestampMillis or 0)
		elseif t == "buffer" then
			return string.format("buffer.create(%d)", buffer.len(v))
		elseif t == "function" then
			return "function() end"
		elseif t == "table" then
			if seen[v] then
				return "{ --[[ cyclic ]] }"
			end
			if depth >= depthLimit then
				return "{ --[[ max depth ]] }"
			end
			seen[v] = true
			local pad = string.rep("\t", depth + 1)
			local closePad = string.rep("\t", depth)
			local parts, count = {}, 0

			end
			seen[v] = nil
			if #parts == 0 then return "{}" end
			return "{\n" .. table.concat(parts, "\n") .. "\n" .. closePad .. "}"
		end
		local desc = string.gsub(tostring(v), "%]%]", "] ]")
		return string.format("nil --[[ %s: %s ]]", t, desc)
	end

	local function formatCallCode(entry)
		local lines = {}
		local dirTag = entry.dir == "S2C" and "S->C" or (entry.dir == "INT" and "INT" or (entry.dir == "ATTR" and "ATTR" or "C->S"))
		local blkTag = entry.blocked and " [BLOCKED]" or ""
		local execTag = entry.isExecutor and " [EXECUTOR]" or ""
		local actorTag = entry.isActor and " [ACTOR]" or ""
		table.insert(lines, string.format("-- [%s] %s:%s | %s%s%s%s", dirTag, entry.className, entry.method, entry.timeStr or "?", blkTag, execTag, actorTag))
		if entry.caller and entry.caller ~= "" then
			table.insert(lines, "-- Caller: " .. entry.caller)
		end
		if entry.sourceLine and entry.sourceLine ~= "" then
			table.insert(lines, string.format("-- Origin: %s (%s)", tostring(entry.funcName or "anon"), entry.sourceLine))
		elseif entry.funcName and entry.funcName ~= "" then
			table.insert(lines, "-- Function: " .. entry.funcName)
		end
		if entry.funcHash and entry.funcHash ~= "" then
			table.insert(lines, "-- Hash: " .. entry.funcHash)
		end
		if flag(N.showStack, false) and entry.stack and entry.stack ~= "" then
			table.insert(lines, "-- Stack:")
			for ln in string.gmatch(entry.stack, "[^\n]+") do
				table.insert(lines, "--   " .. ln)
			end
		end
		if entry.dir == "INT" then
			local fnName = entry.className == "ProximityPrompt" and "fireproximityprompt" or "fireclickdetector"
			table.insert(lines, string.format("%s(%s)", fnName, entry.path))
			return table.concat(lines, "\n")
		end
		if entry.dir == "ATTR" then
			local attrKey = serializeValue(entry.args[1], 1, {})
			local attrVal = serializeValue(entry.args[2], 1, {})
			table.insert(lines, string.format("%s:SetAttribute(%s, %s)", entry.path, attrKey, attrVal))
			return table.concat(lines, "\n")
		end
		local argCount = entry.args.n or #entry.args
		if argCount == 0 then
			if entry.dir == "S2C" then
				table.insert(lines, string.format("-- Received by %s.%s()", entry.path, entry.method))
			else
				table.insert(lines, string.format("%s:%s()", entry.path, entry.method))
			end
		else
			table.insert(lines, "local args = {")
			for i = 1, argCount do
				table.insert(lines, string.format("\t[%d] = %s,", i, serializeValue(entry.args[i], 1, {})))
			end
			table.insert(lines, "}")
			if entry.dir == "S2C" then
				table.insert(lines, string.format("-- Received by %s.%s(table.unpack(args, 1, %d))", entry.path, entry.method, argCount))
			else
				table.insert(lines, string.format("%s:%s(table.unpack(args, 1, %d))", entry.path, entry.method, argCount))
			end
		end
		if entry.returns then
			local rCount = entry.returns.n or #entry.returns
			table.insert(lines, "")
			if rCount == 0 then
				table.insert(lines, "-- Return: (void)")
			else
				table.insert(lines, "-- Return Values:")
				for i = 1, rCount do
					table.insert(lines, string.format("--   [%d] = %s", i, serializeValue(entry.returns[i], 1, {})))
				end
			end
		end
		return table.concat(lines, "\n")
	end

		table.clear(state.groupByLabel)
		table.clear(state.groupCount)
		table.clear(state.groupKeyOf)
		local shownSet = {}
		local function passesFilter(e)
			if isRobloxIgnored(e.remote, e.name, e.callerInst, e.className, e.isExecutor) then return false end
			local classOk
			if filter == "All" then
				classOk = true
			elseif filter == "Interactions" then
				classOk = (e.dir == "INT")
			elseif filter == "Attributes" then
				classOk = (e.dir == "ATTR")
			else
				classOk = (FILTER_MATCH[filter] ~= nil and FILTER_MATCH[filter][e.className] == true)
			end
			if not classOk then return false end
			return needle == ""
				or string.find(string.lower(e.label), needle, 1, true) ~= nil
				or string.find(string.lower(e.path), needle, 1, true) ~= nil
		end
		local function tagOf(e)
			return e.dir == "S2C" and "S->C" or (e.dir == "INT" and "INT" or (e.dir == "ATTR" and "ATTR" or "C->S"))
		end
		if grouped then
			local order, groups = {}, {}
			for i = #state.entries, 1, -1 do
				local e = state.entries[i]
				if passesFilter(e) then
					local gKey = e.path .. "|" .. e.dir .. "|" .. e.method
					local g = groups[gKey]
					if not g then
						g = { latest = e, count = 1, key = gKey }
						groups[gKey] = g
						order[#order + 1] = g
					else
						g.count += 1
					end
				end
			end
			for _, g in ipairs(order) do
				local e = g.latest
				local dirTag = tagOf(e)
				local blkTag = e.blocked and "[BLK] " or ""
				local actTag = e.isActor and "[ACT] " or ""
				local gLabel = string.format("[%s] (x%d) %s%s%s | %s", dirTag, g.count, blkTag, actTag, e.method, e.name)
				if state.groupByLabel[gLabel] then gLabel = gLabel .. " #" .. e.id end
				state.groupByLabel[gLabel] = e
				state.groupCount[gLabel] = g.count
				state.groupKeyOf[gLabel] = g.key
				table.insert(state.shownLabels, gLabel)
				shownSet[gLabel] = true
			end
		else
			for i = #state.entries, 1, -1 do
				local e = state.entries[i]
				if passesFilter(e) then
					table.insert(state.shownLabels, e.label)
					shownSet[e.label] = true
				end
			end
		end
		Zenith.setItems(N.list, table.clone(state.shownLabels))
		local keep
		if selectNewest ~= true then
			if grouped and state.selectedGroupKey then
				for _, lbl in ipairs(state.shownLabels) do
					if state.groupKeyOf[lbl] == state.selectedGroupKey then
						keep = lbl
						break
					end
				end
			elseif state.selectedLabel and shownSet[state.selectedLabel] then
				keep = state.selectedLabel
			end
		end
		local newest = state.shownLabels[1]
		selectLabel((selectNewest == true and newest) or keep or newest)
		Zenith.values[N.list] = state.selectedLabel
		renderSelected()
	end
	local function appendSpyFileLog(_entry) end
	local function addEntry(item)
		local rName = tostring(safeRead(item.remote, "Name") or "?")
		if isRobloxIgnored(item.remote, rName, item.callerInst, item.className, item.isExecutor) then return end
		if flag(N.autoIgnoreSpam, false) and typeof(item.remote) == "Instance" and not item.blocked then
			local now = os.clock()
			local rec = state.spamTrack[item.remote]
			if not rec or (now - rec.t0) > 1 then
				rec = { t0 = now, count = 1 }
				state.spamTrack[item.remote] = rec
			else
				rec.count += 1
				if rec.count >= 30 and not isIgnoredName(rName) then
					local cur = tostring(Zenith.values[N.ignoreList] or "")
					local nextStr = (cur ~= "" and (cur .. ", " .. rName)) or rName
					setVal(N.ignoreList, nextStr)
					refreshIgnoreSet()
					Zenith.log("Info", "RSpy: auto-ignorado por spam -> " .. rName)
					return
				end
			end
		end
		state.seq += 1
		local path = instancePath(item.remote)
		local dirTag = item.dir == "S2C" and "S->C" or (item.dir == "INT" and "INT" or (item.dir == "ATTR" and "ATTR" or "C->S"))
		local blkTag = item.blocked and "[BLK] " or ""
		local actTag = item.isActor and "[ACT] " or ""
		local label = string.format("#%03d [%s] %s%s%s | %s", state.seq, dirTag, blkTag, actTag, item.method, rName)
		local callerStr = ""
		if item.callerInst then callerStr = instancePath(item.callerInst) end
		local entry = {
			id = state.seq,
			label = label,
			remote = item.remote,
			name = rName,
			className = item.className,
			method = item.method,
			dir = item.dir,
			path = path,
			args = item.args or { n = 0 },
			returns = item.returns,
			blocked = item.blocked,
			caller = callerStr,
			callerInst = item.callerInst,
			func = item.func,
			funcName = item.funcName,
			sourceLine = item.sourceLine,
			funcHash = item.funcHash,
			isExecutor = item.isExecutor == true,
			isActor = item.isActor == true,
			stack = item.stack,
			timeStr = item.timeStr,
		}
		table.insert(state.entries, entry)
		state.byLabel[label] = entry
		pcall(appendSpyFileLog, entry)
	end
	local function flushQueue()
		state.flushing = false
		if stopped or #state.queue == 0 then return end
		local batch = state.queue
		state.queue = {}
		for _, item in ipairs(batch) do
			local ok, err = pcall(addEntry, item)
			if not ok then
				Zenith.log("Warn", "RSpy: error registrando llamada: " .. tostring(err))
			end
		end
		local maxCap = math.max(10, math.floor(num(N.maxLogs, 300)))
		while #state.entries > maxCap do
			local oldest = table.remove(state.entries, 1)
			if oldest then
				state.byLabel[oldest.label] = nil
				if state.selectedLabel == oldest.label then state.selectedLabel = nil end
			end
		end
		setVal(N.calls, state.seq)
		applyListFilter(flag(N.autoScroll, true))
	end
	local function enqueue(item)
		if #state.queue >= 2000 then
			return -- flood protection
		end
		table.insert(state.queue, item)
		if not state.flushing then
			state.flushing = true
			task.defer(flushQueue)
		end
	end
	local function extractFuncMeta(fn)
		if type(fn) ~= "function" then return nil, nil, nil end
		local fnName, sourceLine, fnHash = nil, nil, nil
		if type(debug) == "table" and type(debug.info) == "function" then
			local okN, nm, src, ln = pcall(debug.info, fn, "nsl")
			if okN then
				if type(nm) == "string" and nm ~= "" then fnName = nm end
				if type(src) == "string" and src ~= "" then sourceLine = string.format("%s:%s", src, tostring(ln or "?")) end
			end
		end
		local isl = Zenith.capFn("islclosure")
		local gfh = Zenith.capFn("getfunctionhash")
		if gfh and (not isl or (pcall(isl, fn) and isl(fn) == true)) then
			local okH, h = pcall(gfh, fn)
			if okH and h ~= nil then fnHash = tostring(h) end
		end
		return fnName, sourceLine, fnHash
	end
	local OWN_SOURCE
	if type(debug) == "table" and type(debug.info) == "function" then
		local okOwn, srcOwn = pcall(debug.info, 1, "s")
		OWN_SOURCE = okOwn and srcOwn or nil
	end
	local function captureCallerInfo()
		local getCallerFn = Zenith.capFn("getcallingscript")
		local callerInst = nil
		if getCallerFn then
			local ok, s = pcall(getCallerFn)
			if ok and typeof(s) == "Instance" then callerInst = s end
		end
		local ccFn = Zenith.capFn("checkcaller")
		local isExec = false
		if ccFn then
			local okC, resC = pcall(ccFn)
			if okC and resC == true then isExec = true end
		end
		local callerFn, fnName, sourceLine, fnHash = nil, nil, nil, nil
		if type(debug) == "table" and type(debug.info) == "function" then
			for i = 2, 14 do
				local okI, f, src, ln = pcall(debug.info, i, "fsl")
				if not okI or not f or not src then break end
				if src ~= "[C]" and (not OWN_SOURCE or src ~= OWN_SOURCE) then
					callerFn = f
					fnName, sourceLine, fnHash = extractFuncMeta(f)
					if not sourceLine then sourceLine = string.format("%s:%s", tostring(src), tostring(ln or "?")) end
					if not isExec then
						local isExFn = Zenith.capFn("isexecutorclosure")
						if isExFn then
							local okE, rE = pcall(isExFn, f)
							if okE and rE == true then isExec = true end
						end
					end
					break
				end
			end
		end
		local stack = nil
		if flag(N.showStack, false) and type(debug) == "table" and type(debug.traceback) == "function" then
			local ok, tb = pcall(debug.traceback, "", 3)
			if ok and type(tb) == "string" then stack = tb end
		end
		return callerInst, stack, isExec, callerFn, fnName, sourceLine, fnHash
	end
	local function timeStamp()
		local ok, s = pcall(os.date, "%H:%M:%S")
		return ok and s or "00:00:00"
	end

	local MAIN_THREAD = {}
	local function handleOutbound(rec, self, method, ...)
		local origFn = rec.old
		if not rec.active or typeof(self) ~= "Instance" then return origFn(self, ...) end
		local canon = canonMethod(method)
		if not canon then return origFn(self, ...) end
		local thread = coroutine.running() or MAIN_THREAD
		if state.internal[thread] then return origFn(self, ...) end
		local cls = safeRead(self, "ClassName")
		if not cls or not OUTBOUND_CLASSES[canon][cls] then return origFn(self, ...) end
		local isBlocked = state.blocked[self] == true
		if state.depth[thread] then
			if isBlocked then return nil end
			return origFn(self, ...)
		end
		local shouldLog = flag(N.enabled, false) and flag(N.c2s, true)
		if shouldLog and isBindableClass(cls) and flag(N.ignoreBindables, true) then shouldLog = false end
		if not shouldLog and not isBlocked then return origFn(self, ...) end
		local item
		if shouldLog then
			local callerInst, stack, isExec, callerFn, fnName, sourceLine, fnHash = captureCallerInfo()
			if not isBlocked and isRobloxIgnored(self, safeRead(self, "Name"), callerInst, cls, isExec) then
				shouldLog = false
			else
				item = {
					remote = self,
					className = cls,
					method = canon,
					dir = "C2S",
					args = table.pack(...),
					blocked = isBlocked,
					callerInst = callerInst,
					func = callerFn,
					funcName = fnName,
					sourceLine = sourceLine,
					funcHash = fnHash,
					isExecutor = isExec,
					stack = stack,
					timeStr = timeStamp(),
				}
			end
		end
		if not shouldLog and not isBlocked then return origFn(self, ...) end
		if isBlocked then
			if item then
				enqueue(item)
			end
			return nil
		end
		local wantRet = shouldLog and flag(N.logReturn, false) and (canon == "InvokeServer" or canon == "Invoke")
		if shouldLog and not wantRet then
			enqueue(item) -- log before the call so hanging invokes are still visible
		end
		state.depth[thread] = true
		local ok, res = pcall(callOrig, origFn, self, ...)
		state.depth[thread] = nil
		if wantRet then
			if ok then item.returns = res end
			enqueue(item)
		end
		if not ok then
			error(res, 0)
		end
		return table.unpack(res, 1, res.n)
	end

	local function installSignalHook(wrapC)
		if state.signalHook then return true end
		local hookFn = Zenith.capFn("hookfunction")
		local getRMT = Zenith.capFn("getrawmetatable")
		local getConns = Zenith.capFn("getconnections")
		if not hookFn or not getRMT or not getConns then return false end
		local okP, part = pcall(Instance.new, "Part")
		if not okP or not part then return false end
		local sig = safeRead(part, "Touched")
		local okM, sigMt = pcall(getRMT, sig)
		pcall(function() part:Destroy() end)
		if not okM or type(sigMt) ~= "table" or type(sigMt.__index) ~= "function" then return false end
		local targetIdx = sigMt.__index
		local rec = { active = true, target = targetIdx, old = nil }
		local okH, old = pcall(hookFn, targetIdx, wrapC(function(self, key)
			local res = rec.old(self, key)
			if not rec.active or stopped or not SIGNAL_CONN_KEYS[key] or type(res) ~= "function" then return res end
			local inst = state.signalMapping[self]
			if not inst or not state.blocked[inst] then return res end
			return wrapC(function(s, callback)
				local ret = table.pack(res(s, callback))
				if state.blocked[inst] then
					local cls = safeRead(inst, "ClassName")
					local sigName = cls and S2C_SOURCES[cls]
					if sigName then
						local okC, conns = pcall(getConns, safeRead(inst, sigName))
						if okC and type(conns) == "table" then
							for _, c in ipairs(conns) do
								if not safeRead(c, "ForeignState") and safeRead(c, "Function") == callback and type(c.Disable) == "function" then pcall(c.Disable, c) end
							end
						end
					end
				end
				return table.unpack(ret, 1, ret.n)
			end)
		end))
		if okH and type(old) == "function" then
			rec.old = old
			state.signalHook = rec
			return true
		end
		rec.active = false
		return false
	end

	local function installActorHooks()
		if state.actorChannel or not flag(N.logActors, true) then return false end
		local runActor = Zenith.capFn("run_on_actor")
		local getActors = Zenith.capFn("getactors")
		local mkChan = Zenith.capFn("create_comm_channel")
		if not runActor or not getActors or not mkChan then return false end
		local okC, chId, ch = pcall(mkChan)
		if not okC or not chId or not ch then return false end
		state.actorChannel = ch
		local evConn = ch.Event:Connect(function(kind, inst, dir, cls, method, argsPack, callerInst, fnName, srcLine, isExec)
			if kind ~= "ActorCall" or stopped or not (flag(N.enabled, false) and flag(N.logActors, true)) then return end
			if typeof(inst) ~= "Instance" then return end
			if isRobloxIgnored(inst, safeRead(inst, "Name"), callerInst, cls, isExec == true) then return end
			enqueue({
				remote = inst,
				className = cls or safeRead(inst, "ClassName") or "RemoteEvent",
				method = method or "FireServer",
				dir = dir or "C2S",
				args = type(argsPack) == "table" and argsPack or { n = 0 },
				blocked = state.blocked[inst] == true,
				callerInst = callerInst,
				funcName = fnName,
				sourceLine = srcLine,
				isExecutor = isExec == true,
				isActor = true,
				timeStr = timeStamp(),
			})
		end)
		table.insert(state.actorConns, evConn)
		local function hookOneActor(act)
			if typeof(act) ~= "Instance" or state.actorHooked[act] then return end
			state.actorHooked[act] = true
			task.spawn(function()
				pcall(runActor, act, ACTOR_HOOK_SRC, chId, act)
			end)
		end
		local okA, actors = pcall(getActors)
		if okA and type(actors) == "table" then
			for _, a in ipairs(actors) do
				hookOneActor(a)
			end
		end
		local okD, dConn = pcall(function()
			return game.DescendantAdded:Connect(function(inst)
				if typeof(inst) == "Instance" and pcall(inst.IsA, inst, "Actor") and inst:IsA("Actor") then
					hookOneActor(inst)
				end
			end)
		end)
		if okD and dConn then table.insert(state.actorConns, dConn) end
		return true
	end
	local function installHooks()
		if state.installed then return true end
		local wrapC = Zenith.capFn("newcclosure") or function(f) return f end
		local getNCM = Zenith.capFn("getnamecallmethod")
		local hookMM = Zenith.capFn("hookmetamethod")
		local hookFn = Zenith.capFn("hookfunction")
		local othHook = flag(N.preferOth, true) and Zenith.capFn("oth_hook") or nil
		if hookMM and getNCM then
			local rec = { active = true, old = nil }
			local ok, old = pcall(hookMM, game, "__namecall", wrapC(function(self, ...)
				local method = getNCM()
				return handleOutbound(rec, self, method, ...)
			end))
			if ok and type(old) == "function" then
				rec.old = old
				state.oldNamecall = old
				state.namecallRec = rec
				state.installed = true
			else
				rec.active = false
				Zenith.log("Warn", "RSpy: hookmetamethod fallo (" .. tostring(old) .. ")")
			end
		end
		if not state.namecallRec and Zenith.cap("getrawmetatable") and getNCM then
			local getRMT = Zenith.capFn("getrawmetatable")
			local setRO = typeof(setreadonly) == "function" and setreadonly or nil
			local ok, mt = pcall(getRMT, game)
			if ok and type(mt) == "table" and type(mt.__namecall) == "function" then
				local rec = { active = true, old = mt.__namecall }
				if setRO then pcall(setRO, mt, false) end
				local okSet = pcall(function()
					mt.__namecall = wrapC(function(self, ...)
						local method = getNCM()
						return handleOutbound(rec, self, method, ...)
					end)
				end)
				if setRO then pcall(setRO, mt, true) end
				if okSet then
					state.oldNamecall = rec.old
					state.namecallRec = rec
					state.installed = true
				else
					rec.active = false
					Zenith.log("Warn", "RSpy: no se pudo escribir __namecall")
				end
			end
		end
		if hookFn or othHook then
			local specs = {
				{ "RemoteEvent", "FireServer" },
				{ "UnreliableRemoteEvent", "FireServer" },
				{ "RemoteFunction", "InvokeServer" },
				{ "BindableEvent", "Fire" },
				{ "BindableFunction", "Invoke" },
			}
			local hookedTargets = {}
			for _, sp in ipairs(specs) do
				local cls, method = sp[1], sp[2]
				local okI, dummy = pcall(Instance.new, cls)
				if okI and dummy then
					local fn = safeRead(dummy, method)
					if type(fn) == "function" and not hookedTargets[fn] then
						hookedTargets[fn] = true
						local record = { active = true, target = fn, old = nil, usedOth = false }
						local hooked = false
						if othHook then
							local okO, oldO = pcall(othHook, fn, wrapC(function(self, ...)
								return handleOutbound(record, self, method, ...)
							end))
							if okO and type(oldO) == "function" then
								record.old = oldO
								record.usedOth = true
								table.insert(state.directHooks, record)
								state.installed = true
								hooked = true
							end
						end
						if not hooked and hookFn then
							local okH, old = pcall(hookFn, fn, wrapC(function(self, ...)
								return handleOutbound(record, self, method, ...)
							end))
							if okH and type(old) == "function" then
								record.old = old
								table.insert(state.directHooks, record)
								state.installed = true
							else
								record.active = false
							end
						elseif not hooked then
							record.active = false
						end
					end
					pcall(function() dummy:Destroy() end)
				end
			end
		end
		if installNewindexHook(wrapC) then state.installed = true end
		installSignalHook(wrapC)
		installActorHooks()
		return state.installed
	end

			local okC, conn = pcall(function()
				return pps.PromptTriggered:Connect(function(prompt, plr)
					if stopped or not (flag(N.enabled, false) and flag(N.logInteractions, false)) then return end
					if plr and localPlayer and plr ~= localPlayer then return end
					if typeof(prompt) ~= "Instance" then return end
					enqueue({
						remote = prompt,
						className = "ProximityPrompt",
						method = "FireProximityPrompt",
						dir = "INT",
						args = { n = 0 },
						timeStr = timeStamp(),
					})
				end)
			end)
			if okC and conn then table.insert(state.interactionConns, conn) end
		end
		local function watchClickDetector(cd)
			if typeof(cd) ~= "Instance" or state.interactionHooked[cd] then return end
			local okIs, isCD = pcall(cd.IsA, cd, "ClickDetector")
			if not okIs or not isCD then return end
			state.interactionHooked[cd] = true
			local okC, conn = pcall(function()
				return cd.MouseClick:Connect(function(plr)
					if stopped or not (flag(N.enabled, false) and flag(N.logInteractions, false)) then return end
					if plr and localPlayer and plr ~= localPlayer then return end
					enqueue({
						remote = cd,
						className = "ClickDetector",
						method = "FireClickDetector",
						dir = "INT",
						args = { n = 0 },
						timeStr = timeStamp(),
					})
				end)
			end)
			if okC and conn then table.insert(state.interactionConns, conn) end
		end
		local okW, ws = pcall(game.GetService, game, "Workspace")
		if okW and ws then
			local okD, desc = pcall(ws.GetDescendants, ws)
			if okD and type(desc) == "table" then
				for _, d in ipairs(desc) do
					watchClickDetector(d)
				end
			end
			local okA, addConn = pcall(function() return ws.DescendantAdded:Connect(watchClickDetector) end)
			if okA and addConn then table.insert(state.interactionConns, addConn) end
		end
		return #state.interactionConns > 0
	end
	local function uninstallAttributeHooks()
		for i = #state.attrConns, 1, -1 do
			local c = state.attrConns[i]
			pcall(function() c:Disconnect() end)
			state.attrConns[i] = nil
		end
		state.attrHooked = setmetatable({}, { __mode = "k" })
	end
	local function installAttributeHooks()
		uninstallAttributeHooks()
		if stopped or not (flag(N.enabled, false) and flag(N.logAttributes, false)) then return false end
		local watchedCount = 0
		local function watchAttr(inst)
			if typeof(inst) ~= "Instance" or state.attrHooked[inst] or watchedCount >= 4000 then return end
			if not inst.AttributeChanged then return end
			state.attrHooked[inst] = true
			watchedCount += 1
			local okC, conn = pcall(function()
				return inst.AttributeChanged:Connect(function(attrName)
					if stopped or not (flag(N.enabled, false) and flag(N.logAttributes, false)) then return end
					local okV, val = pcall(inst.GetAttribute, inst, attrName)
					enqueue({
						remote = inst,
						className = safeRead(inst, "ClassName") or "Instance",
						method = "SetAttribute",
						dir = "ATTR",
						args = { n = 2, attrName, okV and val or nil },
						timeStr = timeStamp(),
					})
				end)
			end)
			if okC and conn then table.insert(state.attrConns, conn) end
		end
		if localPlayer then
			watchAttr(localPlayer)
			if localPlayer.Character then
				watchAttr(localPlayer.Character)
			end
		end
		for _, svcName in ipairs({ "Workspace", "ReplicatedStorage" }) do
			local okS, svc = pcall(game.GetService, game, svcName)
			if okS and svc then
				watchAttr(svc)
				local okD, desc = pcall(svc.GetDescendants, svc)
				if okD and type(desc) == "table" then
					for _, d in ipairs(desc) do
						watchAttr(d)
					end
				end
				local okA, addConn = pcall(function() return svc.DescendantAdded:Connect(watchAttr) end)
				if okA and addConn then table.insert(state.attrConns, addConn) end
			end
		end
		return #state.attrConns > 0
	end
	local function uninstallHooks()
		if state.namecallRec then state.namecallRec.active = false end
		if state.signalHook then state.signalHook.active = false end
		for _, rec in ipairs(state.directHooks) do
			rec.active = false
		end
		disconnectS2C()
		uninstallNewindexHook()
		uninstallActorHooks()
		uninstallInteractionHooks()
		uninstallAttributeHooks()
		local hookMM = Zenith.capFn("hookmetamethod")
		if state.oldNamecall then
			if hookMM then
				pcall(hookMM, game, "__namecall", state.oldNamecall)
			elseif Zenith.cap("getrawmetatable") then
				local getRMT = Zenith.capFn("getrawmetatable")
				local setRO = typeof(setreadonly) == "function" and setreadonly or nil
				local ok, mt = pcall(getRMT, game)
				if ok and type(mt) == "table" then
					if setRO then pcall(setRO, mt, false) end
					pcall(function() mt.__namecall = state.oldNamecall end)
					if setRO then pcall(setRO, mt, true) end
				end
			end
			state.oldNamecall = nil
		end
		state.namecallRec = nil
		local hookFn = Zenith.capFn("hookfunction")
		if state.signalHook and state.signalHook.target and state.signalHook.old then
			local restoredSig = false
			if typeof(restorefunction) == "function" then restoredSig = pcall(restorefunction, state.signalHook.target) end
			if not restoredSig and hookFn then pcall(hookFn, state.signalHook.target, state.signalHook.old) end
		end
		state.signalHook = nil
		local othUnhook = Zenith.capFn("oth_unhook")
		for _, rec in ipairs(state.directHooks) do
			if rec.target and rec.old then
				local restored = false
				if rec.usedOth and othUnhook then restored = pcall(othUnhook, rec.target) end
				if not restored and typeof(restorefunction) == "function" then restored = pcall(restorefunction, rec.target) end
				if not restored and hookFn then pcall(hookFn, rec.target, rec.old) end
			end
		end
		table.clear(state.directHooks)
		state.installed = false
	end
	Zenith.RSpy = state
	Zenith.register("RSpy", {
		requires = {},
		optional = { "hookmetamethod", "hookfunction", "getrawmetatable", "getnamecallmethod", "newcclosure", "getcallingscript", "setclipboard", "getfunctionhash", "checkcaller", "isexecutorclosure", "islclosure", "setstackhidden", "run_on_actor", "getactors", "create_comm_channel", "oth_hook", "oth_unhook", "fireproximityprompt", "fireclickdetector" },
		init = function()
			stopped = false
			Zenith.RSpy = state
			refreshIgnoreSet()
			local function onEnabled(on)
				if on then
					local ok = installHooks()
					syncS2C(true, nil)
					installInteractionHooks()
					installAttributeHooks()
					Zenith.log(ok and "Info" or "Warn", ok and "RSpy: captura en vivo activada" or "RSpy: sin soporte de hooks (__namecall/hookfunction), solo OnClientEvent")
				else
					disconnectS2C()
					uninstallInteractionHooks()
					uninstallAttributeHooks()
					restoreWrappedCallbacks()
					Zenith.log("Info", "RSpy: captura pausada")
				end
			end
			hookUI(N.enabled, function(on)
				onEnabled(on == true)
			end)
			hookUI(N.s2c, function(v)
				syncS2C(nil, v == true)
			end)
			hookUI(N.list, function(label)
				selectLabel(label)
				renderSelected()
			end)
			hookUI(N.filter, function() applyListFilter() end)
			hookUI(N.search, function() applyListFilter() end)
			hookUI(N.ignoreList, function()
				refreshIgnoreSet()
				applyListFilter()
			end)
			for _, fName in ipairs({ N.ignoreInternals, N.ignoreBindables, N.ignorePlayerMod, N.ignoreServices, N.ignoreRbxScripts, N.ignoreRbxRemotes, N.ignoreExecutor }) do
				hookUI(fName, function()
					refreshIgnoreSet()
					applyListFilter()
					if flag(N.enabled, false) and flag(N.s2c, true) then
						syncS2C(true, true)
					end
				end)
			end
			hookUI(N.logActors, function(on)
				if on and flag(N.enabled, false) then
					installActorHooks()
				elseif not on then
					uninstallActorHooks()
				end
			end)
			hookUI(N.logInteractions, function(on)
				if on and flag(N.enabled, false) then
					installInteractionHooks()
				elseif not on then
					uninstallInteractionHooks()
				end
			end)
			hookUI(N.logAttributes, function(on)
				if on and flag(N.enabled, false) then
					installAttributeHooks()
				elseif not on then
					uninstallAttributeHooks()
				end
			end)
			hookUI(N.fullSerialize, function() renderSelected() end)
			hookUI(N.showStack, function() renderSelected() end)
			hookUI(N.groupByRemote, function() applyListFilter(true) end)
			Zenith.actions[N.clear] = function()
				state.seq = 0
				table.clear(state.entries); table.clear(state.byLabel); table.clear(state.groupByLabel); table.clear(state.groupCount); table.clear(state.groupKeyOf); table.clear(state.shownLabels); table.clear(state.queue)
				selectLabel(nil)
				Zenith.values[N.list] = nil
				setVal(N.calls, 0)
				Zenith.setItems(N.list, {})
				local out = Zenith.outputs and Zenith.outputs[N.log]
				if out then
					out.clear()
				end
				Zenith.log("Info", "RSpy: logs limpiados")
			end
			Zenith.actions[N.copyCode] = function()
				local entry = getSelectedEntry()
				if not entry then return Zenith.log("Warn", "RSpy: selecciona una llamada primero") end
				local ok = Zenith.copy(safeFormat(entry))
				Zenith.log(ok and "Info" or "Warn", ok and "RSpy: codigo copiado" or "Clipboard no soportado")
			end
			Zenith.actions[N.copyPath] = function()
				local entry = getSelectedEntry()
				if not entry then return Zenith.log("Warn", "RSpy: selecciona una llamada primero") end
				local ok = Zenith.copy(entry.path)
				Zenith.log(ok and "Info" or "Warn", ok and "RSpy: ruta copiada" or "Clipboard no soportado")
			end
			Zenith.actions[N.decompileCaller] = function()
				local entry = getSelectedEntry()
				if not entry then return Zenith.log("Warn", "RSpy: selecciona una llamada primero") end
				local callerInst = entry.callerInst
				if not callerInst and type(entry.func) == "function" then callerInst = resolveScriptOrigin(nil, entry.func) end
				if not callerInst then return Zenith.log("Warn", "RSpy: esta llamada no tiene Script caller identificable") end
				if Zenith.decompileInstance then
					Zenith.decompileInstance(callerInst)
				else
					Zenith.log("Warn", "RSpy: modulo Decompiler no cargado")
				end
			end
			Zenith.actions[N.findXRefs] = function()
				local entry = getSelectedEntry()
				if not entry then return Zenith.log("Warn", "RSpy: selecciona una llamada primero") end
				if Zenith.findScriptXRefs then
					Zenith.findScriptXRefs(entry.name)
				else
					Zenith.log("Warn", "RSpy: modulo Decompiler no cargado")
				end
			end
			local function runAsInternal(fn, ...)
				local thread = coroutine.running() or MAIN_THREAD
				local prev = state.internal[thread]
				state.internal[thread] = true
				local ok, err = pcall(fn, ...)
				state.internal[thread] = prev
				return ok, err
			end
			Zenith.actions[N.repeatCall] = function()
				local entry = getSelectedEntry()
				if not entry or not entry.remote then return Zenith.log("Warn", "RSpy: selecciona una llamada valida") end
				if entry.dir == "S2C" then
					return Zenith.log("Warn", "RSpy: no se puede reenviar una llamada Server->Client")
				end
				if entry.dir == "INT" then
					local fnName = entry.className == "ProximityPrompt" and "fireproximityprompt" or "fireclickdetector"
					local fn = Zenith.capFn(fnName)
					if not fn then return Zenith.log("Warn", "RSpy: " .. fnName .. " no soportado en este executor") end
					task.spawn(function()
						local ok, err = runAsInternal(fn, entry.remote)
						Zenith.log(ok and "Info" or "Error", ok and ("RSpy: repetida " .. entry.name) or ("RSpy Repeat error: " .. tostring(err)))
					end)
					return
				end
				if entry.dir == "ATTR" then
					task.spawn(function()
						local ok, err = runAsInternal(function()
							entry.remote:SetAttribute(entry.args[1], entry.args[2])
						end)
						Zenith.log(ok and "Info" or "Error", ok and ("RSpy: atributo restaurado " .. tostring(entry.args[1])) or ("RSpy Attribute error: " .. tostring(err)))
					end)
					return
				end
				task.spawn(function()
					local ok, err = runAsInternal(function()
						local fn = safeRead(entry.remote, entry.method)
						if type(fn) ~= "function" then
							error("metodo no encontrado: " .. tostring(entry.method))
						end
						local n = entry.args.n or #entry.args
						fn(entry.remote, table.unpack(entry.args, 1, n))
					end)
					Zenith.log(ok and "Info" or "Error", ok and ("RSpy: repetida " .. entry.name) or ("RSpy Repeat error: " .. tostring(err)))
				end)
			end
			Zenith.actions[N.replayCustom] = function()
				local entry = getSelectedEntry()
				if not entry or not entry.remote then return Zenith.log("Warn", "RSpy: selecciona una llamada valida") end
				if entry.dir == "S2C" then
					return Zenith.log("Warn", "RSpy: no se puede reenviar una llamada Server->Client")
				end
				if entry.dir == "INT" or entry.dir == "ATTR" then return Zenith.actions[N.repeatCall]() end
				local raw = string.match(tostring(Zenith.values[N.customArgs] or ""), "^%s*(.-)%s*$")
				if raw == "" then return Zenith.actions[N.repeatCall]() end
				local loader = (typeof(loadstring) == "function" and loadstring) or (typeof(load) == "function" and load)
				if not loader then return Zenith.log("Warn", "RSpy: este entorno no soporta loadstring/load para Custom Args") end
				local chunk, compileErr = loader("return " .. raw)
				if not chunk then return Zenith.log("Error", "RSpy Custom Args sintaxis: " .. tostring(compileErr)) end
				local evalPack = table.pack(pcall(chunk))
				if not evalPack[1] then return Zenith.log("Error", "RSpy Custom Args eval: " .. tostring(evalPack[2])) end
				local argCount = math.max(0, (evalPack.n or #evalPack) - 1)
				task.spawn(function()
					local ok, err = runAsInternal(function()
						local fn = safeRead(entry.remote, entry.method)
						if type(fn) ~= "function" then
							error("metodo no encontrado: " .. tostring(entry.method))
						end
						fn(entry.remote, table.unpack(evalPack, 2, 1 + argCount))
					end)
					Zenith.log(ok and "Info" or "Error", ok and string.format("RSpy: replay custom (%d args) -> %s", argCount, entry.name) or ("RSpy Replay Custom error: " .. tostring(err)))
				end)
			end
			Zenith.actions[N.blockRemote] = function()
				local entry = getSelectedEntry()
				if not entry or not entry.remote then return Zenith.log("Warn", "RSpy: selecciona un remote primero") end
				local nowBlocked = not (state.blocked[entry.remote] == true)
				state.blocked[entry.remote] = nowBlocked or nil
				Zenith.log("Info", string.format("RSpy: %s %s", entry.name, nowBlocked and "BLOQUEADO" or "desbloqueado"))
			end
			Zenith.actions[N.ignoreRemote] = function()
				local entry = getSelectedEntry()
				if not entry then return Zenith.log("Warn", "RSpy: selecciona un remote primero") end
				local cur = tostring(Zenith.values[N.ignoreList] or "")
				if not isIgnoredName(entry.name) then
					local nextStr = (cur ~= "" and (cur .. ", " .. entry.name)) or entry.name
					setVal(N.ignoreList, nextStr)
					refreshIgnoreSet()
					applyListFilter()
					Zenith.log("Info", "RSpy: ignorado " .. entry.name)
				end
			end
			if flag(N.enabled, false) then
				onEnabled(true)
			end
			Zenith.log("Info", "RSpy listo: activa 'Remote Spy' para capturar")
		end,
		stop = function()
			stopped = true
			uninstallHooks()
			for name, s in pairs(savedHooks) do
				Zenith.hooks[name] = s.prev
			end
			table.clear(savedHooks)
			for _, k in ipairs({ N.clear, N.copyCode, N.copyPath, N.decompileCaller, N.findXRefs, N.repeatCall, N.replayCustom, N.blockRemote, N.ignoreRemote }) do
				Zenith.actions[k] = nil
			end
			table.clear(state.queue)
			if Zenith.RSpy == state then Zenith.RSpy = nil end
		end,
	})
end
Zenith.startAll()
