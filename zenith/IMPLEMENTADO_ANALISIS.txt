	local function xmlEscape(s)
		return (string.gsub(tostring(s or ""), "[&<>\"']", {
			["&"] = "&amp;",
			["<"] = "&lt;",
			[">"] = "&gt;",
			['"'] = "&quot;",
			["'"] = "&apos;",
		}))
	end

	local function cdataSafe(s)
		return (string.gsub(tostring(s or ""), "]]>", "]]]]><![CDATA[>"))
	end

	local function exportPlaceRbxmx()
		local timeoutSec = math.max(1, num(N.decompileTimeout, 10))
		local saveBc = flag(N.saveBytecode, false)
		local anon = flag(N.anonymizeDump, false)
		local ignoreInternals = flag(N.ignoreInternals, true)
		local svcNames = { "Workspace", "ReplicatedStorage", "Lighting", "StarterGui", "StarterPack", "StarterPlayer" }
		local xml = { '<roblox xmlns:xmime="http://www.w3.org/2005/05/xmlmime" xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance" xsi:noNamespaceSchemaLocation="http://www.roblox.com/roblox.xsd" version="4">' }
		local referent, instCount, scriptCount = 0, 0, 0
		local VALUE_CLASSES = { StringValue = "string", IntValue = "int", NumberValue = "double", BoolValue = "bool" }

		local function emitInst(inst, indent, isRootSvc)
			if instCount >= 12000 then
				return
			end
			if typeof(inst) ~= "Instance" then
				return
			end
			if not isRootSvc then
				local okA, arch = pcall(function() return inst.Archivable end)
				if okA and arch == false then
					return
				end
			end
			local okN, iName = pcall(function() return inst.Name end)
			local okC, iClass = pcall(function() return inst.ClassName end)
			iName = tostring(okN and iName or "Instance")
			iClass = tostring(okC and iClass or "Folder")
			if ignoreInternals and (iName == "Camera" or iName == "Terrain" or iClass == "Player") then
				return
			end

			referent += 1
			instCount += 1
			local pad = string.rep("\t", indent)
			xml[#xml + 1] = string.format('%s<Item class="%s" referent="RBX%d">', pad, xmlEscape(iClass), referent)
			xml[#xml + 1] = string.format('%s\t<Properties>', pad)
			xml[#xml + 1] = string.format('%s\t\t<string name="Name">%s</string>', pad, xmlEscape(iName))
			local isScrOk, isScr = pcall(inst.IsA, inst, "LuaSourceContainer")
			if isScrOk and isScr then
				scriptCount += 1
				local lbl = Index.labelOf[inst] or iName
				local src = decompileEntryToText(inst, lbl, timeoutSec, saveBc, anon)
				xml[#xml + 1] = string.format('%s\t\t<ProtectedString name="Source"><![CDATA[%s]]></ProtectedString>', pad, cdataSafe(src))
			elseif VALUE_CLASSES[iClass] then
				local okV, val = pcall(function() return inst.Value end)
				if okV and val ~= nil then
					local tag = VALUE_CLASSES[iClass]
					xml[#xml + 1] = string.format('%s\t\t<%s name="Value">%s</%s>', pad, tag, xmlEscape(tostring(val)), tag)
				end
			end
			xml[#xml + 1] = string.format('%s\t</Properties>', pad)

			local okCh, ch = pcall(inst.GetChildren, inst)
			if okCh and type(ch) == "table" then
				for _, child in ipairs(ch) do
					emitInst(child, indent + 1, false)
				end
			end
			xml[#xml + 1] = pad .. "</Item>"
			if instCount % 250 == 0 then
				task.wait()
			end
		end

		for _, svcName in ipairs(svcNames) do
			local okS, svc = pcall(game.GetService, game, svcName)
			if okS and svc then
				emitInst(svc, 1, true)
			end
		end
		xml[#xml + 1] = "</roblox>"

		local fullXml = table.concat(xml, "\n")
		local fileName = "Zenith/Zenith_Place_" .. tostring(game.PlaceId or 0) .. ".rbxmx"
		if Zenith.cap("writefile") then
			ensureFolders({ "Zenith" })
			local okW, errW = pcall(Zenith.capFn("writefile"), fileName, fullXml)
			if okW then
				Zenith.log("Info", string.format("Export Place .rbxmx: %d instancias (%d scripts) -> %s", instCount, scriptCount, fileName))
				return fullXml
			end
			Zenith.log("Warn", "writefile Place .rbxmx fallo: " .. tostring(errW))
		end
		if Zenith.copy then
			Zenith.copy(fullXml)
			Zenith.log("Info", string.format("Export Place .rbxmx (%d instancias) copiado al portapapeles", instCount))
		end
		return fullXml
	end

	local xrefCache = {}

	local function getXrefIndex(bytes)
		local hit = xrefCache[bytes]
		if hit then
			return hit
		end
		local ok, bc = pcall(parseCached, bytes)
		if not ok or not bc then
			return nil
		end
		link(bc)
		local imports, strings = {}, {}
		for _, p in ipairs(bc.protos) do
			local fnTag = "#" .. p.index .. (p.name and (" (" .. p.name .. ")") or "")
			for _, k in ipairs(p.k) do
				if k.t == "import" then
					local nm = kstr(p, k, true)
					imports[#imports + 1] = { val = nm, low = string.lower(nm), fnTag = fnTag }
				elseif k.t == "string" and type(k.v) == "string" and k.v ~= "" then
					strings[#strings + 1] = { val = k.v, low = string.lower(k.v), fnTag = fnTag }
				end
			end
		end
		hit = { imports = imports, strings = strings }
		xrefCache[bytes] = hit
		return hit
	end

	function Dec.globalSearch(rawQuery)
		local query = string.match(tostring(rawQuery or Zenith.values[N.globalSearch] or ""), "^%s*(.-)%s*$")
		if query == "" then
			Zenith.log("Warn", "Global Search: escribe un texto o nombre de Remote")
			return 0
		end
		local entries = getExportTargets()
		if #entries == 0 then
			publish({ "-- Global Search: no hay scripts tras filtros" }, "txt")
			return 0
		end

		local needle = string.lower(query)
		local out = {
			"-- Zenith Global Bytecode XRef Search",
			"-- Query: " .. query .. " | Scripts escaneados: " .. tostring(#entries),
			"",
		}
		local matchedLabels, totalHits, skippedNoBc = {}, 0, 0

		for i, e in ipairs(entries) do
			local bytes = fetch(e.inst)
			if not bytes then
				skippedNoBc += 1
			else
				local idx = getXrefIndex(bytes)
				if idx then
					local scriptHits = {}
					for _, imp in ipairs(idx.imports) do
						if string.find(imp.low, needle, 1, true) then
							scriptHits[#scriptHits + 1] = string.format("  [IMPORT %s] %s", imp.fnTag, imp.val)
						end
					end
					for _, st in ipairs(idx.strings) do
						if string.find(st.low, needle, 1, true) then
							local preview = string.gsub(string.gsub(st.val, "\n", "\\n"), "\r", "\\r")
							if #preview > 140 then
								preview = string.sub(preview, 1, 140) .. "..."
							end
							scriptHits[#scriptHits + 1] = string.format("  [STRING %s] %q", st.fnTag, preview)
						end
					end
					if #scriptHits > 0 then
						matchedLabels[#matchedLabels + 1] = e.label
						totalHits += #scriptHits
						if #out < 450 then
							out[#out + 1] = string.format("== %s (%d hits)", e.label, #scriptHits)
							for k = 1, math.min(#scriptHits, 12) do
								out[#out + 1] = scriptHits[k]
							end
							if #scriptHits > 12 then
								out[#out + 1] = string.format("  ... (+%d hits mas en este script)", #scriptHits - 12)
							end
							out[#out + 1] = ""
						end
					end
				end
			end
			if i % 40 == 0 then
				task.wait()
			end
		end

		out[3] = string.format("-- Coincidencias: %d hits en %d scripts (%d sin bytecode omitidos)", totalHits, #matchedLabels, skippedNoBc)
		if #matchedLabels == 0 then
			out[#out + 1] = "-- Sin coincidencias para: " .. query
		else
			table.clear(Index.shown)
			for idx, lbl in ipairs(matchedLabels) do
				if idx <= MAX_SHOWN then
					Index.shown[idx] = lbl
				end
			end
			set(N.scriptsFound, #matchedLabels)
			if Zenith.setItems then
				pcall(Zenith.setItems, N.list, Index.shown)
			end
		end

		Dec.current = { label = "GlobalSearch_" .. sanitize(query), inst = nil, bc = nil, size = 0, class = "XRef" }
		publish(out, "txt")
		Zenith.log("Info", string.format("Global Search '%s': %d hits en %d scripts", query, totalHits, #matchedLabels))
		return #matchedLabels, totalHits
	end

	function Dec.gcSearch(rawQuery)
		local query = string.match(tostring(rawQuery or Zenith.values[N.globalSearch] or ""), "^%s*(.-)%s*$")
		if query == "" then
			Zenith.log("Warn", "GC Search: escribe un texto en Global Search")
			return 0
		end
		local getGC = Zenith.capFn("getgc")
		if not getGC then
			Zenith.log("Warn", "GC Search requiere getgc() en el executor")
			publish({ "-- GC Closure Search requiere getgc() en el executor" }, "txt")
			return 0
		end

		local needle = string.lower(query)
		local isExecFn = Zenith.capFn("isexecutorclosure")
		local isLFn = Zenith.capFn("islclosure")
		local getKs = Zenith.capFn("getconstants") or (type(debug) == "table" and debug.getconstants)
		local getUps = Zenith.capFn("getupvalues") or (type(debug) == "table" and debug.getupvalues)
		local okG, gcList = pcall(getGC, false)
		if not okG or type(gcList) ~= "table" then
			Zenith.log("Error", "getgc fallo: " .. tostring(gcList))
			return 0
		end

		local out = {
			"-- Zenith Live GC Closure Search",
			"-- Query: " .. query .. " | GC objects: " .. tostring(#gcList),
			"",
		}
		local matched, scanned = 0, 0
		for i, obj in ipairs(gcList) do
			if type(obj) == "function" then
				local skip = (isExecFn and pcall(isExecFn, obj) and isExecFn(obj) == true)
					or (isLFn and pcall(isLFn, obj) and isLFn(obj) == false)
				if not skip then
					scanned += 1
					local src, line, fname = "?", "?", ""
					if type(debug) == "table" and type(debug.info) == "function" then
						local okI, s, l, n = pcall(debug.info, obj, "sln")
						if okI then
							src = tostring(s or "?")
							line = tostring(l or "?")
							fname = tostring(n or "")
						end
					end
					if src ~= "[C]" then
						local hits = {}
						if fname ~= "" and string.find(string.lower(fname), needle, 1, true) then
							hits[#hits + 1] = "name=" .. fname
						end
						if string.find(string.lower(src), needle, 1, true) then
							hits[#hits + 1] = "src=" .. src
						end
						if getKs then
							local okK, ks = pcall(getKs, obj)
							if okK and type(ks) == "table" then
								for ki, kv in pairs(ks) do
									if type(kv) == "string" and string.find(string.lower(kv), needle, 1, true) then
										hits[#hits + 1] = string.format("K[%s]=%q", tostring(ki), string.sub(kv, 1, 80))
									end
								end
							end
						end
						if getUps then
							local okU, ups = pcall(getUps, obj)
							if okU and type(ups) == "table" then
								for ui, uv in pairs(ups) do
									if type(uv) == "string" and string.find(string.lower(uv), needle, 1, true) then
										hits[#hits + 1] = string.format("U[%s]=%q", tostring(ui), string.sub(uv, 1, 80))
									end
								end
							end
						end
						if #hits > 0 then
							matched += 1
							if #out < 450 then
								out[#out + 1] = string.format("[GC #%d] %s:%s fn=%s", matched, src, line, fname ~= "" and fname or "anon")
								for k = 1, math.min(#hits, 8) do
									out[#out + 1] = "  " .. hits[k]
								end
								out[#out + 1] = ""
							end
						end
					end
				end
			end
			if i % 400 == 0 then
				task.wait()
			end
		end

		out[3] = string.format("-- Closures Lua escaneadas: %d | Coincidencias: %d", scanned, matched)
		if matched == 0 then
			out[#out + 1] = "-- Sin closures en memoria para: " .. query
		end
		Dec.current = { label = "GCSearch_" .. sanitize(query), inst = nil, bc = nil, size = 0, class = "XRef" }
		publish(out, "txt")
		Zenith.log("Info", string.format("GC Search '%s': %d closures de %d", query, matched, scanned))
		return matched
	end

	local skipGlobalSearchHook = false
	function Zenith.findScriptXRefs(query)
		local q = tostring(query or "")
		if q == "" then
			Zenith.log("Warn", "Find Script XRefs: nombre de remote vacio")
			return 0
		end
		if Zenith.switchTab then
			pcall(Zenith.switchTab, "Decompiler")
		end
		skipGlobalSearchHook = true
		if Zenith.setters and Zenith.setters[N.globalSearch] then
			pcall(Zenith.setters[N.globalSearch], q)
		end
		skipGlobalSearchHook = false
		Zenith.values[N.globalSearch] = q
		return Dec.globalSearch(q)
	end

	local function formatCallCode(entry)
		local lines = {}
		local dirTag = entry.dir == "S2C" and "S->C" or (entry.dir == "INT" and "INT" or "C->S")
		local blkTag = entry.blocked and " [BLOCKED]" or ""
		local execTag = entry.isExecutor and " [EXECUTOR]" or ""
		local actorTag = entry.isActor and " [ACTOR]" or ""
		table.insert(lines, string.format("-- [%s] %s:%s | %s%s%s%s", dirTag, entry.className, entry.method, entry.timeStr or "?", blkTag, execTag, actorTag))
		if entry.caller and entry.caller ~= "" then
			table.insert(lines, "-- Caller: " .. entry.caller)
		end
		if entry.sourceLine and entry.sourceLine ~= "" then
			table.insert(lines, string.format("-- Origin: %s (%s)", tostring(entry.funcName or "anon"), entry.sourceLine))
		elseif entry.funcName and entry.funcName ~= "" then
			table.insert(lines, "-- Function: " .. entry.funcName)
		end
		if entry.funcHash and entry.funcHash ~= "" then
			table.insert(lines, "-- Hash: " .. entry.funcHash)
		end
		if flag(N.showStack, false) and entry.stack and entry.stack ~= "" then
			table.insert(lines, "-- Stack:")
			for ln in string.gmatch(entry.stack, "[^\n]+") do
				table.insert(lines, "--   " .. ln)
			end
		end

		if entry.dir == "INT" then
			local fnName = entry.className == "ProximityPrompt" and "fireproximityprompt" or "fireclickdetector"
			table.insert(lines, string.format("%s(%s)", fnName, entry.path))
			return table.concat(lines, "\n")
		end

		local argCount = entry.args.n or #entry.args
		if argCount == 0 then
			if entry.dir == "S2C" then
				table.insert(lines, string.format("-- Received by %s.%s()", entry.path, entry.method))
			else
				table.insert(lines, string.format("%s:%s()", entry.path, entry.method))
			end
		else
			table.insert(lines, "local args = {")
			for i = 1, argCount do
				table.insert(lines, string.format("\t[%d] = %s,", i, serializeValue(entry.args[i], 1, {})))
			end
			table.insert(lines, "}")
			if entry.dir == "S2C" then
				table.insert(lines, string.format("-- Received by %s.%s(table.unpack(args, 1, %d))", entry.path, entry.method, argCount))
			else
				table.insert(lines, string.format("%s:%s(table.unpack(args, 1, %d))", entry.path, entry.method, argCount))
			end
		end

		if entry.returns then
			local rCount = entry.returns.n or #entry.returns
			table.insert(lines, "")
			if rCount == 0 then
				table.insert(lines, "-- Return: (void)")
			else
				table.insert(lines, "-- Return Values:")
				for i = 1, rCount do
					table.insert(lines, string.format("--   [%d] = %s", i, serializeValue(entry.returns[i], 1, {})))
				end
			end
		end

		return table.concat(lines, "\n")
	end

	local function safeFormat(entry)
		local ok, res = pcall(formatCallCode, entry)
		if ok then
			return res
		end
		return "-- error formatting call: " .. tostring(res)
	end

	local function getSelectedEntry()
		local label = state.selectedLabel or Zenith.values[N.list]
		return label and (state.byLabel[label] or state.groupByLabel[label]) or nil, label
	end

	local function renderSelected()
		local entry, label = getSelectedEntry()
		local out = Zenith.outputs and Zenith.outputs[N.log]
		if not out then
			return
		end
		if not entry then
			out.setText("-- Selecciona una llamada capturada en Remote List.")
			return
		end
		local txt = safeFormat(entry)
		local gCount = label and state.groupCount[label]
		if gCount and gCount > 1 then
			txt = string.format("-- Group: %d llamadas capturadas (mostrando #%03d)\n", gCount, entry.id) .. txt
		end
		out.setText(txt)
	end

	-- selectNewest: jump selection to the newest visible entry (auto scroll)
	local function applyListFilter(selectNewest)
		local filter = tostring(Zenith.values[N.filter] or "All")
		local needle = string.lower(tostring(Zenith.values[N.search] or ""))
		local grouped = flag(N.groupByRemote, false)
		table.clear(state.shownLabels)
		table.clear(state.groupByLabel)
		table.clear(state.groupCount)
		local shownSet = {}

		if grouped then
			local order, groups = {}, {}
			for i = #state.entries, 1, -1 do
				local e = state.entries[i]
				if not isRobloxIgnored(e.remote, e.name, e.callerInst, e.className, e.isExecutor) then
					local classOk = (filter == "All") or (FILTER_MATCH[filter] and FILTER_MATCH[filter][e.className] == true)
					local searchOk = (needle == "")
						or string.find(string.lower(e.label), needle, 1, true) ~= nil
						or string.find(string.lower(e.path), needle, 1, true) ~= nil
					if classOk and searchOk then
						local gKey = tostring(e.remote or e.path) .. "|" .. e.dir .. "|" .. e.method
						local g = groups[gKey]
						if not g then
							g = { latest = e, count = 1 }
							groups[gKey] = g
							order[#order + 1] = g
						else
							g.count += 1
						end
					end
				end
			end
			for _, g in ipairs(order) do
				local e = g.latest
				local dirTag = e.dir == "S2C" and "S->C" or (e.dir == "INT" and "INT" or "C->S")
				local blkTag = e.blocked and "[BLK] " or ""
				local actTag = e.isActor and "[ACT] " or ""
				local gLabel = string.format("[%s] (x%d) %s%s%s | %s", dirTag, g.count, blkTag, actTag, e.method, e.name)
				if state.groupByLabel[gLabel] then
					gLabel = gLabel .. " #" .. e.id
				end
				state.groupByLabel[gLabel] = e
				state.groupCount[gLabel] = g.count
				table.insert(state.shownLabels, gLabel)
				shownSet[gLabel] = true
			end
		else
			for i = #state.entries, 1, -1 do
				local e = state.entries[i]
				if not isRobloxIgnored(e.remote, e.name, e.callerInst, e.className, e.isExecutor) then
					local classOk = (filter == "All") or (FILTER_MATCH[filter] and FILTER_MATCH[filter][e.className] == true)
					local searchOk = (needle == "")
						or string.find(string.lower(e.label), needle, 1, true) ~= nil
						or string.find(string.lower(e.path), needle, 1, true) ~= nil
					if classOk and searchOk then
						table.insert(state.shownLabels, e.label)
						shownSet[e.label] = true
					end
				end
			end
		end

		Zenith.setItems(N.list, table.clone(state.shownLabels))

		if selectNewest == true and state.shownLabels[1] then
			state.selectedLabel = state.shownLabels[1]
		elseif not (state.selectedLabel and shownSet[state.selectedLabel]) then
			state.selectedLabel = state.shownLabels[1]
		end
		Zenith.values[N.list] = state.selectedLabel
		renderSelected()
	end

	local function addEntry(item)
		local rName = tostring(safeRead(item.remote, "Name") or "?")
		if isRobloxIgnored(item.remote, rName, item.callerInst, item.className, item.isExecutor) then
			return
		end

		if flag(N.autoIgnoreSpam, false) and typeof(item.remote) == "Instance" and not item.blocked then
			local now = os.clock()
			local rec = state.spamTrack[item.remote]
			if not rec or (now - rec.t0) > 1 then
				rec = { t0 = now, count = 1 }
				state.spamTrack[item.remote] = rec
			else
				rec.count += 1
				if rec.count >= 30 and not isIgnoredName(rName) then
					local cur = tostring(Zenith.values[N.ignoreList] or "")
					local nextStr = (cur ~= "" and (cur .. ", " .. rName)) or rName
					setVal(N.ignoreList, nextStr)
					refreshIgnoreSet()
					Zenith.log("Info", "RSpy: auto-ignorado por spam -> " .. rName)
					return
				end
			end
		end

		state.seq += 1
		local path = instancePath(item.remote)
		local dirTag = item.dir == "S2C" and "S->C" or (item.dir == "INT" and "INT" or "C->S")
		local blkTag = item.blocked and "[BLK] " or ""
		local actTag = item.isActor and "[ACT] " or ""
		local label = string.format("#%03d [%s] %s%s%s | %s", state.seq, dirTag, blkTag, actTag, item.method, rName)
		local callerStr = ""
		if item.callerInst then
			callerStr = instancePath(item.callerInst)
		end
		local entry = {
			id = state.seq,
			label = label,
			remote = item.remote,
			name = rName,
			className = item.className,
			method = item.method,
			dir = item.dir,
			path = path,
			args = item.args or { n = 0 },
			returns = item.returns,
			blocked = item.blocked,
			caller = callerStr,
			callerInst = item.callerInst,
			func = item.func,
			funcName = item.funcName,
			sourceLine = item.sourceLine,
			funcHash = item.funcHash,
			isExecutor = item.isExecutor == true,
			isActor = item.isActor == true,
			stack = item.stack,
			timeStr = item.timeStr,
		}
		table.insert(state.entries, entry)
		state.byLabel[label] = entry
		pcall(appendSpyFileLog, entry)
	end

	local function installHooks()
		if state.installed then
			return true
		end
		local wrapC = Zenith.capFn("newcclosure") or function(f) return f end
		local getNCM = Zenith.capFn("getnamecallmethod")
		local hookMM = Zenith.capFn("hookmetamethod")
		local hookFn = Zenith.capFn("hookfunction")
		local othHook = flag(N.preferOth, true) and Zenith.capFn("oth_hook") or nil

		if hookMM and getNCM then
			local rec = { active = true, old = nil }
			local ok, old = pcall(hookMM, game, "__namecall", wrapC(function(self, ...)
				local method = getNCM()
				return handleOutbound(rec, self, method, ...)
			end))
			if ok and type(old) == "function" then
				rec.old = old
				state.oldNamecall = old
				state.namecallRec = rec
				state.installed = true
			else
				rec.active = false
				Zenith.log("Warn", "RSpy: hookmetamethod fallo (" .. tostring(old) .. ")")
			end
		end
		if not state.namecallRec and Zenith.cap("getrawmetatable") and getNCM then
			local getRMT = Zenith.capFn("getrawmetatable")
			local setRO = typeof(setreadonly) == "function" and setreadonly or nil
			local ok, mt = pcall(getRMT, game)
			if ok and type(mt) == "table" and type(mt.__namecall) == "function" then
				local rec = { active = true, old = mt.__namecall }
				if setRO then pcall(setRO, mt, false) end
				local okSet = pcall(function()
					mt.__namecall = wrapC(function(self, ...)
						local method = getNCM()
						return handleOutbound(rec, self, method, ...)
					end)
				end)
				if setRO then pcall(setRO, mt, true) end
				if okSet then
					state.oldNamecall = rec.old
					state.namecallRec = rec
					state.installed = true
				else
					rec.active = false
					Zenith.log("Warn", "RSpy: no se pudo escribir __namecall")
				end
			end
		end

		if hookFn or othHook then
			local specs = {
				{ "RemoteEvent", "FireServer" },
				{ "UnreliableRemoteEvent", "FireServer" },
				{ "RemoteFunction", "InvokeServer" },
				{ "BindableEvent", "Fire" },
				{ "BindableFunction", "Invoke" },
			}
			local hookedTargets = {}
			for _, sp in ipairs(specs) do
				local cls, method = sp[1], sp[2]
				local okI, dummy = pcall(Instance.new, cls)
				if okI and dummy then
					local fn = safeRead(dummy, method)
					if type(fn) == "function" and not hookedTargets[fn] then
						hookedTargets[fn] = true
						local record = { active = true, target = fn, old = nil, usedOth = false }
						local hooked = false
						if othHook then
							local okO, oldO = pcall(othHook, fn, wrapC(function(self, ...)
								return handleOutbound(record, self, method, ...)
							end))
							if okO and type(oldO) == "function" then
								record.old = oldO
								record.usedOth = true
								table.insert(state.directHooks, record)
								state.installed = true
								hooked = true
							end
						end
						if not hooked and hookFn then
							local okH, old = pcall(hookFn, fn, wrapC(function(self, ...)
								return handleOutbound(record, self, method, ...)
							end))
							if okH and type(old) == "function" then
								record.old = old
								table.insert(state.directHooks, record)
								state.installed = true
							else
								record.active = false
							end
						elseif not hooked then
							record.active = false
						end
					end
					pcall(function()
						dummy:Destroy()
					end)
				end
			end
		end

		if installNewindexHook(wrapC) then
			state.installed = true
		end
		installSignalHook(wrapC)
		installActorHooks()

		return state.installed
	end

	local function uninstallInteractionHooks()
		for i = #state.interactionConns, 1, -1 do
			local c = state.interactionConns[i]
			pcall(function()
				c:Disconnect()
			end)
			state.interactionConns[i] = nil
		end
		state.interactionHooked = setmetatable({}, { __mode = "k" })
	end

	local function installInteractionHooks()
		uninstallInteractionHooks()
		if stopped or not (flag(N.enabled, false) and flag(N.logInteractions, false)) then
			return false
		end

		local okP, pps = pcall(game.GetService, game, "ProximityPromptService")
		if okP and pps and pps.PromptTriggered then
			local okC, conn = pcall(function()
				return pps.PromptTriggered:Connect(function(prompt, plr)
					if stopped or not (flag(N.enabled, false) and flag(N.logInteractions, false)) then
						return
					end
					if plr and localPlayer and plr ~= localPlayer then
						return
					end
					if typeof(prompt) ~= "Instance" then
						return
					end
					enqueue({
						remote = prompt,
						className = "ProximityPrompt",
						method = "FireProximityPrompt",
						dir = "INT",
						args = { n = 0 },
						timeStr = timeStamp(),
					})
				end)
			end)
			if okC and conn then
				table.insert(state.interactionConns, conn)
			end
		end

		local function watchClickDetector(cd)
			if typeof(cd) ~= "Instance" or state.interactionHooked[cd] then
				return
			end
			local okIs, isCD = pcall(cd.IsA, cd, "ClickDetector")
			if not okIs or not isCD then
				return
			end
			state.interactionHooked[cd] = true
			local okC, conn = pcall(function()
				return cd.MouseClick:Connect(function(plr)
					if stopped or not (flag(N.enabled, false) and flag(N.logInteractions, false)) then
						return
					end
					if plr and localPlayer and plr ~= localPlayer then
						return
					end
					enqueue({
						remote = cd,
						className = "ClickDetector",
						method = "FireClickDetector",
						dir = "INT",
						args = { n = 0 },
						timeStr = timeStamp(),
					})
				end)
			end)
			if okC and conn then
				table.insert(state.interactionConns, conn)
			end
		end

		local okW, ws = pcall(game.GetService, game, "Workspace")
		if okW and ws then
			local okD, desc = pcall(ws.GetDescendants, ws)
			if okD and type(desc) == "table" then
				for _, d in ipairs(desc) do
					watchClickDetector(d)
				end
			end
			local okA, addConn = pcall(function()
				return ws.DescendantAdded:Connect(watchClickDetector)
			end)
			if okA and addConn then
				table.insert(state.interactionConns, addConn)
			end
		end
		return #state.interactionConns > 0
	end

	local function uninstallHooks()
		disconnectS2C()
		uninstallNewindexHook()
		uninstallActorHooks()
		uninstallInteractionHooks()

		-- Deactivate first: any wrapper still alive becomes a pure pass-through
		if state.namecallRec then
			state.namecallRec.active = false
		end
		if state.signalHook then
			state.signalHook.active = false
		end
		for _, rec in ipairs(state.directHooks) do
			rec.active = false
		end

		local hookMM = Zenith.capFn("hookmetamethod")
		if state.oldNamecall then
			if hookMM then
				pcall(hookMM, game, "__namecall", state.oldNamecall)
			elseif Zenith.cap("getrawmetatable") then
				local getRMT = Zenith.capFn("getrawmetatable")
				local setRO = typeof(setreadonly) == "function" and setreadonly or nil
				local ok, mt = pcall(getRMT, game)
				if ok and type(mt) == "table" then
					if setRO then pcall(setRO, mt, false) end
					pcall(function()
						mt.__namecall = state.oldNamecall
					end)
					if setRO then pcall(setRO, mt, true) end
				end
			end
			state.oldNamecall = nil
		end
		state.namecallRec = nil

		local hookFn = Zenith.capFn("hookfunction")
		if state.signalHook and state.signalHook.target and state.signalHook.old then
			local restoredSig = false
			if typeof(restorefunction) == "function" then
				restoredSig = pcall(restorefunction, state.signalHook.target)
			end
			if not restoredSig and hookFn then
				pcall(hookFn, state.signalHook.target, state.signalHook.old)
			end
		end
		state.signalHook = nil

		local othUnhook = Zenith.capFn("oth_unhook")
		for _, rec in ipairs(state.directHooks) do
			if rec.target and rec.old then
				local restored = false
				if rec.usedOth and othUnhook then
					restored = pcall(othUnhook, rec.target)
				end
				if not restored and typeof(restorefunction) == "function" then
					restored = pcall(restorefunction, rec.target)
				end
				if not restored and hookFn then
					pcall(hookFn, rec.target, rec.old)
				end
			end
		end
		table.clear(state.directHooks)
		state.installed = false
	end

			end

	Zenith.actions[N.repeatCall] = function()
		local entry = getSelectedEntry()
		if not entry or not entry.remote then
			return Zenith.log("Warn", "RSpy: selecciona una llamada valida")
		end
		if entry.dir == "S2C" then
			return Zenith.log("Warn", "RSpy: no se puede reenviar una llamada Server->Client")
		end
		if entry.dir == "INT" then
			local fnName = entry.className == "ProximityPrompt" and "fireproximityprompt" or "fireclickdetector"
			local fn = Zenith.capFn(fnName)
			if not fn then
				return Zenith.log("Warn", "RSpy: " .. fnName .. " no soportado en este executor")
			end
			task.spawn(function()
				local thread = coroutine.running() or MAIN_THREAD
				state.internal[thread] = true
				local ok, err = pcall(fn, entry.remote)
				state.internal[thread] = nil
				Zenith.log(ok and "Info" or "Error", ok and ("RSpy: repetida " .. entry.name) or ("RSpy Repeat error: " .. tostring(err)))
			end)
			return
		end
		task.spawn(function()
			local thread = coroutine.running() or MAIN_THREAD
			state.internal[thread] = true -- per-thread: other calls keep being logged
			local ok, err = pcall(function()
				local fn = safeRead(entry.remote, entry.method)
				if type(fn) ~= "function" then
					error("metodo no encontrado: " .. tostring(entry.method))
				end
				local n = entry.args.n or #entry.args
				fn(entry.remote, table.unpack(entry.args, 1, n))
			end)
			state.internal[thread] = nil
			Zenith.log(ok and "Info" or "Error", ok and ("RSpy: repetida " .. entry.name) or ("RSpy Repeat error: " .. tostring(err)))
		end)
	end

	Zenith.actions[N.replayCustom] = function()
		local entry = getSelectedEntry()
		if not entry or not entry.remote then
			return Zenith.log("Warn", "RSpy: selecciona una llamada valida")
		end
		if entry.dir == "S2C" then
			return Zenith.log("Warn", "RSpy: no se puede reenviar una llamada Server->Client")
		end
		if entry.dir == "INT" then
			return Zenith.actions[N.repeatCall]()
		end
		local raw = string.match(tostring(Zenith.values[N.customArgs] or ""), "^%s*(.-)%s*$")
		if raw == "" then
			return Zenith.actions[N.repeatCall]()
		end
		local loader = (typeof(loadstring) == "function" and loadstring) or (typeof(load) == "function" and load)
		if not loader then
			return Zenith.log("Warn", "RSpy: este entorno no soporta loadstring/load para Custom Args")
		end
		local chunk, compileErr = loader("return " .. raw)
		if not chunk then
			return Zenith.log("Error", "RSpy Custom Args sintaxis: " .. tostring(compileErr))
		end
		local evalPack = table.pack(pcall(chunk))
		if not evalPack[1] then
			return Zenith.log("Error", "RSpy Custom Args eval: " .. tostring(evalPack[2]))
		end
		local argCount = math.max(0, (evalPack.n or #evalPack) - 1)
		task.spawn(function()
			local thread = coroutine.running() or MAIN_THREAD
			state.internal[thread] = true
			local ok, err = pcall(function()
				local fn = safeRead(entry.remote, entry.method)
				if type(fn) ~= "function" then
					error("metodo no encontrado: " .. tostring(entry.method))
				end
				fn(entry.remote, table.unpack(evalPack, 2, 1 + argCount))
			end)
			state.internal[thread] = nil
			Zenith.log(ok and "Info" or "Error", ok and string.format("RSpy: replay custom (%d args) -> %s", argCount, entry.name) or ("RSpy Replay Custom error: " .. tostring(err)))
		end)
	end

	Zenith.actions[N.blockRemote] = function()
		local entry = getSelectedEntry()
		if not entry or not entry.remote then
			return Zenith.log("Warn", "RSpy: selecciona un remote primero")
		end
		local nowBlocked = not (state.blocked[entry.remote] == true)
		state.blocked[entry.remote] = nowBlocked or nil
		Zenith.log("Info", string.format("RSpy: %s %s", entry.name, nowBlocked and "BLOQUEADO" or "desbloqueado"))
	end

	Zenith.actions[N.ignoreRemote] = function()
		local entry = getSelectedEntry()
		if not entry then
			return Zenith.log("Warn", "RSpy: selecciona un remote primero")
		end
		local cur = tostring(Zenith.values[N.ignoreList] or "")
		if not isIgnoredName(entry.name) then
			local nextStr = (cur ~= "" and (cur .. ", " .. entry.name)) or entry.name
			setVal(N.ignoreList, nextStr)
			refreshIgnoreSet()
			applyListFilter()
			Zenith.log("Info", "RSpy: ignorado " .. entry.name)
		end
	end

end
