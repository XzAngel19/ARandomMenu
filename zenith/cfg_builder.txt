-- ============================================================================
-- Zenith Decompiler (Fase 2 - Paso 1): Basic Block + CFG Builder + Pretty Printer
-- Archivo independiente para revision y pruebas antes de integrarlo en zenith.txt.
--
-- Contrato de entrada esperado (cumplido por zenith/zenith.txt -> decode()):
--   insn = { { pc, op, name, A, B, C, D, E, len, aux, line }, ... }
--   - ins.pc es el indice de PALABRA base 0 (las palabras AUX no son entradas).
--   - ins.D (16-bit) e ins.E (24-bit) vienen decodificados CON SIGNO.
--   - ins.len = 2 si la instruccion consume palabra AUX, si no 1.
-- ============================================================================

local CFG = {}
local band = bit32.band

-- ----------------------------------------------------------------------------
-- 1. Clasificacion de instrucciones de control de flujo en Luau
-- ----------------------------------------------------------------------------
-- Nota sobre FASTCALL*: aunque C indica el offset al CALL si la builtin tiene
-- exito, semanticamente representan una llamada lineal (fastcall + args + CALL).
-- Por eso no parten bloques; registramos su rango para que el lifter lo use.

local FASTCALL_OPS = {
	FASTCALL = true,
	FASTCALL1 = true,
	FASTCALL2 = true,
	FASTCALL2K = true,
	FASTCALL3 = true,
	FASTPCALL = true,
}

local UNCOND_JUMP = {
	JUMP = true,
	JUMPBACK = true,
	JUMPX = true,
}

local COND_JUMP = {
	JUMPIF = true,
	JUMPIFNOT = true,
	JUMPIFEQ = true,
	JUMPIFLE = true,
	JUMPIFLT = true,
	JUMPIFNOTEQ = true,
	JUMPIFNOTLE = true,
	JUMPIFNOTLT = true,
	JUMPIFEQK = true,      -- Luau v3 compat
	JUMPIFNOTEQK = true,   -- Luau v3 compat
	JUMPXEQKNIL = true,
	JUMPXEQKB = true,
	JUMPXEQKN = true,
	JUMPXEQKS = true,
	CMPPROTO = true,
}

-- LOADB A, B, C (si C ~= 0) y FINCONSTRUCT A, C (si C ~= 0) saltan incondicionalmente
local COND_SKIP_UNCOND = {
	LOADB = true,
	FINCONSTRUCT = true,
}

local FOR_GEN_PREP = {
	FORGPREP = true,
	FORGPREP_INEXT = true,
	FORGPREP_NEXT = true,
}

local FOR_LOOP = {
	FORNLOOP = true,
	FORGLOOP = true,
	FORGLOOP_INEXT = true, -- compat versiones antiguas
	FORGLOOP_NEXT = true,  -- compat versiones antiguas
}

local JD_DEFAULT = {
	JUMP = true, JUMPBACK = true, JUMPIF = true, JUMPIFNOT = true,
	JUMPIFEQ = true, JUMPIFLE = true, JUMPIFLT = true,
	JUMPIFNOTEQ = true, JUMPIFNOTLE = true, JUMPIFNOTLT = true,
	JUMPIFEQK = true, JUMPIFNOTEQK = true,
	FORNPREP = true, FORNLOOP = true, FORGLOOP = true,
	FORGLOOP_INEXT = true, FORGLOOP_NEXT = true,
	FORGPREP_INEXT = true, FORGPREP_NEXT = true, FORGPREP = true,
	JUMPXEQKNIL = true, JUMPXEQKB = true, JUMPXEQKN = true, JUMPXEQKS = true,
	CMPPROTO = true,
}

local function defaultJumpOf(ins)
	if JD_DEFAULT[ins.name] then
		return ins.pc + 1 + (ins.D or 0)
	end
	if ins.name == "JUMPX" then
		return ins.pc + 1 + (ins.E or 0)
	end
	if COND_SKIP_UNCOND[ins.name] and (ins.C or 0) ~= 0 then
		return ins.pc + 1 + ins.C
	end
	return nil
end

-- Describe la polaridad y operandos de un salto condicional para el structurer.
-- Nota Luau/NaN: JUMPIFNOTLT y JUMPIFNOTLE preservan `not (a < b)` y `not (a <= b)`
-- (no deben reescribirse a >= o > sin saber si los operandos pueden ser NaN).
local function describeCond(ins)
	local n = ins.name
	if not COND_JUMP[n] then
		return nil
	end
	local aux = ins.aux or 0
	local notBit = aux >= 0x80000000

	if n == "JUMPIF" then
		return { op = "truthy", regA = ins.A, negated = false }
	elseif n == "JUMPIFNOT" then
		return { op = "truthy", regA = ins.A, negated = true }
	elseif n == "JUMPIFEQ" then
		return { op = "eq", regA = ins.A, regB = band(aux, 255), negated = false }
	elseif n == "JUMPIFNOTEQ" then
		return { op = "eq", regA = ins.A, regB = band(aux, 255), negated = true }
	elseif n == "JUMPIFLT" then
		return { op = "lt", regA = ins.A, regB = band(aux, 255), negated = false }
	elseif n == "JUMPIFNOTLT" then
		return { op = "lt", regA = ins.A, regB = band(aux, 255), negated = true, nanSafe = true }
	elseif n == "JUMPIFLE" then
		return { op = "le", regA = ins.A, regB = band(aux, 255), negated = false }
	elseif n == "JUMPIFNOTLE" then
		return { op = "le", regA = ins.A, regB = band(aux, 255), negated = true, nanSafe = true }
	elseif n == "JUMPIFEQK" then
		return { op = "eqk", regA = ins.A, kIdx = band(aux, 0xFFFFFF), negated = false }
	elseif n == "JUMPIFNOTEQK" then
		return { op = "eqk", regA = ins.A, kIdx = band(aux, 0xFFFFFF), negated = true }
	elseif n == "JUMPXEQKNIL" then
		return { op = "eq_nil", regA = ins.A, negated = notBit }
	elseif n == "JUMPXEQKB" then
		return { op = "eq_bool", regA = ins.A, val = band(aux, 1) == 1, negated = notBit }
	elseif n == "JUMPXEQKN" or n == "JUMPXEQKS" then
		return { op = "eq_k", regA = ins.A, kIdx = band(aux, 0xFFFFFF), negated = notBit }
	elseif n == "CMPPROTO" then
		return { op = "cmpproto", regA = ins.A, aux = aux, negated = false }
	end
	return nil
end

local function isControlFlow(ins, jumpOfFn)
	local name = ins.name
	if name == "RETURN" or UNCOND_JUMP[name] or COND_JUMP[name] or FOR_GEN_PREP[name] or name == "FORNPREP" or FOR_LOOP[name] then
		return true
	end
	if COND_SKIP_UNCOND[name] and jumpOfFn(ins) ~= nil then
		return true
	end
	return false
end

local function outgoingOf(ins, nextPc, jumpOfFn)
	local name = ins.name
	local target = jumpOfFn(ins)

	if name == "RETURN" then
		return {}
	end

	if UNCOND_JUMP[name] then
		if target then
			local kind = (name == "JUMPBACK" or target <= ins.pc) and "jump_back" or "jump"
			return { { targetPc = target, kind = kind } }
		end
		return {}
	end

	if COND_SKIP_UNCOND[name] then
		if target then
			return { { targetPc = target, kind = "jump" } }
		end
		return nextPc and { { targetPc = nextPc, kind = "fallthrough" } } or {}
	end

	if FOR_GEN_PREP[name] then
		-- FORGPREP* salta incondicionalmente hacia adelante al FORGLOOP correspondiente
		if target then
			return { { targetPc = target, kind = "for_prep" } }
		end
		return {}
	end

	if name == "FORNPREP" then
		-- FORNPREP entra al cuerpo (fallthrough) o salta fuera del bucle si 0 iteraciones
		local out = {}
		if nextPc then
			out[#out + 1] = { targetPc = nextPc, kind = "for_enter" }
		end
		if target then
			out[#out + 1] = { targetPc = target, kind = "for_skip" }
		end
		return out
	end

	if FOR_LOOP[name] then
		-- FORNLOOP / FORGLOOP repiten el cuerpo o salen por fallthrough
		local out = {}
		if target then
			out[#out + 1] = { targetPc = target, kind = "for_loop" }
		end
		if nextPc then
			out[#out + 1] = { targetPc = nextPc, kind = "for_exit" }
		end
		return out
	end

	if COND_JUMP[name] then
		local out = {}
		if target then
			out[#out + 1] = { targetPc = target, kind = "taken" }
		end
		if nextPc then
			out[#out + 1] = { targetPc = nextPc, kind = "fallthrough" }
		end
		return out
	end

	if nextPc then
		return { { targetPc = nextPc, kind = "fallthrough" } }
	end
	return {}
end

-- ----------------------------------------------------------------------------
-- 2. Construccion de Basic Blocks y Grafo de Flujo (CFG)
-- ----------------------------------------------------------------------------
function CFG.build(insn, opts)
	opts = opts or {}
	local jumpOfFn = opts.jumpOf or defaultJumpOf

	local cfg = {
		blocks = {},          -- { BasicBlock } ordenados por pc ascendente
		byStartPc = {},       -- [pc] -> BasicBlock
		entry = nil,          -- BasicBlock inicial
		exits = {},           -- { BasicBlock } alcanzables sin sucesores
		rpo = {},             -- { BasicBlock } alcanzables en Reverse Postorder
		loops = {},           -- { LoopInfo } ordenados de mas interno a mas externo
		fastcalls = {},       -- { { pc, callPc, op, fastId } }
		errors = {},          -- { string } objetivos invalidos o inconsistencias
		irreducible = false,  -- true si hay retreating edges no dominadas
		irreducibleEdges = {},
	}

	local n = #insn
	if n == 0 then
		return cfg
	end

	local pcToIdx = {}
	for i, ins in ipairs(insn) do
		pcToIdx[ins.pc] = i
		if FASTCALL_OPS[ins.name] and ins.C and ins.C > 0 then
			cfg.fastcalls[#cfg.fastcalls + 1] = {
				pc = ins.pc,
				callPc = ins.pc + 1 + ins.C,
				op = ins.name,
				fastId = ins.A,
			}
		end
	end

	-- Paso A: Detectar leaders (inicios de bloque)
	local isLeader = {}
	isLeader[insn[1].pc] = true

	for i, ins in ipairs(insn) do
		local nextIns = insn[i + 1]
		local nextPc = nextIns and nextIns.pc or nil
		local target = jumpOfFn(ins)

		if isControlFlow(ins, jumpOfFn) then
			if target and pcToIdx[target] then
				isLeader[target] = true
			end
			if nextPc then
				isLeader[nextPc] = true
			end
		end
	end

	-- Paso B: Cortar instrucciones en BasicBlocks
	local blocks = {}
	local curBlock = nil

	local function flushBlock()
		if curBlock and #curBlock.insns > 0 then
			local lastIns = curBlock.insns[#curBlock.insns]
			curBlock.endPc = lastIns.pc
			curBlock.cond = describeCond(lastIns)
			blocks[#blocks + 1] = curBlock
		end
		curBlock = nil
	end

	for i, ins in ipairs(insn) do
		if isLeader[ins.pc] then
			flushBlock()
			curBlock = {
				id = #blocks + 1,
				startPc = ins.pc,
				endPc = ins.pc,
				insns = {},
				succ = {},        -- { BasicBlock } deduplicado
				pred = {},        -- { BasicBlock } deduplicado
				edgesOut = {},    -- { Edge } (puede haber >1 arista si target == nextPc)
				edgesIn = {},     -- { Edge } (misma tabla Edge que edgesOut)
				cond = nil,       -- metadata de condicion si termina en COND_JUMP
				reachable = false,
				rpoIndex = nil,
				idom = nil,       -- dominador inmediato
				ipdom = nil,      -- post-dominador inmediato (nil = salida/bucle infinito)
				loopHeader = nil, -- cabecera del bucle mas interno
			}
			cfg.byStartPc[ins.pc] = curBlock
		end

		curBlock.insns[#curBlock.insns + 1] = ins

		local nextIns = insn[i + 1]
		if isControlFlow(ins, jumpOfFn) or (nextIns and isLeader[nextIns.pc]) then
			flushBlock()
		end
	end
	flushBlock()

	-- Paso C: Conectar aristas compartidas y deduplicar succ/pred
	local succSeen, predSeen = {}, {}
	for _, b in ipairs(blocks) do
		succSeen[b.id] = {}
		predSeen[b.id] = {}
	end

	for idx, blk in ipairs(blocks) do
		local lastIns = blk.insns[#blk.insns]
		local nextBlk = blocks[idx + 1]
		local nextPc = nextBlk and nextBlk.startPc or nil
		local outs = outgoingOf(lastIns, nextPc, jumpOfFn)

		for _, e in ipairs(outs) do
			local targetBlk = cfg.byStartPc[e.targetPc]
			if targetBlk then
				local edge = {
					from = blk,
					to = targetBlk,
					kind = e.kind,
					isBack = false, -- se marca en _findLoops usando dominancia real
				}
				blk.edgesOut[#blk.edgesOut + 1] = edge
				targetBlk.edgesIn[#targetBlk.edgesIn + 1] = edge

				if not succSeen[blk.id][targetBlk.id] then
					succSeen[blk.id][targetBlk.id] = true
					blk.succ[#blk.succ + 1] = targetBlk
				end
				if not predSeen[targetBlk.id][blk.id] then
					predSeen[targetBlk.id][blk.id] = true
					targetBlk.pred[#targetBlk.pred + 1] = blk
				end
			else
				cfg.errors[#cfg.errors + 1] = string.format(
					"arista sin bloque destino (%s en pc %d -> target pc %d)",
					lastIns.name,
					lastIns.pc,
					e.targetPc
				)
			end
		end
	end

	cfg.blocks = blocks
	cfg.entry = blocks[1]

	-- Paso D: RPO iterativo (seguro contra funciones enormes sin desbordar stack)
	if cfg.entry then
		local visited, postorder = {}, {}
		local stack = { { blk = cfg.entry, i = 1 } }
		visited[cfg.entry.id] = true
		cfg.entry.reachable = true

		while #stack > 0 do
			local top = stack[#stack]
			local nxt = top.blk.succ[top.i]
			if nxt then
				top.i += 1
				if not visited[nxt.id] then
					visited[nxt.id] = true
					nxt.reachable = true
					stack[#stack + 1] = { blk = nxt, i = 1 }
				end
			else
				postorder[#postorder + 1] = top.blk
				stack[#stack] = nil
			end
		end

		for i = #postorder, 1, -1 do
			local blk = postorder[i]
			blk.rpoIndex = #cfg.rpo + 1
			cfg.rpo[blk.rpoIndex] = blk
		end
	end

	for _, blk in ipairs(blocks) do
		if blk.reachable and #blk.succ == 0 then
			cfg.exits[#cfg.exits + 1] = blk
		end
	end

	-- Paso E: Dominadores, Post-dominadores y Bucles Naturales
	CFG._computeDominators(cfg)
	CFG._computePostDominators(cfg)
	CFG._findLoops(cfg)

	return cfg
end

-- ----------------------------------------------------------------------------
-- 3. Dominadores, Post-dominadores y Bucles Naturales
-- ----------------------------------------------------------------------------
function CFG._dominates(a, b)
	local cur = b
	while cur do
		if cur == a then
			return true
		end
		if cur == cur.idom then
			break
		end
		cur = cur.idom
	end
	return false
end

local function intersectBy(b1, b2, domField, rpoField)
	local f1, f2 = b1, b2
	while f1 ~= f2 do
		while f1[rpoField] > f2[rpoField] do
			f1 = f1[domField]
		end
		while f2[rpoField] > f1[rpoField] do
			f2 = f2[domField]
		end
	end
	return f1
end

function CFG._computeDominators(cfg)
	local entry = cfg.entry
	if not entry then
		return
	end
	entry.idom = entry

	local changed = true
	while changed do
		changed = false
		for i = 2, #cfg.rpo do
			local b = cfg.rpo[i]
			local newIdom = nil
			for _, p in ipairs(b.pred) do
				if p.reachable and p.idom ~= nil then
					if newIdom == nil then
						newIdom = p
					else
						newIdom = intersectBy(p, newIdom, "idom", "rpoIndex")
					end
				end
			end
			if newIdom and b.idom ~= newIdom then
				b.idom = newIdom
				changed = true
			end
		end
	end
end

function CFG._computePostDominators(cfg)
	if #cfg.exits == 0 then
		return
	end

	-- Nodo virtual de salida para soportar multiples RETURN
	local virtualExit = {
		id = 0,
		pred = cfg.exits,
		succ = {},
		revRpoIndex = 1,
		ipdom = nil,
	}
	virtualExit.ipdom = virtualExit

	-- RPO sobre el grafo inverso desde virtualExit
	local visited, revPost = { [0] = true }, {}
	local stack = { { blk = virtualExit, i = 1 } }
	while #stack > 0 do
		local top = stack[#stack]
		local prev = top.blk.pred[top.i]
		if prev then
			top.i += 1
			if prev.reachable and not visited[prev.id] then
				visited[prev.id] = true
				stack[#stack + 1] = { blk = prev, i = 1 }
			end
		else
			revPost[#revPost + 1] = top.blk
			stack[#stack] = nil
		end
	end

	local revRpo = {}
	for i = #revPost, 1, -1 do
		local b = revPost[i]
		b.revRpoIndex = #revRpo + 1
		revRpo[b.revRpoIndex] = b
	end

	local isExit = {}
	for _, ex in ipairs(cfg.exits) do
		isExit[ex.id] = true
	end

	local changed = true
	while changed do
		changed = false
		for i = 2, #revRpo do
			local b = revRpo[i]
			local newIpdom = isExit[b.id] and virtualExit or nil
			for _, s in ipairs(b.succ) do
				if s.revRpoIndex and s.ipdom ~= nil then
					if newIpdom == nil then
						newIpdom = s
					else
						newIpdom = intersectBy(s, newIpdom, "ipdom", "revRpoIndex")
					end
				end
			end
			if newIpdom and b.ipdom ~= newIpdom then
				b.ipdom = newIpdom
				changed = true
			end
		end
	end

	for _, b in ipairs(revRpo) do
		if b ~= virtualExit then
			if b.ipdom == virtualExit then
				b.ipdom = nil
			end
			b.revRpoIndex = nil
		end
	end
end

local function classifyLoopKind(loop)
	local headerLast = loop.header.insns[#loop.header.insns]
	if headerLast then
		local n = headerLast.name
		if n == "FORNLOOP" then
			return "numeric_for"
		elseif n == "FORGLOOP" or n == "FORGLOOP_INEXT" or n == "FORGLOOP_NEXT" then
			return "generic_for"
		end
	end
	if #loop.latches == 1 then
		local latch = loop.latches[1]
		local latchLast = latch.insns[#latch.insns]
		if latchLast and COND_JUMP[latchLast.name] then
			return "repeat_until"
		end
	end
	return "while_or_loop"
end

function CFG._findLoops(cfg)
	local byHeader = {}

	for _, u in ipairs(cfg.rpo) do
		for _, e in ipairs(u.edgesOut) do
			local h = e.to
			if h.reachable then
				if CFG._dominates(h, u) then
					e.isBack = true
					local loop = byHeader[h.id]
					if not loop then
						loop = {
							header = h,
							latches = {},
							backEdges = {},
							exits = {},
							body = { [h.id] = true },
							kind = "loop",
							parent = nil,
							depth = 1,
						}
						byHeader[h.id] = loop
						cfg.loops[#cfg.loops + 1] = loop
					end
					loop.backEdges[#loop.backEdges + 1] = e
					loop.latches[#loop.latches + 1] = u

					if not loop.body[u.id] then
						loop.body[u.id] = true
						local stack = { u }
						while #stack > 0 do
							local cur = table.remove(stack)
							for _, p in ipairs(cur.pred) do
								if p.reachable and not loop.body[p.id] then
									loop.body[p.id] = true
									stack[#stack + 1] = p
								end
							end
						end
					end
				elseif h.rpoIndex and u.rpoIndex and h.rpoIndex <= u.rpoIndex then
					-- Arista que retrocede en RPO pero sin dominancia: grafo irreducible
					cfg.irreducible = true
					cfg.irreducibleEdges[#cfg.irreducibleEdges + 1] = e
				end
			end
		end
	end

	-- Ordenar de mas interno a mas externo (mayor rpoIndex primero)
	table.sort(cfg.loops, function(a, b)
		return a.header.rpoIndex > b.header.rpoIndex
	end)

	-- Calcular loopHeader mas interno, exits, kind y jerarquia parent/depth
	for i, loop in ipairs(cfg.loops) do
		loop.kind = classifyLoopKind(loop)

		for blkId in pairs(loop.body) do
			local blk = cfg.blocks[blkId]
			if blk then
				if blk.loopHeader == nil then
					blk.loopHeader = loop.header
				end
				for _, e in ipairs(blk.edgesOut) do
					if not loop.body[e.to.id] then
						loop.exits[#loop.exits + 1] = e
					end
				end
			end
		end

		for j = i + 1, #cfg.loops do
			local outer = cfg.loops[j]
			if outer.body[loop.header.id] then
				loop.parent = outer
				break
			end
		end
	end

	for i = #cfg.loops, 1, -1 do
		local loop = cfg.loops[i]
		loop.depth = loop.parent and (loop.parent.depth + 1) or 1
	end
end

-- ----------------------------------------------------------------------------
-- 4. Pretty Printer legible para inspeccionar el CFG
-- ----------------------------------------------------------------------------
local function defaultFmtIns(ins)
	local parts = {}
	if ins.A ~= nil then parts[#parts + 1] = "A=" .. tostring(ins.A) end
	if ins.B ~= nil then parts[#parts + 1] = "B=" .. tostring(ins.B) end
	if ins.C ~= nil then parts[#parts + 1] = "C=" .. tostring(ins.C) end
	if ins.D ~= nil and ins.D ~= 0 then parts[#parts + 1] = "D=" .. tostring(ins.D) end
	return table.concat(parts, " "), nil
end

function CFG.format(cfg, opts)
	opts = opts or {}
	local fmtInsFn = opts.fmtIns or defaultFmtIns
	local out = {}

	local edgeCount, unreachableCount = 0, 0
	for _, b in ipairs(cfg.blocks) do
		edgeCount += #b.edgesOut
		if not b.reachable then
			unreachableCount += 1
		end
	end

	out[#out + 1] = string.format(
		"-- CFG: %d bloques | %d aristas | %d bucles | %d inalcanzables%s",
		#cfg.blocks,
		edgeCount,
		#cfg.loops,
		unreachableCount,
		cfg.irreducible and " | IRREDUCIBLE" or ""
	)

	if #cfg.errors > 0 then
		for _, err in ipairs(cfg.errors) do
			out[#out + 1] = "-- ERROR: " .. err
		end
	end

	if #cfg.loops > 0 then
		local loopDesc = {}
		for i, loop in ipairs(cfg.loops) do
			local count = 0
			for _ in pairs(loop.body) do
				count += 1
			end
			local backs = {}
			for _, be in ipairs(loop.backEdges) do
				backs[#backs + 1] = "B" .. be.from.id .. "(" .. be.kind .. ")"
			end
			local exits = {}
			for _, ex in ipairs(loop.exits) do
				exits[#exits + 1] = string.format("B%d->B%d", ex.from.id, ex.to.id)
			end
			loopDesc[i] = string.format(
				"L%d[%s d=%d]: header=B%d bloques=%d back=[%s] exits=[%s]",
				i,
				loop.kind,
				loop.depth,
				loop.header.id,
				count,
				table.concat(backs, ", "),
				table.concat(exits, ", ")
			)
		end
		out[#out + 1] = "-- Bucles: " .. table.concat(loopDesc, " ; ")
	end
	out[#out + 1] = ""

	for _, b in ipairs(cfg.blocks) do
		local tags = {}
		if b == cfg.entry then tags[#tags + 1] = "entry" end
		if #b.succ == 0 and b.reachable then tags[#tags + 1] = "exit" end
		if not b.reachable then tags[#tags + 1] = "UNREACHABLE" end
		if b.loopHeader == b then tags[#tags + 1] = "loop_header" end

		local meta = {}
		if b.idom and b ~= cfg.entry then
			meta[#meta + 1] = "idom=B" .. b.idom.id
		end
		if b.ipdom then
			meta[#meta + 1] = "ipdom=B" .. b.ipdom.id
		end
		if b.loopHeader and b.loopHeader ~= b then
			meta[#meta + 1] = "in_loop=B" .. b.loopHeader.id
		end
		if b.cond then
			meta[#meta + 1] = "cond=" .. b.cond.op .. (b.cond.negated and "(not)" or "")
		end

		local preds = {}
		for _, e in ipairs(b.edgesIn) do
			preds[#preds + 1] = string.format("B%d(%s%s)", e.from.id, e.kind, e.isBack and ",BACK" or "")
		end

		local succs = {}
		for _, e in ipairs(b.edgesOut) do
			succs[#succs + 1] = string.format("B%d(%s%s)", e.to.id, e.kind, e.isBack and ",BACK" or "")
		end

		local tagStr = #tags > 0 and (" [" .. table.concat(tags, ", ") .. "]") or ""
		local metaStr = #meta > 0 and (" {" .. table.concat(meta, ", ") .. "}") or ""
		out[#out + 1] = string.format("B%d (pc %04d..%04d)%s%s", b.id, b.startPc, b.endPc, tagStr, metaStr)
		out[#out + 1] = "  pred: " .. (#preds > 0 and table.concat(preds, ", ") or "(none)")
		out[#out + 1] = "  succ: " .. (#succs > 0 and table.concat(succs, ", ") or "(none)")

		for _, ins in ipairs(b.insns) do
			local lineStr = ins.line and string.format("L%-4d ", ins.line) or "      "
			local text, note = fmtInsFn(ins)
			local suffix = (note and note ~= "") and (" ; " .. note) or ""
			out[#out + 1] = string.format("    %04d  %s%-14s %s%s", ins.pc, lineStr, ins.name, text or "", suffix)
		end
		out[#out + 1] = ""
	end

	return out
end

function CFG.formatString(cfg, opts)
	return table.concat(CFG.format(cfg, opts), "\n")
end

return CFG
