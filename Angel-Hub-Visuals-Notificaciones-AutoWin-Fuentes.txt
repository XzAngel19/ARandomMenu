export type Runtime = {
    framework: any,
    entity: any,
    host: any,
    services: any,
}

local Module = {
    Name = "PlayerESP",
    PlaceId = 0,
    Events = {} :: {[string]: any},
    Initialized = false,
}

local activeCleanup: (() -> ())? = nil
local activeCard: any = nil

function Module.init(context: Runtime): any
    local host: any = context.host
    local framework: any = context.framework
    local players: Players = host.Players

    local render: any = context.render

    local container: Frame = render:Layer("PlayerEspLayer")
    local pool: {[Player]: boolean} = {}
    local lastStatusAt: number = -math.huge
    local visibilityCache: any = setmetatable({}, {__mode = "k"})

    local esp: any
    esp = framework.Categories.Visuals:CreateModule({
        Name = "PlayerESP",
        Category = "Render",
        ConfigKey = "Universal.PlayerESP",
        Order = 1,
        Tooltip = "Outlined boxes, skeletons and tracers. Nametags and HP are independent modules.",
        Function = function(enabled: boolean): ()
            container.Visible = enabled
            if not enabled then
                esp:SetStatus(nil)
                render:ReleaseAll(container)
                table.clear(pool)
                return
            end
            lastStatusAt = -math.huge
            esp:SetStatus("0")
            esp:Render(function(): ()
                Module.render(context, container, pool, esp, visibilityCache)
                local now: number = os.clock()
                if now - lastStatusAt >= 0.25 then
                    lastStatusAt = now
                    local count: number = 0
                    for _player: Player in pairs(pool) do
                        count += 1
                    end
                    esp:SetStatus(tostring(count))
                end
            end)
            esp:Event(players.PlayerRemoving, function(player: Player): ()
                render:Release(container, player)
                pool[player] = nil
            end)
            esp:Clean(function(): ()
                render:ReleaseAll(container)
                table.clear(pool)
                container.Visible = false
            end)
        end,
    })

    esp:CreateDropdown({
        Name = "Boxes",
        List = {"Corner", "Full", "Off"},
        Index = 1,
        Tooltip = "Corner draws four brackets, Full draws the whole rectangle.",
    })
    esp:CreateSlider({
        Name = "Box thickness",
        Show = {Option = "Boxes", Values = {"Corner", "Full"}},
        Min = 1,
        Max = 5,
        Default = 1,
    })
    esp:CreateToggle({
        Name = "Filled box",
        Show = {Option = "Boxes", Values = {"Corner", "Full"}},
        Default = false,
        Tooltip = "Wash the inside of the box in the target's colour. Reads "
            .. "at a glance across a map; noisy in a crowd.",
    })
    esp:CreateToggle({
        Name = "Skeleton",
        Default = false,
        Tooltip = "Draw the rig's bones. Works on both R6 and R15.",
    })
    esp:CreateDropdown({
        Name = "Tracers",
        List = {"Off", "Bottom", "Centre", "Cursor"},
        Index = 1,
        Tooltip = "Where the line to each target starts.",
    })
    esp:CreateToggle({
        Name = "Team check",
        Default = false,
        Tooltip = "Skip players on your own team.",
    })
    esp:CreateToggle({
        Name = "Visibility check",
        Default = false,
        Tooltip = "Recolour targets that are behind geometry.",
    })
    esp:CreateSlider({
        Name = "Max distance",
        Min = 50,
        Max = 5000,
        Default = 2000,
    })

    esp:CreateColor({
        Name = "Colour",
        Default = Color3.fromRGB(85, 255, 85),
        Tooltip = "Used for every target the game gives no colour of its own.",
    })
    esp:CreateToggle({
        Name = "Team colours",
        Default = true,
        Tooltip = "Draw each player in their own team's colour where the game "
            .. "has teams.",
    })
    local roles = context.services.gameRoles:Bind(esp)
    Module.roles = roles
    esp.OptionFilter = function(name)
        return not (roles.provider and roles.provider.ReplacesTeams and (name == "Team check" or name == "Team colours"))
    end
    esp:RefreshVisibility()
    local function addExtraRows(extra: any): ()
        if type(extra) ~= "table" or type(extra.Name) ~= "string" or esp.Options[extra.Name] then return end
        esp:CreateToggle({
            Name = extra.Name,
            Default = extra.Default == true,
            Tooltip = extra.Tooltip,
            Function = function(value: boolean): ()
                if type(extra.Toggle) == "function" then
                    pcall(extra.Toggle, value)
                end
            end,
        })
        if extra.Color then
            esp:CreateColor({
                Name = extra.Name .. " colour",
                Show = {Option = extra.Name},
                Default = extra.Color,
                Function = function(value: Color3): ()
                    if type(extra.SetColor) == "function" then
                        pcall(extra.SetColor, value)
                    end
                end,
            })
        end

        for _, definition: any in ipairs(extra.Options or {}) do
            local rowName: string = extra.Name .. " " .. definition.Name
            if definition.Kind == "toggle" then
                esp:CreateToggle({
                    Name = rowName,
                    Show = {Option = extra.Name},
                    Default = definition.Default == true,
                    Tooltip = definition.Tooltip,
                    Function = function(value: boolean): ()
                        if type(definition.Set) == "function" then
                            pcall(definition.Set, value)
                        end
                    end,
                })
            else
                esp:CreateSlider({
                    Name = rowName,
                    Show = {Option = extra.Name},
                    Min = definition.Min or 0,
                    Max = definition.Max or 100,
                    Default = definition.Default or definition.Min or 0,
                    Tooltip = definition.Tooltip,
                    Function = function(value: number): ()
                        if type(definition.Set) == "function" then
                            pcall(definition.Set, value)
                        end
                    end,
                })
            end
        end
    end

    local gameBridge: any = context.services.gameBridge
    if type(gameBridge.onEvent) == "function" then
        Module.bridgeCleanup = gameBridge.onEvent(function(kind: string, payload: any): ()
            if kind == "extra" then
                addExtraRows(payload)
            end
        end)
    end

    activeCleanup = function(): ()
        render:ReleaseAll(container)
        table.clear(pool)
        container.Visible = false
    end
    activeCard = esp
    Module.Initialized = true
    return esp
end

function Module.render(
    context: Runtime,
    container: Frame,
    pool: {[Player]: any},
    esp: any,
    visibilityCache: any
): ()
    local host: any = context.host
    local entityLibrary: any = context.entity
    local render: any = context.render
    local gameBridge: any = context.services.gameBridge
    local currentWorkspace: Workspace = host.workspace or workspace
    local camera: Camera? = currentWorkspace.CurrentCamera
    if not camera then
        container.Visible = false
        return
    end
    container.Visible = true
    local resolvedCamera: Camera = camera :: Camera
    local options: any = esp.Options
    local viewport: Vector2 = resolvedCamera.ViewportSize

    entityLibrary:Refresh()
    local seen: {[Player]: boolean} = {}

    for _, entity: any in ipairs(entityLibrary.List) do
        local player: Player = entity.Player
        if not (Module.roles and Module.roles.provider and Module.roles.provider.ReplacesTeams)
            and options["Team check"].Value and entity.IsFriendly then
            continue
        end
        if entity.Distance > options["Max distance"].Value then

            continue
        end

        local rect: any = render:ModelRect(resolvedCamera, entity.Character)

        seen[player] = true
        pool[player] = true
        local drawings: any = render:Set(container, player)

        if not rect then

            drawings:Show(false)
            continue
        end

        local visible: boolean = true
        if options["Visibility check"].Value then
            local cache: any = visibilityCache[player]
            local now: number = os.clock()
            if not cache or now - cache.at > 0.12 then
                cache = {at = now, value = entityLibrary:VisibleFrom(entity)}
                visibilityCache[player] = cache
            end
            visible = cache.value
        end
        local colour = options["Colour"].Value
        local roles = Module.roles
        local roleColour = roles and roles:Resolve(player, nil)
        if roleColour then
            colour = roleColour
        elseif not (roles and roles.provider and roles.provider.ReplacesTeams)
            and options["Team colours"].Value and player.Team then
            colour = player.Team.TeamColor.Color
        end

        if not visible then
            colour = colour:Lerp(Color3.new(0, 0, 0), 0.45)
        end

        local left: number = rect.left
        local top: number = rect.top
        local bottom: number = rect.bottom
        local height: number = rect.height
        local centreX: number = rect.centreX

        drawings:Show(true)

        local thickness: number = math.round(options["Box thickness"].Value)
        local boxMode: string = options["Boxes"].Value
        drawings:Fill(
            options["Filled box"].Value and boxMode ~= "Off" and rect or nil,
            colour,
            0.82
        )
        drawings:Box(rect, boxMode, thickness, colour)

        if options["Skeleton"].Value then
            local bones: {{BasePart}} = entityLibrary:Rig(entity)
            for index: number = 1, 14 do
                local pair: {BasePart}? = bones[index]
                local first: Vector2? = pair
                    and render:Project(resolvedCamera, pair[1].Position)
                local second: Vector2? = pair
                    and render:Project(resolvedCamera, pair[2].Position)
                if first and second then
                    drawings:Line(
                        "Bone",
                        index,
                        first :: Vector2,
                        second :: Vector2,
                        1,
                        colour
                    )
                else
                    drawings:HideLines("Bone", index, index)
                end
            end
        else
            drawings:HideLines("Bone", 1, 14)
        end

        local tracerMode: string = options["Tracers"].Value
        if tracerMode == "Off" then
            drawings:HideLines("Tracer", 1, 1)
        else
            local origin: Vector2
            if tracerMode == "Bottom" then
                origin = Vector2.new(viewport.X * 0.5, viewport.Y)
            elseif tracerMode == "Centre" then
                origin = viewport * 0.5
            else
                origin = host.UserInputService:GetMouseLocation()
            end
            drawings:Line(
                "Tracer",
                1,
                origin,
                Vector2.new(centreX, bottom),
                1,
                colour
            )
        end

    end

    for player: Player in pairs(pool) do
        if not seen[player] then
            render:Release(container, player)
            pool[player] = nil
        end
    end
end

function Module.destroy(): ()
    if activeCard and activeCard.Enabled then activeCard:Toggle(false) end
    if activeCleanup then
        pcall(activeCleanup)
    end
    if Module.roles then Module.roles:Destroy(); Module.roles = nil end
    if Module.bridgeCleanup then Module.bridgeCleanup(); Module.bridgeCleanup = nil end
    activeCleanup = nil
    activeCard = nil
    Module.Initialized = false
end

return Module

local Module = {Name = "Nametags", Initialized = false}
local activeCard, cleanup, roleBinding
local BLACK = Color3.new(0, 0, 0)

function Module.init(context: any): any
    local host, entity, render = context.host, context.entity, context.render
    local layer = render:Layer("NametagsLayer")
    local tracked = {}
    local store = host.configData
    -- Preserve explicit new choices; migrate only previously enabled overlays.
    if store and store.states["Universal.Nametags"] == nil then
        local states, values = store.states, store.values
        states["Universal.Nametags"] = states["Universal.Chams"] == true and states["Universal.Chams.Nametags"] == true
            or states["Universal.PlayerESP"] == true and states["Universal.PlayerESP.Nametags"] ~= false
        if values["Universal.Nametags.Textsize"] == nil then values["Universal.Nametags.Textsize"] = values["Universal.PlayerESP.Textsize"] end
        for new, old in pairs({Background="Nameplate",Displayname="Displayname",Heldtool="Heldtool",Showrole="Showrole",Rolecolours="Rolecolours"}) do
            local key = "Universal.Nametags." .. new
            if states[key] == nil then states[key] = states["Universal.PlayerESP." .. old] end
        end
        for key, colour in pairs(table.clone(values)) do
            if string.sub(key,1,20) == "Universal.PlayerESP." and string.lower(string.sub(key,-6)) == "colour" then
                local newKey = "Universal.Nametags." .. string.sub(key,21)
                if values[newKey] == nil then values[newKey] = colour end
            end
        end
    end
    local textService = game:GetService("TextService")
    local card
    local function clear()
        render:ReleaseAll(layer); table.clear(tracked); layer.Visible = false
    end
    local function fit(drawing, slot, text, size, viewport)
        local label = drawing.labels[slot]
        if not label then return end
        local key = text .. ":" .. tostring(size) .. ":" .. tostring(viewport.X)
        drawing.textMetrics = drawing.textMetrics or {}
        -- v139.1: GetTextSize + Size + TextTruncate only when the text or the
        -- size really changed. Writing Size every frame re-laid out every
        -- nametag in the scene, which is a visible hitch with many players.
        if not drawing.textMetrics[slot] or drawing.textMetrics[slot].key ~= key then
            local width = #text * size * 0.65 + 8
            pcall(function() width = textService:GetTextSize(text, size, Enum.Font.Arial, Vector2.new(10000,100)).X + 8 end)
            width = math.max(1, math.min(width, viewport.X - 24))
            drawing.textMetrics[slot] = {key = key, width = width}
            label.AutomaticSize = Enum.AutomaticSize.None
            label.Size = UDim2.fromOffset(width, size + 7)
            label.TextTruncate = Enum.TextTruncate.AtEnd
        end
        -- Clamped to screen independently of target position, not a clipped name.
        local half = drawing.textMetrics[slot].width * 0.5
        local clampedX = math.clamp(label.Position.X.Offset, half + 8, math.max(half + 8, viewport.X - half - 8))
        if math.abs(clampedX - label.Position.X.Offset) >= 0.5 then
            label.Position = UDim2.fromOffset(clampedX, label.Position.Y.Offset)
        end
    end
    card = context.framework.Categories.Visuals:CreateModule({Name = "Nametags", Category = "Render",
        ConfigKey = "Universal.Nametags", Order = 3,
        Tooltip = "Readable player nameplates, account names, distance and game roles. Independent of ESP and Chams.",
        Function = function(enabled)
            if not enabled then clear(); card:SetStatus(nil); return end
            layer.Visible = true
            card:Clean(clear)
            card:Render(function()
                local camera = (host.workspace or workspace).CurrentCamera
                if not camera then clear(); return end
                layer.Visible = true
                local seen, count = {}, 0
                local options = card.Options
                for _, target in ipairs(entity:Refresh()) do
                    local player = target.Player
                    if not player or target.Health <= 0 or target.Distance > options["Max distance"].Value
                        or not (roleBinding.provider and roleBinding.provider.ReplacesTeams) and not options.Teammates.Value and target.IsFriendly then continue end
                    local head = target.Head or target.RootPart
                    local point = head and render:Project(camera, head.Position + Vector3.new(0, head.Size.Y * 0.5 + 0.7, 0))
                    if not point or point.X < 0 or point.Y < 0 or point.X > camera.ViewportSize.X or point.Y > camera.ViewportSize.Y
                        or not options["Through walls"].Value and not entity:VisibleFrom(target) then continue end
                    local colour, role = roleBinding:Resolve(player, options.Colour.Value)
                    local shown = options["Display name"].Value and player.DisplayName or player.Name
                    local details = {}
                    if options["Account name"].Value and shown ~= player.Name then table.insert(details, "@" .. player.Name) end
                    if options["Show role"].Value and role then table.insert(details, role) end

                    if options["Held tool"].Value then
                        local tool = target.Character:FindFirstChildOfClass("Tool")
                        if tool then table.insert(details, tool.Name) end
                    end
                    seen[player] = true; tracked[player] = true; count += 1
                    local drawing = render:Set(layer, player)
                    drawing:Show(true)
                    local size = math.round(options["Text size"].Value)
                    if #details > 0 then shown ..= " [" .. table.concat(details, "] [") .. "]" end
                    if options.Distance.Value then shown = "[" .. tostring(math.round(target.Distance)) .. "] " .. shown end
                    local y = math.max(point.Y, size + 16)
                    drawing:Label("Name", shown, Vector2.new(point.X, y), 1, size, colour, options.Background.Value)
                    local label = drawing.labels.Name
                    if label then
                        if not drawing.flatNametag then
                            for _, child in ipairs(label:GetChildren()) do
                                if child:IsA("UICorner") or child:IsA("UIStroke") then child:Destroy()
                                elseif child:IsA("UIPadding") then
                                    child.PaddingLeft, child.PaddingRight = UDim.new(0,4), UDim.new(0,4)
                                end
                            end
                            drawing.flatNametag = true
                        end
                        -- v139.1: these are constant. Re-assigning Font in
                        -- particular re-shapes the glyphs of every label every
                        -- single frame.
                        if label.Font ~= Enum.Font.Arial then
                            label.Font = Enum.Font.Arial
                        end
                        if label.BackgroundColor3 ~= BLACK then
                            label.BackgroundColor3 = BLACK
                        end
                        local plate: number = options.Background.Value and 0.5 or 1
                        if label.BackgroundTransparency ~= plate then
                            label.BackgroundTransparency = plate
                        end
                        if label.TextStrokeTransparency ~= 1 then
                            label.TextStrokeTransparency = 1
                        end
                    end
                    fit(drawing, "Name", shown, size, camera.ViewportSize)
                    drawing:HideLabel("Details")
                end
                for player in pairs(tracked) do
                    if not seen[player] then render:Release(layer,player); tracked[player] = nil end
                end
                card:SetStatus(tostring(count))
            end)
        end})
    card:CreateToggle({Name = "Display name", Default = true})
    card:CreateToggle({Name = "Account name", Default = false, Show = {Option = "Display name"}})
    card:CreateToggle({Name = "Distance", Default = false})
    card:CreateToggle({Name = "Show role", Default = false})
    card:CreateToggle({Name = "Held tool", Default = false})
    card:CreateToggle({Name = "Background", Default = true})
    card:CreateSlider({Name = "Text size", Min = 10, Max = 24, Default = 14})
    card:CreateColor({Name = "Colour", Default = Color3.fromRGB(85, 255, 85)})
    card:CreateSlider({Name = "Max distance", Min = 50, Max = 5000, Default = 1000})
    card:CreateToggle({Name = "Through walls", Default = true})
    card:CreateToggle({Name = "Teammates", Default = true})
    roleBinding = context.services.gameRoles:Bind(card)
    card.OptionFilter = function(name)
        if name == "Show role" then return roleBinding.provider ~= nil end
        return not (name == "Teammates" and roleBinding.provider and roleBinding.provider.ReplacesTeams)
    end
    card:RefreshVisibility()
    activeCard, cleanup = card, clear
    Module.Initialized = true
    return card
end
function Module.destroy()
    if activeCard and activeCard.Enabled then activeCard:Toggle(false) end
    if cleanup then cleanup() end
    if roleBinding then roleBinding:Destroy() end
    activeCard, cleanup, roleBinding = nil, nil, nil
    Module.Initialized = false
end
return Module

local Module = {Name = "HP", Initialized = false}
local activeCard, cleanup, hud, roleBinding
local LOW_HEALTH = Color3.fromRGB(255,70,70)
local MID_HEALTH = Color3.fromRGB(255,220,60)
local HIGH_HEALTH = Color3.fromRGB(85,255,85)

-- Shared by world bars and Self HP; finite/max-health guards avoid NaN UI sizes.
local function healthValues(health, maximum)
    if type(health) ~= "number" or health ~= health or math.abs(health) == math.huge then health = 0 end
    if type(maximum) ~= "number" or maximum ~= maximum or maximum <= 0 or maximum == math.huge then maximum = 100 end
    health = math.max(0, health)
    return health, maximum, math.clamp(health / maximum, 0, 1)
end
local function healthColour(fraction)
    return fraction < 0.5 and LOW_HEALTH:Lerp(MID_HEALTH, fraction * 2)
        or MID_HEALTH:Lerp(HIGH_HEALTH, (fraction - 0.5) * 2)
end

function Module.init(context: any): any
    local host, render, entity = context.host, context.render, context.entity
    local create = host.create
    local layer, tracked = render:Layer("HPLayer"), {}
    local store = host.configData
    if store and store.states["Universal.HP"] == nil then
        store.states["Universal.HP"] = store.states["Universal.PlayerESP"] == true
            and (store.states["Universal.PlayerESP.Healthbar"] ~= false or store.states["Universal.PlayerESP.Healthtext"] == true)
    end
    -- Vape Health is a simple number + heart near the crosshair, not a card.
    hud = create("TextLabel", {Name="SelfHP",Parent=host.ScreenGui,
        AnchorPoint=Vector2.new(0.5,0),Position=UDim2.new(0.5,6,0.5,30),Size=UDim2.fromOffset(140,24),
        BackgroundTransparency=1,BorderSizePixel=0,Visible=false,ZIndex=20,
        Text="",Font=Enum.Font.Arial,TextSize=18,TextStrokeTransparency=1,TextColor3=healthColour(1)})
    local scale = create("UIScale", {Parent=hud,Scale=1})
    local centerPosition, bottomPosition = UDim2.new(0.5,6,0.5,30), UDim2.new(0.5,0,1,-110)
    local card
    local function clear()
        render:ReleaseAll(layer); table.clear(tracked); layer.Visible = false
        hud.Visible = false; hud.Text = ""
    end
    card = context.framework.Categories.Visuals:CreateModule({Name = "HP", Category = "Render", ConfigKey = "Universal.HP", Order = 4,
        Tooltip = "Independent player health bars and optional heart-style Self HP display.", Function = function(enabled)
            if not enabled then clear(); card:SetStatus(nil); return end
            card:Clean(clear)
            card:Render(function(dt)
                local options = card.Options
                local targets = entity:Refresh()
                local camera = (host.workspace or workspace).CurrentCamera
                layer.Visible = true
                local seen, count = {}, 0
                if camera and options.Players.Value then
                    for _, target in ipairs(targets) do
                        if not target.Player or target.Health <= 0 or target.Distance > options["Max distance"].Value
                            or not (roleBinding.provider and roleBinding.provider.ReplacesTeams) and not options.Teammates.Value and target.IsFriendly then continue end
                        local rect = render:ModelRect(camera,target.Character)
                        if not rect or not options["Through walls"].Value and not entity:VisibleFrom(target) then continue end
                        local health, maximum, fraction = healthValues(target.Health,target.MaxHealth)
                        local drawing = render:Set(layer,target.Player)
                        drawing:Show(true); drawing:Bar(Vector2.new(rect.left-2,rect.top), rect.height, fraction)
                        if options["Health text"].Value then
                            drawing:Label("HP", tostring(math.ceil(health)), Vector2.new(rect.left-14,rect.top+(1-fraction)*rect.height), 0.5, 12, healthColour(fraction), false)
                        else drawing:HideLabel("HP") end
                        tracked[target.Player] = true; seen[target.Player] = true; count += 1
                    end
                end
                for player in pairs(tracked) do if not seen[player] then render:Release(layer,player); tracked[player] = nil end end
                local showSelf = options["Self HP"].Value
                if hud.Visible ~= showSelf then hud.Visible = showSelf end
                if showSelf then
                    local character = host.LocalPlayer.Character
                    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
                    local health, maximum, fraction = healthValues(humanoid and humanoid.Health, humanoid and humanoid.MaxHealth)
                    local wanted = humanoid and humanoid.Health > 0 and tostring(math.ceil(health)) .. " ❤️" or ""
                    if hud.Text ~= wanted then hud.Text = wanted end
                    local colour = healthColour(fraction)
                    if hud.TextColor3 ~= colour then hud.TextColor3 = colour end
                    local position = options["Self HP position"].Value == "Bottom" and bottomPosition or centerPosition
                    if hud.Position ~= position then hud.Position = position end
                    local wantedScale = options["Self HP scale"].Value
                    if camera then wantedScale = math.min(wantedScale, math.max(0.5, (camera.ViewportSize.X - 24) / 140)) end
                    if scale.Scale ~= wantedScale then scale.Scale = wantedScale end
                end
                card:SetStatus(options["Self HP"].Value and tostring(count) .. " + self" or tostring(count))
            end)
        end})
    card:CreateToggle({Name = "Players", Default = true})
    card:CreateToggle({Name = "Health text", Default = false, Show = {Option = "Players"}})
    card:CreateSlider({Name = "Max distance", Min = 50, Max = 5000, Default = 2000, Show = {Option = "Players"}})
    card:CreateToggle({Name = "Teammates", Default = true, Show = {Option = "Players"}})
    card:CreateToggle({Name = "Through walls", Default = true, Show = {Option = "Players"}})
    card:CreateToggle({Name = "Self HP", Default = false})
    card:CreateDropdown({Name="Self HP position",List={"Center","Bottom"},Index=1,Show={Option="Self HP"}})
    card:CreateSlider({Name = "Self HP scale", Min = 0.8, Max = 1.6, Decimal = 1, Default = 1, Show = {Option = "Self HP"}})
    roleBinding = context.services.gameRoles:Bind(card, {palette = false})
    card.OptionFilter = function(name)
        return not (name == "Teammates" and roleBinding.provider and roleBinding.provider.ReplacesTeams)
    end
    card:RefreshVisibility()
    activeCard, cleanup = card, clear
    Module.Initialized = true
    return card
end
function Module.destroy()
    if activeCard and activeCard.Enabled then activeCard:Toggle(false) end
    if cleanup then cleanup() end
    if hud then hud:Destroy() end
    if roleBinding then roleBinding:Destroy() end
    activeCard, cleanup, hud, roleBinding = nil, nil, nil, nil
    Module.Initialized = false
end
return Module

-- MM2 Coin ESP: highlights ONLY the real pickups the game itself spawns.
--
-- A real pickup carries three exact signatures from the game's own code,
-- and this module accepts a part when ANY of them matches -- never a
-- generic name search:
--
--   1. The CollectionService tag "CoinVisual". The game's CoinVisualizer
--      script animates exactly GetTagged("CoinVisual"), skips the parts
--      whose "Collected" attribute is true and destroys the "Delete" ones.
--   2. The "CoinID" attribute. The same script keys every pickup by
--      GetAttribute("CoinID") when CoinCollected fires, so only a real
--      pickup carries it.
--   3. The exact spawn structure from the place: an invisible server part
--      named Coin_Server whose visible child is named CoinVisual, with the
--      mesh children (MainCoin / DecalPart / 2Part, or the event look of
--      the season) under it.
--
-- The highlight always sits on the visible CoinVisual part itself -- never
-- on a parent model or container. (An earlier version walked up to the
-- container model and painted the whole map yellow, and counted every
-- part with "coin" in its name; that detection is gone.)
--
-- The family of a pickup (Coin, Candy, SnowToken, BeachBall, Egg, Heart)
-- is read from its own mesh children; when the meshes say nothing
-- specific, the family the server announced applies, because a round only
-- pays one kind of pickup. Candies replace coins during Halloween with
-- this same spawn system ([1](https://murder-mystery-2.fandom.com/wiki/Halloween_Event_2025)),
-- so a candy round is found by the same signatures. The server family
-- comes from the Remotes/Gameplay/CoinsStarted bag table (Coin, Candy,
-- SnowToken, BeachBall, Egg) and the Workspace event attributes.
local Module = {
    Name = "MM2 Coin ESP",
    Events = {} :: {[string]: any},
    Initialized = false,
    Runtime = nil :: any,
}

local activeCleanup: () -> () = function(): () end

-- The exact signatures of a real pickup.
local COIN_TAG: string = "CoinVisual"
local VISUAL_NAME: string = "CoinVisual"
local SERVER_NAME: string = "Coin_Server"

local FAMILY_COIN: string = "Coin"
local FAMILY_CANDY: string = "Candy"
local FAMILY_SNOW: string = "SnowToken"
local FAMILY_BEACH: string = "BeachBall"
local FAMILY_EGG: string = "Egg"
local FAMILY_HEART: string = "Heart"
local FAMILY_ALL: string = "All"
local FAMILY_AUTO: string = "Auto (server)"

-- Most specific first: a SnowToken can carry a mesh named Coin.
local FAMILY_ORDER: {string} = {
    FAMILY_EGG,
    FAMILY_BEACH,
    FAMILY_SNOW,
    FAMILY_CANDY,
    FAMILY_HEART,
    FAMILY_COIN,
}

local COUNT_ORDER: {string} = {
    FAMILY_COIN,
    FAMILY_CANDY,
    FAMILY_SNOW,
    FAMILY_BEACH,
    FAMILY_EGG,
    FAMILY_HEART,
}

local FAMILY_NOUNS: {[string]: {string}} = {
    [FAMILY_COIN] = {"coin", "coins"},
    [FAMILY_CANDY] = {"candy", "candies"},
    [FAMILY_SNOW] = {"snowtoken", "snowtokens"},
    [FAMILY_BEACH] = {"beachball", "beachballs"},
    [FAMILY_EGG] = {"egg", "eggs"},
    [FAMILY_HEART] = {"heart", "hearts"},
}

local FAMILY_CHOICES: {string} = {
    FAMILY_AUTO,
    FAMILY_COIN,
    FAMILY_CANDY,
    FAMILY_SNOW,
    FAMILY_BEACH,
    FAMILY_EGG,
    FAMILY_HEART,
    FAMILY_ALL,

local BAG_TO_FAMILY: {[string]: string} = {
    Coin = FAMILY_COIN,
    Candy = FAMILY_CANDY,
    SnowToken = FAMILY_SNOW,
    BeachBall = FAMILY_BEACH,
    Egg = FAMILY_EGG,
}

-- Words read ONLY from the pickup's own mesh children (never from
-- parents: a container name must not reclassify every coin inside).
-- The tagged part itself is always named CoinVisual, whatever family it
-- pays, so that name must never classify a part as a plain coin either.
local FAMILY_PHRASES: {[string]: {string}} = {
    [FAMILY_EGG] = {"egg", "eggs"},
    [FAMILY_BEACH] = {"beachball", "beach ball", "beachballs", "beach balls"},
    [FAMILY_SNOW] = {"snowtoken", "snow token", "snowtokens", "snow tokens", "snowball", "snow ball"},
    [FAMILY_CANDY] = {"candy", "candies", "candied", "candycorn", "candy corn"},
    [FAMILY_HEART] = {"heart", "hearts"},
    [FAMILY_COIN] = {"maincoin", "coinold", "decalpart"},
}

-- Safety-net rescan; the signals do the real work.
local RESCAN_PERIOD: number = 2

local function lower(value: string): string
    return string.lower(value)
end

local function contains(value: string, needle: string): boolean
    return string.find(value, needle, 1, true) ~= nil
end

function Module.init(runtime: any): any
    if Module.Initialized then
        return Module
    end
    local core: any = state.mm2Core
    assert(type(core) == "table", "MM2 Coin ESP requires the MM2 core module")
    Module.Runtime = runtime
    local CoinEffects: any = core.CoinEffects
    local clearEffects: any = core.clearEffects
    local mm2Settings: any = core.mm2Settings
    local collectionService: any = game:GetService("CollectionService")

    local settings = {
        family = FAMILY_AUTO,
    }
    -- Set by the `CoinsStarted` remote and by the Workspace event attributes.
    local serverFamily: string? = nil
    -- Raw signature counts from the last scan, for the report action.
    local lastRaw = {tagged = 0, coinId = 0, named = 0}

    -- Native per-pickup marker: ONE small BoxHandleAdornment per
    -- pickup, sized exactly to the visible coin mesh. No labels and
    -- no per-frame Lua: a hundred markers cost the engine one
    -- adornment each and nothing else. The adornment renders through
    -- walls regardless of the holder's transparency.
    local coinTags: {[Instance]: {box: Instance}} = {}
    -- Pickup -> family, kept between scans for the tag text.
    local lastFamilies: {[Instance]: string} = {}
    -- Parts currently carrying a tag, for tests and the report action.
    state.mm2CoinTagParts = {}

    local function clearTags(): ()
        for part: Instance, entry in pairs(coinTags) do
            coinTags[part] = nil
            pcall(function(): ()
                entry.box:Destroy()
            end)
        end
        table.clear(lastFamilies)
        state.mm2CoinTagParts = {}
    end

    local function attribute(object: Instance, name: string): any
        return object:GetAttribute(name)
    end

    local function hasTag(object: Instance): boolean
        return collectionService:HasTag(object, COIN_TAG)
    end

    -- The visible half of a pickup. A server part resolves to its
    -- CoinVisual child; anything else is used as-is. Parents are NEVER
    -- walked: the container must stay dark.
    local function visualOf(part: Instance): Instance?
        if part.Name == SERVER_NAME then
            local child: Instance? = part:FindFirstChild(VISUAL_NAME)
            if child and child:IsA("BasePart") then
                return child
            end
            return nil
        end
        return part
    end

    -- The volatile Collected/Delete flags are checked by passesFiltersCached;
    -- this static half rejects anything outside the map or inside a character.
    local function passesFilters(part: Instance): boolean
        if not part:IsA("BasePart") or not part:IsDescendantOf(workspace) then
            return false
        end
        local current: Instance? = part.Parent
        while current and current ~= workspace do
            if current:FindFirstChildOfClass("Humanoid") then
                return false
            end
            current = current.Parent
        end
        return true
    end

    -- The family the server says is in play: the `CoinsStarted` table first,
    -- then the Workspace event attributes.
    local function familyFromWorkspace(): string?
        local eventName: string = lower(tostring(attribute(workspace, "CurrentEvent") or ""))
        if contains(eventName, "halloween") then
            return FAMILY_CANDY
        elseif contains(eventName, "christmas") then
            return FAMILY_SNOW
        elseif contains(eventName, "summer") then
            return FAMILY_BEACH
        elseif contains(eventName, "easter") then
            return FAMILY_EGG
        elseif contains(eventName, "valentine") then
            return FAMILY_HEART
        end
        for attrName: string, family: string in pairs({
            isValentines = FAMILY_HEART,
            isChristmas = FAMILY_SNOW,
            isChristmasEnding = FAMILY_SNOW,
            IsSummer = FAMILY_BEACH,
            IsEaster = FAMILY_EGG,
        }) do
            if attribute(workspace, attrName) == true then
                return family
            end
        end
        return nil
    end

    local function activeFamily(): string?
        return serverFamily or familyFromWorkspace()
    end

    -- Family of one pickup: its mesh children only. With no specific
    -- word, the round's family applies (a round pays a single kind of
    -- pickup), falling back to the classic coin.
    local familyCache: {[Instance]: string} = {}
    local function familyOf(part: Instance): string
        local cached: string? = familyCache[part]
        if cached then
            return cached
        end
        local names: {string} = {}
        for _, object: Instance in ipairs(part:GetDescendants()) do
            table.insert(names, object.Name)
        end
        local haystack: string = lower(table.concat(names, " "))
        -- A round pays a single kind of pickup, and event pickups are
        -- built on the classic coin template (MainCoin/DecalPart meshes
        -- with an event look). So when the server announces an event
        -- family, plain coin mesh names must NOT drag the pickup back
        -- to Coin -- that mismatch is what left Auto showing nothing
        -- while the map was full of candies.
        local active: string? = activeFamily()
        for _, family: string in ipairs(FAMILY_ORDER) do
            if family ~= FAMILY_COIN or active == nil or active == FAMILY_COIN then
                for _, phrase: string in ipairs(FAMILY_PHRASES[family]) do
                    if contains(haystack, phrase) then
                        -- Cache only the non-Coin mesh families:
                        -- they win over the server family in every
                        -- scan, so they can never go stale. A Coin
                        -- mesh classification is NOT cached -- when
                        -- the server announces an event family, the
                        -- coin template must reclassify to it.
                        if family ~= FAMILY_COIN then
                            familyCache[part] = family
                        end
                        return family
                    end
                end
            end
        end
        return active or FAMILY_COIN
    end

    -- Every real pickup on the map, merged from the three signatures.
    -- (v84 tried a "cheap" rescan -- tag list plus previously-known
    -- parts -- and in the live game the ESP stopped reporting coins
    -- entirely; the mock suite passed, the game did not. The full
    -- scan below is the behaviour that demonstrably works in game.
    -- The cost is kept down elsewhere: Auto Farm's polls are cached
    -- (1.5 s), the family classification is cached, and the ancestor
    -- walk inside passesFilters is cached per part.)
    local filterCache: {[Instance]: boolean} = {}
    local function collectAll(): ({[Instance]: string}, {[string]: number})
        local found: {[Instance]: string} = {}
        local counts: {[string]: number} = {}
        lastRaw = {tagged = 0, coinId = 0, named = 0}

        -- The attribute half of the filter is volatile (Collected
        -- flips when a coin is taken) and is always re-read; only
        -- the static half (in the workspace, not inside a character)
        -- is cached -- a pickup's ancestry never changes.
        local function passesFiltersCached(part: Instance): boolean
            if attribute(part, "Collected") == true
                or attribute(part, "Delete") == true then
                return false
            end
            local cached: boolean? = filterCache[part]
            if cached ~= nil then
                return cached
            end
            local result: boolean = passesFilters(part)
            filterCache[part] = result
            return result
        end

        local function consider(part: Instance?): ()
            if not part then
                return
            end
            local visual: Instance? = visualOf(part)
            if not visual or found[visual] then
                return
            end
            if not passesFiltersCached(visual) then
                return
            end
            local family: string = familyOf(visual)
            found[visual] = family
            counts[family] = (counts[family] or 0) + 1
        end

        -- Signature 1: the game's own tag list.
        local tagged: {Instance} = collectionService:GetTagged(COIN_TAG)
        lastRaw.tagged = #tagged
        for _, object: Instance in ipairs(tagged) do
            if object:IsA("BasePart") then consider(object) end
        end

        -- Signatures 2 and 3: CoinID and exact spawn names from one scan.
        for _, object: Instance in ipairs(workspace:GetDescendants()) do
            if object:IsA("BasePart") then
                if attribute(object, "CoinID") ~= nil then
                    lastRaw.coinId += 1
                    consider(object)
                elseif object.Name == VISUAL_NAME or object.Name == SERVER_NAME then
                    lastRaw.named += 1
                    consider(object)
                end
            end
        end
        return found, counts
    end

    local function wanted(family: string): boolean
        local choice: string = settings.family
        if choice == FAMILY_ALL then
            return true
        end
        if choice == FAMILY_AUTO then
            local active: string? = activeFamily()
            return active == nil or family == active
        end
        return family == choice
    end

    local function describe(counts: {[string]: number}): string
        local parts: {string} = {}
        for _, family: string in ipairs(COUNT_ORDER) do
            local amount: number = counts[family] or 0
            if amount > 0 then
                local nouns: {string} = FAMILY_NOUNS[family]
                table.insert(parts, tostring(amount) .. " " .. (amount == 1 and nouns[1] or nouns[2]))
            end
        end
        if #parts == 0 then
            return "no coins yet"
        end
        return table.concat(parts, " · ")
    end

    -- Shared with Auto Farm: the pickup parts of the family in use.
    -- CACHED: this used to run a full collectAll() per call, and Auto
    -- Farm polls every 0.4 s -- each poll re-scanned every descendant
    -- of Workspace and hitch-spiked the frame. The cache refreshes on
    -- the ESP's own scans, or at most every 0.5 s when polled.
    local pickupCache: {at: number, list: {Instance}} = {at = -math.huge, list = {}}
    local function refreshPickupCache(found: {[Instance]: string}): ()
        local list: {Instance} = {}
        for part: Instance, family: string in pairs(found) do
            if wanted(family) then
                table.insert(list, part)
            end
        end
        pickupCache = {at = os.clock(), list = list}
    end
    state.mm2CoinPickups = function(): {Instance}
        if os.clock() - pickupCache.at < 0.5 then
            return pickupCache.list
        end
        local found: {[Instance]: string} = collectAll()
        refreshPickupCache(found)
        return pickupCache.list
    end

    local function toggleCoinEsp(enabled: boolean): ()
        disconnectFeatureConnection("MM2CoinRefresh")
        disconnectFeatureConnection("MM2CoinCollected")
        disconnectFeatureConnection("MM2CoinsStarted")
        disconnectFeatureConnection("MM2CoinTagAdded")
        disconnectFeatureConnection("MM2CoinTagRemoved")
        disconnectFeatureConnection("MM2CoinAdded")
        clearEffects(CoinEffects)
        clearTags()

        if not enabled then
            local card: any = state.mm2CoinFeature
            if card then
                card:SetStatus(nil)
            end
            return
        end

        local elapsed: number = 0
        local lastStatus: string? = nil
        local dirty: boolean = true

        local function setStatus(text: string): ()

            if text == lastStatus then
                return
            end
            lastStatus = text
            local card: any = state.mm2CoinFeature
            if card then
                card:SetStatus(text)
            end
        end

        -- The visible mesh of a pickup: MainCoin under the CoinVisual
        -- holder (the Candy / coin itself), falling back to the pickup
        -- part. The box takes THIS part's exact size, so the marker is
        -- as small as the candy on screen -- never the big invisible
        -- Coin_Server touch volume.
        local function visualOf(part: Instance): BasePart
            local main: Instance? = part:FindFirstChild("MainCoin", true)
            if main and main:IsA("BasePart") then return main :: BasePart end
            local visual: Instance? = part:FindFirstChild("CoinVisual", true)
            if visual and visual:IsA("BasePart") then return visual :: BasePart end
            return part :: BasePart
        end

        -- One small adornment per visible pickup. Creation happens
        -- once per pickup; between rescans Lua does nothing at all.
        local tagsDirty: boolean = true
        -- Opacity guard: the setting's old default (0.8, from the
        -- full-model Highlight era) made a candy-sized adornment
        -- effectively invisible -- v82-v85 "detected nothing" while
        -- the farm, using the same list, walked to the coins fine.
        -- Cap the marker at 0.6 transparency whatever is saved.
        local function markerTransparency(): number
            return math.clamp(mm2Settings.coinTransparency, 0, 0.6)
        end

        -- Creates the box for one pickup. Called only when the pickups change.
        local function makeTag(part: Instance): any
            local visual: BasePart = visualOf(part)
            local box: BoxHandleAdornment = Instance.new("BoxHandleAdornment")
            box.Name = "WurstCoinBox"
            box.AlwaysOnTop = true
            box.ZIndex = 5
            box.Adornee = visual
            box.Color3 = mm2Settings.coinColor
            box.Transparency = markerTransparency()
            box.Size = visual.Size * 1.3
            box.Parent = part
            return {box = box}
        end

        -- v139: reconciliation runs when the pickup set changes, not every
        -- frame. drawOverlay used to allocate a `seen` table and walk every coin
        -- 60 times a second even when nothing had moved, which showed up as GC
        -- spikes with ~100 coins on the map. A BoxHandleAdornment follows its
        -- Adornee by itself, so the per-frame work was pure waste.
        local function syncTags(): ()
            for part: Instance in pairs(lastFamilies) do
                if part.Parent and not coinTags[part] then
                    coinTags[part] = makeTag(part)
                    tagsDirty = true
                end
            end
            for part: Instance, entry: any in pairs(coinTags) do
                if not part.Parent or not lastFamilies[part] then
                    coinTags[part] = nil
                    pcall(function(): ()
                        entry.box:Destroy()
                    end)
                    tagsDirty = true
                end
            end
            if tagsDirty then
                tagsDirty = false
                local published: {[Instance]: boolean} = {}
                for part: Instance in pairs(coinTags) do
                    published[part] = true
                end
                state.mm2CoinTagParts = published
            end
        end

        -- Per-frame work is now only the colour refresh, twice a second.
        local lastTagColourAt: number = -math.huge
        local function drawOverlay(): ()
            local now: number = os.clock()
            if now - lastTagColourAt < 0.5 then
                return
            end
            lastTagColourAt = now
            for part: Instance, entry: any in pairs(coinTags) do
                if not part.Parent then
                    coinTags[part] = nil
                    entry.box:Destroy()
                    tagsDirty = true
                else
                    pcall(function(): ()
                        entry.box.Color3 = mm2Settings.coinColor
                        entry.box.Transparency = markerTransparency()
                    end)
                end
            end
            if tagsDirty then
                tagsDirty = false
                local published: {[Instance]: boolean} = {}
                for part: Instance in pairs(coinTags) do
                    published[part] = true
                end
                state.mm2CoinTagParts = published
            end
        end

        -- One pickup, cham(s) on its visible geometry only.
        local function refresh(): ()
            local found: {[Instance]: string}, counts: {[string]: number} = collectAll()
            local visible: {[Instance]: boolean} = {}
            table.clear(lastFamilies)
            for object: Instance, family: string in pairs(found) do
                if wanted(family) then
                    visible[object] = true
                    lastFamilies[object] = family
                end
            end
            -- Auto follows the server family, but a mismatch between
            -- the classification and the announcement must not blank
            -- the ESP: fall back to showing every pickup found.
            if settings.family == FAMILY_AUTO and activeFamily() ~= nil then
                local anyWanted: boolean = false
                for object: Instance in pairs(visible) do
                    anyWanted = true
                    break
                end
                if not anyWanted then
                    for object: Instance, family: string in pairs(found) do
                        visible[object] = true
                        lastFamilies[object] = family
                    end
                end
            end
            refreshPickupCache(found)
            syncTags()
            local shown: number = 0
            for _ in pairs(visible) do
                shown += 1
            end
            local foundTotal: number = 0
            for _ in pairs(found) do
                foundTotal += 1
            end
            if foundTotal == 0 then
                setStatus(string.format(
                    "no coins yet (tag %d · id %d · named %d)",
                    lastRaw.tagged, lastRaw.coinId, lastRaw.named))
            else
                setStatus(describe(counts) .. " · " .. shown .. " shown")
            end
            dirty = false
        end

        -- A pickup appears: watch the tag signal (the game's own) and new
        -- workspace parts carrying one of the exact signatures.
        featureConnections.MM2CoinTagAdded =
            collectionService:GetInstanceAddedSignal(COIN_TAG):Connect(function(): ()
                dirty = true
            end)
        featureConnections.MM2CoinTagRemoved =
            collectionService:GetInstanceRemovedSignal(COIN_TAG):Connect(function(): ()
                dirty = true
            end)
        featureConnections.MM2CoinAdded = workspace.DescendantAdded:Connect(function(object: Instance): ()
            if object:IsA("BasePart")
                and (object.Name == VISUAL_NAME or object.Name == SERVER_NAME
                    or attribute(object, "CoinID") ~= nil or hasTag(object))
            then
                dirty = true
            end
        end)

        local started: Instance? = core.mm2GameplayRemotes
            and core.mm2GameplayRemotes:FindFirstChild("CoinsStarted")
        if started and started:IsA("RemoteEvent") then
            featureConnections.MM2CoinsStarted =
                (started :: any).OnClientEvent:Connect(function(bagTable: any): ()
                    if type(bagTable) == "table" then
                        for bag: any in pairs(bagTable) do
                            local family: string? = BAG_TO_FAMILY[tostring(bag)]
                            if family then
                                serverFamily = family
                            end
                        end
                    end
                    dirty = true
                end)
        end

        local remotes: any = core.mm2GameplayRemotes
        local collected: Instance? = remotes and remotes:FindFirstChild("CoinCollected")
        if collected and collected:IsA("RemoteEvent") then
            featureConnections.MM2CoinCollected =
                (collected :: any).OnClientEvent:Connect(function(): ()
                    dirty = true
                end)
        end

        refresh()
        featureConnections.MM2CoinRefresh = TaskManager:Connect(
            function(deltaTime: number): ()
                -- v139: the only per-frame work left is the colour refresh
                -- (twice a second). BoxHandleAdornments follow their Adornee
                -- inside the engine; the reconciliation lives in refresh().
                drawOverlay()
                elapsed += deltaTime
                if not dirty and elapsed < RESCAN_PERIOD then
                    return
                end
                elapsed = 0
                refresh()
            end
        , nil, "Coin ESP")
    end

    local CoinFeature: any = createUniversalFeature(
        "Coin ESP",
        "Draws one small box, exactly the size of the coin, candy, snow token, beach ball or egg mesh, on every real pickup and reports how many of each family exist.",
        3,
        toggleCoinEsp,
        {
            categoryName = "Render",
            parent = MM2Scroll,
            registry = mm2Features,
        }
    )
    state.mm2CoinFeature = CoinFeature

    addCycleOption(
        CoinFeature,
        "Coin in use",
        FAMILY_CHOICES,
        1,
        function(value: string): ()
            settings.family = value
            if CoinFeature.enabled then
                toggleCoinEsp(true)
            end
        end,
        "Auto follows the coin family the server announces through CoinsStarted."
    )
    addColorOption(
        CoinFeature,
        "Coin colour",
        mm2Settings.coinColor,
        function(value: Color3): ()
            mm2Settings.coinColor = value
            if CoinFeature.enabled then
                toggleCoinEsp(true)
            end
        end
    )
    addNumberOption(
        CoinFeature,
        "Coin transparency",
        mm2Settings.coinTransparency,
        0,
        1,
        function(value: number): ()
            mm2Settings.coinTransparency = value
            if CoinFeature.enabled then
                toggleCoinEsp(true)
            end
        end
    )
    addInformationOption(
        CoinFeature,
        "Pickups are recognised by the game's own signatures only: the CoinVisual tag, the CoinID attribute and the exact server/visual part pair. Each pickup gets a single BoxHandleAdornment sized exactly to its visible mesh (no labels), so a hundred coins cost no per-frame Lua. Auto follows the family the server announces on CoinsStarted."
    )

    activeCleanup = function(): ()
        state.mm2CoinFeature = nil
        toggleCoinEsp(false)
    end
    Module.Events = featureConnections
    Module.Initialized = true
    return Module
end

function Module.destroy(): ()
    if not Module.Initialized then
        return
    end
    Module.Initialized = false
    pcall(activeCleanup)
    activeCleanup = function(): ()
    end
    Module.Events = {}
    Module.Runtime = nil
end

return Module

function Module.init(runtime: any): any
    if Module.Initialized then return Module end
    local core: any = state.mm2Core
    assert(type(core) == "table", "MM2 Notifications requires the MM2 core")
    Module.Runtime = runtime

    -- Instant Roles used to be an independent Blurt Roles checkbox. Preserve
    -- both its value and its old independent-on behavior, without overriding a
    -- newer Notifications master setting on subsequent boots.
    if type(configData) == "table" and type(configData.states) == "table" then
        local states: any = configData.states
        local oldKey: string = "MM2.BlurtRoles.InstantRoles"
        local newKey: string = "MM2.Notifications.InstantRoles"
        local old: any = states[oldKey]
        local changed: boolean = false
        if states[newKey] == nil and type(old) == "boolean" then
            states[newKey] = old
            if old and states["MM2.Notifications"] == nil then
                states["MM2.Notifications"] = true
            end
            changed = true
        end
        if old ~= nil then states[oldKey] = nil; changed = true end
        if changed and type(queueConfigSave) == "function" then queueConfigSave() end
    end

    local settings: any = {
        instantRoles = false,
        gun = true,
        murderClose = true,
        yourRole = true,
        roundEvents = true,
        murderDefeated = true,
    }
    local enabled: boolean = false
    local phase: string = type(core.getRoundPhase) == "function" and core.getRoundPhase() or "unknown"
    local elapsed: number = 0
    local murdererSeen: boolean = false
    local sheriffSeen: boolean = false
    local lastLocalRole: string? = nil
    local lastMurderer: Player? = nil
    local defeatNotified: boolean = false
    local nearMurderer: boolean = false
    local lastCloseAt: number = -math.huge
    local gunWasDropped: boolean = false
    local gunLastSeenAt: number = -math.huge
    local gunDropPreviousHolder: Player? = nil
    local lastGunHolder: Player? = nil
    local gunHolderInitialized: boolean = false

    local function resetRound(): ()
        murdererSeen = false
        sheriffSeen = false
        lastLocalRole = nil
        lastMurderer = nil
        defeatNotified = false
        nearMurderer = false
        lastCloseAt = -math.huge
        gunWasDropped = false
        gunLastSeenAt = -math.huge
        gunDropPreviousHolder = nil
        lastGunHolder = nil
        gunHolderInitialized = false
    end

    local function inRound(): boolean
        if type(core.isRoundLifecycleActive) == "function"
            and core.isRoundLifecycleActive() == false then return false end
        if phase == "lobby" then return false end
        return phase == "map" or (type(core.hasActiveRoundRoles) == "function" and core.hasActiveRoundRoles())
    end

    local function rootOf(player: Player?): BasePart?
        local character: Model? = player and player.Character
        return character and character:FindFirstChild("HumanoidRootPart") :: BasePart? or nil
    end

    local function isAlive(player: Player?): boolean
        if not player then return false end
        if type(core.isPlayerAlive) == "function" then return core.isPlayerAlive(player) == true end
        return rootOf(player) ~= nil
    end

    local function showRoles(murderer: Player?, sheriff: Player?): ()
        if not enabled or not settings.instantRoles or not inRound() then return end
        if not murdererSeen and not sheriffSeen and murderer and sheriff then
            murdererSeen = true
            sheriffSeen = true
            notify("Roles ready - Murderer: " .. murderer.Name .. " | Sheriff/Hero: " .. sheriff.Name)
            return
        end
        if murderer and not murdererSeen then
            murdererSeen = true
            notify("Murderer: " .. murderer.Name)
        end
        if sheriff and not sheriffSeen then
            sheriffSeen = true
            notify("Sheriff/Hero: " .. sheriff.Name)
        end
    end

    local function gunDistance(drop: Instance?): number?
        local root: BasePart? = rootOf(LocalPlayer)
        if not root or not drop then return nil end
        local ok: boolean, distance: any = pcall(function(): number
            local position: Vector3 = if drop:IsA("BasePart")
                then (drop :: BasePart).Position
                else (drop :: Model):GetPivot().Position
            return (position - root.Position).Magnitude
        end)
        return if ok and type(distance) == "number" then distance else nil
    end

    local function updateGun(sheriff: Player?): ()
        if not settings.gun or type(core.findDroppedGun) ~= "function" then return end
        local now: number = os.clock()
        local drop: Instance? = core.findDroppedGun()
        local holder: Player? = nil
        if sheriff and isAlive(sheriff) and type(core.getPlayerWeapon) == "function"
            and core.getPlayerWeapon(sheriff, "Gun") ~= nil then holder = sheriff end
        if drop then
            if not gunWasDropped then
                -- The previous owner's Tool may still replicate briefly after
                -- the drop appears. It is not evidence they picked it back up.
                gunDropPreviousHolder = lastGunHolder or holder
                local distance: number? = gunDistance(drop)
                notify("Gun dropped" .. (distance and (" · " .. tostring(math.round(distance)) .. " studs away") or ""))
            end
            gunWasDropped = true
            gunLastSeenAt = now
        elseif gunWasDropped and holder and holder ~= gunDropPreviousHolder then
            notify("Gun picked up by " .. holder.Name)
            gunWasDropped = false
            gunDropPreviousHolder = nil
        elseif gunWasDropped and now - gunLastSeenAt >= 2 then
            -- Missing/streamed-out is not proof of a pickup.
            gunWasDropped = false
            gunDropPreviousHolder = nil
        elseif holder and gunHolderInitialized and lastGunHolder and holder ~= lastGunHolder then
            notify("Gun is now held by " .. holder.Name)
        end
        if holder then
            lastGunHolder = holder
            gunHolderInitialized = true
        end
    end

    local function confirmedDead(player: Player?): boolean
        if not player then return false end
        -- getPlayerRole can infer Dead when a character streams out. Only
        -- explicit round data or a visible zero-health humanoid is evidence.
        if type(core.getRoundRole) == "function" and core.getRoundRole(player) == "Dead" then return true end
        local character: Model? = player.Character
        local humanoid: Humanoid? = character and character:FindFirstChildOfClass("Humanoid") :: Humanoid?
        return humanoid ~= nil and humanoid.Health <= 0
    end

    local function updateProximity(murderer: Player?): ()
        if not settings.murderClose or not murderer or murderer == LocalPlayer
            or not isAlive(LocalPlayer) or not isAlive(murderer) then
            nearMurderer = false
            return
        end
        local localRoot: BasePart? = rootOf(LocalPlayer)
        local murderRoot: BasePart? = rootOf(murderer)
        if not localRoot or not murderRoot then nearMurderer = false; return end
        local distance: number = (murderRoot.Position - localRoot.Position).Magnitude
        local now: number = os.clock()
        if distance <= 30 and not nearMurderer and now - lastCloseAt >= 8 then
            nearMurderer = true
            lastCloseAt = now
            notify("Murderer close: " .. murderer.Name .. " · " .. tostring(math.round(distance)) .. " studs")
        elseif distance > 38 then
            nearMurderer = false
        end
    end

    local function sample(): ()
        if not enabled or not inRound() then return end
        local murderer: Player? = type(core.findMurderer) == "function" and core.findMurderer() or nil
        local sheriff: Player? = type(core.findSheriff) == "function" and core.findSheriff() or nil
        if murderer then lastMurderer = murderer end
        showRoles(murderer, sheriff)
        if settings.yourRole and type(core.getPlayerRole) == "function" then
            local myRole: any = core.getPlayerRole(LocalPlayer)
            if (myRole == "Innocent" or myRole == "Sheriff" or myRole == "Hero" or myRole == "Murderer")
                and myRole ~= lastLocalRole then
                lastLocalRole = myRole
                notify("Your role: " .. myRole)
            end
        end
        updateGun(sheriff)
        updateProximity(murderer)
        if settings.murderDefeated and not defeatNotified and confirmedDead(lastMurderer) then
            defeatNotified = true
            notify("Murderer defeated: " .. (lastMurderer :: Player).Name)
        end
    end

    local nextWarningAt: number = -math.huge
    local sampling: boolean = false
    local sampleAgain: boolean = false
    local function trySample(): ()
        -- A role lookup can rebuild the core index and synchronously emit
        -- onRoundRoles. Coalesce that callback instead of recursing into a
        -- partially updated sample; the regular tick handles further changes.
        if sampling then sampleAgain = true; return end
        sampling = true
        for _pass: number = 1, 2 do
            sampleAgain = false
            local ok: boolean, err: any = pcall(sample)
            if not ok and os.clock() >= nextWarningAt then
                nextWarningAt = os.clock() + 5
                warn("MM2 Notifications: " .. tostring(err))
            end
            if not sampleAgain or not enabled then break end
        end
        sampling = false
    end

    local unsubscribeRoles: () -> () = core.onRoundRoles(function(active: boolean): ()
        if active and enabled then trySample() end
    end)
    local unsubscribePhase: () -> () = core.onRoundPhase(function(nextPhase: string, reason: string): ()
        local previous: string = phase
        phase = nextPhase
        if nextPhase == "lobby" and previous ~= "lobby" then
            if enabled and settings.roundEvents and previous == "map" and reason ~= "subscribe" then
                notify("Round ended")
            end
            resetRound()
        elseif nextPhase == "map" and previous ~= "map" then
            resetRound()
            if enabled and settings.roundEvents and reason ~= "subscribe" then
                notify("Entered the map - round starting")
            end
            if enabled then trySample() end
        end
    end)

    local function toggleNotifications(value: boolean): ()
        enabled = value == true
        disconnectFeatureConnection("MM2NotificationsTick")
        elapsed = 0
        if not enabled then return end
        trySample()
        featureConnections.MM2NotificationsTick = TaskManager:Connect(function(deltaTime: number): ()
            elapsed += math.max(0, deltaTime)
            if elapsed < 0.25 then return end
            elapsed = 0
            trySample()
        end, nil, "MM2 Notifications")
    end

    local feature: any = createUniversalFeature(
        "Notifications", "Private MM2 alerts for roles, the gun, nearby danger and rounds.",
        4, toggleNotifications,
        {categoryName = "Utilities", parent = MM2Scroll, registry = mm2Features}
    )
    addToggleOption(feature, "Instant Roles", false, function(value: boolean): ()
        settings.instantRoles = value
        core.mm2Settings.instantRoleNotify = value
        if enabled and value then trySample() end
    end, "Private role alerts as soon as each role is known; never sends public chat.")
    addToggleOption(feature, "Gun Notification", true, function(value: boolean): ()
        settings.gun = value
        if not value then
            gunWasDropped = false
            gunDropPreviousHolder = nil
            lastGunHolder = nil
            gunHolderInitialized = false
        elseif enabled then
            trySample()
        end
    end, "Announces a dropped gun and a confirmed new holder.")
    addToggleOption(feature, "Murder Close", true, function(value: boolean): ()
        settings.murderClose = value
        if enabled and value then trySample() end
    end, "Warns when the living Murderer is within 30 studs; includes a cooldown.")
    addToggleOption(feature, "Your Role", true, function(value: boolean): ()
        settings.yourRole = value
        if enabled and value then trySample() end
    end, "Notifies you when your role becomes known or changes.")
    addToggleOption(feature, "Round Events", true, function(value: boolean): ()
        settings.roundEvents = value
    end, "Private alerts for entering the map and the round ending.")
    addToggleOption(feature, "Murderer Defeated", true, function(value: boolean): ()
        settings.murderDefeated = value
        if enabled and value then trySample() end
    end, "Warns only when the known Murderer is confirmed dead.")
    addInformationOption(feature, "These alerts are private. Blurt Roles is the separate public-chat feature.")

    activeCleanup = function(): ()
        toggleNotifications(false)
        pcall(unsubscribeRoles)
        pcall(unsubscribePhase)
        core.mm2Settings.instantRoleNotify = false
    end
    Module.Events = featureConnections
    Module.Initialized = true
    return Module
end

function Module.destroy(): ()
    if not Module.Initialized then return end
    Module.Initialized = false
    activeCleanup()
    activeCleanup = function(): () end
    Module.Events = {}
    Module.Runtime = nil
end

return Module

===== AUTO WIN MM2 | src/games/MM2/Other/AutoWin.lua =====
local Module = {
    Name = "MM2 Auto Win",
    Events = {} :: {[string]: any},
    Initialized = false,
    Runtime = nil :: any,
}

local activeCleanup: () -> () = function(): () end

function Module.init(runtime: any): any
    if Module.Initialized then
        return Module
    end
    local core: any = state.mm2Core
    assert(type(core) == "table", "MM2 Auto Win requires the MM2 core module")
    local prediction: any = state.mm2Prediction
    Module.Runtime = runtime

    local findDroppedGun: any = core.findDroppedGun
    local findMM2Map: any = core.findMM2Map
    local findMurderer: any = core.findMurderer
    local findSheriff: any = core.findSheriff
    local getGunOriginCFrame: any = core.getGunOriginCFrame
    local getPlayerRole: any = core.getPlayerRole
    local getPlayerWeapon: any = core.getPlayerWeapon
    local isPlayerAlive: any = core.isPlayerAlive
    local isProtectedTarget: any = core.isProtectedTarget
    local isRoundLifecycleActive: any = core.isRoundLifecycleActive

    local ABOVE_MAP_OFFSET: number = 120
    local MAP_FLING_WARMUP_SECONDS: number = 5.0
    local SHERIFF_READY_DELAY_SECONDS: number = 2.0
    local SHERIFF_FLING_COOLDOWN_SECONDS: number = 1.2

    local settings = {
        instaKillMurder = true,
        instaKillSheriff = true,
        flingSheriffGetGun = false,
        flingMurderIfDead = false,
    }

    local autoWinActive: boolean = false
    local AutoWinFeature: any = nil
    local lastStatus: string? = nil

    local sheriffWins: number = 0
    local murderWins: number = 0
    local innocentWins: number = 0
    local sheriffWinsStat: any = nil
    local murderWinsStat: any = nil
    local innocentWinsStat: any = nil

    local wasOnMap: boolean = false
    local mapEnteredAt: number = 0
    local sheriffFirstSeenAt: number = 0
    local lastSheriffFlingAt: number = 0
    local participatedInRound: boolean = false
    local roundMyRole: string? = nil
    local roundSawMurderer: boolean = false
    local roundCountedWin: boolean = false
    local flungMurderThisDeath: boolean = false
    local flingBusy: boolean = false

    local safeAbovePosition: Vector3? = nil
    local savedCollide: {[Instance]: boolean} = {}
    local savedCustomProps: {[BasePart]: PhysicalProperties?} = {}
    local victimSnapshots: {[BasePart]: {cframe: CFrame, anchored: boolean, canCollide: boolean}} = {}
    local toolConnections: {RBXScriptConnection} = {}
    local flingAngle: number = 0

    local function updateWinOverlay(): ()
        if sheriffWinsStat and type(sheriffWinsStat.set) == "function" then
            sheriffWinsStat.set(tostring(sheriffWins))
        end
        if murderWinsStat and type(murderWinsStat.set) == "function" then
            murderWinsStat.set(tostring(murderWins))
        end
        if innocentWinsStat and type(innocentWinsStat.set) == "function" then
            innocentWinsStat.set(tostring(innocentWins))
        end
    end

    local function awardRoundWin(roleCategory: string?): ()
        if roundCountedWin then
            return
        end
        local cat: string = roleCategory or roundMyRole or "Innocent"
        roundCountedWin = true
        if cat == "Murderer" then
            murderWins += 1
        elseif cat == "Sheriff" or cat == "Hero" then
            sheriffWins += 1
        else
            innocentWins += 1
        end
        updateWinOverlay()
    end

    local function setStatus(text: string?): ()
        if text == lastStatus then
            return
        end
        lastStatus = text
        if AutoWinFeature then
            AutoWinFeature:SetStatus(text)
        end
    end

    local function yawOf(frame: CFrame): number
        local look: Vector3 = frame.LookVector
        return math.atan2(-look.X, -look.Z)
    end

    local function characterParts(): (Model?, Humanoid?, BasePart?)
        local character: Model? = LocalPlayer and LocalPlayer.Character
        if not character then
            return nil, nil, nil
        end
        local humanoid: Humanoid? = character:FindFirstChildOfClass("Humanoid")
        local root: BasePart? = character:FindFirstChild("HumanoidRootPart") :: BasePart?
        if not humanoid or not root then
            return nil, nil, nil
        end
        return character, humanoid, root
    end

    local function findLobbyReferencePart(): BasePart?
        for _, child: Instance in ipairs(workspace:GetChildren()) do
            local lowerName: string = string.lower(tostring(child.Name))
            local isLobby: boolean = string.find(lowerName, "lobby", 1, true) ~= nil
                or child:FindFirstChild("VotePads", true) ~= nil
                or child:FindFirstChild("VotePad1", true) ~= nil
                or child:FindFirstChild("VoteIcons", true) ~= nil
            if isLobby then
                local votePads: Instance? = child:FindFirstChild("VotePads", true)
                if votePads then
                    local padPart: BasePart? = votePads:FindFirstChildWhichIsA("BasePart", true)
                    if padPart then
                        return padPart
                    end
                end
                local spawns: Instance? = child:FindFirstChild("Spawns", true)
                if spawns then
                    local spawnPart: BasePart? = spawns:FindFirstChildWhichIsA("BasePart", true)
                    if spawnPart then
                        return spawnPart
                    end
                end
                local anyPart: BasePart? = child:FindFirstChildWhichIsA("BasePart", true)
                if anyPart then
                    return anyPart
                end
            end
        end
        return nil
    end

    -- True as soon as the player's character is on the MM2 map, even during
    -- the pre-round countdown before roles or weapons have been handed out.
    local function isCharacterOnMap(root: BasePart): (boolean, Vector3?)
        if type(findMM2Map) ~= "function" then
            return false, nil
        end
        local map: Instance? = findMM2Map()
        if not map then
            return false, nil
        end
        local rootPos: Vector3 = root.Position
        local lobbyPart: BasePart? = findLobbyReferencePart()
        if lobbyPart and (rootPos - lobbyPart.Position).Magnitude <= 350 then
            return false, nil
        end
        local refPart: BasePart? = nil
        local spawns: Instance? = (map :: Instance):FindFirstChild("Spawns")
        if spawns then
            refPart = spawns:FindFirstChildWhichIsA("BasePart", true)
        end
        if not refPart then
            local coinContainer: Instance? = (map :: Instance):FindFirstChild("CoinContainer")
            if coinContainer then
                refPart = coinContainer:FindFirstChildWhichIsA("BasePart", true)
            end
        end
        if not refPart then
            refPart = (map :: Instance):FindFirstChildWhichIsA("BasePart", true)
        end
        if refPart then
            if (rootPos - refPart.Position).Magnitude > 1500 then
                return false, nil
            end
            return true, refPart.Position
        end
        return true, rootPos
    end

    local function restoreVictims(): ()
        for part: BasePart, snap in pairs(victimSnapshots) do
            if part and part.Parent then
                pcall(function(): ()
                    part.Anchored = snap.anchored
                    part.CanCollide = snap.canCollide
                    part.CFrame = snap.cframe
                end)
            end
        end
        table.clear(victimSnapshots)
    end

    local function restoreLocalCharacter(): ()
        restoreVictims()
        if services and services.activity then
            pcall(function(): ()
                services.activity.set("fling", false)
            end)
        end
        for part: Instance, wasCollide: boolean in pairs(savedCollide) do
            if part and part.Parent then
                pcall(function(): ()
                    (part :: BasePart).CanCollide = wasCollide
                end)
            end
        end
        table.clear(savedCollide)
        for part: BasePart, oldProps: PhysicalProperties? in pairs(savedCustomProps) do
            if part and part.Parent then
                pcall(function(): ()
                    part.CustomPhysicalProperties = oldProps
                end)
            end
        end
        table.clear(savedCustomProps)

        local character: Model?, humanoid: Humanoid?, root: BasePart? = characterParts()
        if humanoid then
            pcall(function(): ()
                humanoid.PlatformStand = false
                humanoid.AutoRotate = true
                humanoid:SetStateEnabled(Enum.HumanoidStateType.FallingDown, true)
                humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, true)
                humanoid:ChangeState(Enum.HumanoidStateType.Running)
            end)
        end
        if root then
            pcall(function(): ()
                root.AssemblyLinearVelocity = Vector3.zero
                root.AssemblyAngularVelocity = Vector3.zero
                if safeAbovePosition then
                    local groundPos: Vector3 = safeAbovePosition - Vector3.new(0, ABOVE_MAP_OFFSET - 4, 0)
                    local targetCF: CFrame = CFrame.new(groundPos)
                        * CFrame.Angles(0, yawOf(root.CFrame), 0)
                    if character then
                        pcall(function(): ()
                            character:PivotTo(targetCF)
                        end)
                    end
                    root.CFrame = targetCF
                end
            end)
        end
    end

    local function disableCharacterCollisions(character: Model): ()
        for _, object: Instance in ipairs(character:GetDescendants()) do
            if object:IsA("BasePart") then
                if savedCollide[object] == nil then
                    savedCollide[object] = (object :: BasePart).CanCollide
                end
                pcall(function(): ()
                    (object :: BasePart).CanCollide = false
                end)
            end
        end
    end

    local function enableCharacterFlingCollisions(character: Model): ()
        local heavyProps: PhysicalProperties = PhysicalProperties.new(100, 0.3, 0.5, 1, 1)
        for _, object: Instance in ipairs(character:GetChildren()) do
            if object:IsA("BasePart") then
                local part: BasePart = object :: BasePart
                if savedCollide[part] == nil then
                    savedCollide[part] = part.CanCollide
                end
                if savedCustomProps[part] == nil then
                    pcall(function(): ()
                        savedCustomProps[part] = part.CustomPhysicalProperties
                    end)
                end
                pcall(function(): ()
                    part.CanCollide = true
                    part.CustomPhysicalProperties = heavyProps
                end)
            end
        end
    end

    local function holdAboveMap(
        character: Model,
        humanoid: Humanoid,
        root: BasePart
    ): ()
        if services and services.activity then
            pcall(function(): ()
                services.activity.set("fling", false)
            end)
        end
        disableCharacterCollisions(character)
        local holdPos: Vector3 = safeAbovePosition
            or (root.Position + Vector3.new(0, ABOVE_MAP_OFFSET, 0))
        pcall(function(): ()
            humanoid.PlatformStand = true
            humanoid.AutoRotate = false
            root.AssemblyLinearVelocity = Vector3.zero
            root.AssemblyAngularVelocity = Vector3.zero
            root.CFrame = CFrame.new(holdPos)
                * CFrame.Angles(0, yawOf(root.CFrame), 0)
        end)
    end

    local function findLocalTool(toolName: string): Tool?
        local character: Model? = LocalPlayer and LocalPlayer.Character
        if type(getPlayerWeapon) == "function" then
            local found: Tool? = getPlayerWeapon(LocalPlayer, toolName, false)
            if found then
                return found
            end
        end
        if character then
            local inChar: Instance? = character:FindFirstChild(toolName)
            if inChar and inChar:IsA("Tool") then
                return inChar
            end
        end
        local backpack: Instance? = LocalPlayer and LocalPlayer:FindFirstChildOfClass("Backpack")
        if backpack then
            local inBp: Instance? = backpack:FindFirstChild(toolName)
            if inBp and inBp:IsA("Tool") then
                return inBp
            end
        end
        return nil
    end

    local function ensureToolEquipped(
        character: Model,
        humanoid: Humanoid,
        tool: Tool
    ): boolean
        if tool.Parent == character then
            return true
        end
        pcall(function(): ()
            humanoid:EquipTool(tool)
        end)
        if tool.Parent ~= character then
            pcall(function(): ()
                tool.Parent = character
            end)
        end
        return tool.Parent == character
    end

    local function findActiveMurderer(): Player?
        local playersService: Players = game:GetService("Players")
        if type(getPlayerWeapon) == "function" then
            for _, player: Player in ipairs(playersService:GetPlayers()) do
                if player ~= LocalPlayer
                    and (type(isPlayerAlive) ~= "function" or isPlayerAlive(player))
                    and getPlayerWeapon(player, "Gun", false) == nil
                    and getPlayerWeapon(player, "Knife", false) ~= nil then
                    return player
                end
            end
        end
        if type(findMurderer) == "function" then
            local target: Player? = findMurderer()
            if target
                and target ~= LocalPlayer
                and (type(isPlayerAlive) ~= "function" or isPlayerAlive(target))
                and (type(getPlayerWeapon) ~= "function"
                    or getPlayerWeapon(target, "Gun", false) == nil) then
                return target
            end
        end
        for _, player: Player in ipairs(playersService:GetPlayers()) do
            if player ~= LocalPlayer
                and (type(isPlayerAlive) ~= "function" or isPlayerAlive(player))
                and (type(getPlayerWeapon) ~= "function"
                    or getPlayerWeapon(player, "Gun", false) == nil) then
                local role: any = type(getPlayerRole) == "function"
                    and getPlayerRole(player)
                    or nil
                if role == "Murderer" then
                    return player
                end
            end
        end
        return nil
    end

    local function findActiveSheriffOrHero(): Player?
        local playersService: Players = game:GetService("Players")
        if type(getPlayerWeapon) == "function" then
            for _, player: Player in ipairs(playersService:GetPlayers()) do
                if player ~= LocalPlayer
                    and (type(isPlayerAlive) ~= "function" or isPlayerAlive(player))
                    and getPlayerWeapon(player, "Knife", false) == nil
                    and getPlayerWeapon(player, "Gun", false) ~= nil then
                    return player
                end
            end
        end
        if type(findSheriff) == "function" then
            local target: Player? = findSheriff()
            if target
                and target ~= LocalPlayer
                and (type(isPlayerAlive) ~= "function" or isPlayerAlive(target))
                and (type(getPlayerWeapon) ~= "function"
                    or getPlayerWeapon(target, "Knife", false) == nil) then
                return target
            end
        end
        return nil
    end

    -- Insta Kill as Murder (Kill All while staying safely above the map).
    local function runInstaKillAsMurder(
        character: Model,
        humanoid: Humanoid,
        root: BasePart,
        knife: Tool
    ): boolean
        holdAboveMap(character, humanoid, root)
        if not ensureToolEquipped(character, humanoid, knife) then
            return false
        end
        local handle: BasePart? = (knife:FindFirstChild("Handle") :: BasePart?)
            or knife:FindFirstChildWhichIsA("BasePart")
        local slashRemote: Instance? = knife:FindFirstChild("Slash")
        local stabRemote: Instance? = knife:FindFirstChild("Stab")
        local eventsFolder: Instance? = knife:FindFirstChild("Events")
        local touchedRemote: Instance? = eventsFolder
            and eventsFolder:FindFirstChild("HandleTouched")

        local attackCF: CFrame = (handle and handle.CFrame)
            or (root.CFrame * CFrame.new(0, 0, -2.5))
        local playersService: Players = game:GetService("Players")
        local hitAny: boolean = false

        for _, player: Player in ipairs(playersService:GetPlayers()) do
            if player ~= LocalPlayer
                and (type(isPlayerAlive) ~= "function" or isPlayerAlive(player))
                and (type(isProtectedTarget) ~= "function" or not isProtectedTarget(player)) then
                local victimChar: Model? = player.Character
                local victimRoot: BasePart? = victimChar
                    and victimChar:FindFirstChild("HumanoidRootPart") :: BasePart?
                if victimChar and victimRoot then
                    local onVictimMap: boolean = isCharacterOnMap(victimRoot)
                    if onVictimMap then
                        hitAny = true
                        if not victimSnapshots[victimRoot] then
                            pcall(function(): ()
                                victimSnapshots[victimRoot] = {
                                    cframe = victimRoot.CFrame,
                                    anchored = victimRoot.Anchored,
                                    canCollide = victimRoot.CanCollide,
                                }
                            end)
                        end
                        pcall(function(): ()
                            victimRoot.Anchored = true
                            victimRoot.CanCollide = false
                            victimRoot.AssemblyLinearVelocity = Vector3.zero
                            victimRoot.AssemblyAngularVelocity = Vector3.zero
                            victimRoot.CFrame = attackCF
                        end)
                        if slashRemote and slashRemote:IsA("RemoteEvent") then
                            pcall(function(): ()
                                (slashRemote :: RemoteEvent):FireServer("Slash")
                            end)
                        end
                        if stabRemote and stabRemote:IsA("RemoteEvent") then
                            pcall(function(): ()
                                (stabRemote :: RemoteEvent):FireServer("Slash")
                            end)
                        end
                        if touchedRemote and touchedRemote:IsA("RemoteEvent") then
                            pcall(function(): ()
                                (touchedRemote :: RemoteEvent):FireServer(victimRoot)
                            end)
                        end
                        if handle and type(state.fireTouchInterest) == "function" then
                            pcall(state.fireTouchInterest, handle, victimRoot, 0)
                            pcall(state.fireTouchInterest, handle, victimRoot, 1)
                        end
                    end
                end
            end
        end
        if not hitAny then
            restoreVictims()
            awardRoundWin("Murderer")
        end
        return hitAny
    end

    -- Insta Kill As Sheriff (uses AutoShoot CFrame immediately upon receiving
    -- the Gun, whether our role is Sheriff, Hero, or Innocent).
    local function runInstaKillAsSheriff(
        character: Model,
        humanoid: Humanoid,
        root: BasePart,
        gun: Tool
    ): boolean
        holdAboveMap(character, humanoid, root)
        if not ensureToolEquipped(character, humanoid, gun) then
            return false
        end
        local murderer: Player? = findActiveMurderer()
        if not murderer then
            if roundSawMurderer then
                awardRoundWin("Sheriff")
            end
            return false
        end
        roundSawMurderer = true
        if type(state.mm2AutoShootCFrameOnce) == "function" then
            local okHelper: boolean, helperRes: any =
                pcall(state.mm2AutoShootCFrameOnce, false)
            if okHelper and helperRes == true then
                return true
            end
        end
        if type(isProtectedTarget) == "function" and isProtectedTarget(murderer) then
            return false
        end
        local targetChar: Model? = murderer.Character
        local targetRoot: BasePart? = targetChar
            and targetChar:FindFirstChild("HumanoidRootPart") :: BasePart?
        if not targetRoot then
            return false
        end
        if type(state.mm2FireSilentShot) == "function" then
            -- v139.4: pass our options so the shared path keeps the wall check.
            local okSilent: boolean, res: any = pcall(state.mm2FireSilentShot, murderer, {
                continuous = true,
                quiet = true,
                wallCheck = true,
            })
            if okSilent and res == true then
                return true
            end
        end
        local shootRemote: Instance? = gun:FindFirstChild("Shoot")
        if not (shootRemote and shootRemote:IsA("RemoteEvent")) then
            return false
        end
        local originCF: CFrame = (type(getGunOriginCFrame) == "function"
            and getGunOriginCFrame(character, gun))
            or root.CFrame
        local endpoint: Vector3 = targetRoot.Position
        if type(prediction) == "table" and type(prediction.gunAim) == "function" then
            local okAim: boolean, aimPt: any = pcall(
                prediction.gunAim,
                murderer,
                originCF.Position,
                "Bullet"
            )
            if okAim and typeof(aimPt) == "Vector3" then
                endpoint = aimPt
            else
                local vel: Vector3 = Vector3.zero
                pcall(function(): ()
                    vel = targetRoot.AssemblyLinearVelocity
                end)
                endpoint = targetRoot.Position + vel * 0.08
            end
        end
        local startPoint: Vector3 = endpoint + Vector3.new(0, 1.5, -2.5)
        local shotOrigin: CFrame = CFrame.lookAt(startPoint, endpoint)
        local shotEnd: CFrame = CFrame.new(endpoint)
        -- v139.4: the hand-rolled fallback must not shoot through a wall.
        if type(isWeaponPathClear) == "function" then
            local okClear: boolean, allowed: any = pcall(
                isWeaponPathClear,
                originCF.Position,
                endpoint,
                targetChar,
                nil,
                {ignoreTarget = true}
            )
            if not okClear or allowed ~= true then
                return false
            end
        end
        pcall(function(): ()
            (shootRemote :: RemoteEvent):FireServer(shotOrigin, shotEnd)
        end)
        return true
    end

    -- Grab the dropped gun ONLY via fireTouchInterest while staying safely
    -- above the map (never risking a teleport down to the map).
    local function grabDroppedGunTouchOnly(
        character: Model,
        humanoid: Humanoid,
        root: BasePart,
        drop: Instance
    ): boolean
        holdAboveMap(character, humanoid, root)
        if type(state.fireTouchInterest) ~= "function" then
            return false
        end
        local touchPart: BasePart? = nil
        if drop:IsA("BasePart") then
            touchPart = drop :: BasePart
        elseif drop:IsA("Model") then
            touchPart = (drop:FindFirstChild("Handle", true) :: BasePart?)
                or drop:FindFirstChildWhichIsA("BasePart", true)
        end
        if not touchPart then
            return false
        end
        pcall(state.fireTouchInterest, root, touchPart, 0)
        pcall(state.fireTouchInterest, root, touchPart, 1)
        return findLocalTool("Gun") ~= nil
    end

    -- Controlled, bounded Fling on the Sheriff: waits for map/character warmup,
    -- uses state.mm2PerformFling with a returnCFrame to safeAbovePosition when
    -- available (or bounded 10000-stud/s impulse fallback so our character
    -- never enters Roblox's floating-point NullZone or void).
    local function flingSheriffStep(
        character: Model,
        humanoid: Humanoid,
        root: BasePart,
        sheriff: Player
    ): boolean
        local sheriffChar: Model? = sheriff.Character
        local sheriffHum: Humanoid? = sheriffChar
            and sheriffChar:FindFirstChildOfClass("Humanoid")
        local sheriffRoot: BasePart? = sheriffChar
            and sheriffChar:FindFirstChild("HumanoidRootPart") :: BasePart?
        if not sheriffChar or not sheriffHum or sheriffHum.Health <= 0 or sheriffHum.Sit or not sheriffRoot then
            holdAboveMap(character, humanoid, root)
            return false
        end
        local destroyHeight: number = workspace.FallenPartsDestroyHeight
        if destroyHeight ~= destroyHeight or math.abs(destroyHeight) > 1e6 then
            destroyHeight = -500
        end
        local sheriffPos: Vector3 = sheriffRoot.Position
        if sheriffPos.Y <= destroyHeight + 60 then
            holdAboveMap(character, humanoid, root)
            return false
        end
        local sheriffOnMap: boolean = isCharacterOnMap(sheriffRoot)
        if not sheriffOnMap then
            holdAboveMap(character, humanoid, root)
            return false
        end

        if type(state.mm2PerformFling) == "function" then
            if os.clock() - lastSheriffFlingAt < SHERIFF_FLING_COOLDOWN_SECONDS then
                holdAboveMap(character, humanoid, root)
                return true
            end
            flingBusy = true
            lastSheriffFlingAt = os.clock()
            local returnPos: Vector3 = safeAbovePosition
                or (root.Position + Vector3.new(0, ABOVE_MAP_OFFSET, 0))
            task.spawn(function(): ()
                pcall(state.mm2PerformFling, sheriff, {
                    silent = true,
                    duration = 2.6,
                    blocking = true,
                    returnCFrame = CFrame.new(returnPos),
                    shouldStop = function(): boolean
                        if not autoWinActive or not settings.flingSheriffGetGun then
                            return true
                        end
                        if findLocalTool("Gun") ~= nil then
                            return true
                        end
                        if type(findDroppedGun) == "function" and findDroppedGun() ~= nil then
                            return true
                        end
                        return false
                    end,
                })
                lastSheriffFlingAt = os.clock()
                local cNow: Model?, hNow: Humanoid?, rNow: BasePart? = characterParts()
                if cNow and hNow and rNow and autoWinActive then
                    holdAboveMap(cNow, hNow, rNow)
                end
                flingBusy = false
            end)
            return true
        end

        if type(state.mm2FlingStep) == "function" then
            local okStep: boolean, res: any = pcall(
                state.mm2FlingStep,
                character,
                humanoid,
                root,
                sheriff,
                savedCollide,
                savedCustomProps
            )
            if okStep and res == true then
                return true
            end
            holdAboveMap(character, humanoid, root)
            return false
        end

        if services and services.activity then
            pcall(function(): ()
                services.activity.set("fling", true)
            end)
        end
        enableCharacterFlingCollisions(character)
        flingAngle = (flingAngle + 100) % 360
        local phase: number = math.floor(flingAngle / 100) % 4
        local sweepOffset: CFrame = if phase == 0
            then CFrame.new(0, 1.5, 0)
            elseif phase == 1 then CFrame.new(0, -1.5, 0)
            elseif phase == 2 then CFrame.new(2.0, 1.2, -2.0)
            else CFrame.new(-2.0, -1.2, 2.0)
        local sheriffVel: Vector3 = Vector3.zero
        pcall(function(): ()
            sheriffVel = sheriffRoot.AssemblyLinearVelocity
        end)
        local targetCF: CFrame = CFrame.new(sheriffPos + sheriffVel * 0.08)
            * sweepOffset
            * CFrame.Angles(math.rad(flingAngle), math.rad(flingAngle), 0)
        pcall(function(): ()
            if humanoid.Sit then
                humanoid.Sit = false
            end
            humanoid.AutoRotate = false
            root.CFrame = targetCF
            character:PivotTo(targetCF)
            root.AssemblyLinearVelocity = Vector3.new(10000, 12000, 10000)
            root.AssemblyAngularVelocity = Vector3.new(12000, 12000, 12000)
        end)
        return true
    end

    local function autoWinTick(): ()
        if not autoWinActive or flingBusy then
            return
        end
        local character: Model?, humanoid: Humanoid?, root: BasePart? = characterParts()
        if not character or not humanoid or not root or humanoid.Health <= 0 then
            return
        end

        local onMap: boolean, mapRefPos: Vector3? = isCharacterOnMap(root)
        local activeMurderer: Player? = findActiveMurderer()
        if activeMurderer then
            roundSawMurderer = true
        end

        if not onMap then
            if wasOnMap then
                wasOnMap = false
                mapEnteredAt = 0
                sheriffFirstSeenAt = 0
                lastSheriffFlingAt = 0
                safeAbovePosition = nil
                restoreLocalCharacter()
            end

            local roundStillRunning: boolean = type(findMM2Map) == "function"
                and findMM2Map() ~= nil
                and activeMurderer ~= nil
                and (type(isRoundLifecycleActive) ~= "function"
                    or isRoundLifecycleActive() ~= false)

            if not roundStillRunning then
                if participatedInRound and not roundCountedWin and roundSawMurderer then
                    local myRoleNow: any = type(getPlayerRole) == "function"
                        and getPlayerRole(LocalPlayer)
                        or nil
                    if myRoleNow ~= "Dead" then
                        awardRoundWin(roundMyRole)
                    end
                end
                participatedInRound = false
                roundMyRole = nil
                roundSawMurderer = false
                roundCountedWin = false
                flungMurderThisDeath = false
                setStatus("Auto Win: waiting for map")
                return
            end

            -- We are in the lobby while the round is still running on the map
            -- (meaning we died / were killed during the round).
            local myRole: any = type(getPlayerRole) == "function"
                and getPlayerRole(LocalPlayer)
                or nil
            local diedInRound: boolean = participatedInRound or myRole == "Dead"
            if settings.flingMurderIfDead
                and diedInRound
                and not flungMurderThisDeath
                and activeMurderer then
                flungMurderThisDeath = true
                if type(state.mm2PerformFling) == "function" then
                    flingBusy = true
                    setStatus("Auto Win: Flinging Murderer (killed us)")
                    task.spawn(function(): ()
                        pcall(state.mm2PerformFling, activeMurderer, {
                            silent = true,
                            duration = 4.0,
                            blocking = true,
                        })
                        flingBusy = false
                    end)
                    return
                end
            end

            setStatus("Auto Win: waiting in lobby")
            return
        end

        if not wasOnMap or not safeAbovePosition then
            wasOnMap = true
            mapEnteredAt = os.clock()
            sheriffFirstSeenAt = 0
            lastSheriffFlingAt = 0
            participatedInRound = true
            flungMurderThisDeath = false
            local anchor: Vector3 = mapRefPos or root.Position
            safeAbovePosition = anchor + Vector3.new(0, ABOVE_MAP_OFFSET, 0)
        end

        -- Check if the Murderer died while we are alive on the map -> count win!
        if roundSawMurderer and not activeMurderer and not roundCountedWin then
            awardRoundWin(roundMyRole)
        end

        -- 1) Murderer / Knife check: Insta Kill as Murder (Kill All).
        local knife: Tool? = findLocalTool("Knife")
        if knife then
            roundMyRole = "Murderer"
            if settings.instaKillMurder then
                runInstaKillAsMurder(character, humanoid, root, knife)
                setStatus("Auto Win: Insta Kill as Murder")
            else
                holdAboveMap(character, humanoid, root)
                setStatus("Auto Win: safe above map (Murderer)")
            end
            return
        end

        -- 2) Gun check (Sheriff OR Hero OR Innocent who received the Gun!):
        --    Dispara al momento que reciba la pistola con AutoShoot CFrame,
        --    incluso si Fling Sheriff and Get Gun está desactivado y el arma
        --    se recogió con Auto Grab Gun.
        local gun: Tool? = findLocalTool("Gun")
        if gun then
            roundMyRole = "Sheriff"
            if settings.instaKillSheriff then
                runInstaKillAsSheriff(character, humanoid, root, gun)
                setStatus("Auto Win: Insta Kill As Sheriff (CFrame)")
            else
                holdAboveMap(character, humanoid, root)
                setStatus("Auto Win: safe above map (Gun ready)")
            end
            return
        end

        if not roundMyRole then
            local roleNow: any = type(getPlayerRole) == "function"
                and getPlayerRole(LocalPlayer)
                or nil
            if roleNow == "Murderer" then
                roundMyRole = "Murderer"
            elseif roleNow == "Sheriff" or roleNow == "Hero" then
                roundMyRole = "Sheriff"
            else
                roundMyRole = "Innocent"
            end
        end

        -- 3) Dropped gun on the map: grab ONLY via fireTouchInterest (no
        --    risky teleport) if Fling Sheriff and Get Gun is ON or Auto Grab
        --    Gun is enabled.
        local autoGrabExternal: boolean = state.mm2GunAuraEnabled == true
            or state.mm2AutoGetGunEnabled == true
        if settings.flingSheriffGetGun or autoGrabExternal then
            local drop: Instance? = type(findDroppedGun) == "function"
                and findDroppedGun()
                or nil
            if drop then
                local grabbed: boolean = grabDroppedGunTouchOnly(
                    character,
                    humanoid,
                    root,
                    drop
                )
                setStatus("Auto Win: touching dropped gun")
                if grabbed then
                    roundMyRole = "Sheriff"
                    local freshGun: Tool? = findLocalTool("Gun")
                    if freshGun and settings.instaKillSheriff then
                        runInstaKillAsSheriff(character, humanoid, root, freshGun)
                    end
                end
                return
            end
        end

        -- 4) Fling Sheriff and Get Gun while Sheriff is still alive.
        --    Wait for the map and characters to finish loading before flinging
        --    so the local character never gets destabilized on spawn.
        if settings.flingSheriffGetGun then
            local nowClock: number = os.clock()
            local mapElapsed: number = nowClock - mapEnteredAt
            if mapElapsed < MAP_FLING_WARMUP_SECONDS then
                holdAboveMap(character, humanoid, root)
                setStatus("Auto Win: loading map before fling...")
                return
            end
            local sheriff: Player? = findActiveSheriffOrHero()
            if sheriff
                and sheriff ~= LocalPlayer
                and (type(isProtectedTarget) ~= "function" or not isProtectedTarget(sheriff)) then
                if sheriffFirstSeenAt == 0 then
                    sheriffFirstSeenAt = nowClock
                end
                if nowClock - sheriffFirstSeenAt < SHERIFF_READY_DELAY_SECONDS then
                    holdAboveMap(character, humanoid, root)
                    setStatus("Auto Win: waiting for Sheriff to settle...")
                    return
                end
                if flingSheriffStep(character, humanoid, root, sheriff) then
                    setStatus("Auto Win: flinging Sheriff for gun")
                    return
                end
            else
                sheriffFirstSeenAt = 0
            end
        end

        -- 5) Default on map (including pre-round before weapons are given):
        --    Stay safely high above the map where nobody can kill us.
        holdAboveMap(character, humanoid, root)
        setStatus("Auto Win: safe above map")
    end

    local function clearToolConnections(): ()
        for _, conn: RBXScriptConnection in ipairs(toolConnections) do
            pcall(function(): ()
                conn:Disconnect()
            end)
        end
        table.clear(toolConnections)
    end

    local function watchToolContainer(container: Instance?): ()
        if not container
            or (type((container :: any).ChildAdded) ~= "table"
                and typeof((container :: any).ChildAdded) ~= "RBXScriptSignal") then
            return
        end
        pcall(function(): ()
            local conn: RBXScriptConnection = (container :: any).ChildAdded:Connect(
                function(child: Instance): ()
                    if autoWinActive
                        and child
                        and (child.Name == "Gun" or child.Name == "Knife") then
                        autoWinTick()
                    end
                end
            )
            table.insert(toolConnections, conn)
        end)
    end

    local function setupInstantWeaponWatchers(): ()
        clearToolConnections()
        if not LocalPlayer then
            return
        end
        watchToolContainer(LocalPlayer.Character)
        watchToolContainer(LocalPlayer:FindFirstChildOfClass("Backpack"))
        pcall(function(): ()
            if LocalPlayer.CharacterAdded then
                table.insert(
                    toolConnections,
                    LocalPlayer.CharacterAdded:Connect(function(newChar: Model): ()
                        watchToolContainer(newChar)
                        watchToolContainer(LocalPlayer:FindFirstChildOfClass("Backpack"))
                    end)
                )
            end
            if LocalPlayer.ChildAdded then
                table.insert(
                    toolConnections,
                    LocalPlayer.ChildAdded:Connect(function(child: Instance): ()
                        if child and child:IsA("Backpack") then
                            watchToolContainer(child)
                        end
                    end)
                )
            end
        end)
    end

    local function toggleAutoWin(enabled: boolean): ()
        autoWinActive = enabled == true
        disconnectFeatureConnection("MM2AutoWin")
        clearToolConnections()
        if not enabled then
            wasOnMap = false
            participatedInRound = false
            roundMyRole = nil
            roundSawMurderer = false
            roundCountedWin = false
            flungMurderThisDeath = false
            restoreLocalCharacter()
            safeAbovePosition = nil
            setStatus(nil)
            return
        end
        setupInstantWeaponWatchers()
        featureConnections.MM2AutoWin = TaskManager:Connect(autoWinTick, nil, "MM2 Auto Win")
        autoWinTick()
    end

    AutoWinFeature = createUniversalFeature(
        "Auto Win",
        "Keeps your character safely above the map as soon as you enter the map (even before the round starts) so nobody can kill you, with Insta Kill as Murder, Insta Kill As Sheriff (AutoShoot CFrame), Fling Sheriff and Get Gun, and Fling Murder if Dead.",
        8,
        toggleAutoWin,
        {
            categoryName = "Utilities",
            parent = MM2Scroll,
            registry = mm2Features,
        }
    )
    state.mm2AutoWinFeature = AutoWinFeature

    if type(state.addSectionOption) == "function" then
        state.addSectionOption(AutoWinFeature, "Overlay Info")
    end
    if type(state.addStatOption) == "function" then
        sheriffWinsStat = state.addStatOption(AutoWinFeature, "Sheriff Wins", "0")
        murderWinsStat = state.addStatOption(AutoWinFeature, "Murder Wins", "0")
        innocentWinsStat = state.addStatOption(AutoWinFeature, "Innocent Wins", "0")
    end

    addToggleOption(AutoWinFeature, "Insta Kill as Murder", true,
        function(value: boolean): ()
            settings.instaKillMurder = value == true
        end,
        "Automatically kills all living players as soon as you receive the Knife as Murderer.")

    addToggleOption(AutoWinFeature, "Insta Kill As Sheriff", true,
        function(value: boolean): ()
            settings.instaKillSheriff = value == true
        end,
        "Automatically equips the Gun and shoots the Murderer using AutoShoot CFrame the instant you receive the Gun (as Sheriff or Hero).")

    addToggleOption(AutoWinFeature, "Fling Sheriff and Get Gun", false,
        function(value: boolean): ()
            settings.flingSheriffGetGun = value == true
        end,
        "Safely flies to the Sheriff with Fling without killing yourself, then automatically grabs the dropped gun via TouchInterest once the Sheriff dies and shoots the Murderer.")

    addToggleOption(AutoWinFeature, "Fling Murder if Dead", false,
        function(value: boolean): ()
            settings.flingMurderIfDead = value == true
        end,
        "If the Murderer kills you and sends you to the lobby while the round is still active, automatically flings the Murderer.")

    activeCleanup = function(): ()
        toggleAutoWin(false)
        state.mm2AutoWinFeature = nil
    end
    Module.Events = featureConnections
    Module.Initialized = true
    return Module
end

function Module.destroy(): ()
    if not Module.Initialized then
        return
    end
    Module.Initialized = false
    pcall(activeCleanup)
    activeCleanup = function(): () end
    Module.Events = {}
    Module.Runtime = nil
end

return Module
