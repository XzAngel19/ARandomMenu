módulos no significan que estén copiados en este archivo.

el motor de lanzamiento y su limpieza, no la lógica externa de seleccionar
víctimas de Auto Farm o Auto Win. Anti Fling es el módulo universal completo.

local Module = {
    Name = "MM2 Auto Shoot",
    Events = {} :: {[string]: any},
    Initialized = false,
    Runtime = nil :: any,
}

local activeCleanup: () -> () = function(): () end

function Module.init(runtime: any): any
    if Module.Initialized then
        return Module
    end
    local core: any = state.mm2Core
    assert(type(core) == "table", "MM2 Auto Shoot requires the MM2 core module")
    local prediction: any = state.mm2Prediction
    assert(type(prediction) == "table", "MM2 Auto Shoot requires MM2 Prediction")
    Module.Runtime = runtime

    local findMurderer: any = core.findMurderer
    local getPlayerRole: any = core.getPlayerRole
    local getPlayerWeapon: any = core.getPlayerWeapon
    local isPlayerAlive: any = core.isPlayerAlive
    local isProtectedTarget: any = core.isProtectedTarget
    local isWeaponPathClear: any = core.isWeaponPathClear
    local getGunOriginCFrame: any = core.getGunOriginCFrame

    local settings = {
        mode = "Bullet",
        wallCheck = true,
        proximity = false,
        aura = 25,
    }

    local autoShootActive: boolean = false
    local AutoShootFeature: any = nil
    local toolConnections: {RBXScriptConnection} = {}
    local lastShotSentAt: number = 0
    local AUTO_SHOT_PACE_SECONDS: number = 0.22

    local function clearToolConnections(): ()
        for _, conn: RBXScriptConnection in ipairs(toolConnections) do
            pcall(function(): ()
                conn:Disconnect()
            end)
        end
        table.clear(toolConnections)
    end

    local function findLocalGun(): (Tool?, Model?, Humanoid?, BasePart?)
        local character: Model? = LocalPlayer and LocalPlayer.Character
        if not character then
            return nil, nil, nil, nil
        end
        local humanoid: Humanoid? = character:FindFirstChildOfClass("Humanoid")
        local root: BasePart? = character:FindFirstChild("HumanoidRootPart") :: BasePart?
        if not humanoid or humanoid.Health <= 0 or not root then
            return nil, character, humanoid, root
        end
        local gun: Tool? = nil
        if type(getPlayerWeapon) == "function" then
            gun = getPlayerWeapon(LocalPlayer, "Gun", false)
        end
        if not gun then
            local charGun: Instance? = character:FindFirstChild("Gun")
            if charGun and charGun:IsA("Tool") then
                gun = charGun
            end
        end
        if not gun then
            local backpack: Instance? = LocalPlayer:FindFirstChildOfClass("Backpack")
            local bpGun: Instance? = backpack and backpack:FindFirstChild("Gun")
            if bpGun and bpGun:IsA("Tool") then
                gun = bpGun
            end
        end
        return gun, character, humanoid, root
    end

    local function ensureGunEquipped(
        character: Model,
        humanoid: Humanoid,
        gun: Tool
    ): boolean
        if gun.Parent == character then
            return true
        end
        pcall(function(): ()
            humanoid:EquipTool(gun)
        end)
        if gun.Parent ~= character then
            pcall(function(): ()
                gun.Parent = character
            end)
        end
        return gun.Parent == character
    end

    local function resolveMurdererTarget(
        localRoot: BasePart,
        enforceProximity: boolean
    ): (Player?, BasePart?)
        local murderer: Player? = nil
        if type(findMurderer) == "function" then
            murderer = findMurderer()
        end
        if not murderer then
            local playersService: Players = game:GetService("Players")
            for _, player: Player in ipairs(playersService:GetPlayers()) do
                if player ~= LocalPlayer then
                    local role: any = type(getPlayerRole) == "function"
                        and getPlayerRole(player)
                        or nil
                    local hasKnife: boolean = type(getPlayerWeapon) == "function"
                        and getPlayerWeapon(player, "Knife", false) ~= nil
                    if role == "Murderer" or hasKnife then
                        murderer = player
                        break
                    end
                end
            end
        end
        if not murderer or murderer == LocalPlayer then
            return nil, nil
        end
        if type(isPlayerAlive) == "function" and not isPlayerAlive(murderer) then
            return nil, nil
        end
        if type(isProtectedTarget) == "function" and isProtectedTarget(murderer) then
            return nil, nil
        end
        local targetChar: Model? = murderer.Character
        local targetRoot: BasePart? = targetChar
            and targetChar:FindFirstChild("HumanoidRootPart") :: BasePart?
        if not targetRoot then
            return nil, nil
        end
        if enforceProximity then
            local okDist: boolean, distance: any = pcall(function(): any
                return (targetRoot.Position - localRoot.Position).Magnitude
            end)
            if okDist and type(distance) == "number" and distance > settings.aura then
                return nil, nil
            end
        end
        return murderer, targetRoot
    end

    local function computeTargetEndpoint(
        target: Player,
        targetRoot: BasePart,
        originPos: Vector3,
        aimMode: string?
    ): Vector3
        if type(prediction) == "table" and type(prediction.gunAim) == "function" then
            local ok: boolean, endpoint: any = pcall(
                prediction.gunAim,
                target,
                originPos,
                aimMode
            )
            if ok and typeof(endpoint) == "Vector3" then
                return endpoint
            end
        end
        local vel: Vector3 = Vector3.zero
        pcall(function(): ()
            vel = targetRoot.AssemblyLinearVelocity
        end)
        return targetRoot.Position + vel * 0.08
    end

    local function fireCFrameShotImmediate(
        target: Player,
        targetRoot: BasePart,
        character: Model,
        localRoot: BasePart,
        gun: Tool
    ): boolean
        local shootRemote: Instance? = gun:FindFirstChild("Shoot")
        if not (shootRemote and shootRemote:IsA("RemoteEvent")) then
            return false
        end
        if gun.Enabled == false
            or gun:GetAttribute("Disabled") == true
            or gun:GetAttribute("Reloading") == true
            or gun:GetAttribute("IsReloading") == true then
            return false
        end
        local originCF: CFrame = (type(getGunOriginCFrame) == "function"
            and getGunOriginCFrame(character, gun))
            or localRoot.CFrame
        local endpoint: Vector3 = computeTargetEndpoint(
            target,
            targetRoot,
            originCF.Position,
            "CFrame"
        )
        local capturedPos: Vector3 = targetRoot.Position
        local chordFlat: Vector3 = Vector3.new(
            endpoint.X - capturedPos.X,
            0,
            endpoint.Z - capturedPos.Z
        )
        local pierceAxis: Vector3
        if chordFlat.Magnitude > 0.15 then
            pierceAxis = chordFlat.Unit
        else
            local toTargetFlat: Vector3 = Vector3.new(
                endpoint.X - originCF.Position.X,
                0,
                endpoint.Z - originCF.Position.Z
            )
            if toTargetFlat.Magnitude > 0.05 then
                pierceAxis = toTargetFlat.Unit
            else
                pierceAxis = Vector3.new(0, 0, -1)
            end
        end
        local startPoint: Vector3 = capturedPos - pierceAxis * 1.5
        local endPoint: Vector3 = endpoint + pierceAxis * 1.5
        if (endPoint - startPoint).Magnitude < 0.2 then
            startPoint = endpoint + Vector3.new(0, 2.0, -1.5)
            endPoint = endpoint
        end
        local shotOrigin: CFrame = CFrame.lookAt(startPoint, endPoint)
        local shotEnd: CFrame = CFrame.new(endPoint)
        local firedSilent: boolean = false
        if type(state.mm2FireSilentShot) == "function" then
            -- v139.4: our own WallCheck has to travel with the request. The
            -- shared constructor used to be called with the target only and
            -- hardcoded continuous=true, so this mode fired with no geometry
            -- check and then, when it was refused, fell back to its own CFrame
            -- -- also unchecked.
            local okSilent: boolean, res: any = pcall(state.mm2FireSilentShot, target, {
                continuous = true,
                quiet = true,
                wallCheck = settings.wallCheck,
            })
            firedSilent = okSilent and res == true
            -- A false result or exception can mean a safety refusal or an
            -- uncertain send. Never bypass it or send a second request. Pace
            -- exceptions too: the send may have happened before the error.
            if not okSilent then lastShotSentAt = os.clock() end
            if not firedSilent then return false end
        end
        if not firedSilent then
            -- Never fire past a wall: the hand-rolled CFrame is only a
            -- transport fallback, not a bypass.
            local clear: boolean = false
            if settings.wallCheck and type(isWeaponPathClear) == "function" then
                local okClear: boolean, allowed: any = pcall(
                    isWeaponPathClear,
                    originCF.Position,
                    endpoint,
                    target.Character,
                    nil,
                    {ignoreTarget = true}
                )
                clear = okClear and allowed == true
            else
                clear = true
            end
            if clear then
                local okFire: boolean = pcall(function(): ()
                    (shootRemote :: RemoteEvent):FireServer(shotOrigin, shotEnd)
                end)
                if not okFire then
                    lastShotSentAt = os.clock()
                    return false
                end
            else
                lastShotSentAt = 0
                return false
            end
        end
        lastShotSentAt = os.clock()
        return true
    end

    local function fireBulletShotImmediate(
        target: Player,
        targetRoot: BasePart,
        character: Model,
        localRoot: BasePart,
        gun: Tool
    ): boolean
        local shootRemote: Instance? = gun:FindFirstChild("Shoot")
        if not (shootRemote and shootRemote:IsA("RemoteEvent")) then
            return false
        end
        if gun.Enabled == false
            or gun:GetAttribute("Disabled") == true
            or gun:GetAttribute("Reloading") == true
            or gun:GetAttribute("IsReloading") == true then
            return false
        end
        local originCF: CFrame = (type(getGunOriginCFrame) == "function"
            and getGunOriginCFrame(character, gun))
            or localRoot.CFrame
        local endpoint: Vector3 = computeTargetEndpoint(
            target,
            targetRoot,
            originCF.Position,
            "Bullet"
        )
        if settings.wallCheck and type(isWeaponPathClear) == "function" then
            local okClear: boolean, clear: any = pcall(
                isWeaponPathClear,
                originCF.Position,
                endpoint,
                target.Character
            )
            if not okClear or clear ~= true then
                return false
            end
        end
        local firedCore: boolean = false
        if type(state.mm2FireGunShot) == "function" then
            local okShot: boolean, res: any = pcall(state.mm2FireGunShot, target, {
                silent = false,
                wallCheck = settings.wallCheck,
                continuous = true,
                quiet = true,
            })
            firedCore = okShot and res == true
            -- Respect a refusal or uncertain send from the shared path.
            if not okShot then lastShotSentAt = os.clock() end
            if not firedCore then return false end
        end
        if not firedCore then
            local shotOrigin: CFrame = CFrame.lookAt(originCF.Position, endpoint)
            local shotEnd: CFrame = CFrame.new(endpoint)
            local okFire: boolean = pcall(function(): ()
                (shootRemote :: RemoteEvent):FireServer(shotOrigin, shotEnd)
            end)
            if not okFire then
                lastShotSentAt = os.clock()
                return false
            end
        end
        lastShotSentAt = os.clock()
        return true
    end

    -- Shared helper for Auto Win ("Insta Kill As Sheriff (Usa AutoShoot CFrame)")
    -- and Auto Shoot CFrame mode: equips the gun the instant the player
    -- receives it (whether as Sheriff, Hero, or Innocent who picked up GunDrop)
    -- and fires Silent AIM CFrame continuously until the Murderer is dead.
    local function executeAutoShootCFrameOnce(enforceProximity: boolean): boolean
        local gun: Tool?, character: Model?, humanoid: Humanoid?,
            localRoot: BasePart? = findLocalGun()
        if not gun or not character or not humanoid or not localRoot then
            lastShotSentAt = 0
            return false
        end
        local target: Player?, targetRoot: BasePart? =
            resolveMurdererTarget(localRoot, enforceProximity)
        if not target or not targetRoot then
            return false
        end
        if not ensureGunEquipped(character, humanoid, gun) then
            return false
        end
        if os.clock() - lastShotSentAt < AUTO_SHOT_PACE_SECONDS then
            return true
        end
        return fireCFrameShotImmediate(target, targetRoot, character, localRoot, gun)
    end
    state.mm2AutoShootCFrameOnce = executeAutoShootCFrameOnce

    local function autoShootTick(): ()
        if not autoShootActive then
            return
        end
        local gun: Tool?, character: Model?, humanoid: Humanoid?,
            localRoot: BasePart? = findLocalGun()
        if not gun or not character or not humanoid or not localRoot then
            lastShotSentAt = 0
            return
        end
        local target: Player?, targetRoot: BasePart? =
            resolveMurdererTarget(localRoot, settings.proximity == true)
        if not target or not targetRoot then
            return
        end
        if not ensureGunEquipped(character, humanoid, gun) then
            return
        end
        if os.clock() - lastShotSentAt < AUTO_SHOT_PACE_SECONDS then
            return
        end
        local isCFrame: boolean = settings.mode == "CFrame"
            or settings.mode == "Cframe"
            or settings.mode == "Silent AIM"
        if settings.mode == "Experimental" then
            -- Share Gun Aimbot's selector and safety path, but use this card's
            -- own WallCheck/Proximity. Never send both on an uncertain result.
            local mode: string = "Bullet"
            if type(state.mm2ChooseExperimentalAim) == "function" then
                local origin: CFrame = (type(getGunOriginCFrame) == "function"
                    and getGunOriginCFrame(character, gun)) or localRoot.CFrame
                local okMode: boolean, chosen: any = pcall(state.mm2ChooseExperimentalAim, target, origin.Position)
                if okMode and chosen == "CFrame" then mode = "CFrame" end
            end
            local sent: boolean = false
            local uncertain: boolean = false
            if type(state.mm2FireGunShot) == "function" then
                local ok: boolean, result: any = pcall(state.mm2FireGunShot, target, {
                    silent = mode == "CFrame", wallCheck = settings.wallCheck,
                    continuous = true, quiet = true,
                })
                sent = ok and result == true
                -- A thrown error can happen after dispatch; pace rather than
                -- repeatedly sending on every frame while the result is unknown.
                uncertain = not ok
                if uncertain then lastShotSentAt = os.clock() end
            end
            if sent then lastShotSentAt = os.clock() end
            if AutoShootFeature then AutoShootFeature:SetStatus("Experimental · " .. mode
                .. (sent and " · request sent" or (uncertain and " · send uncertain" or " · not sent"))) end
        elseif isCFrame then
            fireCFrameShotImmediate(target, targetRoot, character, localRoot, gun)
        else
            fireBulletShotImmediate(target, targetRoot, character, localRoot, gun)
        end
    end

    local function watchToolContainer(container: Instance?): ()
        if not container or type((container :: any).ChildAdded) ~= "table"
            and typeof((container :: any).ChildAdded) ~= "RBXScriptSignal" then
            return
        end
        pcall(function(): ()
            local conn: RBXScriptConnection = (container :: any).ChildAdded:Connect(
                function(child: Instance): ()
                    if autoShootActive and child and child.Name == "Gun" then
                        autoShootTick()
                    end
                end
            )
            table.insert(toolConnections, conn)
        end)
    end

    local function setupInstantGunWatchers(): ()
        clearToolConnections()
        if not LocalPlayer then
            return
        end
        watchToolContainer(LocalPlayer.Character)
        watchToolContainer(LocalPlayer:FindFirstChildOfClass("Backpack"))
        pcall(function(): ()
            if LocalPlayer.CharacterAdded then
                table.insert(
                    toolConnections,
                    LocalPlayer.CharacterAdded:Connect(function(newChar: Model): ()
                        watchToolContainer(newChar)
                        watchToolContainer(LocalPlayer:FindFirstChildOfClass("Backpack"))
                    end)
                )
            end
            if LocalPlayer.ChildAdded then
                table.insert(
                    toolConnections,
                    LocalPlayer.ChildAdded:Connect(function(child: Instance): ()
                        if child and child:IsA("Backpack") then
                            watchToolContainer(child)
                        end
                    end)
                )
            end
        end)
    end

    local function updateStatus(): ()
        -- The feature's UI flag is flipped after its callback returns, so use
        -- our controller state during the initial enable as well.
        if not AutoShootFeature or not autoShootActive then
            return
        end
        if settings.proximity then
            AutoShootFeature:SetStatus(
                settings.mode .. " · " .. tostring(settings.aura) .. " studs"
            )
        else
            AutoShootFeature:SetStatus(settings.mode)
        end
    end

    -- v139: this card and the MM2 "Shoot" card both used the key MM2AutoShoot,
    -- so enabling both lost one handle and left a ghost task firing every frame
    -- after the card was switched off. Each card owns its own key now, and the
    -- connection is dropped before a new one is made.
    local CONNECTION_KEY: string = "MM2AutoShootCard"

    local function toggleAutoShoot(enabled: boolean): ()
        autoShootActive = enabled == true
        if enabled then
            updateStatus()
            setupInstantGunWatchers()
            disconnectFeatureConnection(CONNECTION_KEY)
            featureConnections[CONNECTION_KEY] = TaskManager:Connect(
                autoShootTick,
                nil,
                "Auto Shoot"
            )
            autoShootTick()
        else
            clearToolConnections()
            disconnectFeatureConnection(CONNECTION_KEY)
            if AutoShootFeature then
                AutoShootFeature:SetStatus(nil)
            end
        end
    end

    AutoShootFeature = createUniversalFeature(
        "Auto Shoot",
        "Automatically equips the Gun the instant you receive it (as Sheriff or Hero) and shoots the Murderer continuously using Bullet (respects WallCheck) or CFrame (Silent AIM). Optional Proximity radius filter.",
        4,
        toggleAutoShoot,
        {
            categoryName = "Utilities",
            parent = MM2Scroll,
            registry = mm2Features,
        }
    )
    state.mm2AutoShootFeature = AutoShootFeature

    local auraSlider: GuiObject? = nil
    local states: any = type(configData) == "table" and configData.states or {}
    local values: any = type(configData) == "table" and configData.values or {}
    local initialMode: string = values["MM2.AutoShoot.Mode"] or "Bullet"
    if initialMode ~= "CFrame" and initialMode ~= "Experimental" then initialMode = "Bullet" end
    local function modeConfig(mode: string): any
        local prefix: string = "MM2.AutoShoot." .. mode .. "."
        local legacy: boolean = mode == initialMode
        local wall: any = states[prefix .. "WallCheck"]
        local proximity: any = states[prefix .. "Proximity"]
        local aura: any = values[prefix .. "Aura"]
        if wall == nil and legacy then wall = states["MM2.AutoShoot.WallCheck"] end
        if proximity == nil and legacy then proximity = states["MM2.AutoShoot.Proximity"] end
        if aura == nil and legacy then aura = values["MM2.AutoShoot.Aura"] end
        return {wallCheck = wall == nil or wall == true, proximity = proximity == true,
            aura = type(aura) == "number" and aura or 25}
    end
    local modeConfigs: any = {Bullet = modeConfig("Bullet"), CFrame = modeConfig("CFrame"),
        Experimental = modeConfig("Experimental")}
    local switching: boolean = false
    local optionsReady: boolean = false
    local function refreshProximityOption(): ()
        if auraSlider and type(setOptionVisible) == "function" then
            -- addNumberOption returns the option row itself, not its slider.
            -- Hiding its parent hides every option in the settings window.
            setOptionVisible(AutoShootFeature, auraSlider :: GuiObject, settings.proximity == true)
        end
        updateStatus()
    end
    local function syncMode(): ()
        local cfg: any = modeConfigs[settings.mode] or modeConfigs.Bullet
        switching = true
        settings.wallCheck, settings.proximity, settings.aura = cfg.wallCheck, cfg.proximity, cfg.aura
        if state and type(state.toggleControls) == "table" then
            local wc: any = state.toggleControls["MM2.AutoShoot.WallCheck"]
            local px: any = state.toggleControls["MM2.AutoShoot.Proximity"]
            if wc and wc.setChecked then pcall(wc.setChecked, cfg.wallCheck) end
            if px and px.setChecked then pcall(px.setChecked, cfg.proximity) end
        end
        if state and type(state.valueControls) == "table" then
            local au: any = state.valueControls["MM2.AutoShoot.Aura"]
            if au and au.apply then pcall(au.apply, cfg.aura) end
        end
        switching = false
        refreshProximityOption()
    end
    addCycleOption(AutoShootFeature, "Mode", {"Bullet", "CFrame", "Experimental"}, 1,
        function(value: string): () settings.mode = value; syncMode() end,
        "Experimental selects one path based on range and movement; the heuristic does not guarantee hits. Options are independent per mode.")
    addToggleOption(AutoShootFeature, "WallCheck", true, function(value: boolean): ()
        settings.wallCheck = value == true
        if optionsReady and not switching then
            modeConfigs[settings.mode].wallCheck = settings.wallCheck
            states["MM2.AutoShoot." .. settings.mode .. ".WallCheck"] = settings.wallCheck
        end
    end, "Check for walls; saved independently per mode.")
    addToggleOption(AutoShootFeature, "Proximity", false, function(value: boolean): ()
        settings.proximity = value == true
        if optionsReady and not switching then
            modeConfigs[settings.mode].proximity = settings.proximity
            states["MM2.AutoShoot." .. settings.mode .. ".Proximity"] = settings.proximity
        end
        refreshProximityOption()
    end, "Only shoot inside Aura radius; saved independently per mode.")
    auraSlider = addNumberOption(AutoShootFeature, "Aura", 25, 5, 150, function(value: number): ()
        settings.aura = value
        if optionsReady and not switching then
            modeConfigs[settings.mode].aura = value
            values["MM2.AutoShoot." .. settings.mode .. ".Aura"] = value
        end
        updateStatus()
    end, "Maximum distance with Proximity enabled, saved per mode.", 1)
    optionsReady = true
    syncMode()

    activeCleanup = function(): ()
        toggleAutoShoot(false)
        state.mm2AutoShootFeature = nil
        state.mm2AutoShootCFrameOnce = nil
    end
    Module.Events = featureConnections
    Module.Initialized = true
    return Module
end

function Module.destroy(): ()
    if not Module.Initialized then
        return
    end
    Module.Initialized = false
    pcall(activeCleanup)
    activeCleanup = function(): () end
    Module.Events = {}
    Module.Runtime = nil
end

return Module

    if Module.Initialized then
        return Module
    end
    local core: any = state.mm2Core
    assert(type(core) == "table", "MM2 Shoot requires the MM2 core module")
    local prediction: any = state.mm2Prediction
    assert(type(prediction) == "table", "MM2 Shoot requires MM2 Prediction")
    local trajectoryCalibration: any = core.trajectoryCalibration
    Module.Runtime = runtime
    local connectGunFiredSignal: any = core.connectGunFiredSignal
    local disconnectGunFiredObserver: any = core.disconnectGunFiredObserver
    local findMurderer: any = core.findMurderer
    local getGunOriginCFrame: any = core.getGunOriginCFrame
    local getPlayerRole: any = core.getPlayerRole
    local getPlayerWeapon: any = core.getPlayerWeapon
    local getWeaponServiceModule: any = core.getWeaponServiceModule

    local shootMacro: boolean = false
    -- First thing the untouched ray meets on an allowed shot (nil if nothing
    -- short of the aim). Persisted into the shot's settings.
    local lastGunOccluder: any = nil
    local function macroUnequip(gun: Tool?): ()
        if not shootMacro or not gun then
            return
        end
        task.delay(0.01, function(): ()
            pcall(function(): ()
                local backpack: Instance? = LocalPlayer:FindFirstChildOfClass("Backpack")
                local character: Model? = LocalPlayer.Character
                if backpack and character and gun.Parent == character then
                    gun.Parent = backpack
                end
            end)
        end)
    end

    local function getSelectedShootTarget(): Player?
        local target
        if mm2Settings.shootMode == "Custom"
            and mm2Settings.shootTarget ~= "" then
            target = findPlayerByText(mm2Settings.shootTarget)
        else
            target = findMurderer()
        end
        if not target
            or target == LocalPlayer
            or not isPlayerAlive(target)
            or isProtectedTarget(target) then
            return nil
        end
        if getPlayerRole(target) ~= "Murderer" then
            return nil
        end
        return target
    end

    local SILENT_ORIGIN_STUDS: number = 2.5
    local SILENT_PIERCE_STUDS: number = 1.0
    local SILENT_LIFT_STUDS: number = 1.6
    local MAX_SILENT_SWEEP: number = 1.5
    -- Inside this distance the murderer's body covers the whole line of sight,
    -- so the lead stops helping and can even invert the shot: 17 studs/s of
    -- charge times a 0.1 s lead is 1.7 studs, more than the gap itself.
    local POINT_BLANK_STUDS: number = 6

    -- The Top/fallback origin lift belongs to the target's rig, not to a
    -- universal constant: R6 and R15 put the head at different heights
    -- above the root, and an origin spawned inside the head can be
    -- ignored by the shot. Predicted endpoint minus predicted root is
    -- the same aim offset the predictor sampled from the live rig.
    local function silentLift(info: any, predicted: Vector3): number
        if type(info) == "table"
            and typeof(info.rootPosition) == "Vector3" then
            local aimOffsetY: number = predicted.Y - info.rootPosition.Y
            if aimOffsetY > 0.5 then
                return math.clamp(aimOffsetY + 0.35, 1.2, 2.4)
            end
        end
        return SILENT_LIFT_STUDS
    end

    -- The bypass segment runs along the target's own motion. A fixed extra
    -- length guesses; the confirmed shots already measure how far the aim
    -- lands from the body, so the same measurement decides how much segment
    -- the shot needs. With no confirmations yet the uncertainty is zero and
    -- the configured value is used untouched.
    local function getSilentSweepDistance(speed: number?): number
        local configured: number = tonumber(mm2Settings.silentSweep) or 0
        local sweep: number = math.clamp(configured, 0, MAX_SILENT_SWEEP)
        local uncertainty: number = 0
        if type(prediction) == "table"
            and type(prediction.getLeadUncertainty) == "function" then
            -- v161: the silent path is the CFrame mode; its measurements are
            -- the ones that size the sweep.
            local ok: boolean, seconds: any = pcall(
                prediction.getLeadUncertainty,
                "CFrame"
            )
            if ok and type(seconds) == "number" and seconds == seconds then
                uncertainty = math.clamp(seconds, 0, 0.15)
            end
        end
        if uncertainty <= 0 then
            return sweep
        end
        local travel: number = math.max(tonumber(speed) or 0, 0)
        local extra: number = uncertainty * travel
        return math.clamp(sweep + extra, 0, MAX_SILENT_SWEEP)
    end

    local function buildSilentShot(
        _target: Player,
        info: any,
        muzzleOrigin: CFrame?,
        variant: string
    ): (CFrame?, CFrame?, number, number, string)
        local predicted: Vector3 = info.endpoint
        local liftStuds: number = silentLift(info, predicted)
        local velocity: Vector3 = info.velocity
        local horizontal: Vector3 = Vector3.new(velocity.X, 0, velocity.Z)

        local direction: Vector3? = nil
        local sweep: number = 0
        if horizontal.Magnitude > 1.5 then
            direction = horizontal.Unit

            sweep = getSilentSweepDistance(horizontal.Magnitude)
        end
        local sweepStart: Vector3 =
            direction and -direction * sweep or Vector3.zero
        local sweepEnd: Vector3 =
            direction and direction * sweep or Vector3.zero

        local approach: Vector3? = nil
        local approachOffset: number = SILENT_ORIGIN_STUDS
        if muzzleOrigin then
            local toTarget: Vector3 = predicted - muzzleOrigin.Position
            local flat: Vector3 = Vector3.new(toTarget.X, 0, toTarget.Z)
            if flat.Magnitude > 0.05 then
                approach = flat.Unit
                approachOffset = math.min(
                    SILENT_ORIGIN_STUDS,
                    math.max(1.0, flat.Magnitude * 0.6)
                )
            end
        end

        local startPoint: Vector3
        local endPoint: Vector3
        local resolvedVariant: string = variant
        local pierceAxis: Vector3? = direction or approach
        if (variant == "Through" or variant == nil or variant == "") and pierceAxis then
            resolvedVariant = "Through"
            local captured: Vector3 = info.capturedPosition or predicted
            startPoint = captured - pierceAxis * 1.0 + sweepStart
            endPoint = predicted + pierceAxis * SILENT_PIERCE_STUDS + sweepEnd
        elseif variant == "Top" then
            resolvedVariant = "Top"
            startPoint = predicted + Vector3.new(0, liftStuds, 0)
                + sweepStart
            endPoint = predicted + sweepEnd
        elseif variant == "Behind" and approach then
            resolvedVariant = "Behind"
            startPoint = predicted + approach * approachOffset + sweepStart
            endPoint = predicted + sweepEnd
        elseif approach then
            resolvedVariant = "Front"
            startPoint = predicted - approach * approachOffset + sweepStart
            endPoint = predicted + sweepEnd
        elseif direction then
            resolvedVariant = "Motion"
            startPoint = predicted + sweepStart
            endPoint = predicted + sweepEnd
        else
            resolvedVariant = "Top"
            startPoint = predicted + Vector3.new(0, liftStuds, 0)
            endPoint = predicted
        end

        if (endPoint - startPoint).Magnitude <= 0.05 then
            startPoint = predicted + Vector3.new(0, liftStuds, 0)
            endPoint = predicted
            resolvedVariant = "Top"
            sweep = 0
        end
        return CFrame.lookAt(startPoint, endPoint), CFrame.new(endPoint),
            approachOffset, sweep, resolvedVariant
    end

    type FireOptions = {
        silent: boolean?,
        quiet: boolean?,
        whenReady: boolean?,
        continuous: boolean?,
        wallCheck: boolean?,
        -- Native PC hook: construct the same shot but let the original
        -- FireServer send it once instead of sending a second packet.
        redirect: ((CFrame, CFrame) -> ())?,
        nativeRedirect: boolean?,
        nativeOrigin: CFrame?,
    }

    -- Experimental policy, not a measured hit-rate optimization: prefer the
    -- ordinary muzzle path nearby/stationary and the CFrame path for a fast,
    -- distant target. It selects exactly one path per request; a refusal is
    -- not evidence that a second packet would be safe or accepted.
    local function chooseExperimentalGunMode(target: Player?, origin: Vector3?): string
        local character: Model? = target and target.Character
        local root: BasePart? = character and character:FindFirstChild("HumanoidRootPart") :: BasePart?
        if not root or not origin then return "Bullet" end
        local speed: number = 0
        pcall(function(): ()
            local velocity: Vector3 = root.AssemblyLinearVelocity
            speed = Vector3.new(velocity.X, 0, velocity.Z).Magnitude
        end)
        if (root.Position - origin).Magnitude >= 30 and speed >= 8 then return "CFrame" end
        return "Bullet"
    end

    -- How long an authored request owns the gun. The feedback queue resolves
    -- or expires a shot at 1.6-1.8 s; keeping the window inside that bound
    -- means the gate never outlives the pending state it reads.
    local SHOT_CONFIRM_WINDOW: number = 1.5

    local ShootFeature: any = nil
    local lastRejectReason: string = ""

    -- The parts an aim may land on, best first, for the ghost preference.
    local AIM_BODY_PARTS: {string} = {
        "UpperTorso", "Torso", "HumanoidRootPart", "Head",
        "RightUpperArm", "LeftUpperArm", "RightUpperLeg", "LeftUpperLeg",
        "Right Arm", "Left Arm", "Right Leg", "Left Leg",
    }

    -- v161: the aim can be perfect and still be stopped by a GHOST part --
    -- transparent AND non-colliding, which every client wall check skips. The
    -- v159 session lost exactly that shot: 0.24 studs from the torso, occluder
    -- "Workspace.Workplace.Raggy.HumanoidRootPart", transparency 1,
    -- CanCollide false, 10.3 studs in front of the aim. What the server's own
    -- weapon ray does with such a part is not known, so this never refuses a
    -- shot: it only prefers another part of the same body whose line does not
    -- cross the ghost. If every candidate crosses it, the original point stands.
    local function ghostOnLine(
        muzzle: Vector3,
        endpoint: Vector3,
        targetCharacter: Model?
    ): any
        if type(describeRayHit) ~= "function" then
            return nil
        end
        local ok: boolean, hit: any = pcall(
            describeRayHit,
            muzzle,
            endpoint,
            targetCharacter
        )
        if not ok or type(hit) ~= "table" or hit.onTarget then
            return nil
        end
        if hit.transparency ~= 1 or hit.canCollide ~= false then
            return nil
        end
        local reach: number = (endpoint - muzzle).Magnitude
        if type(hit.distance) ~= "number" or hit.distance >= reach - 0.5 then
            return nil
        end
        return hit
    end

    -- The ghost probe is a preference, not a gate, so it is allowed to be
    -- stale: what is cached is which body part was clean, never a position.
    -- v162: Auto Shoot calls the aim path every frame; probing there cost up
    -- to 13 raycasts per frame. Now it costs at most 5 every 0.15 s and the
    -- cached answer is one FindFirstChild plus the same lead arithmetic.
    local GHOST_PROBE_INTERVAL: number = 0.15
    local GHOST_PROBE_CANDIDATES: number = 4
    local ghostProbe: any = {
        target = nil,
        at = -math.huge,
        partName = nil,
    }

    -- Returns a cleaner endpoint and the part it moved to, or nil.
    local function avoidGhostAim(
        target: Player,
        info: any,
        muzzle: Vector3
    ): (Vector3?, string?)
        local character: Model? = target.Character
        if not character or not info or typeof(info.endpoint) ~= "Vector3" then
            return nil, nil
        end
        local now: number = os.clock()
        if target ~= ghostProbe.target
            or now - ghostProbe.at >= GHOST_PROBE_INTERVAL then
            ghostProbe.target = target
            ghostProbe.at = now
            ghostProbe.partName = nil
            if ghostOnLine(muzzle, info.endpoint, character) then
                local captured: Vector3? = info.capturedPosition
                local probeLead: Vector3 = captured
                        and (info.endpoint - captured)
                    or Vector3.zero
                local tried: number = 0
                for _, partName: string in ipairs(AIM_BODY_PARTS) do
                    if tried >= GHOST_PROBE_CANDIDATES then
                        break
                    end
                    local part: Instance? = character:FindFirstChild(partName)
                    if part and part:IsA("BasePart") then
                        tried += 1
                        local candidate: Vector3 = (part :: BasePart).Position
                            + probeLead
                        if not ghostOnLine(muzzle, candidate, character) then
                            ghostProbe.partName = partName
                            break
                        end
                    end
                end
            end
        end
        local partName: string? = ghostProbe.partName
        if not partName then
            return nil, nil
        end
        local part: Instance? = character:FindFirstChild(partName)
        if not part or not part:IsA("BasePart") then
            ghostProbe.partName = nil
            return nil, nil
        end
        local captured: Vector3? = info.capturedPosition
        local lead: Vector3 = captured and (info.endpoint - captured) or Vector3.zero
        return (part :: BasePart).Position + lead, partName
    end

    local function fireGunAtTarget(
        target: Player?,
        options: FireOptions?
    ): boolean
        local opts: FireOptions = options or {}
        local silent: boolean = mm2Settings.silentAim
        local experimental: boolean = opts.silent == nil and mm2Settings.experimentalAim == true
        if experimental then
            local localCharacter: Model? = LocalPlayer and LocalPlayer.Character
            local ownRoot: BasePart? = localCharacter and localCharacter:FindFirstChild("HumanoidRootPart") :: BasePart?
            silent = chooseExperimentalGunMode(target, ownRoot and ownRoot.Position) == "CFrame"
        elseif opts.silent ~= nil then
            silent = opts.silent
        end
        local quiet: boolean = opts.quiet == true
        local continuous: boolean = opts.continuous == true
        -- An intercepted native FireServer is already being sent by the game.
        -- Rechecking a Tool that the native script just put on cooldown can
        -- only suppress the redirect, not prevent an extra shot.
        local nativeRedirect: boolean = opts.nativeRedirect == true and type(opts.redirect) == "function"
        local useRaw: boolean = mm2Settings.shootRaw
            or (continuous and silent)
        local useWallCheck: boolean = if opts.wallCheck ~= nil
            then opts.wallCheck == true
            else mm2Settings.shootWallCheck
        local function reject(message: string): boolean
            lastRejectReason = message
            if not quiet and ShootFeature then ShootFeature:SetStatus(message) end
            return false
        end

        local character, humanoid = getCharacterParts()
        if not target or not target.Character or target == LocalPlayer
            or not isPlayerAlive(target) or isProtectedTarget(target) or getPlayerRole(target) ~= "Murderer" then
            return reject("No selected shoot target was found.")
        end
        if not character or not humanoid then
            return reject("Your character is not available.")
        end
        local targetCharacter = target.Character
        local gun: Tool? = getPlayerWeapon(LocalPlayer, "Gun", true)
        if not gun or gun.Parent ~= character then
            return reject("Equip the sheriff gun yourself to shoot.")
        end
        if not nativeRedirect and (gun.Enabled == false or gun:GetAttribute("Disabled") == true
            or gun:GetAttribute("Reloading") == true or gun:GetAttribute("IsReloading") == true) then
            return reject("The gun is disabled or reloading.")
        end

        local remote: Instance? = gun and gun:FindFirstChild("Shoot")
        local okOrigin: boolean, foundOrigin: any = pcall(getGunOriginCFrame, character, gun)
        local origin: CFrame? = okOrigin and foundOrigin or nil
        if not origin and nativeRedirect and typeof(opts.nativeOrigin) == "CFrame" then
            origin = opts.nativeOrigin
        end

        -- v160: the two cheap gates move ABOVE the aim solve. Auto Shoot runs
        -- this function every frame; during the 1.5 s confirmation window it
        -- used to pay for the predictor plus wall rays and then
        -- reject at the very end. Refusing before the work leaves the firing
        -- instant unchanged (the gate is time based) and makes an Auto Shoot
        -- that is holding fire cost almost nothing per frame.
        local pendingShot: any = state.mm2ShotFeedback.pending
        if not continuous and not nativeRedirect
            and type(pendingShot) == "table"
            and type(pendingShot.queuedAt) == "number"
            and os.clock() - pendingShot.queuedAt < SHOT_CONFIRM_WINDOW then
            local remaining: number = SHOT_CONFIRM_WINDOW
                - (os.clock() - pendingShot.queuedAt)
            return reject(string.format(
                "Shot gate: waiting for the previous shot (%.1f s left).",
                remaining))
        end
        -- Do not overwrite the feedback record for an earlier shot just to
        -- redirect a second click; telemetry is optional and cannot gate it.
        local trackFeedback: boolean = not continuous
            and not (nativeRedirect and type(pendingShot) == "table")
        if not remote or not remote:IsA("RemoteEvent") or not origin then
            return reject(not origin and "The gun has no usable origin."
                or "The murderer is unavailable or obstructed.")
        end

        local info: any = nil
        local reason: string? = nil
        if origin then
            -- v161: tell the predictor which way this shot will be authored.
            -- Bullet and CFrame keep separate measured lead corrections.
            local aimMode: string = silent and "CFrame" or "Bullet"
            local ok: boolean, endpoint: any, aimInfo: any, aimReason: any =
                pcall(prediction.gunAim, target, origin.Position, aimMode)
            if not ok then return reject("Aim or visibility check failed.") end
            if endpoint == nil then
                reason = aimReason or "no samples"
            else
                info = aimInfo
            end
        end

        -- Raw mode or continuous Auto Shoot / Auto Win: when gunAim has no
        -- samples yet on the first frame, fall back immediately to the
        -- target's current root position (+ short velocity lead) so the shot
        -- fires without waiting even a single second.
        if (useRaw or continuous or nativeRedirect) and not info and target.Character then
            local targetRoot: BasePart? = target.Character:FindFirstChild("HumanoidRootPart")
            if targetRoot then
                local vel: Vector3 = Vector3.zero
                pcall(function(): ()
                    vel = targetRoot.AssemblyLinearVelocity
                end)
                local fallbackEndpoint: Vector3 = targetRoot.Position + vel * 0.08
                info = {
                    endpoint = fallbackEndpoint,
                    velocity = vel,
                    leadSeconds = 0.08,
                    motionSource = "raw",
                    model = "raw",
                    sampleAge = 0,
                    samples = 0,
                    capturedPosition = targetRoot.Position,
                    rtt = 0,
                    echoDelay = 0,
                    rttEngine = "raw",
                    echoSamples = 0,
                    turnRate = 0,
                    brakeSeconds = 0,
                    leadCorrection = 0,
                    rootPosition = targetRoot.Position,
                }
            end
        end

        -- v161: the aim is final at this point; make sure its line does not
        -- cross a ghost body part (transparent + non-colliding) that every
        -- client wall check skips. Preference only: failure keeps the point.
        if info and origin and targetCharacter then
            local cleaner: Vector3?, cleanerPart: string? =
                avoidGhostAim(target, info, origin.Position)
            if cleaner then
                info.endpoint = cleaner
                info.targetPosition = cleaner
                info.ghostAvoided = cleanerPart
            end
        end

        -- v139.3: at arm's length the lead stops helping and starts hurting.
        -- The murderer charging at the measured 17 studs/s covers 1.7 studs in
        -- a 0.1 s lead: from 1.5 studs away the predicted point lands BEHIND
        -- our own muzzle, the authored segment inverts, and the shot is
        -- refused as "Invalid shot geometry" (reproduced in
        -- tools/test_gun_pointblank.py). Inside POINT_BLANK_STUDS the body
        -- covers the whole sight, so aim at it and drop the lead.
        if origin and info then
            local root: BasePart? = nil
            pcall(function(): ()
                root = targetCharacter:FindFirstChild("HumanoidRootPart") :: BasePart?
            end)
            local aimVector: Vector3 = info.endpoint - origin.Position
            local toRoot: Vector3 = root and (root.Position - origin.Position) or aimVector
            local inverted: boolean = aimVector.Magnitude < 0.05
                or aimVector:Dot(toRoot) <= 0
            if root and (inverted or aimVector.Magnitude < POINT_BLANK_STUDS) then
                info.capturedPosition = root.Position
                info.endpoint = root.Position
                info.aimPartChosen = "HumanoidRootPart"
                info.pointBlank = true
            end
        end

        local rejectMessages: {[string]: string} = {
            ["no body"] = "The murderer has no usable body to aim at.",
            ["no samples"] = "The murderer has no fresh motion samples yet.",
            ["stale sample"] = "The murderer's position is stale; waiting for a fresh sample.",
            ["unreachable"] = "The murderer is moving too fast to intercept.",
        }
        if not remote
            or not remote:IsA("RemoteEvent")
            or not origin
            or not info then
            local message: string = rejectMessages[reason or ""]
                or (
                    not origin
                        and "The gun has no usable origin."
                        or "The murderer is unavailable or obstructed."
                )
            return reject(message)
        end

        -- One authored request at a time (v106). Continuous Auto Shoot /
        -- Auto Win bypasses the 1.5s confirmation gate so it never stops
        -- firing while the Murderer is still alive. The gate itself is
        -- checked above, before any aim or raycast work.
        local function safeToSend(
            shotOrigin: CFrame,
            shotEnd: CFrame,
            muzzle: CFrame?
        ): boolean
            if gun.Parent ~= character or (not nativeRedirect and (gun.Enabled == false
                or gun:GetAttribute("Disabled") == true or gun:GetAttribute("Reloading") == true
                or gun:GetAttribute("IsReloading") == true)) then
                return reject("The gun is disabled or reloading.")
            end
            -- v139.4: "raw" means "do not wait for the near-target origin
            -- guard", NOT "do not look at walls". Raw/continuous Silent AIM
            -- used to return true here, so Auto Shoot and Auto Win (which call
            -- this path with continuous=true) fired with no wall check at all:
            -- the only geometry they ever saw was the near-target pass, never
            -- the gun's own line. When WallCheck is enabled, check the muzzle
            -- line too; Auto Farm explicitly opts out of that filter.
            if useRaw then
                if not useWallCheck or not isWeaponPathClear then
                    return true
                end
                local rawMuzzle: Vector3 = (muzzle or shotOrigin).Position
                local rawClear, rawBlocker = isWeaponPathClear(
                    rawMuzzle,
                    shotEnd.Position,
                    targetCharacter,
                    nil,
                    {ignoreTarget = true}
                )
                if not rawClear then
                    if type(rawBlocker) == "table" and trajectoryCalibration
                        and type(trajectoryCalibration.noteBlockedShot) == "function" then
                        pcall(trajectoryCalibration.noteBlockedShot,
                            trajectoryCalibration, rawBlocker)
                    end
                    return reject("A wall or another body blocks the shot.")
                end
                return true
            end
            -- The muzzle is the character's gun, never the authored origin.
            -- validateGunShot checks BOTH rays: the gun's line of sight and
            -- the authored segment. Passing the silent origin twice made them
            -- one ray, so a shot the gun could not see was cleared by a
            -- near-target origin (the v105 silent map hits), and a body
            -- between the gun and the target was never looked for.
            local muzzlePosition: Vector3 = (muzzle or shotOrigin).Position
            local ok, allowed, why, blocker, occluder = pcall(validateGunShot,target,character,targetCharacter,
                muzzlePosition,shotOrigin.Position,shotEnd.Position,useWallCheck)
            if not ok or not allowed then
                if type(blocker) == "table" and trajectoryCalibration
                    and type(trajectoryCalibration.noteBlockedShot) == "function" then
                    pcall(trajectoryCalibration.noteBlockedShot, trajectoryCalibration, blocker)
                end
                return reject(why or "Shot safety check failed.")
            end
            -- Cleared, but keep what the untouched ray met: a transparent solid
            -- short of the aim is exactly the case the client cannot see.
            if type(occluder) == "table" then
                lastGunOccluder = occluder
            else
                lastGunOccluder = nil
            end
            return true
        end

        -- Do not infer a reload from our last REQUEST: a rejected packet
        -- never proves the server fired. Native Tool/attribute readiness is
        -- checked above; no speculative 3.1-second local lockout.
        if silent then

            -- The configured construction, not a hardcoded one: the setting was
            -- only ever read for telemetry, so a saved "Top" or "Front" was
            -- silently replaced by "Through" on every shot.
            local variant: string = "Through"
            local configuredSpawn: any = mm2Settings.silentSpawn
            if configuredSpawn == "Top" or configuredSpawn == "Behind"
                or configuredSpawn == "Front" then
                variant = configuredSpawn
            end
            local shotOrigin: CFrame?, shotEnd: CFrame?, offset: number,
                sweep: number, resolvedVariant: string
            if info.pointBlank and origin then
                -- v139.3: at arm's length the "Through" construction pierces
                -- along the target's own motion, and a murderer charging at us
                -- makes that direction point at US: the authored segment ended
                -- on our own muzzle (muzzle 0.5, endpoint 0.5) and every shot
                -- was refused as "Invalid shot geometry". Point blank needs no
                -- trick at all -- the muzzle already sees the body.
                shotOrigin = origin
                shotEnd = CFrame.new(info.endpoint)
                offset = 0
                sweep = 0
                resolvedVariant = "PointBlank"
            else
                shotOrigin, shotEnd, offset, sweep, resolvedVariant =
                    buildSilentShot(target, info, origin, variant)
            end
            if not shotOrigin or not shotEnd then
                if continuous and target.Character then
                    local tr: BasePart? = target.Character:FindFirstChild("HumanoidRootPart")
                    if tr then
                        local ep: Vector3 = info.endpoint or tr.Position
                        local capPt: Vector3 = info.capturedPosition or tr.Position
                        local startPt: Vector3 = capPt + Vector3.new(0, 1.2, -1.5)
                        shotOrigin = CFrame.lookAt(startPt, ep)
                        shotEnd = CFrame.new(ep)
                        offset = 1.5
                        sweep = 0
                        resolvedVariant = "Through"
                    end
                end
                if not shotOrigin or not shotEnd then
                    return reject("Could not construct the silent shot.")
                end
            end
            variant = resolvedVariant
            -- A near-target origin inside a body can make the engine miss that
            -- containing body. Fall back to the real muzzle, never a bypass.
            -- Raw mode skips this check: the shot fires regardless.
            if not useRaw
                and not isWeaponPathClear(shotOrigin.Position,shotEnd.Position,targetCharacter,nil,{ignoreTarget=true}) then
                shotOrigin,shotEnd = origin,CFrame.new(info.endpoint)
                variant = "Muzzle"
                offset = 0
                sweep = 0
            end
            if not safeToSend(shotOrigin,shotEnd,origin) then return false end
            if trackFeedback then
                pcall(state.mm2ShotFeedback.queue, target, gun, shotOrigin, shotEnd)
            end

            state.lastSilentShotInfo = {
                variant = variant,
                at = os.clock(),
                offset = math.floor(offset * 10) / 10,
            }
            if opts.redirect then
                opts.redirect(shotOrigin, shotEnd)
            else
                remote:FireServer(shotOrigin, shotEnd)
            end

            local function recordSilent(): ()
                pcall(function(): ()
                    if not trajectoryCalibration or type(trajectoryCalibration.noteAuthoredShot) ~= "function" then return end
                    trajectoryCalibration:noteAuthoredShot({
                        kind = "gun",
                        source = nativeRedirect and "native-redirect" or "menu",
                        tool = gun,
                        toolName = gun and gun.Name or "Gun",
                        originPos = shotOrigin.Position,
                        aimPos = shotEnd.Position,
                        target = target,
                        silent = true,
                        variant = variant,
                        offset = offset,
                        settings = {mode = "Silent", spawn = variant,
                            sweep = sweep, wallCheck = useWallCheck,
                            leadSeconds = info.leadSeconds, motionSource = info.motionSource,
                            model = info.model,
                            sampleAge = info.sampleAge, samples = info.samples,
                            referencePoint = info.capturedPosition and {info.capturedPosition.X, info.capturedPosition.Y, info.capturedPosition.Z},
                            predictedPoint = {info.endpoint.X, info.endpoint.Y, info.endpoint.Z},
                            sampledVelocity = {info.velocity.X, info.velocity.Y, info.velocity.Z},
                            rtt = info.rtt, echoDelay = info.echoDelay,
                            rttEngine = info.rttEngine, rttData = info.rttData,
                            occluder = lastGunOccluder,

            if not okCaller or fromExecutor ~= false then

            local gun: Tool? = if okGun then foundGun else nil

                return predecessor(remote, ...)
            end
            local okRemote: boolean, shootRemote: any = pcall(function(): any
                return gun:FindFirstChild("Shoot")
            end)
            if not okRemote or remote ~= shootRemote then return predecessor(remote, ...) end
            local arguments: any = table.pack(...)
            if typeof(arguments[1]) ~= "CFrame" or typeof(arguments[2]) ~= "CFrame" then
                return predecessor(remote, ...)
            end

            local shotEnd: CFrame? = nil

                lastRejectReason = ""
                return fireGunAtTarget(getSelectedShootTarget(), {
                    quiet = true,
                    nativeRedirect = true,
                    nativeOrigin = arguments[1],
                    redirect = function(origin: CFrame, endpoint: CFrame): ()
                        shotOrigin, shotEnd = origin, endpoint

            end)
            if shotOrigin and shotEnd then
                arguments[1], arguments[2] = shotOrigin, shotEnd
                if ShootFeature then
                    pcall(function(): () ShootFeature:SetStatus("Native click redirected") end)
                end
                return predecessor(remote, table.unpack(arguments, 1, arguments.n))
            end

                pcall(function(): ()
                    ShootFeature:SetStatus(if not ok then "Native redirect failed; original click kept"
                        else (lastRejectReason ~= "" and lastRejectReason or "Native click unchanged"))
                end)
            end
            -- No target, blocked line, unsupported geometry or an error: keep
            -- the native click intact instead of blocking it or firing twice.
            return predecessor(remote, ...)
        end

        if type(state.newCClosure) == "function" then
            local ok: boolean, wrapped: any = pcall(state.newCClosure, callback)
            if ok and type(wrapped) == "function" then callback = wrapped end
        end

        if not ok or type(old) ~= "function" then return false end
        predecessor = old
        originalNamecall, shotHookWrapper, shotHookInstalled = old, callback, true
        return true
    end

        return true

    -- Get equipped or carried knife when local player is Murderer
    local function getKnife(): Tool?
        local character: Model? = LocalPlayer and LocalPlayer.Character
        if not character then
            return nil
        end
        if type(getPlayerWeapon) == "function" then
            local equippedKnife: Tool? = getPlayerWeapon(LocalPlayer, "Knife", true)
            if equippedKnife then
                return equippedKnife
            end
            local anyKnife: Tool? = getPlayerWeapon(LocalPlayer, "Knife", false)
            if anyKnife then
                return anyKnife
            end
        end
        for _, item: Instance in ipairs(character:GetChildren()) do
            if item:IsA("Tool") then
                local name: string = item.Name:lower()
                if name:find("knife") or item:FindFirstChild("Events") then
                    local events: Instance? = item:FindFirstChild("Events")
                    if name:find("knife") or (events and events:FindFirstChild("KnifeThrown")) then
                        return item :: Tool
                    end
                end
            end
        end
        local backpack: Instance? = LocalPlayer:FindFirstChildOfClass("Backpack")
        if backpack then
            for _, item: Instance in ipairs(backpack:GetChildren()) do
                if item:IsA("Tool") then
                    local name: string = item.Name:lower()
                    local events: Instance? = item:FindFirstChild("Events")
                    if name:find("knife") or (events and events:FindFirstChild("KnifeThrown")) then
                        return item :: Tool
                    end
                end
            end
        end
        return nil
    end

    local originHandleTool: Tool? = nil
    local originHandle: BasePart? = nil
    local function getThrowOriginCFrame(knife: Tool?): CFrame?
        local character: Model? = LocalPlayer and LocalPlayer.Character
        if not character then
            return nil
        end
        if knife and knife.Parent == character then
            if originHandleTool ~= knife or not originHandle or originHandle.Parent ~= knife then
                local handle: Instance? = knife:FindFirstChild("Handle")
                originHandleTool = knife
                originHandle = handle and handle:IsA("BasePart") and (handle :: BasePart) or nil
            end
            if originHandle then return originHandle.CFrame end
        end
        local hand: Instance? = character:FindFirstChild("RightHand")
            or character:FindFirstChild("Right Arm")
            or character:FindFirstChild("HumanoidRootPart")
        if hand and hand:IsA("BasePart") then
            return (hand :: BasePart).CFrame
        end
        return nil
    end

    -- v159: WallCheck is now the only guarantee, so it has to be honest:
    --  * the shared helper is called with `strict`, which makes a fully
    --    transparent but SOLID barrier stop the blade. The gun's first pass
    --    skips those; the knife must not.
    --  * the segment is tested as a thin corridor, not one mathematical line,
    --    so a railing edge cannot slip between samples.
    --  * the body has to be visible from the hand RIGHT NOW, not only at the
    --    predicted point. A predicted point that peeks around a corner is not
    --    reachable by a straight blade, and aiming at it is what looked like
    --    "knife through the wall".
    local CORRIDOR_STUDS: number = 0.45
    local function raycastClear(
        from: Vector3,
        to: Vector3,
        ignoreList: {Instance},
        requireBody: Model?
    ): boolean
        local ok: boolean, clear: any = pcall(function(): boolean
            local params: RaycastParams = RaycastParams.new()
            params.FilterType = Enum.RaycastFilterType.Exclude
            params.FilterDescendantsInstances = ignoreList
            params.RespectCanCollide = true
            params.IgnoreWater = true
            local hit: RaycastResult? = workspace:Raycast(from, to - from, params)
            if requireBody then
                return hit ~= nil and hit.Instance:IsDescendantOf(requireBody)
            end
            return hit == nil
        end)
        return ok and clear == true
    end

    local function sharedPathClear(
        from: Vector3,
        to: Vector3,
        targetChar: Model?,
        requireBody: boolean
    ): boolean?
        if type(isWeaponPathClear) ~= "function" or not targetChar then
            return nil
        end
        local options: any = requireBody and { requireTarget = true } or { ignoreTarget = true }
        local ok: boolean, clear: any = pcall(isWeaponPathClear, from, to, targetChar, true, options)
        if not ok or type(clear) ~= "boolean" then
            return nil
        end
        return clear
    end

    local function ignoreListFor(targetChar: Model?): {Instance}
        local list: {Instance} = {}
        local localChar: Model? = LocalPlayer and LocalPlayer.Character
        if localChar then table.insert(list, localChar) end
        if targetChar then table.insert(list, targetChar) end
        return list
    end

    -- "I can see that body from the hand": the ray must END on the body.
    local function bodyVisible(originPos: Vector3, bodyPos: Vector3, targetChar: Model?): boolean
        if not settings.wallCheck then
            return true
        end
        local from: Vector3 = originPos
        local to: Vector3 = bodyPos
        if (to - from).Magnitude < 0.05 then
            return true
        end
        local shared: boolean? = sharedPathClear(from, to, targetChar, true)
        if shared ~= nil then
            return shared
        end
        return raycastClear(from, to, ignoreListFor(nil), targetChar)
    end

    -- "Nothing solid between the hand and that point": corridor of five rays.
    local function checkLineOfSight(originPos: Vector3, targetPos: Vector3, targetChar: Model?): boolean
        if not settings.wallCheck then
            return true
        end
        local direction: Vector3 = targetPos - originPos
        local length: number = direction.Magnitude
        if length < 0.05 then
            return true
        end
        local unit: Vector3 = direction / length
        local side: Vector3 = unit:Cross(Vector3.new(0, 1, 0))
        if side.Magnitude < 0.05 then
            side = Vector3.new(1, 0, 0)
        end
        side = side.Unit
        local up: Vector3 = side:Cross(unit).Unit
        local offsets: {Vector3} = {
            Vector3.zero,
            side * CORRIDOR_STUDS,
            side * -CORRIDOR_STUDS,
            up * CORRIDOR_STUDS,
            up * -CORRIDOR_STUDS,
        }
        for _index: number, offset: Vector3 in ipairs(offsets) do
            local from: Vector3 = originPos + offset
            local to: Vector3 = targetPos + offset
            -- Every corridor ray uses the same strict rule as the centre one.
            -- A raw raycast here would let an invisible decoration that the
            -- shared helper skips block the throw, which made WallCheck say no
            -- for geometry the blade would have flown through.
            local shared: boolean? = sharedPathClear(from, to, targetChar, false)
            if shared ~= nil then
                if not shared then
                    return false
                end
            elseif not raycastClear(from, to, ignoreListFor(targetChar), nil) then
                return false
            end
        end
        return true
    end

    -- Predict target intercept position for MM2 ThrowingKnife (96 studs/s linear projectile)
    -- Returns the point and the blade's flight time, which is what makes one
    -- target a sooner hit than another.
    -- `cheap` is for target SELECTION: it still requires the body to be
    -- visible, but skips the five-ray corridor, which is re-checked with the
    -- live origin right before the packet leaves.
    local function resolveKnifeAim(
        target: Player,
        originPos: Vector3,
        knife: Tool?,
        cheap: boolean?
    ): (Vector3?, number?)
        local character: Model? = target.Character
        if not character then
            return nil, nil
        end
        local root: BasePart? = character:FindFirstChild("HumanoidRootPart") :: BasePart?
        if not root then
            return nil, nil
        end
        -- Check the live body before doing the more expensive intercept/ground
        -- work; hidden candidates cannot become speculative shots.
        if not bodyVisible(originPos, root.Position, character) then
            return nil, nil
        end

        local speed: number = 96
        if knife then
            pcall(function(): ()
                local attrSpeed: any = knife:GetAttribute("ThrowSpeed")
                if type(attrSpeed) == "number" and attrSpeed >= 24 and attrSpeed <= 300 then
                    speed = attrSpeed
                end
            end)
        end

        local predicted: Vector3? = nil
        local aimInfo: any = nil
        if type(prediction) == "table" and type(prediction.knifeAim) == "function" then
            -- v139.4: the blade speed HAS to reach the predictor. It was read
            -- from the tool above and then only used by the local fallback, so
            -- every knife that is not the default 96 studs/s was solved with
            -- the wrong flight time even though the correct number was in
            -- hand.
            local okPred: boolean, aimPoint: any, info: any =
                pcall(prediction.knifeAim, target, originPos, speed)
            if okPred and typeof(aimPoint) == "Vector3" then
                predicted = aimPoint :: Vector3
                aimInfo = info
            end
        end

        if not predicted then
            local velocity: Vector3 = root.AssemblyLinearVelocity
            if velocity.X ~= velocity.X or velocity.Y ~= velocity.Y or velocity.Z ~= velocity.Z
                or math.abs(velocity.X) == math.huge or math.abs(velocity.Y) == math.huge
                or math.abs(velocity.Z) == math.huge then
                velocity = Vector3.zero
            end
            local latency: number = 0.06
            if type(getEstimatedLatency) == "function" then
                local okLat: boolean, l: any = pcall(getEstimatedLatency)
                if okLat and type(l) == "number" and l == l then
                    latency = math.clamp(l, 0, 0.5)
                end
            end
            -- This is only the fallback when the shared predictor is absent or
            -- cannot solve. Previously `velocity * latency` applied unbounded
            -- vertical lead BEFORE the later 3.5 stud drop clamp: a replicated
            -- -120 Y velocity at 0.5 s already aimed 60 studs below the body.
            -- Keep horizontal transport and cap the *entire* vertical lead.
            -- The body's actual height is never clamped against the thrower:
            -- an enemy on a lower floor still produces a downward shot.
            local horizVel: Vector3 = Vector3.new(velocity.X, 0, velocity.Z)
            local observed: Vector3 = root.Position + horizVel * latency
            local rel: Vector3 = observed - originPos
            local a: number = horizVel:Dot(horizVel) - speed * speed
            local b: number = 2 * rel:Dot(horizVel)
            local c: number = rel:Dot(rel)
            local flight: number? = nil
            if c < 0.0025 then
                flight = 0
            elseif math.abs(a) < 1e-6 then
                if b < -1e-6 then
                    flight = -c / b
                end
            else
                local disc: number = b * b - 4 * a * c
                if disc >= 0 then
                    local sqrtDisc: number = math.sqrt(disc)
                    local t1: number = (-b - sqrtDisc) / (2 * a)
                    local t2: number = (-b + sqrtDisc) / (2 * a)
                    if t1 > 0 then
                        flight = t1
                    end
                    if t2 > 0 and (not flight or t2 < flight) then
                        flight = t2
                    end
                end
            end
            if flight and flight == flight then
                flight = math.clamp(flight, 0, 1.5)
                local yLead: number = math.clamp(
                    velocity.Y * math.min(latency + flight, 0.28), -2.0, 4.0
                )
                predicted = observed + horizVel * flight + Vector3.new(0, yLead, 0)
            else
                local yLead: number = math.clamp(velocity.Y * math.min(latency, 0.28), -2.0, 4.0)
                predicted = observed + Vector3.new(0, yLead, 0)
            end
        end

        if not predicted then
            return nil, nil
        end

        local flight: number? = nil
        if aimInfo and type(aimInfo.flightSeconds) == "number" then
            flight = math.max(0, aimInfo.flightSeconds)
        else
            flight = (predicted - originPos).Magnitude / speed
        end

        -- The current body was already checked before prediction: a future
        -- point peeking around a corner cannot make a hidden body eligible.
        if cheap or checkLineOfSight(originPos, predicted, character) then
            return predicted, flight
        end

        -- Fallback visibility scan across torso/head/limbs if WallCheck blocks primary point
        --
        -- v139.4: only the TIME part of the predictor's displacement may be
        -- reused on another body part. `predicted - root.Position` is lead
        -- PLUS the torso/head offset the predictor itself sampled
        -- (`capturedPosition = last.position + aimOffset`), so adding it to
        -- UpperTorso or Head counted the torso height twice and aimed 2-4
        -- studs above the part it named. Using the predictor's own reference
        -- point keeps the lead and drops the duplicated height.
        local leadOffset: Vector3 = predicted - root.Position
        if aimInfo and typeof(aimInfo.capturedPosition) == "Vector3" then
            leadOffset = predicted - (aimInfo.capturedPosition :: Vector3)
        end
        local candidateNames: {string} = {
            "UpperTorso", "Torso", "HumanoidRootPart", "Head",
            "RightUpperArm", "LeftUpperArm", "Right Arm", "Left Arm",
        }
        for _, partName: string in ipairs(candidateNames) do
            local part: Instance? = character:FindFirstChild(partName)
            if part and part:IsA("BasePart") then
                local partPosition: Vector3 = (part :: BasePart).Position
                local candidatePoint: Vector3 = partPosition + leadOffset
                if bodyVisible(originPos, partPosition, character)
                    and (cheap or checkLineOfSight(originPos, candidatePoint, character)) then
                    return candidatePoint, flight
                end
            end
        end

        return nil, nil
    end

    -- Find best target for knife throw
    local function findTarget(): Player?
        local localChar: Model? = LocalPlayer and LocalPlayer.Character
        if not localChar then
            return nil
        end
        local localRoot: BasePart? = localChar:FindFirstChild("HumanoidRootPart") :: BasePart?
        if not localRoot then
            return nil
        end
        local knife: Tool? = getKnife()
        local originCF: CFrame? = getThrowOriginCFrame(knife)
        local originPos: Vector3 = originCF and originCF.Position or localRoot.Position

        local bestTarget: Player? = nil
        local bestScore: number = math.huge

        for _, player: Player in ipairs(Players:GetPlayers()) do
            if isValidTarget(player) then
                local targetChar: Model? = player.Character
                local targetRoot: BasePart? = targetChar
                    and targetChar:FindFirstChild("HumanoidRootPart") :: BasePart?
                if targetRoot then
                    local distance: number = (targetRoot.Position - originPos).Magnitude
                    if distance <= 250 then
                        local aimPoint: Vector3?, flight: number? =
                            resolveKnifeAim(player, originPos, knife, true)
                        if aimPoint then
                            -- v159: score by the blade's FLIGHT TIME, not raw
                            -- distance. The nearest body is not the soonest hit:
                            -- a target running towards the throw intercepts in
                            -- a fraction of the time a farther one standing
                            -- still does, and the sooner the blade lands, the
                            -- smaller every prediction error is.
                            local score: number = flight or (distance / 96)
                            if type(getPlayerRole) == "function" then
                                local role: string? = getPlayerRole(player)
                                if role == "Sheriff" or role == "Hero" then
                                    score -= 0.08
                                end
                            end
                            if score < bestScore then
                                bestScore = score
                                bestTarget = player
                            end
                        end
                    end
                end
            end
        end

        return bestTarget
    end

    -- The solve is throttled, but its result used to jump at each update. This is
    -- visual-only: the native redirects and throwKnife always solve separately
    -- and never read this smoothed point.
    local TRACER_SMOOTH_RATE: number = 26
    local TRACER_SNAP_DISTANCE: number = 20
    local tracerPoint: Vector3? = nil
    local tracerTarget: Player? = nil
    local tracerShown: boolean = false
    local tracerOpacity: number = 0
    local drawnOpacity: number = -1
    local drawnOrigin: Vector3? = nil
    local drawnPoint: Vector3? = nil

    local function hideTracerParts(): ()
        if tracerPart and tracerPart.Parent then tracerPart.Transparency = 1 end
        if targetDotPart and targetDotPart.Parent then targetDotPart.Transparency = 1 end
    end

    local function hideTracerVisuals(): ()
        -- Do not tween a stale point after the next visibility check fails.
        -- Reuse the two invisible parts if a target reappears; destroy them
        -- when the option/module is turned off.
        if tracerShown then pcall(hideTracerParts) end
        tracerShown = false
        tracerOpacity = 0
        drawnOpacity = -1
        tracerPoint = nil
        tracerTarget = nil
        drawnOrigin = nil
        drawnPoint = nil
    end

    local function drawTracerParts(
        originPos: Vector3, point: Vector3, delta: Vector3, dist: number,
        updateGeometry: boolean, opacity: number, updateOpacity: boolean
    ): ()
        if not tracerPart or not tracerPart.Parent then
            local beamPart: Part = Instance.new("Part")
            beamPart.Name = "WurstKnifeTracer"
            beamPart.Anchored = true
            beamPart.CanCollide = false
            beamPart.CanQuery = false
            beamPart.CanTouch = false
            beamPart.Material = Enum.Material.Neon
            beamPart.Color = Color3.fromRGB(255, 65, 65)
            beamPart.Transparency = 1
            beamPart.Parent = workspace
            tracerPart = beamPart
            tracerShown = false
        end
        if not targetDotPart or not targetDotPart.Parent then
            local dot: Part = Instance.new("Part")
            dot.Name = "WurstKnifeAimDot"
            dot.Shape = Enum.PartType.Ball
            dot.Size = Vector3.new(1.1, 1.1, 1.1)
            dot.Anchored = true
            dot.CanCollide = false
            dot.CanQuery = false
            dot.CanTouch = false
            dot.Material = Enum.Material.Neon
            dot.Color = Color3.fromRGB(255, 220, 60)
            dot.Transparency = 1
            dot.Parent = workspace
            targetDotPart = dot
            tracerShown = false
        end
        if updateGeometry then
            tracerPart.Size = Vector3.new(0.14, 0.14, dist)
            tracerPart.CFrame = CFrame.lookAt(originPos + delta * 0.5, point)
            targetDotPart.CFrame = CFrame.new(point)
        end
        if updateOpacity or not tracerShown then
            tracerPart.Transparency = 1 - 0.75 * opacity
            targetDotPart.Transparency = 1 - 0.85 * opacity
            tracerShown = true
        end
    end

    local function updateTracerVisual(
        originPos: Vector3?, aimPos: Vector3?, target: Player?, deltaTime: number?
    ): ()
        if not settings.enabled or not settings.knifeTracer then
            hideTracerVisuals()
            destroyTracerVisuals()
            return
        end
        if not originPos or not aimPos or not target then
            hideTracerVisuals()
            return
        end
        local dt: number = tonumber(deltaTime) or (1 / 60)
        if dt ~= dt then dt = 1 / 60 end
        local alpha: number = 1 - math.exp(-TRACER_SMOOTH_RATE * math.clamp(dt, 0, 0.25))
        local previous: Vector3? = tracerPoint
        if not previous or tracerTarget ~= target
            or (aimPos - previous).Magnitude > TRACER_SNAP_DISTANCE then
            -- Never sweep a line across different players or a teleport. A
            -- short fade-in still softens the appearance of the new point.
            tracerPoint = aimPos
            tracerOpacity = 0
        else
            local nextPoint: Vector3 = previous + (aimPos - previous) * alpha
            tracerPoint = (aimPos - nextPoint).Magnitude < 0.002 and aimPos or nextPoint
        end
        tracerOpacity += (1 - tracerOpacity) * alpha
        if tracerOpacity > 0.998 then tracerOpacity = 1 end
        tracerTarget = target
        local point: Vector3 = tracerPoint :: Vector3
        local delta: Vector3 = point - originPos
        local dist: number = delta.Magnitude
        if dist < 0.25 or dist > 400 then
            hideTracerVisuals()
            return
        end
        local missing: boolean = not tracerPart or not tracerPart.Parent
            or not targetDotPart or not targetDotPart.Parent
        local geometryChanged: boolean = missing or not drawnOrigin or not drawnPoint
            or (originPos - (drawnOrigin :: Vector3)).Magnitude >= 0.002
            or (point - (drawnPoint :: Vector3)).Magnitude >= 0.002
        local opacityChanged: boolean = missing or not tracerShown
            or math.abs(tracerOpacity - drawnOpacity) >= 0.002
        if not geometryChanged and not opacityChanged then return end
        -- Named callback: unlike the old per-frame pcall(closure), this does
        -- not allocate a fresh closure every frame. No raycasts/tweens here.
        local ok: boolean = pcall(
            drawTracerParts, originPos, point, delta, dist,
            geometryChanged, tracerOpacity, opacityChanged
        )
        if ok then
            if geometryChanged then
                drawnOrigin = originPos
                drawnPoint = point
            end
            if opacityChanged then drawnOpacity = tracerOpacity end
        else
            pcall(hideTracerParts)
            tracerShown = false
            drawnOpacity = -1
            drawnOrigin = nil
            drawnPoint = nil
        end
    end

    -- Throw knife at predicted target using MM2's Knife.Events.KnifeThrown remote
    -- v139.2: the equip is a round trip, the throw is not. Firing the remote
    -- in the same frame as EquipTool sends a throw from a tool the server still
    -- believes is in the backpack: in the saved session 13 auto throws produced
    -- 9 observed blades (4 vanished) and the 4 that did fire used the RightHand
    -- fallback origin instead of the knife handle. So: equip once, wait for the
    -- tool to actually be in the character, and only then throw.
    local equipRequestedAt: number = -math.huge
    local equipRequestedTool: Tool? = nil

    local function ensureEquipped(knife: Tool, humanoid: Humanoid): boolean
        local character: Model? = LocalPlayer and LocalPlayer.Character
        if not character then
            return false
        end
        if knife.Parent == character then
            equipRequestedAt = -math.huge
            equipRequestedTool = nil
            return true
        end
        local now: number = os.clock()
        -- Re-ask every 0.25 s at most; EquipTool while the previous request is
        -- still travelling just stacks another equip request.
        if equipRequestedTool ~= knife or now - equipRequestedAt >= 0.25 then
            equipRequestedAt = now
            equipRequestedTool = knife
            pcall(function(): ()
                humanoid:EquipTool(knife)
            end)
        end
        return false
    end

    local function throwKnife(target: Player?, knife: Tool): boolean
        local character: Model? = LocalPlayer and LocalPlayer.Character
        if not character then
            return false
        end
        local humanoid: Humanoid? = character:FindFirstChildOfClass("Humanoid")
        if not humanoid or humanoid.Health <= 0 or not ensureEquipped(knife, humanoid) then
            return false
        end
        local originCF: CFrame? = getThrowOriginCFrame(knife)
        if not originCF then
            return false
        end
        -- v159: no lane fallback at all. Without a visible target there is
        -- nothing to throw at: a blade is not a homing projectile, and a throw
        -- into an opening only spent the knife.
        if not target or not isValidTarget(target) then
            return false
        end
        local aimPos: Vector3? = resolveKnifeAim(target, originCF.Position, knife)
        if not aimPos then
            return false
        end

        local fired: boolean = false
        local noted: boolean = false
        local function noteThrowAttempt(): ()
            if noted then return end
            noted = true
            local recorder: any = core and core.trajectoryCalibration
            if recorder and type(recorder.noteAuthoredShot) == "function" then
                local root: BasePart? = target.Character
                    and target.Character:FindFirstChild("HumanoidRootPart") :: BasePart?
                pcall(recorder.noteAuthoredShot, recorder, {
                    kind = "knife", tool = knife, toolName = knife.Name,
                    originPos = originCF.Position, aimPos = aimPos, target = target,
                    aimVelocity = root and root.AssemblyLinearVelocity or nil,
                    source = "menu-auto-throw",
                    settings = {mode = "Auto Knife Throw", wallCheck = settings.wallCheck},
                })
            end
        end
        local eventsFolder: Instance? = knife:FindFirstChild("Events")
        local knifeThrownRemote: Instance? = eventsFolder and eventsFolder:FindFirstChild("KnifeThrown")
        if knifeThrownRemote and knifeThrownRemote:IsA("RemoteEvent") then
            noteThrowAttempt()
            local okFire: boolean = pcall(function(): ()
                (knifeThrownRemote :: RemoteEvent):FireServer(originCF, CFrame.new(aimPos))
            end)
            if okFire then
                fired = true
            end
        end

        if not fired then
            pcall(function(): ()
                local rs: Instance? = game:GetService("ReplicatedStorage")
                local remotes: Instance? = rs and rs:FindFirstChild("Remotes")
                local throwRemote: Instance? = remotes and remotes:FindFirstChild("Event")
                if throwRemote and throwRemote:IsA("RemoteEvent") then
                    noteThrowAttempt()
                    local dir: Vector3 = (aimPos - originCF.Position).Unit
                    local rem: RemoteEvent = throwRemote :: RemoteEvent
                    rem:FireServer("Throw", aimPos, dir)
                    fired = true
                end
            end)
        end

        return fired
    end

    -- Hook ReplicatedStorage.ClientServices.WeaponService so native PC & Mobile
    -- knife throws (Mouse, Keyboard E, Gamepad, Mobile TouchControls.RightBar.Throw)
    -- automatically redirect to the Knife Predictor's aim point!
    local function installWeaponServiceRedirect(): ()
        if hookedWeaponService then
            return
        end
        pcall(function(): ()
            local rs: Instance? = game:GetService("ReplicatedStorage")
            local clientServices: Instance? = rs and rs:FindFirstChild("ClientServices")
            local wsModule: Instance? = clientServices and clientServices:FindFirstChild("WeaponService")
            if not wsModule then
                return
            end
            local okReq: boolean, wsTable: any = pcall(require, wsModule)
            if not okReq or type(wsTable) ~= "table" then
                return
            end
            origGetMouseTargetCFrame = wsTable.GetMouseTargetCFrame
            origGetTargetPosition = wsTable.GetTargetPosition
            hookedWeaponService = wsTable

            if type(origGetMouseTargetCFrame) == "function" then
                wsTable.GetMouseTargetCFrame = function(self: any, ...: any): any
                    if settings.enabled then
                        local knife: Tool? = getKnife()
                        local character: Model? = LocalPlayer and LocalPlayer.Character
                        if knife and character and knife.Parent == character then
                            local originCF: CFrame? = getThrowOriginCFrame(knife)
                            local target: Player? = findTarget()
                            if originCF and target then
                                local aim: Vector3? = resolveKnifeAim(target, originCF.Position, knife)
                                if aim then return CFrame.new(aim) end
                            end
                        end
                    end
                    return origGetMouseTargetCFrame(self, ...)
                end
            end

            if type(origGetTargetPosition) == "function" then
                wsTable.GetTargetPosition = function(self: any, ...: any): any
                    if settings.enabled then
                        local knife: Tool? = getKnife()
                        local character: Model? = LocalPlayer and LocalPlayer.Character
                        if knife and character and knife.Parent == character then
                            local originCF: CFrame? = getThrowOriginCFrame(knife)
                            local target: Player? = findTarget()
                            if originCF and target then
                                local aim: Vector3? = resolveKnifeAim(target, originCF.Position, knife)
                                -- The captured MM2 WeaponService returns a CFrame here,
                                -- despite the method name. Preserve that native contract
                                -- just as GetMouseTargetCFrame does.
                                if aim then return CFrame.new(aim) end
                            end
                        end
                    end
                    return origGetTargetPosition(self, ...)
                end
            end
        end)
    end

    local function uninstallWeaponServiceRedirect(): ()
        if not hookedWeaponService then
            return
        end
        pcall(function(): ()
            if origGetMouseTargetCFrame then
                hookedWeaponService.GetMouseTargetCFrame = origGetMouseTargetCFrame
            end
            if origGetTargetPosition then
                hookedWeaponService.GetTargetPosition = origGetTargetPosition
            end
        end)
        hookedWeaponService = nil
        origGetMouseTargetCFrame = nil
        origGetTargetPosition = nil
    end

    -- v139.2: this loop used to run every frame and, for EVERY player in the
    -- lobby, ran the aim solver -- and the aim solver casts line-of-sight rays
    -- against the map, then up to 8 more per body part when the first one is
    -- blocked. Ten players meant dozens of raycasts per frame for a number
    -- that only changes a few times per second. The solve is now cached for
    -- 1/15 s when an automatic throw is due; a visual-only preview can use
    -- 1/10 s because the displayed point follows the observed root between
    -- solves. The actual throw always re-solves from the live handle.
    local AIM_CACHE_SECONDS: number = 1 / 15
    local TRACER_CACHE_SECONDS: number = 1 / 10
    local cachedTarget: Player? = nil
    local cachedAim: Vector3? = nil
    local cachedOrigin: CFrame? = nil
    local cachedVisualRoot: BasePart? = nil
    local cachedVisualPosition: Vector3? = nil
    local cachedAt: number = -math.huge

    local function solveAim(knife: Tool, throwDue: boolean): (Player?, Vector3?, CFrame?)
        local now: number = os.clock()
        local interval: number = throwDue and AIM_CACHE_SECONDS or TRACER_CACHE_SECONDS
        if now - cachedAt >= interval then
            cachedAt = now
            cachedTarget = findTarget()
            cachedAim = nil
            cachedVisualRoot = nil
            cachedVisualPosition = nil
            cachedOrigin = getThrowOriginCFrame(knife)
            if cachedOrigin and cachedTarget then
                cachedAim = resolveKnifeAim(cachedTarget, cachedOrigin.Position, knife)
                if cachedAim and settings.knifeTracer then
                    local character: Model? = cachedTarget.Character
                    local root: BasePart? = character
                        and character:FindFirstChild("HumanoidRootPart") :: BasePart?
                    if root then
                        cachedVisualRoot = root
                        cachedVisualPosition = root.Position
                    end
                end
            end
        elseif cachedOrigin then
            -- Origin only: the target's predicted point stays cached, but the
            -- handle moves with the character and the tracer must follow it.
            cachedOrigin = getThrowOriginCFrame(knife) or cachedOrigin
        end
        return cachedTarget, cachedAim, cachedOrigin
    end

    local function invalidateAim(): ()
        cachedAt = -math.huge
        cachedTarget = nil
        cachedAim = nil
        cachedOrigin = nil
        cachedVisualRoot = nil
        cachedVisualPosition = nil
    end

    -- getPlayerWeapon walks the character/backpack's children. A tracer-only
    -- heartbeat must not do that 60-240 times a second just to keep the line
    -- attached to an already equipped handle. Actual throws and native input
    -- still use a fresh getKnife(), and parent changes invalidate this cache.
    local loopKnife: Tool? = nil
    local loopKnifeParent: Instance? = nil
    local loopKnifeCharacter: Model? = nil
    local loopKnifeAt: number = -math.huge
    local function resetLoopKnife(): ()
        loopKnife = nil
        loopKnifeParent = nil
        loopKnifeCharacter = nil
        loopKnifeAt = -math.huge
        originHandleTool = nil
        originHandle = nil
    end
    local function getLoopKnife(now: number, throwDue: boolean): Tool?
        local character: Model? = LocalPlayer and LocalPlayer.Character
        if character ~= loopKnifeCharacter then resetLoopKnife() end
        loopKnifeCharacter = character
        if not throwDue then
            if loopKnife and loopKnife.Parent == character and character then
                return loopKnife
            end
            local age: number = now - loopKnifeAt
            if age >= 0 and age < TRACER_CACHE_SECONDS then
                if not loopKnife then return nil end
                if loopKnifeParent and loopKnife.Parent == loopKnifeParent then
                    return loopKnife
                end
            end
        end
        loopKnife = getKnife()
        loopKnifeParent = loopKnife and loopKnife.Parent or nil
        loopKnifeAt = now
        return loopKnife
    end

    local function tracerGoal(target: Player?, aimPos: Vector3?): Vector3?
        if not target or not aimPos then return nil end
        local root: BasePart? = cachedVisualRoot
        local captured: Vector3? = cachedVisualPosition
        if not root or not captured then return aimPos end
        if target ~= cachedTarget or root.Parent ~= target.Character then
            invalidateAim()
            return nil
        end
        -- Follow the already-observed root between the throttled solves. This is
        -- only a visual offset, not another prediction or a per-frame raycast.
        local moved: Vector3 = root.Position - captured
        if moved.Magnitude > 8 then
            invalidateAim()
            return nil
        end
        return aimPos + moved
    end

    -- Main loop for Knife Tracer and Auto Knife Throw
    local function knifeThrowLoop(deltaTime: number?): ()
        if not settings.enabled then
            setKnifeStatus(nil)
            hideTracerVisuals()
            destroyTracerVisuals()
            invalidateAim()
            return
        end

        local now: number = os.clock()
        local cooldown: number = math.clamp(tonumber(settings.delay) or 0, 0, 10)
        local wantThrow: boolean = settings.autoThrow
            and now - lastThrowAttempt >= cooldown

        -- Manual redirects solve on the native input path. Do not enumerate
        -- players, predict and raycast merely to update a status when both
        -- Auto Knife Throw and the tracer are off. Likewise, without a tracer
        -- no preview is needed during a successful throw's cooldown.
        if not settings.knifeTracer and not wantThrow then
            hideTracerVisuals()
            destroyTracerVisuals()
            if not settings.autoThrow then
                setKnifeStatus(nil)
                invalidateAim()
            end
            return
        end

        local knife: Tool? = getLoopKnife(now, wantThrow)
        if not knife then
            setKnifeStatus("Waiting for knife")
            hideTracerVisuals()
            invalidateAim()
            return
        end

        -- The cached pick only decides WHO gets the blade. throwKnife
        -- re-solves the aim from the live handle when it actually fires.
        local target, aimPos, originCF = solveAim(knife, wantThrow)
        if not aimPos then
            setKnifeStatus(target and "Target blocked · WallCheck" or "No visible target")
            hideTracerVisuals()
            return
        end
        setKnifeStatus("Predicted target")

        if settings.knifeTracer and originCF then
            updateTracerVisual(originCF.Position, tracerGoal(target, aimPos), target, deltaTime)
        else
            hideTracerVisuals()
            destroyTracerVisuals()
        end

        if wantThrow then
            if throwKnife(target, knife) then
                setKnifeStatus("Throw sent · hit unconfirmed")
                lastThrowAttempt = now
                -- A throw consumes the aim: the next frame re-solves instead of
                -- reusing a point measured before the blade left.
                invalidateAim()
            end
        end
    end

    -- Toggle function
    local function toggleKnifeThrow(enabled: boolean): ()
        settings.enabled = enabled
        disconnectFeatureConnection("MM2KnifeThrow")

        if not enabled then
            uninstallWeaponServiceRedirect()
            setKnifeStatus(nil)

            resetLoopKnife()
            return
        end

        installWeaponServiceRedirect()
        featureConnections.MM2KnifeThrow =
            TaskManager:Connect(knifeThrowLoop, nil, "MM2 Knife Throw")
    end

    -- Create the feature in Combat category
    KnifeThrowFeature = createUniversalFeature(

    addToggleOption(

    )

    addToggleOption(

        "Delay in seconds between automatic knife throws (0 to 10 seconds).",

    )

    addInformationOption(

    )

    addToggleOption(

        "Knife Tracer",

            if not settings.knifeTracer then
                hideTracerVisuals()
                destroyTracerVisuals()
            end
        end,
        "Smooth 3D preview from your hand to the predicted point. The actual throw is re-solved separately; the preview hides when no target passes WallCheck."
    )

    activeCleanup = function(): ()
        toggleKnifeThrow(false)
    end
    Module.Events = featureConnections
    Module.Initialized = true
    return Module
end

-- Internal MM2 engine only. Auto Farm and Auto Win keep their explicit fling

            return false

        return ok and result == true
    end

    -- SkidFling (zqyDSUWX / YARHM), the fling every working menu copies:
    --
    --   RootPart.CFrame      = CFrame.new(targetPart.Position) * offset * spin
    --   RootPart.Velocity    = Vector3.new(9e7, 9e7 * 10, 9e7)
    --   RootPart.RotVelocity = Vector3.new(9e8, 9e8, 9e8)
    --
    -- The local character is parked on the victim and given an absurd velocity
    -- while a finite-force BodyVelocity reinforces contact. The large impulse is
    -- applied briefly; the local body and other temporary state are restored
    -- afterwards. Actual transfer still depends on the server's physics.
    -- SkidFling adapted from the YARHM hub the user supplied
    -- (YARHM_Hub_KEYLESS_2026.txt, "Fling" -> SkidFling, originally by zqyDSUWX).
    -- MM2 callers supply a target; the engine keeps its power, prediction and retries.
    --
    -- The parts that matter and that our own port had wrong:
    --
    --   * The BodyVelocity named "EpixVel" drives at Vector3.new(9e8, 9e8, 9e8)
    --     with finite 9e9 MaxForce. It is a catapult, not a pin: the local body is
    --     thrown into the victim. With Velocity zero (our old version) nothing is
    --     transferred, the character just sits there and the camera flickers.
    --   * The pose is written through RootPart.CFrame *and*
    --     Character:SetPrimaryPartCFrame, then RootPart.Velocity/RotVelocity.
    --     No AssemblyLinearVelocity, no CustomPhysicalProperties: writing those
    --     every frame is what made our character look like it was fading.
    --   * Each pose is separated by task.wait() (Heartbeat, after physics), not
    --     RenderStepped.
    --   * Two pose sets: six poses while the victim is still (< 50 studs/s) and
    --     nine while it is already moving.
    -- Power 3 reproduces YARHM's numbers exactly (9e7 / 9e8); anything above
    -- scales BOTH textures of the trick (the pose velocity and the EpixVel pin),
    -- which is what makes the fling survive a laggy or fast-moving victim.
    local FLING_VELOCITY_BASE: Vector3 = Vector3.new(9e7, 9e7 * 10, 9e7)
    local FLING_ROT_VELOCITY_BASE: Vector3 = Vector3.new(9e8, 9e8, 9e8)
    local PIN_VELOCITY_BASE: Vector3 = Vector3.new(9e8, 9e8, 9e8)
    -- v180: finite force, like YARHM and SkidFling use. `math.huge` makes the
    -- solver fight itself on some servers (the pin "wins" every contact and the
    -- victim never receives the impulse), which is exactly the low success
    -- rate the former card was reporting.
    local PIN_FORCE: Vector3 = Vector3.new(9e9, 9e9, 9e9)

    -- v180's default 6 doubles the pose velocity against YARHM's baseline.
    -- The universal card/slider are gone; MM2 retains this fixed default.
    local FLING_POWER: number = 6
    local FLING_DURATION: number = 3
    local FLING_PREDICTION: number = 0.15
    local flingRetries: number = 2
    -- Temporary boost used by the second pass only (never persisted).
    local flingBoost: number = 1

    local function powerScale(): number
        return (FLING_POWER / 3) * flingBoost
    end

    local function scaledVelocity(base: Vector3): Vector3
        local scale: number = powerScale()
        return Vector3.new(base.X * scale, base.Y * scale, base.Z * scale)
    end

    local flingRunning: boolean = false
    local engineActive: boolean = true
    local fullFlingPin: any = nil
    local stepPhase: number = 0
    local stepAngle: number = 0
    local lastOldPos: CFrame? = nil
    local stepPin: any = nil
    local stepPinToken: number = 0

    local function setFlingRunning(running: boolean): ()
        flingRunning = running
        if activity and type(activity.set) == "function" then
            activity.set("fling", running)
        end
    end

    local function targetParts(targetPlayer: Player?): (Model?, Humanoid?, BasePart?, BasePart?, BasePart?)
        local targetCharacter: Model? = targetPlayer and targetPlayer.Character
        if not targetCharacter then
            return nil, nil, nil, nil, nil
        end
        local targetHumanoid: Humanoid? = targetCharacter:FindFirstChildOfClass("Humanoid")
        local targetRoot: BasePart? = targetHumanoid and targetHumanoid.RootPart or nil
        local targetHead: BasePart? = targetCharacter:FindFirstChild("Head") :: BasePart?
        return targetCharacter, targetHumanoid, targetRoot, targetHead, nil
    end

    -- FPos: the whole trick. Position on the victim part, then hand the assembly
    -- 9e7 and 9e8 so the contact solver launches the victim.
    local function fpos(
        character: Model,
        root: BasePart,
        basePart: BasePart,
        offset: CFrame,
        angles: CFrame,
        lead: Vector3?
    ): ()
        local origin: Vector3 = basePart.Position
        if lead then
            origin = origin + lead
        end
        local targetCFrame: CFrame = CFrame.new(origin) * offset * angles
        root.CFrame = targetCFrame
        character:SetPrimaryPartCFrame(targetCFrame)
        root.Velocity = scaledVelocity(FLING_VELOCITY_BASE)
        root.RotVelocity = scaledVelocity(FLING_ROT_VELOCITY_BASE)
    end

    -- Round trip / 2: how stale the server's copy of the victim is by the time it
    -- applies the pose we just wrote. With 300 ms of ping that is 150 ms of
    -- movement we have to lead with our own.
    local function transportSeconds(): number
        local ping: any = 0
        pcall(function(): ()
            local player: any = LocalPlayer
            if player and type(player.GetNetworkPing) == "function" then
                ping = player:GetNetworkPing()
            end
        end)
        if type(ping) ~= "number" or ping ~= ping or ping < 0 then
            ping = 0
        end
        -- GetNetworkPing returns seconds in current builds; older ones returned
        -- milliseconds. Anything above 2 is milliseconds.
        if ping > 2 then
            ping = ping / 1000
        end
        return math.clamp(ping * 0.5, 0, 0.5)
    end

    -- Where the victim will be when the server applies this pose: planar lead
    -- from its own velocity. Vertical lead is left out on purpose, it is the
    -- floor contact that carries the fling.
    local function leadFor(targetPart: BasePart?): Vector3?
        if not targetPart then
            return nil
        end
        local velocity: any = Vector3.zero
        pcall(function(): ()
            velocity = targetPart.AssemblyLinearVelocity
        end)
        if type(velocity) ~= "userdata" and type(velocity) ~= "table" then
            return nil
        end
        local seconds: number = math.clamp(
            transportSeconds() + FLING_PREDICTION,
            0,
            0.6
        )
        if seconds <= 0.001 then
            return nil
        end
        local lead: any = Vector3.new(velocity.X, 0, velocity.Z) * seconds
        return lead :: Vector3
    end

    -- The victim's own parts can be replaced (respawn, ragdoll, tool swap) while
    -- the fling runs; re-read them every pose instead of caching the first pair.
    local function liveTargetParts(
        targetPlayer: Player,
        targetCharacter: Model
    ): (Model?, Humanoid?, BasePart?, BasePart?)
        local character: Model? = targetPlayer.Character or targetCharacter
        if not character or not character.Parent then
            return nil, nil, nil, nil
        end
        local humanoid: Humanoid? = character:FindFirstChildOfClass("Humanoid")
        local root: BasePart? = humanoid and humanoid.RootPart or nil
        local head: BasePart? = character:FindFirstChild("Head") :: BasePart?
        return character, humanoid, root, head
    end

    local function partSpeed(part: BasePart?): number
        return if part then part.Velocity.Magnitude else 0
    end

    -- cloneReference() gives the host a DIFFERENT reference for the same
    -- service, so "targetPlayer.Parent ~= Players" can stop the fling early.
    -- Resolve the player through the service instead.
    local function playerStillHere(player: Player): boolean
        return Players:FindFirstChild(player.Name) ~= nil
    end

    -- SFBasePart: YARHM's choreography (six poses while the victim is slow, ten
    -- while it is already moving), with the menu's adaptations:
    --
    --   * MIN_POSES before the "it is flying, stop" check, so one stale
    --     replication frame does not end the fling early;
    --   * the victim's parts are re-read on every pose (respawn, ragdoll);
    --   * each pose is led by the victim's own velocity times half the ping, so a
    --     300 ms connection still lands the contact.
    local MIN_POSES: number = 12

    local function sfBasePart(
        character: Model,
        humanoid: Humanoid,
        root: BasePart,
        basePart: BasePart,
        targetCharacter: Model,
        targetPlayer: Player,
        targetHumanoid: Humanoid?,
        targetRoot: BasePart?,
        timeToWait: number,
        shouldStop: (() -> boolean)?,
        stats: any
    ): ()
        local startedAt: number = os.clock()
        local angle: number = 0
        local liveBase: BasePart = basePart

        local function currentBasePart(): BasePart?
            local characterNow: Model?, humanoidNow: Humanoid?, rootNow: BasePart?, headNow: BasePart? =
                liveTargetParts(targetPlayer, targetCharacter)
            if not characterNow then
                return nil
            end
            targetCharacter = characterNow
            targetHumanoid = humanoidNow or targetHumanoid
            targetRoot = rootNow or targetRoot
            local candidate: BasePart? = rootNow
            if rootNow and headNow then
                if (rootNow.Position - headNow.Position).Magnitude > 5 then
                    candidate = headNow
                end
            elseif headNow then
                candidate = headNow
            end
            if candidate and candidate.Parent then
                liveBase = candidate
            end
            return liveBase
        end

        local function pose(offset: CFrame, angles: CFrame): boolean
            -- Stop between individual yielded poses, not only once per group.
            if shouldStop and shouldStop() then return false end
            local part: BasePart? = currentBasePart()
            if not part then
                return false
            end
            fpos(character, root, part, offset, angles, leadFor(part))
            stats.poses += 1
            task.wait()
            return true
        end

        repeat
            if not root or not targetHumanoid then
                break
            end
            if shouldStop and shouldStop() then
                break
            end
            local part: BasePart? = currentBasePart()
            if not part then
                break
            end
            local speed: number = partSpeed(part)
            local moveDirection: Vector3 = Vector3.zero
            local walkSpeed: number = 16
            pcall(function(): ()
                moveDirection = targetHumanoid.MoveDirection
                walkSpeed = targetHumanoid.WalkSpeed
            end)
            if speed < 50 then
                angle = angle + 100
                local lead: Vector3 = moveDirection * (speed / 1.25)
                local spin: CFrame = CFrame.Angles(math.rad(angle), 0, 0)
                pose(CFrame.new(0, 1.5, 0) + lead, spin)
                pose(CFrame.new(0, -1.5, 0) + lead, spin)
                pose(CFrame.new(2.25, 1.5, -2.25) + lead, spin)
                pose(CFrame.new(-2.25, -1.5, 2.25) + lead, spin)
                pose(CFrame.new(0, 1.5, 0) + moveDirection, spin)
                pose(CFrame.new(0, -1.5, 0) + moveDirection, spin)
            else
                local downLead: number = partSpeed(targetRoot) / 1.25
                pose(CFrame.new(0, 1.5, walkSpeed), CFrame.Angles(math.rad(90), 0, 0))
                pose(CFrame.new(0, -1.5, -walkSpeed), CFrame.Angles(0, 0, 0))
                pose(CFrame.new(0, 1.5, walkSpeed), CFrame.Angles(math.rad(90), 0, 0))
                pose(CFrame.new(0, 1.5, downLead), CFrame.Angles(math.rad(90), 0, 0))
                pose(CFrame.new(0, -1.5, -downLead), CFrame.Angles(0, 0, 0))
                pose(CFrame.new(0, 1.5, downLead), CFrame.Angles(math.rad(90), 0, 0))
                pose(CFrame.new(0, -1.5, 0), CFrame.Angles(math.rad(90), 0, 0))
                pose(CFrame.new(0, -1.5, 0), CFrame.Angles(0, 0, 0))
                pose(CFrame.new(0, -1.5, 0), CFrame.Angles(math.rad(-90), 0, 0))
                pose(CFrame.new(0, -1.5, 0), CFrame.Angles(0, 0, 0))
            end
        until (stats.poses >= MIN_POSES and partSpeed(liveBase) > 500)
            or not liveBase.Parent
            or targetPlayer.Character ~= targetCharacter
            or not playerStillHere(targetPlayer)
            or (targetHumanoid ~= nil and targetHumanoid.Sit)
            or humanoid.Health <= 0
            or os.clock() > startedAt + timeToWait
            or stats.poses >= 600
    end

    local function createPin(root: BasePart): any
        local pin: any = Instance.new("BodyVelocity")
        pin.Name = "EpixVel"
        pin.Velocity = scaledVelocity(PIN_VELOCITY_BASE)
        pin.MaxForce = PIN_FORCE
        pin.Parent = root
        return pin
    end

    local function destroyPin(pin: any): ()
        if pin then pin:Destroy() end
    end

    -- Per-frame caller (Auto Win's Fling Sheriff). Same FPos pose, with a pin
    -- that is cleaned up 0.5 s after the last call so the local body is never
    -- left with an infinite-force catapult attached.
    local function flingSingleStep(
        character: Model,
        humanoid: Humanoid,
        root: BasePart,
        targetPlayer: Player,
        _savedCollide: any,
        _savedProps: any
    ): boolean
        local targetCharacter: Model?, targetHumanoid: Humanoid?, targetRoot: BasePart?, targetHead: BasePart? =
            targetParts(targetPlayer)
        if not targetCharacter or not targetHead and not targetRoot then
            return false
        end
        if targetHumanoid and (targetHumanoid.Health <= 0 or targetHumanoid.Sit) then
            return false
        end
        if not targetHumanoid then
            return false
        end
        local basePart: BasePart = targetRoot or targetHead :: BasePart
        if targetRoot and targetHead then
            if (targetRoot.Position - targetHead.Position).Magnitude > 5 then
                basePart = targetHead
            end
        end
        if activity and type(activity.set) == "function" then
            activity.set("fling", true)
        end
        if not stepPin then
            stepPin = createPin(root)
        end
        stepPinToken += 1
        local myToken: number = stepPinToken
        task.delay(0.5, function(): ()
            if myToken == stepPinToken then
                destroyPin(stepPin)
                stepPin = nil
                if not flingRunning and activity and type(activity.set) == "function" then
                    activity.set("fling", false)
                end
            end
        end)
        stepAngle = (stepAngle + 100) % 360
        local speed: number = 0
        pcall(function(): ()
            speed = basePart.Velocity.Magnitude
        end)
        if speed < 50 then
            stepPhase = (stepPhase + 1) % 6
            local offsets: {CFrame} = {
                CFrame.new(0, 1.5, 0),
                CFrame.new(0, -1.5, 0),
                CFrame.new(2.25, 1.5, -2.25),
                CFrame.new(-2.25, -1.5, 2.25),
                CFrame.new(0, 1.5, 0),
                CFrame.new(0, -1.5, 0),
            }
            fpos(character, root, basePart, offsets[stepPhase + 1], CFrame.Angles(math.rad(stepAngle), 0, 0))
        else
            fpos(character, root, basePart, CFrame.new(0, 1.5, 0), CFrame.Angles(math.rad(90), 0, 0))
        end
        return true
    end

    local function performFling(targetPlayer: Player?, options: {[string]: any}?): boolean
        if not Module.Initialized or not targetPlayer then
            return false
        end
        if protectedCheck(targetPlayer) then
            notify("player protected")
            return false
        end
        local silent: boolean = type(options) == "table" and options.silent == true
        local maxDuration: number = (type(options) == "table" and tonumber(options.duration))
            or FLING_DURATION
        local requestedStop: (() -> boolean)? = type(options) == "table"
            and type(options.shouldStop) == "function"
            and options.shouldStop
            or nil
        local function shouldStop(): boolean
            return not engineActive or (requestedStop ~= nil and requestedStop())
        end
        -- Auto Farm opts out of returning to an old round after cancellation.
        -- Other callers (including Auto Win) retain their existing return policy.
        local skipReturnOnStop: boolean = type(options) == "table" and options.skipReturnOnStop == true
        local returnCFrame: CFrame? = type(options) == "table"
            and typeof(options.returnCFrame) == "CFrame"
            and options.returnCFrame
            or nil

        if flingRunning then
            if not silent then
                notify("fling already running")
            end
            return false
        end

        setFlingRunning(true)

        local stats: any = {poses = 0, speed = 0, pinned = false, retries = 0}

        local function runFling(): boolean
            local character: Model?, humanoid: Humanoid?, root: BasePart? = getCharacterParts()
            if not character or not humanoid or not root then
                setFlingRunning(false)
                if not silent then
                    notify("character not ready")
                end
                return false
            end

            local targetCharacter: Model?, targetHumanoid: Humanoid?, targetRoot: BasePart?, targetHead: BasePart? =
                targetParts(targetPlayer)
            if not targetCharacter or not targetCharacter:FindFirstChildWhichIsA("BasePart") then
                setFlingRunning(false)
                if not silent then
                    notify("No valid character of said target player. May have died.")
                end
                return false
            end
            if not targetHumanoid then
                setFlingRunning(false)
                if not silent then
                    notify("the target has no humanoid")
                end
                return false
            end

            if partSpeed(root) < 50 then
                lastOldPos = root.CFrame
            end
            local oldPos: CFrame = returnCFrame or lastOldPos or root.CFrame

            local savedDestroyHeight: any = nil
            pcall(function(): ()
                savedDestroyHeight = workspace.FallenPartsDestroyHeight
            end)
            local camera: Camera? = workspace.CurrentCamera

            local pin: any = nil
            local success: boolean, errorMessage: any = pcall(function(): ()
                if camera then
                    pcall(function(): ()
                        local accessory: any = targetCharacter:FindFirstChildOfClass("Accessory")
                        camera.CameraSubject = targetHead
                            or (accessory and accessory:FindFirstChild("Handle"))
                            or targetHumanoid
                    end)
                end

                -- FallenPartsDestroyHeight = 0/0 while the local body crosses the
                -- map at 9e8 studs/s; always restored below.
                pcall(function(): ()
                    workspace.FallenPartsDestroyHeight = 0 / 0
                end)

                pin = createPin(root)
                fullFlingPin = pin
                stats.pinned = true
                pcall(function(): ()
                    humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, false)
                end)

                local basePart: BasePart? = nil
                if targetRoot and targetHead then
                    if (targetRoot.Position - targetHead.Position).Magnitude > 5 then
                        basePart = targetHead
                    else
                        basePart = targetRoot
                    end
                elseif targetRoot then
                    basePart = targetRoot
                elseif targetHead then
                    basePart = targetHead
                end

                -- Retries can switch parts and temporarily boost force.
                local maxPasses: number = math.clamp(
                    math.floor(tonumber(flingRetries) or 0),
                    0,
                    3
                ) + 1
                local used: BasePart? = nil
                local pass: number = 0
                while pass < maxPasses do
                    if shouldStop and shouldStop() then break end
                    pass += 1
                    local partNow: BasePart? = nil
                    if pass == 1 then
                        partNow = basePart
                    else
                        local _c: Model?, _h: Humanoid?, rootNow: BasePart?, headNow: BasePart? =
                            liveTargetParts(targetPlayer, targetCharacter)
                        targetHumanoid = _h or targetHumanoid
                        targetRoot = rootNow or targetRoot
                        local primary: BasePart? = rootNow
                        if rootNow and headNow then
                            if (rootNow.Position - headNow.Position).Magnitude > 5 then
                                primary = headNow
                            end
                        elseif headNow then
                            primary = headNow
                        end
                        if primary == used then
                            partNow = if primary == rootNow then headNow else rootNow
                        else
                            partNow = primary
                        end
                        if partNow then
                            flingBoost = 1.5
                            stats.retries = (stats.retries or 0) + 1
                        end
                    end
                    if not partNow or not partNow.Parent then
                        flingBoost = 1
                        break
                    end
                    used = partNow
                    sfBasePart(
                        character,
                        humanoid,
                        root,
                        partNow,
                        targetCharacter,
                        targetPlayer,
                        targetHumanoid,
                        targetRoot,
                        -- Retries are half as long: they exist to catch a late
                        -- server contact, not to double the time spent posing.
                        if pass > 1 then maxDuration * 0.5 else maxDuration,
                        shouldStop,
                        stats
                    )
                    flingBoost = 1
                    stats.speed = math.max(
                        stats.speed or 0,
                        partSpeed(targetRoot),
                        partSpeed(targetHead)
                    )
                    if stats.speed > 100 then
                        break
                    end
                end
                stats.speed = math.max(
                    stats.speed or 0,
                    partSpeed(targetRoot),
                    partSpeed(targetHead)
                )
            end)

            -- The pin and temporary world state must be restored even if a pose
            -- failed. No secondary impulse is applied to the target.
            flingBoost = 1
            destroyPin(pin)
            if fullFlingPin == pin then fullFlingPin = nil end
            pcall(function(): ()
                humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, true)
            end)
            pcall(function(): ()
                workspace.FallenPartsDestroyHeight = savedDestroyHeight
            end)
            local finalCharacter: Model?, finalHumanoid: Humanoid?, finalRoot: BasePart? =
                getCharacterParts()
            local cancelled: boolean = not engineActive or (skipReturnOnStop and shouldStop())
            if camera then
                pcall(function(): ()
                    camera.CameraSubject = if cancelled then (finalHumanoid or humanoid) else humanoid
                end)
            end

            local function stopBodyMotion(body: Model): ()
                for _, descendant: Instance in ipairs(body:GetChildren()) do
                    if descendant:IsA("BasePart") then
                        local bodyPart: BasePart = descendant :: BasePart
                        bodyPart.Velocity = Vector3.zero
                        bodyPart.RotVelocity = Vector3.zero
                    end
                end
            end
            if cancelled then
                -- A server teleport/new round may already have moved the body.
                -- Release the old fling velocity, but never move a new body or
                -- return the lobby body to the previous map's saved CFrame.
                pcall(stopBodyMotion, character)
            elseif finalCharacter and finalRoot and finalRoot.Parent then
                local attempts: number = 0
                repeat
                    if not engineActive or (skipReturnOnStop and shouldStop()) then
                        pcall(stopBodyMotion, character)
                        break
                    end
                    attempts += 1
                    pcall(function(): ()
                        finalRoot.CFrame = oldPos * CFrame.new(0, 0.5, 0)
                        finalCharacter:SetPrimaryPartCFrame(oldPos * CFrame.new(0, 0.5, 0))
                        if finalHumanoid then
                            pcall(function(): ()
                                finalHumanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
                            end)
                        end
                        stopBodyMotion(finalCharacter)
                    end)
                    task.wait()
                until (finalRoot.Position - oldPos.Position).Magnitude < 25 or attempts >= 60
            end

            setFlingRunning(false)
            local touchedTarget: boolean = stats.speed > 100
            pcall(function(): ()
                state.mm2LastFling = {
                    target = tostring(targetPlayer.Name),
                    poses = stats.poses,
                    speed = stats.speed,
                    pinned = stats.pinned,
                    retries = stats.retries,
                    power = powerScale(),
                    ok = success,
                }
            end)
            if not Module.Initialized then
                return touchedTarget
            end
            if not silent then
                local speedText: string = tostring(math.floor(stats.speed))
                if not success then
                    notify("fling failed: " .. tostring(errorMessage))
                elseif touchedTarget then
                    notify(string.format(
                        "fling: %s sent flying (%s studs/s, %d poses%s)",
                        tostring(targetPlayer.Name),
                        speedText,
                        stats.poses,
                        (stats.retries or 0) > 0
                            and string.format(", %d retries", stats.retries)
                            or ""
                    ))
                else
                    notify(string.format(
                        "fling: %s did not move (%d poses, %s studs/s)",
                        tostring(targetPlayer.Name),
                        stats.poses,
                        speedText
                    ))
                end
            end
            return touchedTarget
        end

        -- Any error must reset flingRunning, otherwise every later fling answers
        -- "fling already running" for the rest of the session.
        local function guardedRun(): boolean
            local ok: boolean, result: any = pcall(runFling)
            if not ok then
                setFlingRunning(false)
                if not silent then
                    notify("fling error: " .. tostring(result))
                end
                return false
            end
            return result == true
        end

        if type(options) == "table" and options.blocking == true then
            return guardedRun()
        end
        task.spawn(guardedRun)
        return true
    end

    state.mm2PerformFling = performFling
    state.mm2FlingStep = flingSingleStep
    activeCleanup = function(): ()
        engineActive = false
        destroyPin(fullFlingPin)
        fullFlingPin = nil
        flingRunning = false
        stepPinToken += 1
        destroyPin(stepPin)
        stepPin = nil
        setFlingRunning(false)
        state.mm2PerformFling = nil
        state.mm2FlingStep = nil
    end
    Module.Initialized = true
    return Module
end

function Module.destroy(): ()
    if activeCleanup then pcall(activeCleanup) end
    activeCleanup = nil
    Module.Initialized = false
end

return Module

export type Runtime = {
    framework: any,
    entity: any,
    host: any,
    services: any,
}

local Module = {
    Name = "AntiFling",
    PlaceId = 0,
    Events = {} :: {[string]: any},
    Initialized = false,
}

local activeCleanup: (() -> ())? = nil

function Module.init(context: Runtime): any
    local host: any = context.host
    local services: any = context.services
    local activity: any = services and services.activity
    local framework: any = context.framework
    local players: any = host.Players
    local localPlayer: any = host.LocalPlayer
    local world: any = host.workspace or workspace
    local featureConnections: any = host.featureConnections
    local disconnect: any = host.disconnectFeatureConnection

    local enabled: boolean = false
    local playerShield: boolean = true
    local impactShield: boolean = true
    local feature: any = nil
    local character: Model? = nil
    local links: {[BasePart]: {[BasePart]: any}} = {}
    local hazardSeen: {[BasePart]: number} = {}
    local nextScan: number = 0
    local dangerUntil: number = 0
    local suspended: boolean = false
    local warnedOverlap: boolean = false

    -- Limits are deliberately conservative. The shield is local physics, not
    -- server authority; do not change other people's CanCollide or global groups.
    local SCAN_INTERVAL: number = 0.15
    local RADIUS: number = 30
    local HAZARD_HOLD: number = 0.75
    local LINEAR_LIMIT: number = 80
    local SURFACE_LIMIT: number = 70
    local MAX_LINKS: number = 2048
    local MAX_PARTS: number = 200

    local function isBusy(): boolean
        return activity ~= nil and type(activity.isActive) == "function"
            and (activity.isActive("fling") or activity.isActive("walkfling"))
    end

    local function destroyLinks(): ()
        for _, perLocal: {[BasePart]: any} in pairs(links) do
            for _, constraint: any in pairs(perLocal) do
                pcall(function(): () constraint:Destroy() end)
            end
        end
        table.clear(links)
        table.clear(hazardSeen)
        dangerUntil = 0
    end

    local function isToolPart(part: Instance): boolean
        local ok: boolean, result: any = pcall(function(): any
            return part:FindFirstAncestorOfClass("Tool")
        end)
        return ok and result ~= nil
    end

    local function listBodyParts(model: Model?, onlyCollidable: boolean): {BasePart}
        local parts: {BasePart} = {}
        if not model or not model.Parent then return parts end
        local ok: boolean, descendants: any = pcall(function(): any return model:GetDescendants() end)
        if not ok or type(descendants) ~= "table" then return parts end
        for _, object: Instance in ipairs(descendants) do
            if object:IsA("BasePart") and not isToolPart(object) then
                local part: BasePart = object :: BasePart
                if not onlyCollidable or part.CanCollide then table.insert(parts, part) end
            end
        end
        return parts
    end

    local function fastPart(part: BasePart, scale: number?): boolean
        local ok: boolean, fast: any = pcall(function(): boolean
            -- Relative speed alone flags a stationary player when *we* sprint;
            -- arm the fallback only for motion driven by the other assembly.
            local linear: Vector3 = part.AssemblyLinearVelocity
            local angular: Vector3 = part.AssemblyAngularVelocity
            local radius: number = math.min(part.Size.Magnitude * 0.5, 20)
            local multiplier: number = scale or 1
            return linear.Magnitude > LINEAR_LIMIT * multiplier
                or angular.Magnitude * radius > SURFACE_LIMIT * multiplier
        end)
        return ok and fast == true
    end

    local function supportAssembly(root: BasePart, humanoid: Humanoid, owner: Model): BasePart?
        local ok: boolean, result: any = pcall(function(): any
            local params: RaycastParams = RaycastParams.new()
            params.FilterType = Enum.RaycastFilterType.Exclude
            params.FilterDescendantsInstances = {owner}
            params.IgnoreWater = true
            params.RespectCanCollide = true
            local distance: number = math.max(4, (tonumber(humanoid.HipHeight) or 2) + root.Size.Y * 0.5 + 2)
            return world:Raycast(root.Position, Vector3.new(0, -distance, 0), params)
        end)
        local ground: any = ok and result and result.Instance
        if ground and ground:IsA("BasePart") then
            return ground.AssemblyRootPart or ground
        end
        return nil
    end

    local function synchronize(owner: Model, humanoid: Humanoid, root: BasePart, now: number): ()
        local ownParts: {BasePart} = listBodyParts(owner, true)
        local desired: {[BasePart]: boolean} = {}
        local candidates: {any} = {}
        local characterModels: {[Model]: boolean} = {}
        local roster: any = players:GetPlayers()

        -- Track every visible player body, not the shared world collision group.
        -- Nearest bodies get the bounded constraint budget first on large servers.
        for _, player: Player in ipairs(roster) do
            local other: Model? = player ~= localPlayer and player.Character or nil
            if other and other ~= owner and other.Parent then
                characterModels[other] = true
                local otherRoot: BasePart? = other:FindFirstChild("HumanoidRootPart") :: BasePart?
                local distance: number = otherRoot and (otherRoot.Position - root.Position).Magnitude or math.huge
                table.insert(candidates, {model = other, distance = distance,
                    parts = if playerShield then listBodyParts(other, false) else {}})
            end
        end
        table.sort(candidates, function(a: any, b: any): boolean return a.distance < b.distance end)
        if playerShield then
            for _, candidate: any in ipairs(candidates) do
                for _, part: BasePart in ipairs(candidate.parts) do
                    desired[part] = true
                    if candidate.distance <= RADIUS and fastPart(part, nil) then
                        dangerUntil = math.max(dangerUntil, now + HAZARD_HOLD)
                    end
                end
            end
        end

        -- The query is local and bounded. Anchored geometry, the supporting
        -- platform/assembly, seats and tools keep their normal interactions.
        -- A fast loose part is suppressed only briefly while it remains risky.
        if impactShield and not humanoid.Sit and not humanoid.SeatPart then
            local overlap: any = nil
            pcall(function(): ()
                overlap = OverlapParams.new()
                overlap.FilterType = Enum.RaycastFilterType.Exclude
                overlap.FilterDescendantsInstances = {owner}
                overlap.MaxParts = MAX_PARTS
                overlap.RespectCanCollide = true
            end)
            local ok: boolean, nearby: any = pcall(function(): any
                return world:GetPartBoundsInRadius(root.Position, RADIUS, overlap)
            end)
            if not ok or type(nearby) ~= "table" then
                if not warnedOverlap and type(host.notify) == "function" then
                    warnedOverlap = true
                    host.notify("Anti Fling: moving-object scan unavailable; player protection remains active.")
                end
            else
                local floorAssembly: BasePart? = supportAssembly(root, humanoid, owner)
                for _, instance: Instance in ipairs(nearby) do
                    if instance:IsA("BasePart") then
                        local part: BasePart = instance :: BasePart
                        local isCharacterPart: boolean = false
                        local ancestor: Instance? = part.Parent
                        while ancestor and ancestor ~= world do
                            if characterModels[ancestor :: any] then isCharacterPart = true break end
                            ancestor = ancestor.Parent
                        end
                        local inOwner: boolean = part:IsDescendantOf(owner)
                        local isTool: boolean = isToolPart(part)
                        local assembly: BasePart? = part.AssemblyRootPart or part
                        -- Ordinary weapon/tools remain interactive. An extreme
                        -- spinning/flying handle is treated as a hazard, at 3x
                        -- the loose-object threshold, only while it is nearby.
                        if part.Parent and part.CanCollide and not part.Anchored and not inOwner
                            and (not isCharacterPart or isTool)
                            and (not floorAssembly or assembly ~= floorAssembly)
                            and fastPart(part, if isTool then 3 else nil) then
                            hazardSeen[part] = now
                            dangerUntil = math.max(dangerUntil, now + HAZARD_HOLD)
                        end
                    end
                end
            end
            for part: BasePart, seen: number in pairs(hazardSeen) do
                if not part.Parent or now - seen > HAZARD_HOLD then
                    hazardSeen[part] = nil
                else
                    desired[part] = true
                end
            end
        else
            table.clear(hazardSeen)
        end

        -- Allocate the finite budget afresh in priority order: newly dangerous
        -- objects first, then nearest players. Otherwise 2,048 older/farther
        -- links could permanently starve a new incoming threat.
        local ordered: {BasePart} = {}
        local included: {[BasePart]: boolean} = {}
        local function include(part: BasePart): ()
            if desired[part] and part.Parent and not included[part] then
                included[part] = true
                table.insert(ordered, part)
            end
        end
        local hazards: {BasePart} = {}
        for part: BasePart in pairs(hazardSeen) do table.insert(hazards, part) end
        table.sort(hazards, function(a: BasePart, b: BasePart): boolean
            return (a.Position - root.Position).Magnitude < (b.Position - root.Position).Magnitude
        end)
        for _, part: BasePart in ipairs(hazards) do include(part) end
        if playerShield then
            for _, candidate: any in ipairs(candidates) do
                for _, part: BasePart in ipairs(candidate.parts) do include(part) end
            end
        end
        local accepted: {[BasePart]: {[BasePart]: boolean}} = {}
        local allocated: number = 0
        for _, other: BasePart in ipairs(ordered) do
            if allocated >= MAX_LINKS then break end
            local perLocal: {[BasePart]: boolean} = {}
            for _, own: BasePart in ipairs(ownParts) do
                if allocated >= MAX_LINKS then break end
                if own ~= other then
                    perLocal[own] = true
                    allocated += 1
                end
            end
            if next(perLocal) then accepted[other] = perLocal end
        end
        -- Remove old or lower-priority pairs before filling the new budget.
        for other: BasePart, perLocal: {[BasePart]: any} in pairs(links) do
            local wanted: {[BasePart]: boolean}? = accepted[other]
            for own: BasePart, constraint: any in pairs(perLocal) do
                if not wanted or not wanted[own] or not own.Parent or not other.Parent then
                    pcall(function(): () constraint:Destroy() end)
                    perLocal[own] = nil
                end
            end
            if next(perLocal) == nil then links[other] = nil end
        end
        for other: BasePart, wanted: {[BasePart]: boolean} in pairs(accepted) do
            local perLocal: {[BasePart]: any} = links[other] or {}
            links[other] = perLocal
            for own: BasePart in pairs(wanted) do
                if not perLocal[own] then
                    local ok: boolean, constraint: any = pcall(function(): any
                        local link: NoCollisionConstraint = Instance.new("NoCollisionConstraint")
                        link.Part0, link.Part1 = own, other
                        link.Parent = own
                        return link
                    end)
                    if ok and constraint then perLocal[own] = constraint end
                end
            end
            if next(perLocal) == nil then links[other] = nil end
        end
    end

    local function step(): ()
        if not enabled then return end
        local owner: Model?, humanoid: Humanoid?, root: BasePart? = host.getCharacterParts()
        if owner ~= character then
            destroyLinks()
            character = owner
            nextScan = 0
        end
        if not owner or not humanoid or not root or not owner.Parent or humanoid.Health <= 0 then
            destroyLinks()
            return
        end
        if isBusy() then
            if not suspended then destroyLinks() end
            suspended = true
            return
        end
        if suspended then suspended = false nextScan = 0 end
        local now: number = os.clock()
        if now >= nextScan then
            nextScan = now + SCAN_INTERVAL
            local ok: boolean, err: any = pcall(synchronize, owner, humanoid, root, now)
            if not ok then
                -- A streaming object may disappear during the scan. Retry on
                -- the next interval, without leaving partial constraints stuck.
                destroyLinks()
                if type(host.warn) == "function" then host.warn("Anti Fling: " .. tostring(err)) end
            end
        end
        if now > dangerUntil or root.Anchored or humanoid.Sit or humanoid.SeatPart then return end
        -- Emergency damping is only armed by a nearby extreme mover. Normal
        -- sprinting, jumps, vehicles and game launch pads are not capped merely
        -- because this feature is on.
        local walkSpeed: number = math.max(16, tonumber(humanoid.WalkSpeed) or 16)
        local speed: any = framework and framework:GetModule("Speed Hack")
        if speed and speed.Enabled and speed.Options.Speed then
            walkSpeed = math.max(walkSpeed, tonumber(speed.Options.Speed.Value) or 0)
        end
        local jumpPower: number = math.max(50, tonumber(humanoid.JumpPower) or 50)
        local velocity: Vector3 = root.AssemblyLinearVelocity
        local planar: Vector3 = Vector3.new(velocity.X, 0, velocity.Z)
        local limit: number = math.max(120, walkSpeed * 4)
        local riseLimit: number = math.max(180, jumpPower * 4)
        local nextVelocity: Vector3 = velocity
        local changed: boolean = false
        if planar.Magnitude > limit then
            local move: Vector3 = humanoid.MoveDirection
            local direction: Vector3 = move.Magnitude > 0.05 and move.Unit * walkSpeed or Vector3.zero
            nextVelocity = Vector3.new(direction.X, nextVelocity.Y, direction.Z)
            changed = true
        end
        if nextVelocity.Y > riseLimit then
            nextVelocity = Vector3.new(nextVelocity.X, riseLimit, nextVelocity.Z)
            changed = true
        end
        pcall(function(): ()
            if root.AssemblyAngularVelocity.Magnitude > 25 then
                root.AssemblyAngularVelocity = Vector3.zero
            end
            if changed then root.AssemblyLinearVelocity = nextVelocity end
        end)
    end

    local function toggleAntiFling(value: boolean): ()
        enabled = value == true
        disconnect("AntiFling")
        destroyLinks()
        character = nil
        nextScan = 0
        suspended = false
        if enabled then
            featureConnections.AntiFling = host.TaskManager:Connect(step, nil, "Anti Fling")
        end
    end

    feature = host.createUniversalFeature(
        "Anti Fling",
        "Collision Shield: local player and extreme-object protection",
        2,
        toggleAntiFling,
        {categoryName = "Utilities"}
    )
    host.addToggleOption(feature, "Player Shield", true, function(value: boolean): ()
        playerShield = value == true
        nextScan = 0
        if not playerShield and not impactShield then destroyLinks() end
    end, "Prevent local physical collisions with other character bodies without changing their collision settings.")
    host.addToggleOption(feature, "Impact Shield", true, function(value: boolean): ()
        impactShield = value == true
        nextScan = 0
        if not playerShield and not impactShield then destroyLinks() end
    end, "Temporarily avoid nearby extreme-speed loose objects; support platforms, seated vehicles and ordinary tools are left alone.")
    host.addFeatureTooltip(feature,
        "Collision Shield only changes local collision pairs. Extremely fast/streamed objects and server-owned physics can still launch you.")

    activeCleanup = function(): ()
        toggleAntiFling(false)
    end
    Module.Initialized = true
    return feature
end

function Module.destroy(): ()
    if activeCleanup then pcall(activeCleanup) end
    activeCleanup = nil
    Module.Initialized = false
end

return Module

    -- The visible half of a pickup. A server part resolves to its
    -- CoinVisual child; anything else is used as-is. Parents are NEVER
    -- walked: the container must stay dark.
    local function visualOf(part: Instance): Instance?
        if part.Name == SERVER_NAME then
            local child: Instance? = part:FindFirstChild(VISUAL_NAME)
            if child and child:IsA("BasePart") then
                return child
            end
            return nil
        end

    end

    -- The volatile Collected/Delete flags are checked by passesFiltersCached;
    -- this static half rejects anything outside the map or inside a character.
    local function passesFilters(part: Instance): boolean
        if not part:IsA("BasePart") or not part:IsDescendantOf(workspace) then
            return false
        end
        local current: Instance? = part.Parent
        while current and current ~= workspace do
            if current:FindFirstChildOfClass("Humanoid") then
                return false
            end
            current = current.Parent
        end
        return true
    end

    -- The family the server says is in play: the `CoinsStarted` table first,

        local eventName: string = lower(tostring(attribute(workspace, "CurrentEvent") or ""))
        if contains(eventName, "halloween") then

    -- Family of one pickup: its mesh children only. With no specific
    -- word, the round's family applies (a round pays a single kind of
    -- pickup), falling back to the classic coin.
    local familyCache: {[Instance]: string} = {}
    local function familyOf(part: Instance): string
        local cached: string? = familyCache[part]
        if cached then
            return cached
        end
        local names: {string} = {}
        for _, object: Instance in ipairs(part:GetDescendants()) do
            table.insert(names, object.Name)
        end
        local haystack: string = lower(table.concat(names, " "))
        -- A round pays a single kind of pickup, and event pickups are
        -- built on the classic coin template (MainCoin/DecalPart meshes
        -- with an event look). So when the server announces an event
        -- family, plain coin mesh names must NOT drag the pickup back
        -- to Coin -- that mismatch is what left Auto showing nothing
        -- while the map was full of candies.
        local active: string? = activeFamily()
        for _, family: string in ipairs(FAMILY_ORDER) do
            if family ~= FAMILY_COIN or active == nil or active == FAMILY_COIN then
                for _, phrase: string in ipairs(FAMILY_PHRASES[family]) do
                    if contains(haystack, phrase) then
                        -- Cache only the non-Coin mesh families:
                        -- they win over the server family in every
                        -- scan, so they can never go stale. A Coin
                        -- mesh classification is NOT cached -- when
                        -- the server announces an event family, the
                        -- coin template must reclassify to it.
                        if family ~= FAMILY_COIN then
                            familyCache[part] = family
                        end
                        return family
                    end
                end
            end
        end
        return active or FAMILY_COIN
    end

    -- Every real pickup on the map, merged from the three signatures.
    -- (v84 tried a "cheap" rescan -- tag list plus previously-known
    -- parts -- and in the live game the ESP stopped reporting coins
    -- entirely; the mock suite passed, the game did not. The full
    -- scan below is the behaviour that demonstrably works in game.
    -- The cost is kept down elsewhere: Auto Farm's polls are cached
    -- (1.5 s), the family classification is cached, and the ancestor
    -- walk inside passesFilters is cached per part.)
    local filterCache: {[Instance]: boolean} = {}
    local function collectAll(): ({[Instance]: string}, {[string]: number})
        local found: {[Instance]: string} = {}
        local counts: {[string]: number} = {}
        lastRaw = {tagged = 0, coinId = 0, named = 0}

        -- The attribute half of the filter is volatile (Collected
        -- flips when a coin is taken) and is always re-read; only
        -- the static half (in the workspace, not inside a character)
        -- is cached -- a pickup's ancestry never changes.
        local function passesFiltersCached(part: Instance): boolean
            if attribute(part, "Collected") == true
                or attribute(part, "Delete") == true then
                return false
            end
            local cached: boolean? = filterCache[part]
            if cached ~= nil then
                return cached
            end
            local result: boolean = passesFilters(part)
            filterCache[part] = result
            return result
        end

        local function consider(part: Instance?): ()
            if not part then
                return
            end
            local visual: Instance? = visualOf(part)
            if not visual or found[visual] then
                return
            end
            if not passesFiltersCached(visual) then
                return
            end
            local family: string = familyOf(visual)
            found[visual] = family
            counts[family] = (counts[family] or 0) + 1
        end

        -- Signature 1: the game's own tag list.
        local tagged: {Instance} = collectionService:GetTagged(COIN_TAG)
        lastRaw.tagged = #tagged
        for _, object: Instance in ipairs(tagged) do
            if object:IsA("BasePart") then consider(object) end
        end

        -- Signatures 2 and 3: CoinID and exact spawn names from one scan.
        for _, object: Instance in ipairs(workspace:GetDescendants()) do
            if object:IsA("BasePart") then
                if attribute(object, "CoinID") ~= nil then
                    lastRaw.coinId += 1
                    consider(object)
                elseif object.Name == VISUAL_NAME or object.Name == SERVER_NAME then
                    lastRaw.named += 1
                    consider(object)
                end
            end
        end
        return found, counts
    end

    local function wanted(family: string): boolean
        local choice: string = settings.family
        if choice == FAMILY_ALL then
            return true
        end
        if choice == FAMILY_AUTO then
            local active: string? = activeFamily()
            return active == nil or family == active
        end
        return family == choice
    end

    local function describe(counts: {[string]: number}): string
        local parts: {string} = {}
        for _, family: string in ipairs(COUNT_ORDER) do
            local amount: number = counts[family] or 0
            if amount > 0 then
                local nouns: {string} = FAMILY_NOUNS[family]
                table.insert(parts, tostring(amount) .. " " .. (amount == 1 and nouns[1] or nouns[2]))
            end
        end
        if #parts == 0 then
            return "no coins yet"
        end
        return table.concat(parts, " · ")
    end

    -- Shared with Auto Farm: the pickup parts of the family in use.
    -- CACHED: this used to run a full collectAll() per call, and Auto
    -- Farm polls every 0.4 s -- each poll re-scanned every descendant
    -- of Workspace and hitch-spiked the frame. The cache refreshes on
    -- the ESP's own scans, or at most every 0.5 s when polled.
    local pickupCache: {at: number, list: {Instance}} = {at = -math.huge, list = {}}
    local function refreshPickupCache(found: {[Instance]: string}): ()
        local list: {Instance} = {}
        for part: Instance, family: string in pairs(found) do
            if wanted(family) then
                table.insert(list, part)
            end
        end
        pickupCache = {at = os.clock(), list = list}
    end
    state.mm2CoinPickups = function(): {Instance}
        if os.clock() - pickupCache.at < 0.5 then
            return pickupCache.list
        end
        local found: {[Instance]: string} = collectAll()
        refreshPickupCache(found)
        return pickupCache.list
    end

    local function coinPickups(): {Instance}
        local map: Instance? = type(findMM2Map) == "function" and findMM2Map()
        local out: {Instance} = {}
        local function consider(object: Instance): ()
            if not object.Parent or not object:IsA("BasePart") then return end
            -- A real Instance can verify ancestry; do not chase a lobby or an
            -- old map's pickup just because it still has the CoinVisual tag.
            if map and type(object.IsDescendantOf) == "function"
                and not object:IsDescendantOf(map) then return end
            local ok: boolean, usable: any = pcall(function(): boolean
                local parent: Instance? = object.Parent
                local server: Instance = if parent and parent.Name == "Coin_Server" then parent else object
                return object:GetAttribute("Collected") ~= true
                    and object:GetAttribute("Delete") ~= true
                    and server:GetAttribute("Collected") ~= true
                    and server:GetAttribute("Delete") ~= true
            end)
            if ok and usable then table.insert(out, object) end
        end
        if type(state.mm2CoinPickups) == "function" then
            local ok: boolean, list: any = pcall(state.mm2CoinPickups)
            if ok and type(list) == "table" then
                -- Coin ESP scans the entire workspace: a non-empty list can
                -- still consist solely of old-map or already-collected items.
                for _, object: Instance in ipairs(list) do consider(object) end
                if #out > 0 then return out end
            end
        end
        local now: number = os.clock()
        if map == fallbackScanMap and now - fallbackScanAt < 0.5 then
            return fallbackPickups
        end
        fallbackScanAt = now
        fallbackScanMap = map
        local okTagged: boolean, tagged: any = pcall(function(): any
            return game:GetService("CollectionService"):GetTagged("CoinVisual")
        end)
        if okTagged and type(tagged) == "table" then
            for _, object: Instance in ipairs(tagged) do consider(object) end
        end
        if #out == 0 then
            local ok: boolean, descendants: any = pcall(function(): any
                -- Pickups live under the active map's CoinContainer; avoid
                -- walking the lobby and every unrelated streamed character.
                if map and type(map.GetDescendants) == "function" then return map:GetDescendants() end
                return workspace:GetDescendants()
            end)
            if ok and type(descendants) == "table" then
                for _, object: Instance in ipairs(descendants) do
                    if object.Name == "Coin_Server" then consider(object) end
                end
            end
        end
        fallbackPickups = out
        return out
    end

        -- The Coin_Server part that pays for a pickup entry (ESP hands
        -- out the visual, the server part is its parent).
        local function serverPartOf(part: Instance): Instance?
            if part.Name == "Coin_Server" then
                return part
            end
            local parent: Instance? = part.Parent
            if parent and parent.Name == "Coin_Server" then
                return parent
            end
            return part
        end

        -- Still there to be collected? A coin somebody already took
        -- keeps its entry in the throttled scan for a moment, and the
        -- old farm flew up to that ghost and hovered over it. The
        -- attribute/parent check runs on the chosen coin every tick,
        -- so a collected coin stops being a target at once.
        local function stillCollectable(part: Instance): boolean
            if not part.Parent then
                return false
            end
            local server: Instance? = serverPartOf(part)
            local ok: boolean, usable: any = pcall(function(): boolean
                return part:GetAttribute("Collected") ~= true
                    and part:GetAttribute("Delete") ~= true
                    and server ~= nil
                    and server:GetAttribute("Collected") ~= true
                    and server:GetAttribute("Delete") ~= true
            end)
            return ok and usable == true
        end

        -- Walk mode movement: point the humanoid at the coin and let the
        -- engine's own locomotion do the walking. Returns the horizontal distance
        -- left. If the distance stops shrinking (a wall, a ledge, or a game whose
        -- control module ignores Humanoid:Move) a short horizontal velocity carry
        -- is used for a second, and the caller's stuck timer blacklists coins
        -- walking truly cannot reach.
        local function walkTowards(
            humanoid: Humanoid,
            root: BasePart,
            destination: Vector3
        ): number
            -- Walking never uses the flight controller.
            hoverActive = false
            hoverTarget = nil
            local delta: Vector3 = destination - root.Position
            local flat: Vector3 = Vector3.new(delta.X, 0, delta.Z)
            local distance: number = flat.Magnitude
            local now: number = os.clock()
            if distance < 1.4 then
                walkCarryDirection = nil
                walkCarryUntil = 0
                pcall(function(): ()
                    humanoid:Move(Vector3.zero, false)
                end)
                return distance
            end
            if distance < walkLastDistance - 0.3 then
                walkLastDistance = distance
                walkStuckSince = now
                walkCarryUntil = 0
            elseif now - walkStuckSince > 2.0 then
                walkStuckSince = now
                walkCarryUntil = now + 1.0
                walkCarryDirection = flat / distance
                pcall(function(): ()
                    humanoid.Jump = true
                end)
            end
            if walkCarryUntil > now and walkCarryDirection then
                local speed: number = math.clamp(settings.speed, 5, 60)
                local direction: Vector3 = walkCarryDirection :: Vector3
                pcall(function(): ()
                    local current: Vector3 = root.AssemblyLinearVelocity
                    root.AssemblyLinearVelocity = Vector3.new(
                        direction.X * speed,
                        math.min(current.Y, 0),
                        direction.Z * speed
                    )
                    humanoid:Move(direction, false)
                end)
            else
                walkCarryDirection = nil
                pcall(function(): ()
                    humanoid:Move(flat.Unit, false)
                end)
            end
            return distance
        end
