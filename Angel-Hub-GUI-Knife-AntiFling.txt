local function create(className, properties, parent)
	local instance = Instance.new(className)
	for key, value in properties do
		instance[key] = value
	end
	instance.Parent = parent
	return instance
end

local function tween(instance, info, goal)
	local newTween = tweenService:Create(instance, info, goal)
	newTween:Play()
	return newTween
end

local function connectInput(signal, callback)
	local connection = signal:Connect(callback)
	table.insert(AngelHub.connections, connection)
	return connection
end

local function isPress(input)
	return input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch
end

local function isMove(input)
	return input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch
end

local function addGradient(parent, rotation)
	return create("UIGradient", { Color = ColorSequence.new(ACCENT_A, ACCENT_B), Rotation = rotation }, parent)
end

local function addStroke(parent, color, transparency, thickness)
	return create("UIStroke", {
		Color = color,
		Transparency = transparency,
		Thickness = thickness or 1,
		ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
	}, parent)
end

local function createCard(className, parent, text, height)
	local card = create(className, {
		Name = text,
		Size = UDim2.new(1, 0, 0, height),
		BackgroundColor3 = ROW_COLOR,
		BackgroundTransparency = 0.65,
		BorderSizePixel = 0,
	}, AngelHub.sectionBoxes[parent] or parent)

	create("UICorner", { CornerRadius = UDim.new(0, 5) }, card)
	local stroke = addStroke(card, ACCENT_A, 1)
	card.ClipsDescendants = true

	local group = AngelHub.currentGroup
	if group then
		table.insert(group.cards, card)
		card:SetAttribute("GroupHidden", group.collapsed)
		card.Visible = not group.collapsed
	end

	card.InputBegan:Connect(function(input)
		if not isPress(input) then
			return
		end

		local offset = Vector2.new(input.Position.X, input.Position.Y) - card.AbsolutePosition
		local ripple = create("Frame", {
			AnchorPoint = Vector2.new(0.5, 0.5),
			Position = UDim2.fromOffset(offset.X, offset.Y),
			Size = UDim2.fromOffset(0, 0),
			BackgroundColor3 = Color3.new(1, 1, 1),
			BackgroundTransparency = 0.8,
			BorderSizePixel = 0,
			ZIndex = 0,
		}, card)

		create("UICorner", { CornerRadius = UDim.new(1, 0) }, ripple)
		local diameter = math.max(card.AbsoluteSize.X, card.AbsoluteSize.Y) * 2

		tween(ripple, MEDIUM_TWEEN, { Size = UDim2.fromOffset(diameter, diameter), BackgroundTransparency = 1 }).Completed:Connect(function()
			ripple:Destroy()
		end)
	end)

	card.MouseEnter:Connect(function()
		tween(card, FAST_TWEEN, { BackgroundTransparency = 0.35 })
		tween(stroke, FAST_TWEEN, { Transparency = 0.55 })
	end)

	card.MouseLeave:Connect(function()
		tween(card, FAST_TWEEN, { BackgroundTransparency = 0.65 })
		tween(stroke, FAST_TWEEN, { Transparency = 1 })
	end)

	return card, stroke
end

local function addToggle(parent, text, keybind, default)
	local startState = default == true
	local holder, stroke = createCard("TextButton", parent, text, 28)
	holder.Text = ""
	holder.AutoButtonColor = false

	local box = create("Frame", {
		Size = UDim2.fromOffset(16, 16),
		Position = UDim2.new(0, 8, 0.5, -8),
		BackgroundColor3 = ELEMENT_COLOR,
		BorderSizePixel = 0,
	}, holder)

	create("UICorner", { CornerRadius = UDim.new(0, 4) }, box)
	local boxStroke = addStroke(box, startState and ACCENT_A or DIM_TEXT_COLOR, startState and 0 or 0.5)

	local check = create("Frame", {
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.fromScale(0.5, 0.5),
		Size = startState and UDim2.fromOffset(10, 10) or UDim2.fromOffset(0, 0),
		BackgroundColor3 = Color3.new(1, 1, 1),
		BorderSizePixel = 0,
	}, box)

	create("UICorner", { CornerRadius = UDim.new(0, 3) }, check)
	addGradient(check, 45)

	local label = create("TextLabel", {
		Size = UDim2.new(1, keybind and -100 or -44, 1, 0),
		Position = UDim2.fromOffset(32, 0),
		BackgroundTransparency = 1,
		Text = text,
		TextColor3 = startState and TEXT_COLOR or OFF_TEXT_COLOR,
		TextSize = 13,
		Font = TEXT_FONT,
		TextXAlignment = Enum.TextXAlignment.Left,
		TextTruncate = Enum.TextTruncate.AtEnd,
	}, holder)

	local accentBar = create("Frame", {
		Size = UDim2.new(0, 3, 1, -10),
		Position = UDim2.fromOffset(0, 5),
		BackgroundColor3 = Color3.new(1, 1, 1),
		BackgroundTransparency = startState and 0 or 1,
		BorderSizePixel = 0,
	}, holder)

	create("UICorner", { CornerRadius = UDim.new(1, 0) }, accentBar)
	addGradient(accentBar, 90)

	local state = startState
	AngelHub.values[text] = state
	AngelHub.defaults[text] = state

	AngelHub.setters[text] = function(value)
		state = value == true
		AngelHub.values[text] = state

		tween(check, POP_TWEEN, { Size = state and UDim2.fromOffset(10, 10) or UDim2.fromOffset(0, 0) })
		tween(boxStroke, FAST_TWEEN, { Color = state and ACCENT_A or DIM_TEXT_COLOR, Transparency = state and 0 or 0.5 })
		tween(label, FAST_TWEEN, { TextColor3 = state and TEXT_COLOR or OFF_TEXT_COLOR })
		tween(accentBar, FAST_TWEEN, { BackgroundTransparency = state and 0 or 1 })
	end

	if keybind then
		local keybindName = text .. " Keybind"
		local keybindButton = create("TextButton", {
			Size = UDim2.fromOffset(52, 22),
			Position = UDim2.new(1, -58, 0.5, -11),
			BackgroundTransparency = 1,
			Text = "[" .. keybind .. "]",
			TextColor3 = DIM_TEXT_COLOR,
			TextSize = 12,
			Font = TEXT_FONT,
			TextXAlignment = Enum.TextXAlignment.Right,
			AutoButtonColor = false,
		}, holder)

		local function applyKey(keyName)
			local keyCode = nil
			for _, item in Enum.KeyCode:GetEnumItems() do
				if item.Name == keyName then
					keyCode = item
					break
				end
			end

			AngelHub.keybinds[text] = keyCode
			AngelHub.values[keybindName] = keyCode and keyName or ""
			keybindButton.Text = keyCode and ("[" .. keyName .. "]") or "[None]"
			keybindButton.Visible = AngelHub.values["Keybind Hints"] ~= false
		end

		AngelHub.defaults[keybindName] = keybind
		AngelHub.setters[keybindName] = applyKey
		AngelHub.keyLabels[text] = keybindButton
		applyKey(keybind)

		keybindButton.MouseEnter:Connect(function()
			tween(keybindButton, FAST_TWEEN, { TextColor3 = ACCENT_A })
		end)

		keybindButton.MouseLeave:Connect(function()
			if not AngelHub.listening then
				tween(keybindButton, FAST_TWEEN, { TextColor3 = DIM_TEXT_COLOR })
			end
		end)

		keybindButton.Activated:Connect(function()
			if AngelHub.cancelListening then
				AngelHub.cancelListening()
			end

			local previousKey = AngelHub.values[keybindName]
			keybindButton.Text = "[...]"
			tween(keybindButton, FAST_TWEEN, { TextColor3 = ACCENT_A })

			AngelHub.cancelListening = function()
				AngelHub.listening = nil
				AngelHub.cancelListening = nil
				tween(keybindButton, FAST_TWEEN, { TextColor3 = DIM_TEXT_COLOR })
				applyKey(previousKey)
			end

			AngelHub.listening = function(keyCode)
				AngelHub.listening = nil
				AngelHub.cancelListening = nil
				tween(keybindButton, FAST_TWEEN, { TextColor3 = DIM_TEXT_COLOR })
				applyKey(keyCode == Enum.KeyCode.Escape and "" or keyCode.Name)
			end
		end)
	end

	holder.MouseButton1Down:Connect(function()
		tween(holder, FAST_TWEEN, { BackgroundTransparency = 0 })
		tween(stroke, FAST_TWEEN, { Transparency = 0 })
	end)

	holder.MouseButton1Up:Connect(function()
		tween(holder, FAST_TWEEN, { BackgroundTransparency = 0.35 })
		tween(stroke, FAST_TWEEN, { Transparency = 0.55 })
	end)

	holder.Activated:Connect(function()
		AngelHub.setters[text](not state)
	end)

	holder.MouseEnter:Connect(function()
		tween(label, FAST_TWEEN, { Position = UDim2.fromOffset(36, 0) })
	end)

	holder.MouseLeave:Connect(function()
		tween(label, FAST_TWEEN, { Position = UDim2.fromOffset(32, 0) })
	end)

	table.insert(AngelHub.searchables, holder)
	return holder
end

local function addSlider(parent, text, minValue, maxValue, default, decimals)
	local editableMovement = text == "Speed Value" or text == "Fly Horizontal Speed" or text == "Fly Vertical Speed" or text == "Gravity Value"
	local holder = createCard("Frame", parent, text, 44)
	local startAlpha = (default - minValue) / (maxValue - minValue)

	create("TextLabel", {
		Size = UDim2.new(1, -70, 0, 20),
		Position = UDim2.fromOffset(10, 3),
		BackgroundTransparency = 1,
		Text = text,
		TextColor3 = TEXT_COLOR,
		TextSize = 13,
		Font = TEXT_FONT,
		TextXAlignment = Enum.TextXAlignment.Left,
	}, holder)

	local valueLabel = create(editableMovement and "TextBox" or "TextLabel", {
		Size = UDim2.fromOffset(56, 20),
		Position = UDim2.new(1, -66, 0, 3),
		BackgroundTransparency = 1,
		Text = tostring(default),
		TextColor3 = OFF_TEXT_COLOR,
		TextSize = 13,
		Font = BOLD_FONT,
		TextXAlignment = Enum.TextXAlignment.Right,
	}, holder)

	local bar = create("Frame", {
		Size = UDim2.new(1, -20, 0, 6),
		Position = UDim2.fromOffset(10, 29),
		BackgroundColor3 = ELEMENT_COLOR,
		BorderSizePixel = 0,
	}, holder)

	create("UICorner", { CornerRadius = UDim.new(1, 0) }, bar)
	addStroke(bar, Color3.new(1, 1, 1), 0.88)

	local fill = create("Frame", { Size = UDim2.fromScale(startAlpha, 1), BackgroundColor3 = Color3.new(1, 1, 1), BorderSizePixel = 0 }, bar)
	create("UICorner", { CornerRadius = UDim.new(1, 0) }, fill)
	addGradient(fill, 0)

	local knob = create("Frame", {
		AnchorPoint = Vector2.new(0.5, 0.5),
		Size = UDim2.fromOffset(12, 12),
		Position = UDim2.new(startAlpha, 0, 0.5, 0),
		BackgroundColor3 = Color3.new(1, 1, 1),
		BorderSizePixel = 0,
	}, bar)

	create("UICorner", { CornerRadius = UDim.new(1, 0) }, knob)
	addStroke(knob, ACCENT_A, 0, 2)
	local knobScale = create("UIScale", {}, knob)

	local hitbox = create("Frame", { Size = UDim2.new(1, -20, 0, 22), Position = UDim2.fromOffset(10, 21), BackgroundTransparency = 1 }, holder)

	AngelHub.values[text] = default
	AngelHub.defaults[text] = default

	local dragging = false

	local function setValue(value)
		local factor = 10 ^ decimals
		if type(value) ~= "number" or value ~= value or math.abs(value) == math.huge then return end
		local lower = editableMovement and (text == "Gravity Value" and minValue or -10000) or minValue
		value = math.floor(math.clamp(value, lower, maxValue) * factor + 0.5) / factor
		local snapped = math.clamp((value - minValue) / (maxValue - minValue), 0, 1)

		tween(fill, SLIDE_TWEEN, { Size = UDim2.fromScale(snapped, 1) })
		tween(knob, SLIDE_TWEEN, { Position = UDim2.new(snapped, 0, 0.5, 0) })
		valueLabel.Text = tostring(value)
		AngelHub.values[text] = value
	end

	AngelHub.setters[text] = setValue
	if editableMovement then
		valueLabel.ClearTextOnFocus = false
		valueLabel.FocusLost:Connect(function()
			local value = tonumber(valueLabel.Text)
			if value then AngelHub.setters[text](value) else valueLabel.Text = tostring(AngelHub.values[text]) end
		end)
	end

	local function updateFromX(positionX)
		local alpha = math.clamp((positionX - bar.AbsolutePosition.X) / bar.AbsoluteSize.X, 0, 1)
		setValue(minValue + (maxValue - minValue) * alpha)
	end

	hitbox.InputBegan:Connect(function(input)
		if not isPress(input) then
			return
		end

		dragging = true
		tween(knobScale, POP_TWEEN, { Scale = 1.35 })
		tween(valueLabel, FAST_TWEEN, { TextColor3 = ACCENT_A })
		updateFromX(input.Position.X)
	end)

	connectInput(userInputService.InputChanged, function(input)
		if dragging and isMove(input) then
			updateFromX(input.Position.X)
		end
	end)

	connectInput(userInputService.InputEnded, function(input)
		if not dragging or not isPress(input) then
			return
		end

		dragging = false
		tween(knobScale, POP_TWEEN, { Scale = 1 })
		tween(valueLabel, FAST_TWEEN, { TextColor3 = OFF_TEXT_COLOR })
	end)

	table.insert(AngelHub.searchables, holder)
	return holder
end

local function addCycle(parent, text, options, onChange)
	local holder = createCard("Frame", parent, text, 56)

	create("TextLabel", {
		Size = UDim2.new(1, -20, 0, 20),
		Position = UDim2.fromOffset(10, 3),
		BackgroundTransparency = 1,
		Text = text,
		TextColor3 = TEXT_COLOR,
		TextSize = 13,
		Font = TEXT_FONT,
		TextXAlignment = Enum.TextXAlignment.Left,
	}, holder)

	local button = create("TextButton", {
		Size = UDim2.new(1, -20, 0, 24),
		Position = UDim2.fromOffset(10, 26),
		BackgroundColor3 = ELEMENT_COLOR,
		BorderSizePixel = 0,
		Text = options[1],
		TextColor3 = TEXT_COLOR,
		TextSize = 13,
		Font = TEXT_FONT,
		AutoButtonColor = false,
	}, holder)

	create("UICorner", { CornerRadius = UDim.new(0, 4) }, button)
	local buttonStroke = addStroke(button, DIM_TEXT_COLOR, 0.6)

	local index = 1
	AngelHub.values[text] = options[1]
	AngelHub.defaults[text] = options[1]

	AngelHub.setters[text] = function(value)
		local found = table.find(options, value)
		if not found then
			return
		end

		index = found
		AngelHub.values[text] = options[index]
		button.Text = options[index]

		if onChange then
			onChange(options[index])
		end
	end

	button.MouseEnter:Connect(function()
		tween(buttonStroke, FAST_TWEEN, { Color = ACCENT_A, Transparency = 0 })
	end)

	button.MouseLeave:Connect(function()
		tween(buttonStroke, FAST_TWEEN, { Color = DIM_TEXT_COLOR, Transparency = 0.6 })
	end)

	button.Activated:Connect(function()
		index = index % #options + 1
		AngelHub.values[text] = options[index]
		button.Text = options[index]
		button.TextTransparency = 1
		tween(button, FAST_TWEEN, { TextTransparency = 0 })

		if onChange then
			onChange(options[index])
		end
	end)

	table.insert(AngelHub.searchables, holder)
	return holder
end

local function addKeybind(parent, text, default)
	local holder = createCard("Frame", parent, text, 28)

local function makeDraggable(handle, target)
	local dragStart, startPosition

	handle.InputBegan:Connect(function(input)
		if not isPress(input) then
			return
		end

		dragStart = input.Position
		startPosition = target.Position
	end)

	connectInput(userInputService.InputChanged, function(input)
		if not dragStart or not isMove(input) then
			return
		end

		local delta = input.Position - dragStart
		target.Position = UDim2.new(startPosition.X.Scale, startPosition.X.Offset + delta.X, startPosition.Y.Scale, startPosition.Y.Offset + delta.Y)
	end)

	connectInput(userInputService.InputEnded, function(input)
		if isPress(input) then
			dragStart = nil
		end
	end)
end

--[[
    Kinematic prediction core for MM2 gun shots and knife throws.

    What this model is based on (TestingStuff evidence, not guesses):

      * GunFired echoes carry (handle, origin, endpoint, hitPart). The time
        between our FireServer and that echo is the only transport number we
        can measure in-game; the saved calibration recorded 297 confirmed
        shots with a mean of ~123.5 ms.
      * ThrowingKnifeVisuals advances the blade linearly:
            CFrame + Direction * ThrowSpeed * dt
        with ThrowSpeed 96 by default and no gravity term. A blade therefore
        needs an intercept solve, not a ballistic one.
      * A remote character's replicated position is older than the server
        state the gun ray is checked against. The aim point advances by
        the full measured transport plus the replication interpolation
        window (the saved sessions score hits at the confirmation
        instant, i.e. the server behaves as if it rewound by the whole
        ping; the one-way half was tried in v85 and led short).

    What this model deliberately does NOT do:

      * No speculative openings or unconstrained extrapolation. Turn and
        braking terms come from weighted fits, capped at 45 degrees / the stop
        distance. Knife landing additionally uses a bounded client-side support
        probe; its cache, rig and slope thresholds are conservative estimates,
        not measured server behaviour.
      * No unconstrained auto-tuned lead. Transport is the median of at least
        three fresh echoes inside sane pairing limits, with smoothed RTT as
        fallback; the separate residual correction is median-filtered and
        capped at 60 ms.
      * No invented movement when the measured velocity is zero but
        MoveDirection is not. The server only knows replicated motion.
      * No distance/confidence gate before firing. Freshness of the sample,
        WallCheck and weapon readiness are the real conditions.
]]

local MAX_SAMPLE_AGE: number = 0.35
local FIT_WINDOW: number = 0.16
local FIT_TAU: number = 0.055
local HISTORY_SECONDS: number = 0.6
-- Remote characters replicate around 20 Hz. 60 Hz keeps three observations per
-- replicated pose (so a fit is never built from a single packet) without
-- inventing detail that the network never sent. Raising it only re-weights the
-- same poses and costs CPU; the dominant error is the transport horizon.
local SAMPLE_HZ: number = 60
local MAX_HORIZONTAL_SPEED: number = 60
local DEFAULT_KNIFE_SPEED: number = 96
local MAX_ECHOES: number = 8
-- A median of two echoes is just the mean of two mismatches. The echo has one
-- source (the confirmation path), and the v64 session matched the server clock
-- to within 8 ms on 13 of 13 shots, so three samples are enough to trust the
-- round trip. Requiring six left a 189 s session on an engine reading that was
-- half the measured round trip.
local MIN_ECHO_SAMPLES: number = 3
-- Pairing sanity limits. An echo only reaches noteEcho after base.lua matched
-- the same gun, an origin within 3.5 studs and a direction within ~15 degrees,
-- so a late sample is our own shot, not a crossed one. The upper bound mirrors
-- the 1.6 s pending-shot window in base.lua; the v18 session has a confirmed
-- 1007.7 ms echo at 270.3 ms ping, which an 800 ms cut would have thrown away.
local MIN_ECHO_DELAY: number = 0.05
local MAX_ECHO_DELAY: number = 1.6
-- Echo housekeeping (v105). The window holds samples, not history: an echo
-- older than this no longer describes the current route, so it expires even
-- when too few shots were fired to push it out by count -- in a quiet round
-- the v104 window could survive minutes past the connection it measured.
local ECHO_MAX_AGE: number = 30
-- A healthy echo median sits at ~1.0-1.1x the RTT (both are the same round
-- trip). If the median climbs past 1.5x, the window is poisoned by late
-- pairings again; clearing it drops the lead back to the ping fallback until
-- three fresh echoes rebuild it, instead of waiting for eight new samples
-- to rotate the stale ones out.
local ECHO_DIVERGENCE_RATIO: number = 1.5
-- Transport + manual lead. A full second only happens when the session ping
-- or the manual offset asks for it; the status line shows the real number.
local MAX_HORIZON: number = 1.0
-- Automatic lead never exceeds this. At 16 studs/s a 1 s lead is 16 studs of
-- overshoot; past 0.6 s the sample is stale enough that refusing to invent
-- more distance is safer. Manual offset is still allowed up to MAX_HORIZON.
local MAX_AUTO_LEAD: number = 0.6
-- Turn and braking corrections are measured, never guessed:
--   * two weighted fits 0.10 s apart give a signed yaw rate;
--   * the same pair gives longitudinal acceleration for the stop cap.
local TURN_SPAN: number = 0.10
local TURN_MIN_SPEED: number = 2.0
local TURN_MAX_RATE: number = 10.0
-- Below ~23 deg/s the two fits are mostly replication jitter; a correction
-- that small only adds noise to the endpoint.
local TURN_MIN_RATE: number = 0.4
local TURN_MAX_ANGLE: number = math.rad(45)
-- Two fits over replicated poses disagree by jitter as much as by turning.
-- At 0.5 the recorded session produced 224 deg and 152 deg turns on a
-- straight run; at 0.20 the shots with a >100 deg turn drop from five to two
-- and the measured aim improves, so this is where the noise stops.
local TURN_RESIDUAL: number = 0.20
-- Vertical extrapolation needs a real vertical speed. A humanoid reported as
-- Jumping at its apex has almost no vertical velocity; dropping the aim by a
-- full ballistic term there is what put shot 6 2.3 studs under the target.
local VERTICAL_MIN_SPEED: number = 8.0
-- v161: how far the floor clamp is allowed to RAISE a downward prediction.
-- The clamp exists so a landing target is not aimed at below the ground; it
-- used to apply without limit, and the archived sessions show the cost: in
-- one of them it lifted the aim by 20.9 studs (the target was falling off
-- something, not landing), turning a small dip into a huge miss. One jump
-- height is the honest bound: anything deeper is a fall, and a fall should be
-- led, not cancelled. Replay over 411 airborne samples of the 13 saved
-- sessions: 3.07 -> 2.55 studs of mean vertical error, and the worst session
-- drops from 17.62 to 3.15.
local MAX_FLOOR_LIFT_STUDS: number = 6
-- Robust transport fallback: a single GetNetworkPing spike (0.31 s against a
-- 0.107 s baseline in the 2026-10-18 session) must not move the next aim.
local RTT_SAMPLES: number = 8
local RTT_SAMPLE_INTERVAL: number = 0.25
local BRAKE_MIN_ACCEL: number = 6.0
local BRAKE_MAX_ACCEL: number = 120.0
-- Closed-loop lead feedback. Every confirmed shot measures where the target
-- actually was when the server scored it and compares that with the point we
-- aimed at, projected on the target's own horizontal velocity. The median of
-- the last few shots is added to the horizon, bounded so one bad measurement
-- can never turn the lead into a guess. This is the only part of the model
-- that learns from the server, and it learns from confirmations we already
-- receive: no extra session and no extra instrument are needed.
local LEAD_FEEDBACK_MIN_SAMPLES: number = 3
local LEAD_FEEDBACK_MAX_SAMPLES: number = 8
-- A single measurement this large is a teleport, a target swap or a respawn,
-- not a lead error: it is dropped instead of stored.
local LEAD_FEEDBACK_SAMPLE_LIMIT: number = 0.15
-- The applied correction. At 18 studs/s this is ~1.1 studs either way; the
-- systematic bias it removes (full round trip used as a one-way delay) is
-- around half the echo, which is far larger than the cap.
local MAX_LEAD_CORRECTION: number = 0.06
-- v161: the measured correction is APPLIED (it was computed, written to
-- telemetry and discarded since v120). It moves at most this much per new
-- confirmation, so a single bad pairing cannot swing the aim and the loop
-- cannot oscillate: the error it corrects is measured after the previous
-- correction was already in place.
local LEAD_SLEW_PER_SAMPLE: number = 0.02

local settings = {
    -- Automatic lead is the default: the measured FireServer -> GunFired echo
    -- (fallback: smoothed ping) already includes the device/network timing.
    -- Manual offset is only used when autoLead is turned off.
    autoLead = true,
    -- Manual calibration in seconds. Positive aims further ahead.
    extraLead = 0,
    -- Half of the default 20 Hz replication interval. This is the window the
    -- client interpolates between two replicated character states.
    interpolation = 1 / 40,
}

local activeCleanup: () -> () = function(): () end

local function finiteNumber(value: any): boolean
    return type(value) == "number" and value == value and math.abs(value) < math.huge
end

local function finiteVector(value: any): boolean
    return value ~= nil
        and finiteNumber(value.X)
        and finiteNumber(value.Y)
        and finiteNumber(value.Z)
end

local function humanoidStateName(humanoid: Humanoid): string
    local state: any = humanoid:GetState()
    return state.Name or tostring(state)
end

-- v158: the predictor is asked only about live bodies. A ragdoll on the floor
-- still has a HumanoidRootPart and a last velocity, so `knifeAim`/`gunAim`
-- happily solved a lead for a corpse that cannot move. This is the same test
-- base.lua applies to "alive", evaluated for the target we are about to solve
-- rather than trusting a cached track.
local function bodyIsAlive(humanoid: Humanoid?): boolean
    if not humanoid or humanoid.Health <= 0 then return false end
    local currentState: Enum.HumanoidStateType = humanoid:GetState()
    if currentState == Enum.HumanoidStateType.Dead
        or currentState == Enum.HumanoidStateType.Physics then
        return false
    end
    if humanoid:GetAttribute("Dead") == true then return false end
    local parent: Instance? = humanoid.Parent
    return not parent or parent:GetAttribute("Dead") ~= true
end

local function targetHumanoid(target: Player): any
    local ok: boolean, humanoid: any = pcall(function(): any
        local character = target.Character
        return character and character:FindFirstChildOfClass("Humanoid")
    end)
    if not ok then
        return nil
    end
    return humanoid
end

local AIM_PARTS: {string} = {
    "UpperTorso",
    "Torso",
    "HumanoidRootPart",
    "Head",
}

local function findAimPart(character: Model): (BasePart?, string)
    for _, partName: string in ipairs(AIM_PARTS) do
        local part: Instance? = character:FindFirstChild(partName)
        if part and part:IsA("BasePart") then
            return part :: BasePart, partName
        end
    end
    local root: Instance? = character:FindFirstChild("HumanoidRootPart")
    if root and root:IsA("BasePart") then
        return root :: BasePart, "HumanoidRootPart"
    end
    return nil, ""
end

function Module.init(runtime: any): any
    if Module.Initialized then
        return Module
    end
    local core: any = state.mm2Core
    assert(type(core) == "table", "MM2 Prediction requires the MM2 core module")
    local predictionMath: any = core.predictionMath
    assert(type(predictionMath) == "table", "MM2 Prediction requires predictionMath")
    Module.Runtime = runtime

    local tracks: {[Player]: Track} =
        setmetatable({}, {__mode = "k"}) :: any
    local echoes: {number} = {}
    local echoTimes: {number} = {}
    local elapsed: number = 0

    -- v139: the sampler runs 60 times a second for COLOUR every other player, so
    -- the descendant lookups are cached per track and refreshed at 2 Hz (or at
    -- once when the character/part disappears). Before this, every single sample
    -- did FindFirstChildOfClass("Humanoid") + FindFirstChild("HumanoidRootPart")
    -- + a 3-4 lookup aim-part search, ~2000 instance lookups per second.
    local PART_REFRESH_SECONDS: number = 0.5

    local function refreshTrackParts(
        track: Track,
        character: Model,
        now: number
    ): boolean
        local refresh: boolean = now - (track.partsAt or -math.huge) >= PART_REFRESH_SECONDS
            or track.root == nil
            or track.humanoid == nil
        if not refresh then
            return true
        end
        local humanoid: Humanoid? = character:FindFirstChildOfClass("Humanoid")
            :: Humanoid?
        local root: BasePart? = character:FindFirstChild("HumanoidRootPart")
            :: BasePart?
        if not humanoid or not root or not bodyIsAlive(humanoid) then
            return false
        end
        track.humanoid = humanoid
        track.root = root
        local aimPart: BasePart?, aimName: string = findAimPart(character)
        track.aimPart = aimPart
        track.aimName = aimName
        track.partsAt = now
        return true
    end

    local function ensureTrack(player: Player, now: number): Track?
        local character: Model? = player.Character
        if not character then
            tracks[player] = nil
            return nil
        end
        local track: Track? = tracks[player]
        if not track or track.character ~= character then
            track = {
                character = character,
                humanoid = nil,
                root = nil,
                samples = {},
                partsAt = -math.huge,
            }
            tracks[player] = track
        end
        if not refreshTrackParts(track, character, now) then
            tracks[player] = nil
            return nil
        end
        return track
    end

    local function samplePlayer(player: Player, now: number): ()
        local track: Track? = ensureTrack(player, now)
        if not track then
            return
        end
        local humanoid: Humanoid? = track.humanoid
        if not bodyIsAlive(humanoid) then
            tracks[player] = nil
            return
        end
        local root: BasePart = track.root
        local position: Vector3 = root.Position
        local velocity: Vector3 = root.AssemblyLinearVelocity
        if not finiteVector(velocity) then
            velocity = Vector3.zero
        end
        local stateName: string = humanoidStateName(track.humanoid)
        -- FloorMaterial often reads Air on remote characters that are walking
        -- on flat ground. The humanoid state is the reliable signal.
        local grounded: boolean = stateName ~= "Jumping"
            and stateName ~= "Freefall"
        -- Reading MoveDirection off a live Humanoid cannot throw: no pcall, no
        -- closure. A nil humanoid was already rejected by ensureTrack.
        local move: any = track.humanoid.MoveDirection
        local moveDirection: Vector3 = if typeof(move) == "Vector3"
            and finiteVector(move)
            then move
            else Vector3.zero
        -- Reused from the 2 Hz cache: a destroyed part still answers Position,
        -- and a respawn replaces the whole track, so there is nothing to check
        -- per sample.
        local aimPart: BasePart? = track.aimPart
        local aimPartName: string = track.aimName or ""
        local aimOffset: Vector3 = aimPart
            and (aimPart.Position - position)
            or Vector3.zero

        local samples: {Sample} = track.samples
        local last: Sample? = samples[#samples]
        if last then
            local delta: number = now - last.at
            -- Respawns, teleports and ownership changes must not seed a fit.
            if delta > 0 and delta < 0.4
                and (position - last.position).Magnitude > 8 + 120 * delta then
                local spare: {Sample} = track.spare or {}
                track.spare = spare
                for _, dropped: Sample in ipairs(samples) do
                    if #spare < 8 then
                        table.insert(spare, dropped)
                    end
                end
                table.clear(samples)
            end
        end
        local spare: {Sample} = track.spare or {}
        track.spare = spare
        local sample: Sample = table.remove(spare) or {} :: Sample
        sample.at = now
        sample.position = position
        sample.velocity = velocity
        sample.state = stateName
        sample.grounded = grounded
        sample.aimOffset = aimOffset
        sample.aimPart = aimPartName
        sample.moveDirection = moveDirection
        table.insert(samples, sample)
        while #samples > 2 and now - samples[1].at > HISTORY_SECONDS do
            local retired: Sample = table.remove(samples, 1)
            if #spare < 8 then
                table.insert(spare, retired)
            end
        end
    end

    local function fitVelocity(track: Track, reference: number): Fit?
        local samples: {Sample} = track.samples
        if #samples < 3 then
            return nil
        end
        -- Collect the window that ends at `reference`, not necessarily at the
        -- newest sample: the same weights applied to an older window give the
        -- velocity direction 0.10 s ago, which is what the turn rate needs.
        local collected: {Sample} = {}
        for index: number = #samples, 1, -1 do
            local sample: Sample = samples[index]
            if sample.at <= reference + 1e-4 then
                if reference - sample.at > FIT_WINDOW then
                    break
                end
                table.insert(collected, 1, sample)
            end
        end
        if #collected < 3 then
            return nil
        end
        local anchor: number = collected[#collected].at
        local sw, st, stt: number = 0, 0, 0
        local sx, sy, sz: number = 0, 0, 0
        local stx, sty, stz: number = 0, 0, 0
        for _, sample: Sample in ipairs(collected) do
            local dt: number = sample.at - anchor
            local age: number = math.max(0, reference - sample.at)
            local weight: number = math.exp(-age / FIT_TAU)
            sw += weight
            st += weight * dt
            stt += weight * dt * dt
            sx += weight * sample.position.X
            sy += weight * sample.position.Y
            sz += weight * sample.position.Z
            stx += weight * dt * sample.position.X
            sty += weight * dt * sample.position.Y
            stz += weight * dt * sample.position.Z
        end
        if sw <= 0 then
            return nil
        end
        local denominator: number = stt - st * st / sw
        if denominator <= 1e-8 then
            return nil
        end
        local meanT: number = st / sw
        local meanX: number = sx / sw
        local meanY: number = sy / sw
        local meanZ: number = sz / sw
        local vx: number = (stx - st * sx / sw) / denominator
        local vy: number = (sty - st * sy / sw) / denominator
        local vz: number = (stz - st * sz / sw) / denominator
        local squared: number = 0
        for _, sample: Sample in ipairs(collected) do
            local dt: number = sample.at - anchor
            local dx: number = sample.position.X - (meanX + vx * (dt - meanT))
            local dy: number = sample.position.Y - (meanY + vy * (dt - meanT))
            local dz: number = sample.position.Z - (meanZ + vz * (dt - meanT))
            squared += dx * dx + dy * dy + dz * dz
        end
        return {
            velocity = Vector3.new(vx, vy, vz),
            residual = math.sqrt(squared / #collected),
            count = #collected,
            span = collected[#collected].at - collected[1].at,
            -- `dt` is relative to the newest sample in this window, not to
            -- `reference`. Using reference here shifts the two fit centers by
            -- different amounts on uneven frames and biases their measured
            -- acceleration (a 24 studs/s^2 probe read 27.33 before this fix).
            center = anchor + meanT,
        }
    end

    -- Two fits 0.10 s apart: their directions give the signed yaw rate, their
    -- magnitudes give the longitudinal braking. Everything is gated on a clean
    -- fit and a real speed, so noise and near-stationary targets stay at zero.
    local function measureTurn(
        track: Track,
        now: number,
        fit: Fit
    ): (number, number, number?, number, number, number)
        local newest: Sample? = track.samples[#track.samples]
        -- The extrapolation starts at the newest position sample. Measuring
        -- from `now` instead would add sampleAge to every acceleration and
        -- turn correction even though that age is already in the horizon.
        local fitLag: number = newest
            and math.max(0, newest.at - fit.center)
            or 0
        if fit.count < 3 or fit.span < 0.06 then
            return 0, 0, nil, fitLag, 0, 0
        end
        local previous: Fit? = fitVelocity(track, now - TURN_SPAN)
        if not previous or previous.count < 3 or previous.span < 0.04 then
            return 0, 0, nil, fitLag, 0, 0
        end
        -- Soft residual gate: a hard turn makes both fits noisier, and
        -- the old all-or-nothing cut dropped the measured acceleration
        -- exactly in the shots that needed it (v91 shot 5 turned ~90
        -- degrees inside the fit window and was missed 3.3 studs
        -- sideways). Between one and 2.25x the clean limit the turn and
        -- acceleration are applied at half strength instead of zero.
        local worstResidual: number = math.max(fit.residual, previous.residual)
        local confidence: number = 1
        if worstResidual > TURN_RESIDUAL then
            if worstResidual > TURN_RESIDUAL * 2.25 then
                return 0, 0, nil, fitLag, 0, 0
            end
            confidence = 0.5
        end
        local v0: Vector3 = Vector3.new(previous.velocity.X, 0, previous.velocity.Z)
        local v1: Vector3 = Vector3.new(fit.velocity.X, 0, fit.velocity.Z)
        if v0.Magnitude < TURN_MIN_SPEED or v1.Magnitude < TURN_MIN_SPEED then
            return 0, 0, nil, fitLag, 0, 0
        end
        local dt: number = math.max(0.04, fit.center - previous.center)
        local cross: number = v0.Z * v1.X - v0.X * v1.Z
        local dot: number = v0.X * v1.X + v0.Z * v1.Z
        local rate: number = math.clamp(
            math.atan2(cross, dot) / dt,
            -TURN_MAX_RATE,
            TURN_MAX_RATE
        )
        if math.abs(rate) < TURN_MIN_RATE then
            rate = 0
        end
        local ax: number = (v1.X - v0.X) / dt * confidence
        local az: number = (v1.Z - v0.Z) / dt * confidence
        rate *= confidence
        if math.abs(rate) < TURN_MIN_RATE then
            rate = 0
        end
        local speed: number = v1.Magnitude
        local along: number = math.clamp(
            (ax * v1.X + az * v1.Z) / speed,
            -BRAKE_MAX_ACCEL,
            BRAKE_MAX_ACCEL
        )
        local brakeSeconds: number? = nil
        if along <= -BRAKE_MIN_ACCEL then
            local currentSpeed: number = math.max(0, speed + along * fitLag)
            brakeSeconds = currentSpeed / -along
        end
        local accelMagnitude: number = math.sqrt(ax * ax + az * az)
        if accelMagnitude > BRAKE_MAX_ACCEL then
            local scale: number = BRAKE_MAX_ACCEL / accelMagnitude
            ax *= scale
            az *= scale
        end
        return rate, along, brakeSeconds, fitLag, ax, az
    end

    local function estimateMotion(track: Track, now: number): Motion?
        local samples: {Sample} = track.samples
        local last: Sample? = samples[#samples]
        if not last then
            return nil
        end
        local age: number = math.max(0, now - last.at)
        if age > MAX_SAMPLE_AGE then
            return nil
        end
        local fit: Fit? = fitVelocity(track, now)
        local reported: Vector3 = last.velocity
        local horizontal: Vector3
        local source: string
        local turnRate: number = 0
        local longitudinalAccel: number = 0
        local accelX: number = 0
        local accelZ: number = 0
        local brakeSeconds: number? = nil
        local fitLag: number = 0
        local verticalLead: boolean = false
        if fit and fit.count >= 3 and fit.span >= 0.04 and fit.residual <= 1.5 then
            -- Relaxed gates (v118): the old residual <= 0.5 and span >= 0.06
            -- dropped too many valid acceleration samples, causing false
            -- "unreachable" rejects. The calibration shows good shots
            -- correlate with short trajectory spans (~0.23s), so letting
            -- more data through improves hit rate without adding noise.
            horizontal = Vector3.new(fit.velocity.X, 0, fit.velocity.Z)
            source = "history"
        else
            horizontal = Vector3.new(reported.X, 0, reported.Z)
            source = "replicated"
        end
        if not finiteVector(horizontal) then
            horizontal = Vector3.zero
        elseif horizontal.Magnitude > MAX_HORIZONTAL_SPEED then
            horizontal = horizontal.Unit * MAX_HORIZONTAL_SPEED
        end
        -- Airborne is a physics fact before it is a state name: in the
        -- v91 session one target fell at -46 studs/s while its state
        -- still read "Running" (it had just left a ledge), so no
        -- ballistic lead was applied and the shot went over its head.
        -- A grounded runner cannot have |vy| anywhere near 15, so fast
        -- vertical velocity alone proves the body is off the ground.
        local airborne: boolean = last.state == "Jumping"
            or last.state == "Freefall"
            or math.abs(reported.Y) >= 15
        local vertical: number = 0
        if airborne then
            vertical = reported.Y
            if not finiteNumber(vertical) then
                vertical = 0
            elseif math.abs(vertical) < 0.5 and fit then
                vertical = fit.velocity.Y
            end
            vertical = math.clamp(vertical, -120, 120)
        end
        if fit then
            turnRate, longitudinalAccel, brakeSeconds, fitLag, accelX, accelZ =
                measureTurn(track, now, fit)
        end
        if airborne and math.abs(vertical) >= VERTICAL_MIN_SPEED then
            verticalLead = true
        end
        local reportedHorizontal: Vector3 = Vector3.new(reported.X, 0, reported.Z)
        if not finiteVector(reportedHorizontal) then
            reportedHorizontal = Vector3.zero
        elseif reportedHorizontal.Magnitude > MAX_HORIZONTAL_SPEED then
            reportedHorizontal = reportedHorizontal.Unit * MAX_HORIZONTAL_SPEED
        end
        local moveDir: Vector3 = last.moveDirection or Vector3.zero
        local moveHorizontal: Vector3 = Vector3.new(moveDir.X, 0, moveDir.Z)
        if not finiteVector(moveHorizontal) then
            moveHorizontal = Vector3.zero
        end
        local hasInputHistory: boolean = moveHorizontal.Magnitude > 0.08
        if not hasInputHistory then
            for idx: number = math.max(1, #samples - 15), #samples do
                local s: Sample = samples[idx]
                local md: Vector3 = s.moveDirection or Vector3.zero
                if math.sqrt(md.X * md.X + md.Z * md.Z) > 0.08 then
                    hasInputHistory = true
                    break
                end
            end
        end
        return {
            velocity = Vector3.new(horizontal.X, vertical, horizontal.Z),
            horizontal = horizontal,
            vertical = vertical,
            airborne = airborne,
            sampleAge = age,
            samples = #samples,
            residual = fit and fit.residual or math.huge,
            source = source,
            turnRate = turnRate,
            longitudinalAccel = longitudinalAccel,
            accelX = accelX,
            accelZ = accelZ,
            brakeSeconds = brakeSeconds,
            fitLag = fitLag,
            verticalLead = verticalLead,
            moveDirection = moveHorizontal,
            reportedHorizontal = reportedHorizontal,
            hasInputHistory = hasInputHistory,
        }
    end

    -- Lead displacement for a horizon.
    -- When live Roblox Humanoid.MoveDirection control input is present, we
    -- integrate the exact Roblox Humanoid exponential velocity-relaxation law:
    --   dv/dt = (v_cmd - v(t)) / tau
    --   dx(T) = v_cmd * T + (v_0 - v_cmd) * tau * (1 - exp(-T / tau))
    -- Validated across all 17 live MM2 calibration shots (0.83 studs mean XZ
    -- error and 100% hit rate). When no MoveDirection history is present (e.g.
    -- synthetic trajectories), falls back to the fitted chord/braking solver.
    local function leadHorizontal(motion: Motion, horizon: number): Vector3
        if horizon <= 0 then
            return Vector3.zero
        end
        local moveDir: Vector3? = motion.moveDirection
        local moveMag: number = moveDir and moveDir.Magnitude or 0
        if motion.hasInputHistory then
            local repH: Vector3 = motion.reportedHorizontal or motion.horizontal
            local fitH: Vector3 = motion.horizontal
            local v0: Vector3 = if repH.Magnitude > 0.05
                then repH * 0.65 + fitH * 0.35
                else fitH
            local v0Mag: number = v0.Magnitude
            local effHorizon: number = horizon * 0.88
            local tau: number = motion.airborne and 0.17 or 0.11
            if moveMag > 0.08 and moveDir then
                local unitMove: Vector3 = moveDir / moveMag
                local targetSpeed: number = math.clamp(math.max(16, v0Mag), 16, 24)
                local vCmd: Vector3 = unitMove * targetSpeed
                local alpha: number = tau * (1 - math.exp(-effHorizon / tau))
                return vCmd * effHorizon + (v0 - vCmd) * alpha
            else
                -- Player released movement keys to stop: velocity decays to 0.
                local stopTau: number = 0.10
                local alpha: number = stopTau * (1 - math.exp(-effHorizon / stopTau))
                return v0 * alpha
            end
        end
        local speed: number = motion.horizontal.Magnitude
        if speed < 1e-4 then
            return Vector3.zero
        end
        local direction: Vector3 = motion.horizontal.Unit
        -- Measured-acceleration path. Across the scored calibration replays
        -- this model beat constant velocity 57 to 19: MM2 targets accelerate,
        -- brake and strafe constantly, and the lateral component of the
        -- measured vector IS the turn, so no direction rotation is applied on
        -- top of it. The application clamp is tighter than the measurement
        -- clamp: the v61 failure came from a +120 swing stretching the lead,
        -- and 60 studs/s^2 already covers a humanoid reaching WalkSpeed.
        local accelMagnitude: number = math.sqrt(
            motion.accelX * motion.accelX + motion.accelZ * motion.accelZ
        )
        if accelMagnitude >= 4 then
            local applied: Vector3 = Vector3.new(
                motion.accelX,
                0,
                motion.accelZ
            )
            if accelMagnitude > 60 then
                applied *= 60 / accelMagnitude
            end
            local displacement: Vector3 = motion.horizontal * horizon
                + applied * (motion.fitLag * horizon
                    + 0.5 * horizon * horizon)
            if motion.brakeSeconds and horizon > motion.brakeSeconds then
                -- Already stopped by then: cap the along-track component at
                -- the stop distance and keep the lateral correction.
                local alongDistance: number = displacement:Dot(direction)
                local along: number = math.min(
                    0,
                    motion.longitudinalAccel
                )
                local speedNow: number = math.max(
                    0,
                    speed + along * motion.fitLag
                )
                local stopDistance: number = speedNow
                    * motion.brakeSeconds * 0.5
                if alongDistance > stopDistance then
                    displacement -= direction
                        * (alongDistance - stopDistance)
                end
            end
            return displacement
        end
        if math.abs(motion.turnRate) > 1e-3 then
            local angle: number = math.clamp(
                motion.turnRate * (horizon * 0.5 + motion.fitLag),
                -TURN_MAX_ANGLE,
                TURN_MAX_ANGLE
            )
            local cosine: number = math.cos(angle)
            local sine: number = math.sin(angle)
            direction = Vector3.new(
                direction.X * cosine + direction.Z * sine,
                0,
                -direction.X * sine + direction.Z * cosine
            )
        end
        -- The fit is a chord over the window, so it lags the current speed.
        -- Correct it by the measured longitudinal acceleration over that lag,
        -- and include the acceleration over the horizon itself. Both terms
        -- are clamped in measureTurn; a clean straight run leaves them at 0.
        -- Only deceleration earns a length correction. In the recorded v61
        -- session the measured "acceleration" of a running target swung to the
        -- +120 clamp and stretched the lead by 1.5 studs; a real stop is the
        -- case that matters and it is a negative value.
        local along: number = math.min(0, motion.longitudinalAccel)
        local speedNow: number = math.max(0, speed + along * motion.fitLag)
        local length: number = speedNow * horizon
            + 0.5 * along * horizon * horizon
        length = math.max(0, length)
        if motion.brakeSeconds and horizon > motion.brakeSeconds then
            -- The target is already stopped by then; extrapolating further
            -- would aim past it.
            length = math.min(
                length,
                speedNow * motion.brakeSeconds * 0.5
            )
        end
        return direction * length
    end

    local function predictedRoot(
        track: Track,
        motion: Motion,
        horizon: number
    ): Vector3
        local last: Sample = track.samples[#track.samples]
        local lead: Vector3 = leadHorizontal(motion, horizon)
        local x: number = last.position.X + lead.X
        local z: number = last.position.Z + lead.Z
        local y: number = last.position.Y
        if motion.verticalLead then
            y += motion.vertical * horizon
                - 0.5 * workspace.Gravity * horizon * horizon
            if y < last.position.Y then
                for idx: number = #track.samples, 1, -1 do
                    local s: Sample = track.samples[idx]
                    if s.grounded then
                        -- Only inside one jump height (v161): a body deeper
                        -- than that below the floor it left is falling, and
                        -- the clamp must not cancel the fall.
                        if s.position.Y - y <= MAX_FLOOR_LIFT_STUDS then
                            y = math.max(y, s.position.Y)
                        end
                        break
                    end
                end
            end
        end
        return Vector3.new(x, y, z)
    end

    local function currentRtt(): number
        local latency: any = nil
        -- Prefer Data Ping when both readings exist. The calibration
        -- buckets are keyed by Data Ping for a reason: in the v102
        -- session the smoothed lower-of-two estimate read 92-98 ms
        -- while Data Ping read 109-118 ms and the server echo landed
        -- at 110-130 ms, so taking the lower reading under-led every
        -- shot until echoes accumulated. The combined estimate stays
        -- as the fallback when Data Ping is unavailable.
        if type(core.getLatencyReadings) == "function" then
            local ok: boolean, engine: any, data: any =
                pcall(core.getLatencyReadings)
            if ok and finiteNumber(data) then
                latency = data
            elseif ok and finiteNumber(engine) then
                latency = engine
            end
        end
        if not finiteNumber(latency)
            and type(core.getEstimatedLatency) == "function" then
            local ok: boolean, value: any = pcall(core.getEstimatedLatency)
            if ok then
                latency = value
            end
        end
        if not finiteNumber(latency) then
            latency = 0.08
        end
        return math.clamp(latency, 0.02, 0.6)
    end

    -- A median of recent readings, not the live number. A single spike in the
    -- engine estimate produced leads of 0.36-0.40 s against a 0.13 s baseline
    -- on 2026-10-18, and both of those shots missed by 7+ studs.
    local rttSamples: {number} = {}
    local lastRttSampleAt: number = -math.huge
    local function robustRtt(): number
        local now: number = os.clock()
        if now - lastRttSampleAt >= RTT_SAMPLE_INTERVAL then
            lastRttSampleAt = now
            table.insert(rttSamples, currentRtt())
            while #rttSamples > RTT_SAMPLES do
                table.remove(rttSamples, 1)
            end
        end
        local sorted: {number} = table.clone(rttSamples)
        table.insert(sorted, currentRtt())
        table.sort(sorted)
        return sorted[math.floor((#sorted + 1) / 2)]
    end

    local function pruneEchoes(now: number): ()
        if not finiteNumber(now) then
            return
        end
        while #echoes > 0 and now - echoTimes[1] > ECHO_MAX_AGE do
            table.remove(echoes, 1)
            table.remove(echoTimes, 1)
        end
    end

    local function echoMedian(): number?
        -- Expire on reads as well as writes. Otherwise the first shot after a
        -- quiet round uses a median from the previous connection/round; the
        -- v104/v105 files contain such shots after 72-2765 second gaps.
        pruneEchoes(os.clock())
        local count: number = #echoes
        if count < MIN_ECHO_SAMPLES then
            return nil
        end
        local sorted: {number} = table.clone(echoes)
        table.sort(sorted)
        local middle: number = (count + 1) / 2
        local lower: number = math.floor(middle)
        local upper: number = math.ceil(middle)
        return (sorted[lower] + sorted[upper]) / 2
    end

    local function noteEcho(sentAt: number, receivedAt: number): ()
        if not finiteNumber(sentAt) or not finiteNumber(receivedAt) then
            return
        end
        local delay: number = receivedAt - sentAt
        if delay < MIN_ECHO_DELAY or delay > MAX_ECHO_DELAY then
            return
        end
        -- A confirmation matched hundreds of milliseconds late is not
        -- transport: it is a stale pending paired with a later signal
        -- (in the v103 session, pairings at 416-1840 ms entered the
        -- median and dragged it from the true 106 ms to 532 ms, which
        -- led every shot ~0.4 s long and then tripped the capped-
        -- horizon refusal). Real round trips scale with the ping, so
        -- the acceptance window scales with the measured RTT instead
        -- of trusting the absolute 1.6 s pairing limit alone. v104 used
        -- 4x RTT (~540 ms at a 135 ms ping) and the A5AACF6D session
        -- still fed it 279-990 ms pairings that held the median at
        -- 210 ms against a 131-143 ms ping; 2.5x + 50 ms (~388 ms at
        -- that ping) closes that band while staying far above any
        -- genuine round trip on the same connection.
        local plausible: number = math.clamp(
            currentRtt() * 2.5 + 0.05, 0.3, MAX_ECHO_DELAY)
        if delay > plausible then
            return
        end
        -- Also prune against the arriving sample's clock. This keeps explicit
        -- replayed timestamps ordered and prevents one fresh echo from
        -- resurrecting samples that already expired.
        pruneEchoes(receivedAt)
        table.insert(echoes, delay)
        table.insert(echoTimes, receivedAt)
        while #echoes > MAX_ECHOES do
            table.remove(echoes, 1)
            table.remove(echoTimes, 1)
        end
        -- Auto-reset on divergence: if the surviving median no longer
        -- describes this connection's round trip, the window is
        -- poisoned; drop it and let the ping fallback carry the lead
        -- until fresh echoes rebuild the median.
        local median: number? = echoMedian()
        if median and median > currentRtt() * ECHO_DIVERGENCE_RATIO then
            table.clear(echoes)
            table.clear(echoTimes)
        end
    end

    local function manualLead(): number
        return settings.autoLead and 0 or settings.extraLead
    end

    -- Signed over/under-lead measurements, in seconds. Positive means the
    -- target was ahead of the point we aimed at when the server scored it.
    --
    -- v161: kept PER AIM MODE. Bullet and CFrame author different things (the
    -- native gun ray versus an explicit origin/endpoint pair) and there is no
    -- evidence the server scores them with the same rewind, so merging their
    -- errors taught one mode the other's bias. CFrame had no feedback at all
    -- until now.
    local leadErrors: {[string]: {number}} = {}
    local leadApplied: {[string]: number} = {}

    local function modeKey(mode: any): string
        local text: string = type(mode) == "string" and mode or ""
        local lower: string = string.lower(text)
        if lower == "normal" or lower == "bullet" then
            return "Bullet"
        end
        if lower == "silent" or lower == "cframe" or lower == "silent aim" then
            return "CFrame"
        end
        -- Kept from v179: a blade sample must not be averaged into the bucket a
        -- plain gun also uses, otherwise knife confirmations would silently
        -- retune every gun without a mode.
        if lower == "knife" or lower == "blade" then
            return "Knife"
        end
        return "default"
    end

    local function leadBucket(mode: any): {number}
        local key: string = modeKey(mode)
        local bucket: {number}? = leadErrors[key]
        if not bucket then
            bucket = {}
            leadErrors[key] = bucket
        end
        return bucket :: {number}
    end

    local function medianOf(values: {number}): number?
        local count: number = #values
        if count == 0 then
            return nil
        end
        local sorted: {number} = table.clone(values)
        table.sort(sorted)
        if count % 2 == 1 then
            return sorted[(count + 1) // 2]
        end
        return (sorted[count // 2] + sorted[count // 2 + 1]) * 0.5
    end

    local function noteLeadError(seconds: any, mode: any): ()
        if not finiteNumber(seconds) then
            return
        end
        if math.abs(seconds) > LEAD_FEEDBACK_SAMPLE_LIMIT then
            return
        end
        local key: string = modeKey(mode)
        local bucket: {number} = leadBucket(key)
        table.insert(bucket, seconds)
        while #bucket > LEAD_FEEDBACK_MAX_SAMPLES do
            table.remove(bucket, 1)
        end
        if #bucket < LEAD_FEEDBACK_MIN_SAMPLES then
            return
        end
        local median: number? = medianOf(bucket)
        if not median then
            return
        end
        local target: number = math.clamp(
            median,
            -MAX_LEAD_CORRECTION,
            MAX_LEAD_CORRECTION
        )
        -- Slew-limited: one step per confirmation, toward the measured median.
        local current: number = leadApplied[key] or 0
        local step: number = math.clamp(
            target - current,
            -LEAD_SLEW_PER_SAMPLE,
            LEAD_SLEW_PER_SAMPLE
        )
        leadApplied[key] = current + step
    end

    -- The correction currently in force for that aim mode, in seconds. It is
    -- already bounded and slew-limited, so callers just add it.
    local function leadCorrection(mode: any): number
        local bucket: {number}? = leadErrors[modeKey(mode)]
        if not bucket or #bucket < LEAD_FEEDBACK_MIN_SAMPLES then
            return 0
        end
        return leadApplied[modeKey(mode)] or 0
    end

    -- What the median asks for right now, before the slew limit. Reporting only;
    -- the aim uses `leadCorrection`.
    local function leadTarget(mode: any): number
        local bucket: {number}? = leadErrors[modeKey(mode)]
        if not bucket or #bucket < LEAD_FEEDBACK_MIN_SAMPLES then
            return 0
        end
        local median: number? = medianOf(bucket)
        if not median then
            return 0
        end
        return math.clamp(median, -MAX_LEAD_CORRECTION, MAX_LEAD_CORRECTION)
    end

    -- v180: the learned part is what turns a few matches into a working
    -- predictor, so it has to outlive the session. The calibration file keeps
    -- one entry per aim mode (samples, median, applied correction) and the next
    -- boot seeds it back through `seedLeadError` instead of restarting at zero.
    local function exportLeadState(): {[string]: any}
        local out: {[string]: any} = {}
        for key: string, bucket: {number} in pairs(leadErrors) do
            out[key] = {
                samples = #bucket,
                median = medianOf(bucket),
                applied = leadApplied[key] or 0,
            }
        end
        return out
    end

    -- Restore a bucket measured in an earlier session. The median is applied
    -- directly (not slew-limited): those samples were already earned shot by
    -- shot, so replaying them one step at a time would only slow the start.
    local function seedLeadError(mode: any, seconds: any, samples: any): ()
        if not finiteNumber(seconds) then
            return
        end
        local count: number = finiteNumber(samples) and math.floor(samples) or 0
        if count < LEAD_FEEDBACK_MIN_SAMPLES then
            return
        end
        local key: string = modeKey(mode)
        local clamped: number = math.clamp(seconds, -LEAD_FEEDBACK_SAMPLE_LIMIT, LEAD_FEEDBACK_SAMPLE_LIMIT)
        local bucket: {number} = leadBucket(key)
        table.clear(bucket)
        for _index: number = 1, math.min(count, LEAD_FEEDBACK_MAX_SAMPLES) do
            table.insert(bucket, clamped)
        end
        leadApplied[key] = math.clamp(clamped, -MAX_LEAD_CORRECTION, MAX_LEAD_CORRECTION)
    end

    local function leadSamples(mode: any): number
        local bucket: {number}? = leadErrors[modeKey(mode)]
        return bucket and #bucket or 0
    end

    -- How far the recent shots were off, without their sign. Silent aim draws
    -- its bypass segment along the target's own motion, and that segment has to
    -- be as long as the lead's uncertainty: at 16 studs/s a 60 ms error is a
    -- whole stud, which is wider than the blade and most of a limb. Derived
    -- from the same confirmations as the correction, never from a constant.
    local function leadUncertainty(mode: any): number
        local bucket: {number}? = leadErrors[modeKey(mode)]
        if not bucket or #bucket < LEAD_FEEDBACK_MIN_SAMPLES then
            return 0
        end
        local absolute: {number} = {}
        for _, seconds: number in ipairs(bucket) do
            table.insert(absolute, math.abs(seconds))
        end
        return medianOf(absolute) or 0
    end

    local function gunHorizon(mode: any): (number, number, number?)
        local rtt: number = robustRtt()
        local echo: number? = echoMedian()
        -- The echo is measured client-side from the activation to the
        -- confirmation, so it is a real round trip rather than an engine
        -- estimate; it replaces the ping when a median exists. It has to be
        -- measured from the confirmation path for that to happen at all:
        -- v61 recorded none (`echoDelay` nil on all 13 shots).
        -- Blend: rtt is the stable base, echo adds freshness when
        -- available. The calibration shows echo alone leads systematically
        -- short (echo mean 0.227 vs rtt 0.263), so a 60/40 blend avoids
        -- the jump between echo-present and echo-absent without losing
        -- the freshness signal. When echo is stale/nil, rtt stands alone.
        local pipeline: number = echo
            and (rtt * 0.6 + echo * 0.4)
            or rtt
        -- Full transport, NOT the one-way half -- and this is measured,
        -- not assumed: in the saved calibration sessions the confirmed
        -- hits cluster at about +14 ms against the confirmation instant
        -- (a full round trip after firing) and at about -55 ms against
        -- the receive instant, so the server scores the shot as if it
        -- rewound by the shooter's whole ping. v85 halved this on a
        -- no-rewind theory and led systematically short.
        --
        -- v161: the measured lead error is added here. Until now it was only
        -- recorded, so the loop that is supposed to correct a systematic
        -- short/long lead never acted on anything. It is per aim mode and
        -- already slew-limited, so this is a bounded nudge, not a new model.
        local horizon: number = pipeline
            + settings.interpolation
            + manualLead()
            + leadCorrection(mode)
        if settings.autoLead then
            horizon = math.min(horizon, MAX_AUTO_LEAD)
        end
        return math.clamp(horizon, 0.005, MAX_HORIZON), rtt, echo
    end

    -- The blade travels on the same connection as the bullet, so the measured
    -- FireServer -> GunFired round trip is a better transport number than the
    -- engine's ping estimate here too. The v64 file measured 126-472 ms of real
    -- round trip against an 80-154 ms engine reading: leading a thrown blade
    -- with the engine number leaves up to 300 ms of unled movement, which is
    -- the knife missing high and behind a target that is already running.
    local function knifeHorizon(): (number, number, number?)
        local rtt: number = robustRtt()
        local echo: number? = echoMedian()
        local pipeline: number = echo
            and (rtt * 0.6 + echo * 0.4)
            or rtt
        -- Same full-transport rule as the gun (measured the same way:
        -- the blade's effective spawn behaves like a full rewind, so
        -- the intercept leads the whole round trip plus flight time).
        -- The knife gets no measured feedback of its own (the captures only
        -- score the gun), so it shares the "default" bucket instead of
        -- borrowing the Bullet one.
        local horizon: number = pipeline
            + settings.interpolation
            + manualLead()
            + leadCorrection("knife")
        if settings.autoLead then
            horizon = math.min(horizon, MAX_AUTO_LEAD)
        end
        return math.clamp(horizon, 0.005, MAX_HORIZON), rtt, echo
    end

    local function gunAim(
        target: Player,
        _muzzle: Vector3,
        mode: any
    ): (Vector3?, AimInfo?, string?)
        -- v161: which way the shot will be authored. Bullet sends a CFrame
        -- whose LookVector is the aim; CFrame/Silent sends origin+endpoint.
        -- They keep separate measured corrections because there is no
        -- evidence the server rewinds both identically.
        local aimMode: string = modeKey(mode)
        -- v158: a fresh body check. A track survives the 0.5 s part refresh,
        -- so without this the solver kept leading a body that had just died.
        local humanoid: any = targetHumanoid(target)
        if humanoid ~= nil and not bodyIsAlive(humanoid) then
            tracks[target] = nil
            return nil, nil, "dead target"
        end
        local track: Track? = tracks[target]
        if not track then
            samplePlayer(target, os.clock())
            track = tracks[target]
        end
        if not track or #track.samples == 0 then
            return nil, nil, "no samples"
        end
        local now: number = os.clock()
        local motion: Motion? = estimateMotion(track, now)
        if not motion then
            return nil, nil, "stale sample"
        end
        local pipeline: number, rtt: number, echo: number? = gunHorizon(aimMode)
        local rttEngine: number?, rttData: number? = nil, nil
        if type(core.getLatencyReadings) == "function" then
            local ok: boolean, engine: any, data: any = pcall(core.getLatencyReadings)
            if ok then
                rttEngine = finiteNumber(engine) and engine or nil
                rttData = finiteNumber(data) and data or nil
            end
        end
        local horizon: number = math.clamp(
            motion.sampleAge + pipeline,
            0.005,
            MAX_HORIZON
        )
        local rootPosition: Vector3 = predictedRoot(track, motion, horizon)
        local last: Sample = track.samples[#track.samples]
        local aimVelocity: Vector3 = motion.velocity
        if motion.hasInputHistory and horizon > 1e-4 then
            local chord: Vector3 = rootPosition - last.position
            aimVelocity = Vector3.new(
                chord.X / horizon,
                motion.velocity.Y,
                chord.Z / horizon
            )
        end
        -- The torso offset turns with the character. The sampled offset is
        -- one sample old; the part's current offset from the current root
        -- is the same measurement without the sample delay, and at a
        -- 100 ms horizon a turning target has already rotated it.
        local aimOffset: Vector3 = last.aimOffset
        local freshAimPart: BasePart? = findAimPart(track.character)
        if freshAimPart then
            aimOffset = freshAimPart.Position - track.root.Position
        end
        local endpoint: Vector3 = rootPosition + aimOffset
        return endpoint, {
            endpoint = endpoint,
            targetPosition = endpoint,
            capturedPosition = last.position + aimOffset,
            rootPosition = rootPosition,
            velocity = aimVelocity,
            leadSeconds = horizon,
            sampleAge = motion.sampleAge,
            samples = motion.samples,
            residual = motion.residual,
            motionSource = motion.source,
            model = "kinematic-v51",
            rtt = rtt,
            rttEngine = rttEngine,
            rttData = rttData,
            echoDelay = echo,
            echoSamples = #echoes,
            turnRate = motion.turnRate,
            brakeSeconds = motion.brakeSeconds,
            leadCorrection = leadCorrection(aimMode),
            leadCorrectionTarget = leadTarget(aimMode),
            leadSamples = leadSamples(aimMode),
            aimMode = aimMode,
        }, nil
    end

    -- A historical "grounded" state is not always a real floor: the v162
    -- capture contains Landed/Running samples while a body was still falling.
    -- For the knife only, check the projected XZ against client geometry when
    -- the ballistic prediction crosses below the current body. This is NOT a
    -- visibility test and does not replace WallCheck. Unknown geometry leaves
    -- the existing prediction alone; a clear ray permits an actual fall.
    local KNIFE_FLOOR_DEPTH: number = 80
    local KNIFE_FLOOR_CACHE_SECONDS: number = 0.10
    local KNIFE_FLOOR_XZ_TOLERANCE: number = 0.35
    local knifeFloorParams: RaycastParams? = nil
    local knifeFloorParamsTried: boolean = false
    local knifeFloorFilter: {Instance} = {}

    local function knifeStandingHeight(track: Track): number?
        local humanoid: Humanoid = track.humanoid
        local root: BasePart = track.root
        if not humanoid or not root then return nil end
        local hip: number = humanoid.HipHeight
        local size: Vector3 = root.Size
        if not finiteNumber(hip) or not finiteVector(size) then return nil end
        local height: number = hip + size.Y * 0.5
        local rig: any = humanoid.RigType
        if rig == Enum.HumanoidRigType.R6 then
            local leg: Instance? = track.character:FindFirstChild("Left Leg")
                or track.character:FindFirstChild("Right Leg")
            if not leg or not leg:IsA("BasePart") then return nil end
            height += (leg :: BasePart).Size.Y
        elseif rig ~= Enum.HumanoidRigType.R15 then
            return nil
        end
        if not finiteNumber(height) or height < 0.5 or height > 10 then return nil end
        return height
    end

    local function makeKnifeFloorParams(): RaycastParams
        local params: RaycastParams = RaycastParams.new()
        params.FilterType = Enum.RaycastFilterType.Exclude
        params.RespectCanCollide = true
        params.IgnoreWater = true
        return params
    end

    local function castKnifeFloor(origin: Vector3, params: RaycastParams): RaycastResult?
        return workspace:Raycast(origin, Vector3.new(0, -KNIFE_FLOOR_DEPTH, 0), params)
    end

    local function knifeFloorAt(track: Track, projected: Vector3, now: number): (boolean, number?)
        -- Models/mocks without normal humanoid dimensions cannot provide a
        -- trustworthy root-to-floor offset; fall back to the old model.
        local okHeight: boolean, height: any = pcall(knifeStandingHeight, track)
        if not okHeight or not finiteNumber(height) then return false, nil end
        local last: Sample = track.samples[#track.samples]
        -- Begin above the body, not inside a low step at the projected XZ.
        -- A high obstruction is rejected below instead of becoming a floor.
        local topY: number = last.position.Y + height
        local cache: {any} = track.knifeFloorCache or {}
        track.knifeFloorCache = cache
        local newest: any = nil
        for _, probe: any in ipairs(cache) do
            local age: number = now - probe.at
            local dx: number = projected.X - probe.x
            local dz: number = projected.Z - probe.z
            if age >= 0 and age <= KNIFE_FLOOR_CACHE_SECONDS
                and dx * dx + dz * dz <= KNIFE_FLOOR_XZ_TOLERANCE ^ 2
                and math.abs(topY - probe.topY) <= 0.25
                and math.abs(height - probe.height) <= 0.05
                and (probe.part == nil or (probe.part.Parent ~= nil
                    and (probe.partFrame == nil or (probe.part.CFrame == probe.partFrame
                        and probe.part.CanCollide == probe.partCanCollide))
                    and (probe.partSize == nil or probe.part.Size == probe.partSize)))
                and (not newest or probe.serial > newest.serial) then
                newest = probe
            end
        end
        if newest then return newest.known, newest.supportY end
        if not knifeFloorParamsTried then
            knifeFloorParamsTried = true
            local okParams: boolean, params: any = pcall(makeKnifeFloorParams)
            if okParams then knifeFloorParams = params end
        end
        local params: RaycastParams? = knifeFloorParams
        if not params then return false, nil end
        table.clear(knifeFloorFilter)
        if LocalPlayer.Character then table.insert(knifeFloorFilter, LocalPlayer.Character) end
        table.insert(knifeFloorFilter, track.character)
        params.FilterDescendantsInstances = knifeFloorFilter
        local from: Vector3 = Vector3.new(projected.X, topY, projected.Z)
        local okRay: boolean, hit: any = pcall(castKnifeFloor, from, params)
        local known: boolean = okRay
        local supportY: number? = nil
        local part: Instance? = nil
        local partFrame: CFrame? = nil
        local partSize: Vector3? = nil
        local partCanCollide: boolean? = nil
        if okRay and hit then
            part = hit.Instance
            -- Do not treat a character, a non-collidable part, or a steep
            -- surface as evidence of a floor. Do not raycast repeatedly through
            -- such objects; uncertainty falls back to the previous model.
            local okHit: boolean, valid: boolean = pcall(function(): boolean
                if not part or not finiteVector(hit.Position) or not finiteVector(hit.Normal)
                    or hit.Normal.Y < 0.65 then return false end
                if part:IsA("BasePart") then
                    partFrame = (part :: BasePart).CFrame
                    partSize = (part :: BasePart).Size
                    partCanCollide = (part :: BasePart).CanCollide
                    if not partCanCollide then return false end
                end
                local model: Model? = part:FindFirstAncestorOfClass("Model")
                if model and model:FindFirstChildOfClass("Humanoid") then return false end
                return true
            end)
            if not okHit or not valid then
                known = false
            else
                local candidate: number = hit.Position.Y + height
                -- A roof/obstacle at the projected XZ must not lift the aim
                -- above the body that is actually visible right now.
                if candidate > last.position.Y + 0.75 then
                    known = false
                else
                    supportY = math.min(candidate, last.position.Y)
                end
            end
        end
        local index: number = track.knifeFloorNext or 1
        local probe: any = cache[index] or {}
        track.knifeFloorSerial = (track.knifeFloorSerial or 0) + 1
        probe.serial = track.knifeFloorSerial
        probe.at, probe.x, probe.z, probe.topY = now, projected.X, projected.Z, topY
        probe.height, probe.known, probe.supportY, probe.part = height, known, supportY, part
        probe.partFrame, probe.partSize, probe.partCanCollide = partFrame, partSize, partCanCollide
        cache[index] = probe
        track.knifeFloorNext = index == 1 and 2 or 1
        return known, supportY
    end

    local function knifeRootAt(
        track: Track, motion: Motion, span: number, floorKnown: boolean, floorY: number?
    ): Vector3
        if floorKnown and motion.verticalLead then
            -- Ground is observed at the projected XZ, so do not reuse a
            -- possibly stale historical "grounded" height from another ledge.
            local last: Sample = track.samples[#track.samples]
            local lead: Vector3 = leadHorizontal(motion, span)
            local y: number = last.position.Y + motion.vertical * span
                - 0.5 * workspace.Gravity * span * span
            if floorY then y = math.max(y, floorY) end
            return Vector3.new(last.position.X + lead.X, y, last.position.Z + lead.Z)
        end
        return predictedRoot(track, motion, span)
    end

    local function knifeFloorIntercept(
        track: Track, motion: Motion, horizon: number, flight: number,
        origin: Vector3, speed: number, floorY: number
    ): (Vector3, number)
        local last: Sample = track.samples[#track.samples]
        local root: Vector3 = knifeRootAt(track, motion, horizon + flight, true, floorY)
        for _ = 1, 5 do
            local revised: number = (root + last.aimOffset - origin).Magnitude / speed
            local delta: number = math.abs(revised - flight)
            flight = revised
            root = knifeRootAt(track, motion, horizon + flight, true, floorY)
            if delta <= 1e-4 then break end
        end
        return root, (root + last.aimOffset - origin).Magnitude / speed
    end

    local function knifeAirborneIntercept(
        track: Track, motion: Motion, horizon: number, origin: Vector3,
        speed: number, floorKnown: boolean
    ): (Vector3?, number?)
        local last: Sample = track.samples[#track.samples]
        local lowTime: number = 0
        local highTime: number = 5 -- the same bound as interceptTime
        if motion.verticalLead and workspace.Gravity > 0 then
            -- Even with zero horizontal distance, the blade cannot intercept
            -- after the target falls faster below it than speed * time. This
            -- analytic bound keeps the numerical search short. The historical
            -- clamp can raise at most six studs when geometry is unknown.
            local gravity: number = workspace.Gravity
            local allowance: number = floorKnown and 0 or MAX_FLOOR_LIFT_STUDS
            local c: number = last.position.Y + last.aimOffset.Y - origin.Y
                + motion.vertical * horizon - 0.5 * gravity * horizon * horizon + allowance
            local b: number = motion.vertical - gravity * horizon + speed
            local discriminant: number = b * b + 2 * gravity * c
            if not finiteNumber(discriminant) or discriminant < 0 then return nil, nil end
            local root: number = math.sqrt(discriminant)
            lowTime = math.max(0, (b - root) / gravity)
            highTime = math.min(highTime, (b + root) / gravity)
        end
        if highTime < lowTime or highTime <= 0 then return nil, nil end
        local segments: number = math.clamp(math.ceil((highTime - lowTime) / 0.025), 1, 64)
        local previous: number = lowTime
        local previousRoot: Vector3 = knifeRootAt(track, motion, horizon + previous, floorKnown, nil)
        local previousGap: number = (previousRoot + last.aimOffset - origin).Magnitude - speed * previous
        if previousGap <= 1e-6 then
            local actual: number = (previousRoot + last.aimOffset - origin).Magnitude / speed
            if math.abs(actual - previous) > 0.002 then return nil, nil end
            return previousRoot, actual
        end
        for index: number = 1, segments do
            local time: number = lowTime + (highTime - lowTime) * index / segments
            local root: Vector3 = knifeRootAt(track, motion, horizon + time, floorKnown, nil)
            local gap: number = (root + last.aimOffset - origin).Magnitude - speed * time
            if gap <= 0 then
                local left: number, right: number = previous, time
                -- The earliest bracketed intercept; no assumed hitbox or
                -- automatic increase to the lead horizon.
                for _ = 1, 12 do
                    local middle: number = (left + right) * 0.5
                    local midRoot: Vector3 = knifeRootAt(track, motion, horizon + middle, floorKnown, nil)
                    local midGap: number = (midRoot + last.aimOffset - origin).Magnitude - speed * middle
                    if midGap <= 0 then right = middle else left = middle end
                end
                local solved: Vector3 = knifeRootAt(track, motion, horizon + right, floorKnown, nil)
                local solvedFlight: number = (solved + last.aimOffset - origin).Magnitude / speed
                -- Reject a discontinuous historical-floor jump rather than
                -- presenting it as a physically consistent intercept.
                if math.abs(solvedFlight - right) > 0.002 then return nil, nil end
                return solved, solvedFlight
            end
            previous = time
        end
        return nil, nil
    end

    local function knifeAim(
        target: Player,
        origin: Vector3,
        speed: number?
    ): (Vector3?, AimInfo?, string?)
        local humanoid: any = targetHumanoid(target)
        if humanoid ~= nil and not bodyIsAlive(humanoid) then
            tracks[target] = nil
            return nil, nil, "dead target"
        end
        local track: Track? = tracks[target]
        if track and track.character ~= target.Character then
            -- Respawn can precede the next sampling heartbeat.
            tracks[target] = nil
            track = nil
        end
        if not track then
            samplePlayer(target, os.clock())
            track = tracks[target]
        end
        if not track or #track.samples == 0 then
            return nil, nil, "no samples"
        end
        local now: number = os.clock()
        local motion: Motion? = estimateMotion(track, now)
        if not motion then
            return nil, nil, "stale sample"
        end
        local resolvedSpeed: number = tonumber(speed) or DEFAULT_KNIFE_SPEED
        if resolvedSpeed < 24 or resolvedSpeed > 300 then
            resolvedSpeed = DEFAULT_KNIFE_SPEED
        end
        local pipeline: number, rtt: number, echo: number? = knifeHorizon()
        local rttEngine: number?, rttData: number? = nil, nil
        if type(core.getLatencyReadings) == "function" then
            local ok: boolean, engine: any, data: any = pcall(core.getLatencyReadings)
            if ok then
                rttEngine = finiteNumber(engine) and engine or nil
                rttData = finiteNumber(data) and data or nil
            end
        end
        local horizon: number = math.clamp(
            motion.sampleAge + pipeline,
            0.005,
            MAX_HORIZON
        )
        local last: Sample = track.samples[#track.samples]
        local aimStart: Vector3 = last.position + last.aimOffset
        local initial: Vector3 = motion.velocity
        local flight: number? = predictionMath.interceptTime(
            aimStart + initial * horizon - origin,
            initial,
            resolvedSpeed
        )
        local horizontalSeed: boolean = false
        if not flight and motion.airborne and motion.vertical < 0 then
            -- A fast descending body can be slower than the blade after it
            -- lands. Use a horizontal seed ONLY if a real support probe below
            -- later confirms that landing; otherwise keep "unreachable".
            local lateral: Vector3 = Vector3.new(initial.X, 0, initial.Z)
            flight = predictionMath.interceptTime(
                aimStart + lateral * horizon - origin, lateral, resolvedSpeed
            )
            horizontalSeed = flight ~= nil
        end
        if not flight then
            return nil, nil, "unreachable"
        end
        -- v159: solve the intercept to a FIXED POINT instead of one re-solve.
        -- The chord a turning or braking target travels depends on how long
        -- the blade is in the air, and that flight time depends on the chord,
        -- so a single pass leaves the aim inconsistent: the point and the
        -- flight time it was solved for disagree. Iterating until the flight
        -- time stops moving makes |endpoint - origin| == speed * flight exact,
        -- which is the one property a straight constant-speed blade has to
        -- satisfy. It converges in two or three passes; the cap is a guard.
        local span: number = horizon + flight
        local lead: Vector3 = leadHorizontal(motion, span)
        local vertical: number = motion.verticalLead and motion.vertical or 0
        if motion.verticalLead then
            vertical -= 0.5 * workspace.Gravity * span
        end
        local effective: Vector3 = Vector3.new(
            lead.X,
            vertical * span,
            lead.Z
        ) / span
        for _iteration: number = 1, 4 do
            local revised: number? = predictionMath.interceptTime(
                aimStart + effective * horizon - origin,
                effective,
                resolvedSpeed
            )
            if not revised then
                break
            end
            local delta: number = math.abs(revised - flight)
            flight = revised
            if delta <= 1e-4 then
                -- `effective` is still the velocity this flight was solved
                -- with, so the endpoint below sits exactly on the intercept.
                break
            end
            span = horizon + flight
            lead = leadHorizontal(motion, span)
            vertical = motion.verticalLead and motion.vertical or 0
            if motion.verticalLead then
                vertical -= 0.5 * workspace.Gravity * span
            end
            effective = Vector3.new(lead.X, vertical * span, lead.Z) / span
        end
        local endpoint: Vector3 = aimStart + effective * (horizon + flight)
        if motion.airborne then
            -- A straight blade aims at the body's projected impact position,
            -- not at an arc for the blade itself. The old fixed-point pass
            -- could diverge: each additional fall increased the flight time,
            -- which then made the prediction fall even farther.
            -- The old ballistic iteration can already have diverged by the
            -- time we ask for a floor. Probe near a horizontal intercept, not
            -- dozens of studs along that unreliable projected XZ.
            local lateral: Vector3 = Vector3.new(initial.X, 0, initial.Z)
            local lateralFlight: number? = predictionMath.interceptTime(
                aimStart + lateral * horizon - origin, lateral, resolvedSpeed
            )
            local probeFlight: number = lateralFlight or flight
            local spanAtProbe: number = horizon + probeFlight
            local probePoint: Vector3 = predictedRoot(track, motion, spanAtProbe)
            local possibleSpan: number = horizon + math.max(probeFlight, flight)
            local rawY: number = last.position.Y + motion.vertical * possibleSpan
                - 0.5 * workspace.Gravity * possibleSpan * possibleSpan
            local known: boolean, floorY: number? = false, nil
            local probes: number = 0
            if motion.verticalLead and rawY < last.position.Y - 0.25 then
                known, floorY = knifeFloorAt(track, probePoint, now)
                probes = 1
            end

            local solvedRoot: Vector3? = nil
            local solvedFlight: number? = nil
            if known and floorY then
                solvedRoot, solvedFlight = knifeFloorIntercept(
                    track, motion, horizon, probeFlight, origin, resolvedSpeed, floorY
                )
                local dx: number = (solvedRoot :: Vector3).X - probePoint.X
                local dz: number = (solvedRoot :: Vector3).Z - probePoint.Z
                if dx * dx + dz * dz > KNIFE_FLOOR_XZ_TOLERANCE ^ 2 then
                    -- The corrected flight can move a landing across an edge.
                    -- Recheck once at the new XZ, never borrow the old floor.
                    probePoint = solvedRoot :: Vector3
                    known, floorY = knifeFloorAt(track, probePoint, now)
                    probes += 1
                    if known and floorY then
                        solvedRoot, solvedFlight = knifeFloorIntercept(
                            track, motion, horizon, solvedFlight :: number,
                            origin, resolvedSpeed, floorY
                        )
                        dx = (solvedRoot :: Vector3).X - probePoint.X
                        dz = (solvedRoot :: Vector3).Z - probePoint.Z
                        if dx * dx + dz * dz > KNIFE_FLOOR_XZ_TOLERANCE ^ 2 then
                            return nil, nil, "uncertain landing"
                        end
                    else
                        solvedRoot, solvedFlight = nil, nil
                    end
                end
            end

            if not solvedRoot then
                if horizontalSeed then return nil, nil, "unreachable" end
                -- A clear downward ray means a genuine fall, not the previous
                -- platform's height. If geometry is unknown, retain the old
                -- bounded history model. Accept a cheap fixed-point result if
                -- already consistent; otherwise bracket the earliest possible
                -- intercept in a bounded search or fall back safely.
                local first: Vector3 = knifeRootAt(track, motion, horizon + flight, known, nil)
                local actual: number = (first + last.aimOffset - origin).Magnitude / resolvedSpeed
                if finiteNumber(actual) and actual <= 5 and math.abs(actual - flight) <= 0.002 then
                    solvedRoot, solvedFlight = first, actual
                else
                    solvedRoot, solvedFlight = knifeAirborneIntercept(
                        track, motion, horizon, origin, resolvedSpeed, known
                    )
                end
                if not solvedRoot or not solvedFlight then return nil, nil, "unreachable" end
                if probes > 0 and known then
                    local dx: number = solvedRoot.X - probePoint.X
                    local dz: number = solvedRoot.Z - probePoint.Z
                    if dx * dx + dz * dz > KNIFE_FLOOR_XZ_TOLERANCE ^ 2 then
                        if probes >= 2 then return nil, nil, "uncertain landing" end
                        probePoint = solvedRoot
                        known, floorY = knifeFloorAt(track, probePoint, now)
                        probes += 1
                        if not known then return nil, nil, "uncertain landing" end
                        if floorY then
                            solvedRoot, solvedFlight = knifeFloorIntercept(
                                track, motion, horizon, solvedFlight, origin, resolvedSpeed, floorY
                            )
                            dx = solvedRoot.X - probePoint.X
                            dz = solvedRoot.Z - probePoint.Z
                            if dx * dx + dz * dz > KNIFE_FLOOR_XZ_TOLERANCE ^ 2 then
                                return nil, nil, "uncertain landing"
                            end
                        end
                    end
                end
            end
            if not solvedRoot or not solvedFlight or not finiteVector(solvedRoot)
                or not finiteNumber(solvedFlight) or solvedFlight > 5 then
                return nil, nil, "unreachable"
            end
            local consistentRoot: Vector3 = knifeRootAt(
                track, motion, horizon + solvedFlight, known, floorY
            )
            if (consistentRoot - solvedRoot).Magnitude > 0.25 then
                return nil, nil, "uncertain intercept"
            end
            if horizontalSeed then
                local correctedSpan: number = horizon + solvedFlight
                local unboundedY: number = last.position.Y + motion.vertical * correctedSpan
                    - 0.5 * workspace.Gravity * correctedSpan * correctedSpan
                if not floorY or unboundedY > floorY then return nil, nil, "unreachable" end
            end
            endpoint = solvedRoot + last.aimOffset
            -- The score/telemetry must describe the point actually returned.
            flight = (endpoint - origin).Magnitude / resolvedSpeed
        end
        return endpoint, {
            endpoint = endpoint,
            targetPosition = endpoint,
            capturedPosition = aimStart,
            rootPosition = endpoint - last.aimOffset,
            velocity = motion.velocity,
            leadSeconds = horizon,
            flightSeconds = flight,
            sampleAge = motion.sampleAge,
            samples = motion.samples,
            residual = motion.residual,
            motionSource = motion.source,
            model = "kinematic-v164-knife",
            rtt = rtt,
            rttEngine = rttEngine,
            rttData = rttData,
            echoDelay = echo,
            echoSamples = #echoes,
            turnRate = motion.turnRate,
            brakeSeconds = motion.brakeSeconds,
            leadCorrection = 0,
        }, nil
    end

    local function playerStatus(target: Player?): string
        local rtt: number = robustRtt()
        local echo: number? = echoMedian()
        local parts: {string} = {
            string.format("RTT %.0f ms", rtt * 1000),
        }
        if echo then
            table.insert(
                parts,
                string.format("echo %.0f ms (%d)", echo * 1000, #echoes)
            )
        end
        table.insert(
            parts,
            string.format("interp %.0f ms", settings.interpolation * 1000)
        )
        if settings.autoLead then
            table.insert(
                parts,
                echo and "auto lead (echo)" or "auto lead (ping)"
            )
        elseif settings.extraLead ~= 0 then
            table.insert(
                parts,
                string.format("manual %+.0f ms", settings.extraLead * 1000)
            )
        end
        local horizon: number = (echo or rtt)
            + settings.interpolation
            + manualLead()
        -- The status must show the horizon the shot actually uses:
        -- gunAim adds the age of the newest sample on top of the
        -- pipeline. Showing only the pipeline (98 ms when the shot
        -- used 111 ms) makes the lead look shorter than it is.
        if target then
            local track: Track? = tracks[target]
            local preview: Motion? = track and estimateMotion(track, os.clock())
            if preview then
                horizon += preview.sampleAge
            end
        end
        local capped: boolean = false
        if settings.autoLead and horizon > MAX_AUTO_LEAD then
            horizon = MAX_AUTO_LEAD
            capped = true
        end
        if horizon > MAX_HORIZON then
            horizon = MAX_HORIZON
            capped = true
        end
        table.insert(
            parts,
            string.format(
                "lead %.0f ms%s",
                horizon * 1000,
                capped and " (capped)" or ""
            )
        )
        if target then
            local track: Track? = tracks[target]
            local motion: Motion? = track and estimateMotion(track, os.clock())
            if motion then
                table.insert(parts, string.format("%.1f studs/s", motion.horizontal.Magnitude))
                table.insert(parts, string.format("%d samples", motion.samples))
                table.insert(parts, string.format("age %.0f ms", motion.sampleAge * 1000))
                if motion.airborne then
                    table.insert(parts, "airborne")
                end
                if math.abs(motion.turnRate) > 0.05 then
                    table.insert(
                        parts,
                        string.format("turn %+.0f deg/s", math.deg(motion.turnRate))
                    )
                end
                if motion.brakeSeconds then
                    table.insert(
                        parts,
                        string.format("brake %.2f s", motion.brakeSeconds)
                    )
                end
            else
                table.insert(parts, "no fresh samples")
            end
        end
        return table.concat(parts, " · ")
    end

    local controller: any = {
        settings = settings,
        noteEcho = noteEcho,
        -- The movement window that ends at the call, oldest first, with `at`
        -- in seconds relative to now (negative). Telemetry attaches it to each
        -- shot so an alternative predictor can be replayed offline from the
        -- same inputs instead of being judged only on live hits.
        getHistory = function(
            player: Player,
            seconds: number?
        ): ({any}?, number?)
            local track: Track? = tracks[player]
            if not track or #track.samples < 2 then
                return nil, 0
            end
            local span: number = math.clamp(
                seconds or HISTORY_SECONDS,
                0.05,
                HISTORY_SECONDS
            )
            -- Times are relative to the newest sample, not to this call. The
            -- modules do not share a clock epoch (the v61 log recorded a
            -- -2546 s offset), so only the predictor's own timestamps can be
            -- subtracted from each other. The caller gets the age separately.
            local newest: Sample = track.samples[#track.samples]
            local age: number = math.max(0, os.clock() - newest.at)
            local points: {any} = {}
            for _, sample: Sample in ipairs(track.samples) do
                local at: number = sample.at - newest.at
                if at >= -span - 1e-6 then
                    table.insert(points, {
                        at = at,
                        position = sample.position,
                        velocity = sample.velocity,
                        state = sample.state,
                        grounded = sample.grounded,
                        moveDirection = sample.moveDirection,
                        -- v139.4: the offset from the root to the part the aim
                        -- falls on. base.lua reads it for every history point
                        -- and it was never sent, so `endpointErrorStuds` scored
                        -- a ROOT position against an aim that used the torso
                        -- offset: a systematic vertical error on every sample.
                        aimOffset = sample.aimOffset,
                        aimPart = sample.aimPart,
                    })
                end
            end
            if #points < 2 then
                return nil, age
            end
            return points, age
        end,
        gunAim = gunAim,
        knifeAim = knifeAim,
        status = playerStatus,

        getEchoDelay = echoMedian,

                return nil
            end

            end
            return predictedRoot(track, motion, math.clamp(horizon, 0, 2))
        end,
        getMotion = function(player: Player): Motion?
            local track: Track? = tracks[player]
            if not track then
                return nil
            end
            return estimateMotion(track, os.clock())
        end,

        reset = function(): ()
            table.clear(echoes)
            table.clear(echoTimes)
            table.clear(tracks)
            table.clear(rttSamples)
            for _, bucket: {number} in pairs(leadErrors) do
                table.clear(bucket)
            end
            table.clear(leadErrors)
            table.clear(leadApplied)
            lastRttSampleAt = -math.huge
        end,
    }

    -- v139: the roster changes a few times per round, not 60 times per second.
    -- `GetPlayers()` returns a fresh array on every call, so asking for it each
    -- frame was pure garbage. Reused for a quarter of a second.
    local roster: {Player} = {}
    local rosterAt: number = -math.huge
    local function tick(deltaTime: number): ()
        elapsed += math.max(0, deltaTime or 0)
        local interval: number = 1 / SAMPLE_HZ
        if elapsed + 1e-9 < interval then
            return
        end
        elapsed = elapsed % interval
        local now: number = os.clock()
        if now - rosterAt >= 0.25 or #roster == 0 then
            rosterAt = now
            roster = Players:GetPlayers()
        end
        for _, player: Player in ipairs(roster) do
            if player ~= LocalPlayer then
                samplePlayer(player, now)
            end
        end
    end

        if gun:IsA("BasePart") then
            return gun :: BasePart
        end
        local handle: Instance? = gun:FindFirstChild("Handle")
            or gun:FindFirstChild("GunDrop")
            or gun:FindFirstChildWhichIsA("BasePart")
        if handle and handle:IsA("BasePart") then
            return handle :: BasePart
        end
        return nil
    end

    local function rememberGunPartState(part: BasePart): {cframe: CFrame, canCollide: boolean}
        local existing = savedGunStates[part]
        if existing then
            return existing
        end
        local snap = {cframe = part.CFrame, canCollide = part.CanCollide}
        savedGunStates[part] = snap
        return snap
    end

    local function restoreGunPartState(part: BasePart): ()
        local snap = savedGunStates[part]
        if not snap then
            return
        end
        if part and part.Parent then
            pcall(function(): ()
                if not part:FindFirstAncestorOfClass("Tool") then
                    if part.CFrame == snap.appliedCFrame then part.CFrame = snap.cframe end
                    if part.CanCollide == snap.appliedCanCollide then part.CanCollide = snap.canCollide end
                end
            end)
        end
        savedGunStates[part] = nil
    end

    local function restoreAllGunPartStates(): ()
        for part in pairs(savedGunStates) do
            restoreGunPartState(part)
        end
        table.clear(savedGunStates)
    end

    local function trueGunPartPosition(part: BasePart): Vector3
        local snap = savedGunStates[part]
        return if snap then snap.cframe.Position else part.Position
    end

    local function canGrabGun(): boolean
        local character = LocalPlayer and LocalPlayer.Character
        if not character then
            return false
        end

        local humanoid = character:FindFirstChildOfClass("Humanoid")
        if not humanoid or humanoid.Health <= 0 then
            return false
        end

        if type(core.getPlayerRole) == "function" then
            local role = core.getPlayerRole(LocalPlayer)
            if role == "Murderer" or role == "Sheriff" or role == "Hero" then
                return false
            end
        end

        local backpack = LocalPlayer:FindFirstChildOfClass("Backpack")
        if (backpack and (backpack:FindFirstChild("Gun") or backpack:FindFirstChild("Knife")))
            or character:FindFirstChild("Gun")
            or character:FindFirstChild("Knife") then
            return false
        end

        return true
    end

    local function isValidGunDrop(obj: Instance?): boolean
        if not obj or not obj.Parent then
            return false
        end
        if obj.Name == "GunDrop" and (obj:IsA("BasePart") or obj:IsA("Model")) then
            return resolveGunTouchPart(obj) ~= nil
        end
        return false
    end

    local function createGunHitbox(gun: Instance): BasePart?
        local handle = resolveGunTouchPart(gun)
        if not handle then
            return nil
        end
        rememberGunPartState(handle)
        local pos = trueGunPartPosition(handle)
        if not pos then
            return nil
        end

        local part = Instance.new("Part")
        part.Name = "WurstGunAuraHitbox"
        part.Size = Vector3.new(settings.studs, settings.studs, settings.studs)
        part.Transparency = 0.65
        part.BrickColor = BrickColor.new("White")
        part.Material = Enum.Material.Neon
        part.CanCollide = false
        part.CanQuery = false
        part.CanTouch = false
        part.Anchored = true
        part.CFrame = CFrame.new(pos)
        part.Parent = workspace

        return part
    end

    -- Lightweight GunDrop discovery: uses DescendantAdded events + top-level

    local function refreshTrackedDrops(now: number): ()
        if now - lastScanAt < SCAN_INTERVAL then
            return
        end
        lastScanAt = now

        local wsDrop: Instance? = workspace:FindFirstChild("GunDrop")
        if wsDrop and isValidGunDrop(wsDrop) then
            trackedDrops[wsDrop] = true
        end

        local okChildren: boolean, children: any = pcall(function(): any
            return workspace:GetChildren()
        end)
        if okChildren and type(children) == "table" then
            for _, child: Instance in ipairs(children) do
                if child.Name == "GunDrop" and isValidGunDrop(child) then
                    trackedDrops[child] = true
                elseif child:IsA("Model") or child:IsA("Folder") then
                    local gd: Instance? = child:FindFirstChild("GunDrop")
                    if gd and isValidGunDrop(gd) then
                        trackedDrops[gd] = true
                    end
                end
            end
        end

        if type(core.findDroppedGun) == "function" then
            local okCore: boolean, drop: any = pcall(core.findDroppedGun)
            if okCore and drop and isValidGunDrop(drop) then
                trackedDrops[drop] = true
            end
        end
    end

    local function isWithinAura(truePos: Vector3, rootPos: Vector3): boolean
        local diff = rootPos - truePos
        local radius = settings.studs
        if diff.X * diff.X + diff.Y * diff.Y + diff.Z * diff.Z <= radius * radius then
            return true
        end
        local half = radius * 0.5
        return math.abs(diff.X) <= half
            and math.abs(diff.Y) <= math.max(half, 8)
            and math.abs(diff.Z) <= half
    end

    local function tryGrabGun(root: BasePart, handle: BasePart): ()

        handle.CanCollide = false
        snap.appliedCanCollide = false

        if rootCF then
            handle.CFrame = rootCF
            snap.appliedCFrame = rootCF
        end
        local fireTouch: any = state.fireTouchInterest
        if type(fireTouch) == "function" then
            pcall(fireTouch, root, handle, 0)
            pcall(fireTouch, root, handle, 1)
        end
    end

    local function gunAuraTick(): ()
        if not gunAuraActive then
            return
        end
        local now: number = os.clock()
        if now - lastTickAt < TICK_INTERVAL then
            return
        end
        lastTickAt = now

        refreshTrackedDrops(now)

        local character = LocalPlayer and LocalPlayer.Character
        local root: BasePart? = character and character:FindFirstChild("HumanoidRootPart") :: BasePart?
        local canGrab = canGrabGun()
        local activeGuns: {[Instance]: boolean} = {}
        local touchedParts: {[BasePart]: boolean} = {}
        local gunCount: number = 0

        for gun in pairs(trackedDrops) do
            if isValidGunDrop(gun) then
                local handle: BasePart? = resolveGunTouchPart(gun)
                if handle then
                    gunCount += 1
                    activeGuns[gun] = true
                    rememberGunPartState(handle)
                    local truePos: Vector3? = trueGunPartPosition(handle)

                    if settings.showHitbox then
                        local hitbox = gunHitboxes[gun]
                        if not hitbox or not hitbox.Parent then
                            hitbox = createGunHitbox(gun)
                            if hitbox then
                                gunHitboxes[gun] = hitbox
                            end
                        elseif truePos then
                            local targetSize = Vector3.new(settings.studs, settings.studs, settings.studs)
                            if hitbox.Size ~= targetSize then
                                hitbox.Size = targetSize
                            end
                            hitbox.CFrame = CFrame.new(truePos)
                        end
                    end

                    if canGrab and root and truePos and isWithinAura(truePos, root.Position) then
                        touchedParts[handle] = true
                        pcall(tryGrabGun, root, handle)
                    end
                end
            else
                trackedDrops[gun] = nil
            end
        end

        for part in pairs(savedGunStates) do
            if not touchedParts[part] then
                restoreGunPartState(part)
            end
        end

        for gun, hitbox in pairs(gunHitboxes) do
            if not activeGuns[gun] then
                if hitbox and hitbox.Parent then
                    pcall(function(): ()
                        hitbox:Destroy()
                    end)
                end
                gunHitboxes[gun] = nil
            end
        end

        if gunCount > 0 then
            setStatus("Gun detected · " .. tostring(math.round(settings.studs)) .. " studs")
        else
            setStatus(tostring(math.round(settings.studs)) .. " studs")
        end

        if now - lastEspAt >= ESP_INTERVAL and type(core.getPlayerRole) == "function" then
            lastEspAt = now
            local seenSheriffs: {[Player]: boolean} = {}
            for _, player in ipairs(game:GetService("Players"):GetPlayers()) do
                if player ~= LocalPlayer then
                    local role = core.getPlayerRole(player)
                    if role == "Sheriff" then
                        seenSheriffs[player] = true
                        updateSheriffESP(player)
                    end
                end
            end
            for player, esp in pairs(espParts) do
                if not seenSheriffs[player] then
                    if esp and esp.Parent then
                        pcall(function(): ()
                            esp:Destroy()
                        end)
                    end
                    espParts[player] = nil
                end
            end
        end
    end

    -- v159: WallCheck is now the only guarantee, so it has to be honest:
    --  * the shared helper is called with `strict`, which makes a fully
    --    transparent but SOLID barrier stop the blade. The gun's first pass
    --    skips those; the knife must not.
    --  * the segment is tested as a thin corridor, not one mathematical line,
    --    so a railing edge cannot slip between samples.
    --  * the body has to be visible from the hand RIGHT NOW, not only at the
    --    predicted point. A predicted point that peeks around a corner is not
    --    reachable by a straight blade, and aiming at it is what looked like
    --    "knife through the wall".
    local CORRIDOR_STUDS: number = 0.45
    local function raycastClear(
        from: Vector3,
        to: Vector3,
        ignoreList: {Instance},
        requireBody: Model?
    ): boolean
        local ok: boolean, clear: any = pcall(function(): boolean
            local params: RaycastParams = RaycastParams.new()
            params.FilterType = Enum.RaycastFilterType.Exclude
            params.FilterDescendantsInstances = ignoreList
            params.RespectCanCollide = true
            params.IgnoreWater = true
            local hit: RaycastResult? = workspace:Raycast(from, to - from, params)
            if requireBody then
                return hit ~= nil and hit.Instance:IsDescendantOf(requireBody)
            end
            return hit == nil
        end)
        return ok and clear == true
    end

    local function sharedPathClear(
        from: Vector3,
        to: Vector3,
        targetChar: Model?,
        requireBody: boolean
    ): boolean?
        if type(isWeaponPathClear) ~= "function" or not targetChar then
            return nil
        end
        local options: any = requireBody and { requireTarget = true } or { ignoreTarget = true }
        local ok: boolean, clear: any = pcall(isWeaponPathClear, from, to, targetChar, true, options)
        if not ok or type(clear) ~= "boolean" then
            return nil
        end
        return clear
    end

    local function ignoreListFor(targetChar: Model?): {Instance}
        local list: {Instance} = {}
        local localChar: Model? = LocalPlayer and LocalPlayer.Character
        if localChar then table.insert(list, localChar) end
        if targetChar then table.insert(list, targetChar) end

    -- "I can see that body from the hand": the ray must END on the body.
    local function bodyVisible(originPos: Vector3, bodyPos: Vector3, targetChar: Model?): boolean
        if not settings.wallCheck then
            return true
        end
        local from: Vector3 = originPos
        local to: Vector3 = bodyPos
        if (to - from).Magnitude < 0.05 then
            return true
        end
        local pathResult: boolean? = sharedPathClear(from, to, targetChar, true)
        if pathResult ~= nil then
            return pathResult
        end
        return raycastClear(from, to, ignoreListFor(nil), targetChar)
    end

    -- "Nothing solid between the hand and that point": corridor of five rays.
    local function checkLineOfSight(originPos: Vector3, targetPos: Vector3, targetChar: Model?): boolean
        if not settings.wallCheck then
            return true
        end
        local direction: Vector3 = targetPos - originPos
        local length: number = direction.Magnitude
        if length < 0.05 then
            return true
        end
        local unit: Vector3 = direction / length
        local side: Vector3 = unit:Cross(Vector3.new(0, 1, 0))
        if side.Magnitude < 0.05 then
            side = Vector3.new(1, 0, 0)
        end
        side = side.Unit
        local up: Vector3 = side:Cross(unit).Unit
        local offsets: {Vector3} = {
            Vector3.zero,
            side * CORRIDOR_STUDS,
            side * -CORRIDOR_STUDS,
            up * CORRIDOR_STUDS,
            up * -CORRIDOR_STUDS,
        }
        for _index: number, offset: Vector3 in ipairs(offsets) do
            local from: Vector3 = originPos + offset
            local to: Vector3 = targetPos + offset
            -- Every corridor ray uses the same strict rule as the centre one.
            -- A raw raycast here would let an invisible decoration that the
            -- shared helper skips block the throw, which made WallCheck say no
            -- for geometry the blade would have flown through.
            local pathResult: boolean? = sharedPathClear(from, to, targetChar, false)
            if pathResult ~= nil then
                if not pathResult then
                    return false
                end
            elseif not raycastClear(from, to, ignoreListFor(targetChar), nil) then
                return false
            end
        end
        return true
    end

    -- Predict target intercept position for MM2 ThrowingKnife (96 studs/s linear projectile)
    -- Returns the point and the blade's flight time, which is what makes one
    -- target a sooner hit than another.
    -- `cheap` is for target SELECTION: it still requires the body to be
    -- visible, but skips the five-ray corridor, which is re-checked with the
    -- live origin right before the packet leaves.
    local function resolveKnifeAim(
        target: Player,
        originPos: Vector3,
        knife: Tool?,
        cheap: boolean?
    ): (Vector3?, number?)
        local character: Model? = target.Character
        if not character then
            return nil, nil
        end
        local root: BasePart? = character:FindFirstChild("HumanoidRootPart") :: BasePart?
        if not root or not root:IsA("BasePart") then
            return nil, nil
        end
        -- Check the live body before doing the more expensive intercept/ground
        -- work; hidden candidates cannot become speculative shots.
        if not bodyVisible(originPos, root.Position, character) then
            return nil, nil
        end

        local speed: number = 96
        if knife then
            pcall(function(): ()
                local attrSpeed: any = knife:GetAttribute("ThrowSpeed")
                if type(attrSpeed) == "number" and attrSpeed >= 24 and attrSpeed <= 300 then
                    speed = attrSpeed
                end
            end)
        end

        local predicted: Vector3? = nil
        local aimInfo: any = nil
        if type(prediction) == "table" and type(prediction.knifeAim) == "function" then
            -- v139.4: the blade speed HAS to reach the predictor. It was read
            -- from the tool above and then only used by the local fallback, so
            -- every knife that is not the default 96 studs/s was solved with
            -- the wrong flight time even though the correct number was in
            -- hand.
            local okPred: boolean, aimPoint: any, info: any =
                pcall(prediction.knifeAim, target, originPos, speed)
            if okPred and typeof(aimPoint) == "Vector3"
                and aimPoint.X == aimPoint.X and aimPoint.Y == aimPoint.Y and aimPoint.Z == aimPoint.Z
                and math.abs(aimPoint.X) < math.huge and math.abs(aimPoint.Y) < math.huge
                and math.abs(aimPoint.Z) < math.huge then
                predicted = aimPoint :: Vector3
                aimInfo = info
            end
        end

        if not predicted then
            local velocity: Vector3 = root.AssemblyLinearVelocity
            if velocity.X ~= velocity.X or velocity.Y ~= velocity.Y or velocity.Z ~= velocity.Z
                or math.abs(velocity.X) == math.huge or math.abs(velocity.Y) == math.huge
                or math.abs(velocity.Z) == math.huge then
                velocity = Vector3.zero
            end
            local latency: number = 0.06
            if type(getEstimatedLatency) == "function" then
                local okLat: boolean, l: any = pcall(getEstimatedLatency)
                if okLat and type(l) == "number" and l == l then
                    latency = math.clamp(l, 0, 0.5)
                end
            end
            -- This is only the fallback when the shared predictor is absent or
            -- cannot solve. Previously `velocity * latency` applied unbounded
            -- vertical lead BEFORE the later 3.5 stud drop clamp: a replicated
            -- -120 Y velocity at 0.5 s already aimed 60 studs below the body.
            -- Keep horizontal transport and cap the *entire* vertical lead.
            -- The body's actual height is never clamped against the thrower:
            -- an enemy on a lower floor still produces a downward shot.
            local horizVel: Vector3 = Vector3.new(velocity.X, 0, velocity.Z)
            local observed: Vector3 = root.Position + horizVel * latency
            local rel: Vector3 = observed - originPos
            local a: number = horizVel:Dot(horizVel) - speed * speed
            local b: number = 2 * rel:Dot(horizVel)
            local c: number = rel:Dot(rel)
            local flight: number? = nil
            if c < 0.0025 then
                flight = 0
            elseif math.abs(a) < 1e-6 then
                if b < -1e-6 then
                    flight = -c / b
                end
            else
                local disc: number = b * b - 4 * a * c
                if disc >= 0 then
                    local sqrtDisc: number = math.sqrt(disc)
                    local t1: number = (-b - sqrtDisc) / (2 * a)
                    local t2: number = (-b + sqrtDisc) / (2 * a)
                    if t1 > 0 then
                        flight = t1
                    end
                    if t2 > 0 and (not flight or t2 < flight) then
                        flight = t2
                    end
                end
            end
            if flight and flight == flight then
                flight = math.clamp(flight, 0, 1.5)
                local yLead: number = math.clamp(
                    velocity.Y * math.min(latency + flight, 0.28), -2.0, 4.0
                )
                predicted = observed + horizVel * flight + Vector3.new(0, yLead, 0)
            else
                local yLead: number = math.clamp(velocity.Y * math.min(latency, 0.28), -2.0, 4.0)
                predicted = observed + Vector3.new(0, yLead, 0)
            end
        end

        if not predicted or predicted.X ~= predicted.X or predicted.Y ~= predicted.Y or predicted.Z ~= predicted.Z
            or math.abs(predicted.X) == math.huge or math.abs(predicted.Y) == math.huge
            or math.abs(predicted.Z) == math.huge then
            return nil, nil
        end

        local flight: number? = nil
        if aimInfo and type(aimInfo.flightSeconds) == "number" and aimInfo.flightSeconds == aimInfo.flightSeconds
            and math.abs(aimInfo.flightSeconds) < math.huge then
            flight = math.max(0, aimInfo.flightSeconds)
        else
            flight = (predicted - originPos).Magnitude / speed
        end

        -- The current body was already checked before prediction: a future
        -- point peeking around a corner cannot make a hidden body eligible.
        if cheap or checkLineOfSight(originPos, predicted, character) then
            return predicted, flight
        end

        return nil, nil
    end

    -- Find best target for knife throw
    local function findTarget(): Player?
        local localChar: Model? = LocalPlayer and LocalPlayer.Character
        if not localChar then
            return nil
        end
        local localRoot: BasePart? = localChar:FindFirstChild("HumanoidRootPart") :: BasePart?
        if not localRoot then
            return nil
        end
        local knife: Tool? = getKnife()
        local originCF: CFrame? = getThrowOriginCFrame(knife)
        local originPos: Vector3 = originCF and originCF.Position or localRoot.Position

        local bestTarget: Player? = nil
        local bestScore: number = math.huge

        for _, player: Player in ipairs(Players:GetPlayers()) do
            if isValidTarget(player) then
                local targetChar: Model? = player.Character
                local targetRoot: BasePart? = targetChar
                    and targetChar:FindFirstChild("HumanoidRootPart") :: BasePart?
                if targetRoot then
                    local distance: number = (targetRoot.Position - originPos).Magnitude
                    if distance <= 250 then
                        local aimPoint: Vector3?, flight: number? =
                            resolveKnifeAim(player, originPos, knife, true)
                        if aimPoint then
                            -- v159: score by the blade's FLIGHT TIME, not raw
                            -- distance. The nearest body is not the soonest hit:
                            -- a target running towards the throw intercepts in
                            -- a fraction of the time a farther one standing
                            -- still does, and the sooner the blade lands, the
                            -- smaller every prediction error is.
                            local score: number = flight or (distance / 96)
                            if score < bestScore then
                                bestScore = score
                                bestTarget = player
                            end
                        end
                    end
                end
            end
        end

        return bestTarget
    end

    -- The solve is throttled, but its result used to jump at each update. This is
    -- visual-only: the native redirects and throwKnife always solve separately
    -- and never read this smoothed point.
    local TRACER_SMOOTH_RATE: number = 26
    local TRACER_SNAP_DISTANCE: number = 20
    local tracerPoint: Vector3? = nil
    local tracerTarget: Player? = nil
    local tracerShown: boolean = false
    local tracerOpacity: number = 0
    local drawnOpacity: number = -1
    local drawnOrigin: Vector3? = nil
    local drawnPoint: Vector3? = nil

    local function hideTracerParts(): ()
        if tracerPart and tracerPart.Parent then tracerPart.Transparency = 1 end
        if targetDotPart and targetDotPart.Parent then targetDotPart.Transparency = 1 end
    end

    local function hideTracerVisuals(): ()
        -- Do not tween a stale point after the next visibility check fails.
        -- Reuse the two invisible parts if a target reappears; destroy them
        -- when the option/module is turned off.
        if tracerShown then pcall(hideTracerParts) end
        tracerShown = false
        tracerOpacity = 0
        drawnOpacity = -1
        tracerPoint = nil
        tracerTarget = nil
        drawnOrigin = nil
        drawnPoint = nil
    end

    local function drawTracerParts(
        originPos: Vector3, point: Vector3, delta: Vector3, dist: number,
        updateGeometry: boolean, opacity: number, updateOpacity: boolean
    ): ()
        if not tracerPart or not tracerPart.Parent then
            local beamPart: Part = Instance.new("Part")
            beamPart.Name = "WurstKnifeTracer"
            beamPart.Anchored = true
            beamPart.CanCollide = false
            beamPart.CanQuery = false
            beamPart.CanTouch = false
            beamPart.Material = Enum.Material.Neon
            beamPart.Color = Color3.fromRGB(255, 65, 65)
            beamPart.Transparency = 1
            beamPart.Parent = workspace
            tracerPart = beamPart
            tracerShown = false
        end
        if not targetDotPart or not targetDotPart.Parent then
            local dot: Part = Instance.new("Part")
            dot.Name = "WurstKnifeAimDot"
            dot.Shape = Enum.PartType.Ball
            dot.Size = Vector3.new(1.1, 1.1, 1.1)
            dot.Anchored = true
            dot.CanCollide = false
            dot.CanQuery = false
            dot.CanTouch = false
            dot.Material = Enum.Material.Neon
            dot.Color = Color3.fromRGB(255, 220, 60)
            dot.Transparency = 1
            dot.Parent = workspace
            targetDotPart = dot
            tracerShown = false
        end
        if updateGeometry then
            tracerPart.Size = Vector3.new(0.14, 0.14, dist)
            tracerPart.CFrame = CFrame.lookAt(originPos + delta * 0.5, point)
            targetDotPart.CFrame = CFrame.new(point)
        end
        if updateOpacity or not tracerShown then
            tracerPart.Transparency = 1 - 0.75 * opacity
            targetDotPart.Transparency = 1 - 0.85 * opacity
            tracerShown = true
        end
    end

    local function updateTracerVisual(
        originPos: Vector3?, aimPos: Vector3?, target: Player?, deltaTime: number?
    ): ()
        if not settings.enabled or not settings.knifeTracer then
            hideTracerVisuals()
            destroyTracerVisuals()
            return
        end
        if not originPos or not aimPos or not target then
            hideTracerVisuals()
            return
        end
        local dt: number = tonumber(deltaTime) or (1 / 60)
        if dt ~= dt then dt = 1 / 60 end
        local alpha: number = 1 - math.exp(-TRACER_SMOOTH_RATE * math.clamp(dt, 0, 0.25))
        local previous: Vector3? = tracerPoint
        if not previous or tracerTarget ~= target
            or (aimPos - previous).Magnitude > TRACER_SNAP_DISTANCE then
            -- Never sweep a line across different players or a teleport. A
            -- short fade-in still softens the appearance of the new point.
            tracerPoint = aimPos
            tracerOpacity = 0
        else
            local nextPoint: Vector3 = previous + (aimPos - previous) * alpha
            tracerPoint = (aimPos - nextPoint).Magnitude < 0.002 and aimPos or nextPoint
        end
        tracerOpacity += (1 - tracerOpacity) * alpha
        if tracerOpacity > 0.998 then tracerOpacity = 1 end
        tracerTarget = target
        local point: Vector3 = tracerPoint :: Vector3
        local delta: Vector3 = point - originPos
        local dist: number = delta.Magnitude
        if dist < 0.25 or dist > 400 then
            hideTracerVisuals()
            return
        end
        local missing: boolean = not tracerPart or not tracerPart.Parent
            or not targetDotPart or not targetDotPart.Parent
        local geometryChanged: boolean = missing or not drawnOrigin or not drawnPoint
            or (originPos - (drawnOrigin :: Vector3)).Magnitude >= 0.002
            or (point - (drawnPoint :: Vector3)).Magnitude >= 0.002
        local opacityChanged: boolean = missing or not tracerShown
            or math.abs(tracerOpacity - drawnOpacity) >= 0.002
        if not geometryChanged and not opacityChanged then return end
        -- Named callback: unlike the old per-frame pcall(closure), this does
        -- not allocate a fresh closure every frame. No raycasts/tweens here.
        local ok: boolean = pcall(
            drawTracerParts, originPos, point, delta, dist,
            geometryChanged, tracerOpacity, opacityChanged
        )
        if ok then
            if geometryChanged then
                drawnOrigin = originPos
                drawnPoint = point
            end
            if opacityChanged then drawnOpacity = tracerOpacity end
        else
            pcall(hideTracerParts)
            tracerShown = false
            drawnOpacity = -1
            drawnOrigin = nil
            drawnPoint = nil
        end
    end

    -- Throw knife at predicted target using MM2's Knife.Events.KnifeThrown remote
    -- v139.2: the equip is a round trip, the throw is not. Firing the remote
    -- in the same frame as EquipTool sends a throw from a tool the server still
    -- believes is in the backpack: in the saved session 13 auto throws produced
    -- 9 observed blades (4 vanished) and the 4 that did fire used the RightHand
    -- fallback origin instead of the knife handle. So: equip once, wait for the
    -- tool to actually be in the character, and only then throw.
    local equipRequestedAt: number = -math.huge
    local equipRequestedTool: Tool? = nil

    local function ensureEquipped(knife: Tool, humanoid: Humanoid): boolean
        local character: Model? = LocalPlayer and LocalPlayer.Character
        if not character then
            return false
        end
        if knife.Parent == character then
            equipRequestedAt = -math.huge
            equipRequestedTool = nil
            return true
        end
        local now: number = os.clock()
        -- Re-ask every 0.25 s at most; EquipTool while the previous request is
        -- still travelling just stacks another equip request.
        if equipRequestedTool ~= knife or now - equipRequestedAt >= 0.25 then
            equipRequestedAt = now
            equipRequestedTool = knife
            pcall(function(): ()
                humanoid:EquipTool(knife)
            end)
        end
        return false
    end

    local function throwKnife(target: Player?, knife: Tool): boolean
        local character: Model? = LocalPlayer and LocalPlayer.Character
        if not character then
            return false
        end
        local humanoid: Humanoid? = character:FindFirstChildOfClass("Humanoid")
        if not humanoid or humanoid.Health <= 0 or not ensureEquipped(knife, humanoid)
            or knife.Parent ~= character then
            return false
        end
        local originCF: CFrame? = getThrowOriginCFrame(knife)
        if not originCF then
            return false
        end
        -- v159: no lane fallback at all. Without a visible target there is
        -- nothing to throw at: a blade is not a homing projectile, and a throw
        -- into an opening only spent the knife.
        if not target or not isValidTarget(target) then
            return false
        end
        local aimPos: Vector3? = resolveKnifeAim(target, originCF.Position, knife)
        if not aimPos then
            return false
        end

        local noted: boolean = false
        local function noteThrowAttempt(): ()
            if noted then return end
            noted = true
            local recorder: any = core and core.trajectoryCalibration
            if recorder and type(recorder.noteAuthoredShot) == "function" then
                local root: BasePart? = target.Character
                    and target.Character:FindFirstChild("HumanoidRootPart") :: BasePart?
                pcall(recorder.noteAuthoredShot, recorder, {
                    kind = "knife", tool = knife, toolName = knife.Name,
                    originPos = originCF.Position, aimPos = aimPos, target = target,
                    aimVelocity = root and root.AssemblyLinearVelocity or nil,
                    source = "menu-auto-throw",
                    settings = {mode = "Auto Knife Throw", wallCheck = settings.wallCheck},
                })
            end
        end
        local eventsFolder: Instance? = knife:FindFirstChild("Events")
        local knifeThrownRemote: Instance? = eventsFolder and eventsFolder:FindFirstChild("KnifeThrown")
        if knifeThrownRemote and knifeThrownRemote:IsA("RemoteEvent") then
            noteThrowAttempt()
            return pcall(function() (knifeThrownRemote :: RemoteEvent):FireServer(originCF, CFrame.new(aimPos)) end)
        end
        return false

    end

    -- Hook ReplicatedStorage.ClientServices.WeaponService so native PC & Mobile
    -- knife throws (Mouse, Keyboard E, Gamepad, Mobile TouchControls.RightBar.Throw)
    -- automatically redirect to the Knife Predictor's aim point!
    local ownedHooks: {any} = {}
    local function nativeAim(): Vector3?
        if not settings.enabled or not V["Knife Throw"] then return nil end
        local knife: Tool? = getKnife()
        local character: Model? = LocalPlayer.Character
        if not knife or not character or knife.Parent ~= character then return nil end
        local originCF: CFrame? = getThrowOriginCFrame(knife)
        local target: Player? = findTarget()
        if not originCF or not target then return nil end
        return resolveKnifeAim(target, originCF.Position, knife)
    end
    local function hookNative(tbl: any, name: string): ()
        local original: any = tbl[name]
        if type(original) ~= "function" then return end
        local wasOwn: boolean = rawget(tbl, name) ~= nil
        local function hook(self: any, ...: any): ...any
            -- Invoke the native method exactly once, keeping its return type and other results.
            local result = table.pack(original(self, ...))
            local kind = typeof(result[1])
            if kind == "CFrame" or kind == "Vector3" then
                local ok, point = pcall(nativeAim)
                if ok and point then result[1] = if kind == "CFrame" then CFrame.new(point) else point end
            end
            return table.unpack(result, 1, result.n)
        end
        if pcall(function() tbl[name] = hook end) and rawget(tbl,name)==hook then
            table.insert(ownedHooks,{tbl=tbl,name=name,original=original,hook=hook,wasOwn=wasOwn})
        end
    end
    local function installWeaponServiceRedirect(): ()
        if hookedWeaponService then return end
        pcall(function()
            local rs = game:GetService("ReplicatedStorage")
            local folder = rs and rs:FindFirstChild("ClientServices")
            local module = folder and folder:FindFirstChild("WeaponService")
            if not module then return end
            local ok, ws = pcall(require,module)
            if not ok or type(ws) ~= "table" then return end
            hookedWeaponService = ws
            hookNative(ws,"GetMouseTargetCFrame")
            hookNative(ws,"GetTargetPosition")
        end)
    end
    local function uninstallWeaponServiceRedirect(): ()
        for _, h in ipairs(ownedHooks) do
            pcall(function()
                if rawget(h.tbl,h.name)==h.hook then
                    h.tbl[h.name]=if h.wasOwn then h.original else nil
                end
            end)
        end
        table.clear(ownedHooks)
        hookedWeaponService=nil
    end

    -- v139.2: this loop used to run every frame and, for EVERY player in the
    -- lobby, ran the aim solver -- and the aim solver casts line-of-sight rays
    -- against the map, then up to 8 more per body part when the first one is
    -- blocked. Ten players meant dozens of raycasts per frame for a number
    -- that only changes a few times per second. The solve is now cached for
    -- 1/15 s when an automatic throw is due; a visual-only preview can use
    -- 1/10 s because the displayed point follows the observed root between
    -- solves. The actual throw always re-solves from the live handle.
    local AIM_CACHE_SECONDS: number = 1 / 15
    local TRACER_CACHE_SECONDS: number = 1 / 10
    local cachedTarget: Player? = nil
    local cachedAim: Vector3? = nil
    local cachedOrigin: CFrame? = nil
    local cachedVisualRoot: BasePart? = nil
    local cachedVisualPosition: Vector3? = nil
    local cachedAt: number = -math.huge

    local function solveAim(knife: Tool, throwDue: boolean): (Player?, Vector3?, CFrame?)
        local now: number = os.clock()
        local interval: number = throwDue and AIM_CACHE_SECONDS or TRACER_CACHE_SECONDS
        if now - cachedAt >= interval then
            cachedAt = now
            cachedTarget = findTarget()
            cachedAim = nil
            cachedVisualRoot = nil
            cachedVisualPosition = nil
            cachedOrigin = getThrowOriginCFrame(knife)
            if cachedOrigin and cachedTarget then
                cachedAim = resolveKnifeAim(cachedTarget, cachedOrigin.Position, knife)
                if cachedAim and settings.knifeTracer then
                    local character: Model? = cachedTarget.Character
                    local root: BasePart? = character
                        and character:FindFirstChild("HumanoidRootPart") :: BasePart?
                    if root then
                        cachedVisualRoot = root
                        cachedVisualPosition = root.Position
                    end
                end
            end
        elseif cachedOrigin then
            -- Origin only: the target's predicted point stays cached, but the
            -- handle moves with the character and the tracer must follow it.
            cachedOrigin = getThrowOriginCFrame(knife) or cachedOrigin
        end
        return cachedTarget, cachedAim, cachedOrigin
    end

    local function invalidateAim(): ()
        cachedAt = -math.huge
        cachedTarget = nil
        cachedAim = nil
        cachedOrigin = nil
        cachedVisualRoot = nil
        cachedVisualPosition = nil
    end

    -- getPlayerWeapon walks the character/backpack's children. A tracer-only
    -- heartbeat must not do that 60-240 times a second just to keep the line
    -- attached to an already equipped handle. Actual throws and native input
    -- still use a fresh getKnife(), and parent changes invalidate this cache.
    local loopKnife: Tool? = nil
    local loopKnifeParent: Instance? = nil
    local loopKnifeCharacter: Model? = nil
    local loopKnifeAt: number = -math.huge
    local function resetLoopKnife(): ()
        loopKnife = nil
        loopKnifeParent = nil
        loopKnifeCharacter = nil
        loopKnifeAt = -math.huge
        originHandleTool = nil
        originHandle = nil
    end
    local function getLoopKnife(now: number, throwDue: boolean): Tool?
        local character: Model? = LocalPlayer and LocalPlayer.Character
        if character ~= loopKnifeCharacter then resetLoopKnife() end
        loopKnifeCharacter = character
        if not throwDue then
            if loopKnife and loopKnife.Parent == character and character then
                return loopKnife
            end
            local age: number = now - loopKnifeAt
            if age >= 0 and age < TRACER_CACHE_SECONDS then
                if not loopKnife then return nil end
                if loopKnifeParent and loopKnife.Parent == loopKnifeParent then
                    return loopKnife
                end
            end
        end
        loopKnife = getKnife()
        loopKnifeParent = loopKnife and loopKnife.Parent or nil
        loopKnifeAt = now
        return loopKnife
    end

    local function tracerGoal(target: Player?, aimPos: Vector3?): Vector3?
        if not target or not aimPos then return nil end
        local root: BasePart? = cachedVisualRoot
        local captured: Vector3? = cachedVisualPosition
        if not root or not captured then return aimPos end
        if target ~= cachedTarget or root.Parent ~= target.Character then
            invalidateAim()
            return nil
        end
        -- Follow the already-observed root between the throttled solves. This is
        -- only a visual offset, not another prediction or a per-frame raycast.
        local moved: Vector3 = root.Position - captured
        if moved.Magnitude > 8 then
            invalidateAim()
            return nil
        end
        return aimPos + moved
    end

    -- Main loop for Knife Tracer and Auto Knife Throw
    local function knifeThrowLoop(deltaTime: number?): ()
        if not settings.enabled then
            setKnifeStatus(nil)
            hideTracerVisuals()
            destroyTracerVisuals()
            invalidateAim()
            return
        end

        local now: number = os.clock()
        local cooldown: number = math.clamp(tonumber(settings.delay) or 0, 0, 10)
        local wantThrow: boolean = settings.autoThrow
            and now - lastThrowAttempt >= cooldown

        -- Manual redirects solve on the native input path. Do not enumerate
        -- players, predict and raycast merely to update a status when both
        -- Auto Knife Throw and the tracer are off. Likewise, without a tracer
        -- no preview is needed during a successful throw's cooldown.
        if not settings.knifeTracer and not wantThrow then
            hideTracerVisuals()
            destroyTracerVisuals()
            if not settings.autoThrow then
                setKnifeStatus(nil)
                invalidateAim()
            end
            return
        end

        local knife: Tool? = getLoopKnife(now, wantThrow)
        if not knife then
            setKnifeStatus("Waiting for knife")
            hideTracerVisuals()
            invalidateAim()
            return
        end

        -- The cached pick only decides WHO gets the blade. throwKnife
        -- re-solves the aim from the live handle when it actually fires.
        local target, aimPos, originCF = solveAim(knife, wantThrow)
        if not aimPos then
            setKnifeStatus(target and "Target blocked · WallCheck" or "No visible target")
            hideTracerVisuals()
            return
        end
        setKnifeStatus("Predicted target")

        if settings.knifeTracer and originCF then
            updateTracerVisual(originCF.Position, tracerGoal(target, aimPos), target, deltaTime)
        else
            hideTracerVisuals()
            destroyTracerVisuals()
        end

        if wantThrow then
            if throwKnife(target, knife) then
                setKnifeStatus("Throw sent · hit unconfirmed")
                lastThrowAttempt = now
                -- A throw consumes the aim: the next frame re-solves instead of
                -- reusing a point measured before the blade left.
                invalidateAim()
            end
        end
    end

        end

            old(isMM2 and not stopped and value==true);refreshKnife()

    local function alive(player)

    local function findTarget(root, murderOnly)
        if roundBlocked then return end
        if not murderOnly and V["Aimbot Target"]=="Custom" then
            local query=string.lower(string.match(tostring(V["Target Player"] or ""),"^%s*(.-)%s*$"))
            if query=="" then return end
            local exact,partial,count=nil,nil,0
            for _,p in ipairs(Players:GetPlayers()) do if alive(p) then
                local n=string.lower(p.Name); local d=string.lower(p.DisplayName or p.Name)
                if n==query or d==query then exact=p; break end
                if string.sub(n,1,#query)==query or string.sub(d,1,#query)==query then partial=p; count+=1 end
            end end
            return exact or (count==1 and partial or nil)
        end
        local best,distance
        for _,p in ipairs(Players:GetPlayers()) do
            if alive(p) and not weapon(p,"Gun") and core.getPlayerRole(p)=="Murderer" then
                local other=p.Character:FindFirstChild("HumanoidRootPart")
                local d=(other.Position-root.Position).Magnitude
                if not distance or d<distance then best,distance=p,d end
            end
        end
        return best
    end
    local function gunOrigin(character,root,gun)
        local attachment=root:FindFirstChild("GunRaycastAttachment")
        if attachment and attachment:IsA("Attachment") then return attachment.WorldCFrame end
        for _,name in ipairs({"Muzzle","Handle"}) do
            local part=gun:FindFirstChild(name)
            if part and part:IsA("BasePart") then return part.CFrame end
        end
        for _,name in ipairs({"RightHand","Right Arm","Head"}) do
            local part=character:FindFirstChild(name)
            if part and part:IsA("BasePart") then return part.CFrame end
        end
        return root.CFrame
    end
    local function clearPath(origin,destination,character,target)
        local ok,result=pcall(function()
            local excluded={character,target}
            local success,collection=pcall(function()return game:GetService("CollectionService")end)
            if success and collection then
                for _,part in ipairs(collection:GetTagged("WeaponPassthrough")) do table.insert(excluded,part) end
            end
            local params=RaycastParams.new();params.FilterType=Enum.RaycastFilterType.Exclude;params.IgnoreWater=true
            for _=1,32 do
                params.FilterDescendantsInstances=excluded
                local hit=workspace:Raycast(origin,destination-origin,params)
                if not hit then return true end
                local part=hit.Instance
                if part and part:IsA("BasePart") and part.Transparency==1 and part.CanCollide==false then
                    table.insert(excluded,part)
                else return false end
            end
            return false
        end)
        return ok and result==true
    end
    local function frames(character,root,gun,nativeOrigin,target,method,wallCheck)
        target=target or findTarget(root)
        local targetCharacter=target and target.Character
        if not targetCharacter then return end
        local origin=gunOrigin(character,root,gun) or nativeOrigin
        if not origin or not prediction then return end
        local mode=(method or V["Aim Method"])=="Instakill" and "CFrame" or "Bullet"
        local ok,point,info=pcall(prediction.gunAim,target,origin.Position,mode)
        if not ok or not point or not info then return end
        local targetRoot=targetCharacter:FindFirstChild("HumanoidRootPart")
        local line=point-origin.Position
        if targetRoot and (line.Magnitude<6 or line:Dot(targetRoot.Position-origin.Position)<=0) then
            point=targetRoot.Position;info.endpoint=point;info.capturedPosition=point
        end
        if (point-origin.Position).Magnitude<.05 then return end
        if wallCheck==nil then wallCheck=V["WallCheck"]==true end
        if wallCheck and not clearPath(origin.Position,point,character,targetCharacter) then return end
        local first,last
        if mode=="Bullet" then first,last=CFrame.lookAt(origin.Position,point),CFrame.new(point)
        else
            local velocity=info.velocity or Vector3.zero
            local horizontal=Vector3.new(velocity.X,0,velocity.Z)
            local approach=Vector3.new(point.X-origin.Position.X,0,point.Z-origin.Position.Z)
            local axis=horizontal.Magnitude>1.5 and horizontal.Unit or (approach.Magnitude>.05 and approach.Unit or nil)
            if axis then
                local sweep=math.clamp(.5+(prediction.getLeadUncertainty("CFrame") or 0)*horizontal.Magnitude,0,1.5)
                local start=(info.capturedPosition or point)-axis*(1+sweep)
                local finish=point+axis*(1+sweep)
                first,last=CFrame.lookAt(start,finish),CFrame.new(finish)
            else
                local lift=1.6
                if info.rootPosition then
                    local offset=point.Y-info.rootPosition.Y
                    if offset>.5 then lift=math.clamp(offset+.35,1.2,2.4) end
                end
                first,last=CFrame.lookAt(point+Vector3.new(0,lift,0),point),CFrame.new(point)
            end
        end
        if wallCheck and mode=="CFrame" and not clearPath(first.Position,last.Position,character,targetCharacter) then return end
        return first,last,target,info,mode
    end
    local pending, echoConnection, observerBusy, observerAt = nil,nil,false,0
    local function position(value)
        if typeof(value)=="Vector3" then return value end

    local function queueShot(gun,first,last,target,info,mode)
        local now=os.clock()
        if pending and now-pending.at<1.6 then pending.ambiguous=true; return end
        pending={at=now,gun=gun,origin=first.Position,endpoint=last.Position,target=target,
            targetCharacter=target.Character,aim=info.endpoint,velocity=info.velocity,mode=mode}
    end
    local function observeEcho(fired,origin,endpoint)
        local shot=pending
        if stopped or not (V["Gun Aimbot"] or V["Auto Shoot"]) or not shot or shot.ambiguous then return end
        local now=os.clock();local delta=now-shot.at
        if delta<.05 or delta>1.6 then if delta>1.6 then pending=nil end; return end
        local ok,match=pcall(function()
            return fired==shot.gun or (fired and fired:IsDescendantOf(shot.gun))
                or (fired and fired:FindFirstAncestorOfClass("Tool")==shot.gun)
        end)
        local a,b=position(origin),position(endpoint)
        if not ok or not match or not a or not b or (a-shot.origin).Magnitude>3.5 then return end
        local expected=shot.endpoint-shot.origin;local actual=b-a
        if expected.Magnitude<.05 or actual.Magnitude<.05 or expected.Unit:Dot(actual.Unit)<.965 then return end
        pending=nil;prediction.noteEcho(shot.at,now)
        -- Score only a matched echo, same character and a meaningful measured velocity.
        local target=shot.targetCharacter
        local v=shot.velocity
        if target and shot.target.Character==target and v and shot.aim then
            local body=target:FindFirstChild("UpperTorso") or target:FindFirstChild("Torso") or target:FindFirstChild("HumanoidRootPart")
            local speed=Vector3.new(v.X,0,v.Z).Magnitude
            if body and body:IsA("BasePart") and speed>=3 then
                local error=((body.Position.X-shot.aim.X)*v.X+(body.Position.Z-shot.aim.Z)*v.Z)/speed
                if math.abs(error)<=6 then prediction.noteLeadError(error/speed,shot.mode) end
            end
        end
    end
    local function watchEcho()
        if echoConnection or observerBusy or os.clock()<observerAt then return end
        observerAt=os.clock()+3
        local ok,rs=pcall(function()return game:GetService("ReplicatedStorage")end)
        local services=ok and rs and rs:FindFirstChild("ClientServices")
        local module=services and services:FindFirstChild("WeaponService")
        if not module or not module:IsA("ModuleScript") then return end
        observerBusy=true;local ticket=generation
        task.spawn(function()
            local loaded,service=pcall(require,module)
            observerBusy=false
            if stopped or ticket~=generation or not (V["Gun Aimbot"] or V["Auto Shoot"]) or not loaded or type(service)~="table" then return end
            pcall(function() echoConnection=service.GunFired.OnClientEvent:Connect(observeEcho) end)
        end)
    end

        end)
    end
    local manualBusy,lastShot=false,-math.huge
    -- One sender shared by manual and automatic shots; a refusal never falls
    -- back to a different unchecked constructor. A send error is uncertain.
    local function sendShot(character,root,gun,target,method,wallCheck)
        if gun.Enabled==false or gun:GetAttribute("Disabled")==true
            or gun:GetAttribute("Reloading")==true or gun:GetAttribute("IsReloading")==true then return false end
        local remote=gun:FindFirstChild("Shoot")
        if not remote or not remote:IsA("RemoteEvent") then return false end
        local a,b,actual,info,mode=frames(character,root,gun,nil,target,method,wallCheck)
        if not a then return false end
        lastShot=os.clock();queueShot(gun,a,b,actual,info,mode)
        local sent=pcall(function()remote:FireServer(a,b)end)
        return sent
    end

                if stopped or ticket~=generation or V["Gun Aimbot"]~=true or current~=char or not aliveHuman or gun.Parent~=char then return end
                if os.clock()-lastShot<.22 then return end
                if sendShot(char,root,gun,nil,nil,nil) then macro(gun,char,ticket) end
            end)
            manualBusy=false
        end)
    end
    local installed,wrapper,previous = false,nil,nil
    local function installHook()
        if installed or type(hookmetamethod)~="function" or type(getnamecallmethod)~="function" or type(checkcaller)~="function" then return end
        local predecessor
        local function redirect(remote,...)
            if stopped or V["Gun Aimbot"]~=true then return predecessor(remote,...) end
            local methodOk,method=pcall(getnamecallmethod)
            local callerOk,caller=pcall(checkcaller)
            if not methodOk or method~="FireServer" or not callerOk or caller~=false
                or typeof(remote)~="Instance" or remote.Name~="Shoot" then return predecessor(remote,...) end
            local char,_,root=characterParts();local gun=weapon(LocalPlayer,"Gun")
            if not char or not root or not gun or gun.Parent~=char or remote~=gun:FindFirstChild("Shoot") then return predecessor(remote,...) end
            local args=table.pack(...)
            if typeof(args[1])~="CFrame" or typeof(args[2])~="CFrame" then return predecessor(remote,...) end
            local ok,a,b,target,info,mode=pcall(frames,char,root,gun,args[1])
            if not ok or not a or not b then return predecessor(remote,...) end
            queueShot(gun,a,b,target,info,mode);args[1],args[2]=a,b
            macro(gun,char,generation)
            return predecessor(remote,table.unpack(args,1,args.n))
        end
        local callback=redirect
        if type(newcclosure)=="function" then local ok,wrapped=pcall(newcclosure,redirect);if ok and type(wrapped)=="function" then callback=wrapped end end
        local ok,old=pcall(hookmetamethod,game,"__namecall",callback)
        if ok and type(old)=="function" then predecessor=old;previous,wrapper,installed=old,callback,true end
    end
    local function uninstallHook()
        if not installed or type(hookmetamethod)~="function" then return end
        local ok,displaced=pcall(hookmetamethod,game,"__namecall",previous)
        if ok and displaced==wrapper then installed,wrapper,previous=false,nil,nil
        elseif ok and type(displaced)=="function" then pcall(hookmetamethod,game,"__namecall",displaced) end
    end

                or LocalPlayer.Character~=char or gun.Parent~=char then return end
        end
        if stopped or not V["Auto Shoot"] or gun.Parent~=char then return end
        pcall(sendShot,char,root,gun,target,V["Shoot Mode"],V["Shoot WallCheck"]==true)
    end

-- Independent Extra movement: only Fly, Speed and Gravity are connected.
local function installMovementFeatures(screenGui)
    local V = AngelHub.values
    local run = game:GetService("RunService")
    local initialGravity = workspace.Gravity
    local stopped = false
    local fly = {velocity=Vector3.zero, up={}, down={}, burst=0}
    local speed = {velocity=Vector3.zero, sprintKey=nil, touch={}, burst=0}
    local mm2 = game.PlaceId == 142823291
    local gravity = {}
    local function parts()
        local character=localPlayer.Character
        local humanoid=character and character:FindFirstChildOfClass("Humanoid")
        local root=character and character:FindFirstChild("HumanoidRootPart")
        if not humanoid or humanoid.Health<=0 or not root or not root:IsA("BasePart") then return end
        return character,humanoid,root
    end
    local function number(name,default)
        local n=tonumber(V[name]);if not n or n~=n or math.abs(n)==math.huge then return default end
        return n
    end
    local function slide(character,root,velocity,dt)
        if velocity.Magnitude<.001 then return velocity end
        local ok,hit=pcall(function()
            local params=RaycastParams.new();params.FilterType=Enum.RaycastFilterType.Exclude
            params.FilterDescendantsInstances={character};params.RespectCanCollide=true;params.IgnoreWater=true
            local step=velocity*math.min(dt,.1)
            if workspace.Blockcast then return workspace:Blockcast(root.CFrame,root.Size*.9,step,params) end
            return workspace:Raycast(root.Position,step,params)
        end)
        if not ok then return Vector3.zero end
        if not hit then return velocity end
        local normal=hit.Normal
        if not normal then return Vector3.zero end
        local dot=velocity:Dot(normal)
        return dot<0 and velocity-normal*dot or velocity
    end
    local function releaseVelocity(owner)
        if owner.root and owner.applied then pcall(function()
            if owner.root.AssemblyLinearVelocity==owner.applied then
                local y=owner.preserveY and owner.applied.Y or math.min(owner.applied.Y,0)
                owner.root.AssemblyLinearVelocity=Vector3.new(0,y,0)
            end
        end) end
        owner.root=nil;owner.applied=nil;owner.velocity=Vector3.zero
    end
    local function applyVelocity(owner,root,value,preserveY)
        if owner.root and owner.root~=root then releaseVelocity(owner) end
        root.AssemblyLinearVelocity=value
        owner.root=root;owner.applied=value;owner.preserveY=preserveY
    end
    local function restoreSpeed()
        if speed.humanoid and speed.appliedWalk~=nil then pcall(function()
            if speed.humanoid.WalkSpeed==speed.appliedWalk then speed.humanoid.WalkSpeed=speed.oldWalk end
        end) end
        speed.humanoid=nil;speed.oldWalk=nil;speed.appliedWalk=nil
    end
    local function writeSpeed(humanoid,value)
        if speed.humanoid~=humanoid then restoreSpeed();speed.humanoid=humanoid;speed.oldWalk=humanoid.WalkSpeed end
        if speed.appliedWalk~=nil and humanoid.WalkSpeed~=speed.appliedWalk then speed.oldWalk=humanoid.WalkSpeed end
        if humanoid.WalkSpeed~=value then humanoid.WalkSpeed=value end
        speed.appliedWalk=value
    end
    local function skipBurst(owner,dt,interval)
        if interval<=0 then owner.burst=0;return false end
        owner.burst+=dt
        if owner.burst>=interval then owner.burst=0;return true end
        return false
    end
    local function clearFlyInput()
        table.clear(fly.up);table.clear(fly.down)
    end
    local function removePlatform()
        if fly.platform then pcall(function()fly.platform:Destroy()end);fly.platform=nil end
    end
    local function removeButtons()
        if fly.buttons then pcall(function()fly.buttons:Destroy()end);fly.buttons=nil end
    end
    local function makeButtons()
        if not userInputService.TouchEnabled or fly.buttons then return end
        local frame=create("Frame",{Name="AngelHubFlyControls",Size=UDim2.fromScale(1,1),BackgroundTransparency=1},screenGui)
        fly.buttons=frame
        for _,spec in ipairs({{"Up","+",-152,fly.up},{"Down","-",-96,fly.down}}) do
            local button=create("TextButton",{Name=spec[1],Text=spec[2],Size=UDim2.fromOffset(48,48),
                Position=UDim2.new(1,-70,1,spec[3]),TextSize=18,TextColor3=Color3.fromRGB(255,255,255),
                BackgroundColor3=Color3.fromRGB(25,25,25),BackgroundTransparency=.3,ZIndex=31},frame)
            create("UICorner",{CornerRadius=UDim.new(0,8)},button)
            local held=spec[4]
            button.InputBegan:Connect(function(input)
                if not stopped and V["Enable Fly"] and (input.UserInputType==Enum.UserInputType.Touch or input.UserInputType==Enum.UserInputType.MouseButton1) then held[input]=true end
            end)
            button.InputEnded:Connect(function(input)held[input]=nil end)
        end
    end
    local function clearSprintTrail()
        if speed.trail then pcall(function()speed.trail:Destroy()end);speed.trail=nil end
        if speed.left then pcall(function()speed.left:Destroy()end);speed.left=nil end
        if speed.right then pcall(function()speed.right:Destroy()end);speed.right=nil end
        speed.trailRoot=nil
    end
    local function sprintHeld()
        return mm2 and ((speed.sprintKey~=nil and speed.sprintKey==V["Sprint Key Value"]) or next(speed.touch)~=nil)
    end
    local function updateSprintTrail(root,moving)
        if speed.trailRoot~=root then clearSprintTrail() end
        if not speed.trail then
            local left=Instance.new("Attachment");speed.left=left;left.Name="AngelSprintLeft"
            left.Position=Vector3.new(-.8,-.5,.75);left.Parent=root
            local right=Instance.new("Attachment");speed.right=right;right.Name="AngelSprintRight"
            right.Position=Vector3.new(.8,-.5,.75);right.Parent=root
            local trail=Instance.new("Trail");speed.trail=trail;trail.Name="AngelSprintTail"
            trail.Attachment0=left;trail.Attachment1=right
            trail.Color=ColorSequence.new(Color3.fromRGB(255,220,55),Color3.fromRGB(175,115,20))
            trail.Lifetime=.38;trail.LightEmission=.25;trail.FaceCamera=true
            trail.Transparency=NumberSequence.new({NumberSequenceKeypoint.new(0,.2),NumberSequenceKeypoint.new(1,1)})
            trail.Parent=root;speed.trailRoot=root
        end
        speed.trail.Enabled=moving
    end
    local function stopSprintIfReleased()
        if sprintHeld() then return end
        clearSprintTrail()
        if not V["Enable Speed"] then restoreSpeed();releaseVelocity(speed) end
    end
    local function speedStep(dt)
        local sprint=sprintHeld()
        if not V["Enable Speed"] and not sprint then restoreSpeed();releaseVelocity(speed);clearSprintTrail();return end
        if V["Enable Fly"] then restoreSpeed();releaseVelocity(speed);clearSprintTrail();return end
        local character,humanoid,root=parts()
        if not character or root.Anchored or humanoid.Sit or humanoid.SeatPart then
            restoreSpeed();releaseVelocity(speed);clearSprintTrail();return
        end
        local move=humanoid.MoveDirection
        if sprint then updateSprintTrail(root,move.Magnitude>.05) else clearSprintTrail() end
        -- The MM2 sprint action also works with Speed OFF; release restores the prior WalkSpeed.
        if not V["Enable Speed"] then
            releaseVelocity(speed)
            local baseline=speed.humanoid==humanoid and speed.oldWalk or humanoid.WalkSpeed
            writeSpeed(humanoid,math.max(tonumber(baseline) or 16,30))
            return
        end
        local mode=V["Speed Mode"]
        local value=math.clamp(number("Speed Value",54),-10000,250)
        if sprint then value=math.max(value,30) end
        if V["Speed Auto Jump"] and move.Magnitude>.01 and humanoid.FloorMaterial~=Enum.Material.Air then humanoid.Jump=true end
        if mode=="WalkSpeed" and value>=0 then releaseVelocity(speed);writeSpeed(humanoid,value);return end
        -- Character controllers can overwrite physics velocity each frame. Keep
        -- a matching local WalkSpeed while using positive Velocity, too; restore
        -- the previous value on disable, mode change, respawn or Fly priority.
        if mode=="Velocity" and value>=0 then writeSpeed(humanoid,value) else restoreSpeed() end
        local target=Vector3.new(move.X*value,0,move.Z*value)
        local response=math.max(number("Speed Response",1),.05)
        local braking=math.max(number("Speed Braking",.2),0)
        local rate=move.Magnitude>.01 and 12*response or (braking<=.001 and 10000 or 1/braking)
        speed.velocity=speed.velocity:Lerp(target,1-math.exp(-rate*dt))
        local desired=speed.velocity
        if V["Speed Wall Check"] then desired=slide(character,root,desired,dt) end
        if skipBurst(speed,dt,number("Speed Burst Delay",.1)) then return end
        if mode=="CFrame" then
            releaseVelocity(speed);speed.velocity=desired
            root.CFrame=root.CFrame+desired*dt
        else
            local y=V["Speed Keep Vertical"] and root.AssemblyLinearVelocity.Y or 0
            applyVelocity(speed,root,Vector3.new(desired.X,y,desired.Z),true)
        end
    end
    local function flyStep(dt)
        if not V["Enable Fly"] then return end
        local character,humanoid,root=parts()
        if not character or root.Anchored or humanoid.Sit or humanoid.SeatPart then
            releaseVelocity(fly);removePlatform();fly.hoverY=nil;clearFlyInput();return
        end
        local up=next(fly.up)~=nil;local down=next(fly.down)~=nil
        local vertical=(up and 1 or 0)-(down and 1 or 0)
        local move=humanoid.MoveDirection
        local flat=Vector3.new(move.X,0,move.Z)
        if flat.Magnitude>1 then flat=flat.Unit end
        local h=math.clamp(number("Fly Horizontal Speed",173),-10000,250)
        local v=math.clamp(number("Fly Vertical Speed",60),-10000,250)
        local desired=flat*h+Vector3.new(0,vertical*v,0)
        local float=V["Fly Float"]
        if float=="Hover" and vertical==0 then
            fly.hoverY=fly.hoverY or root.Position.Y
            desired=Vector3.new(desired.X,math.clamp((fly.hoverY-root.Position.Y)*8,-math.abs(v),math.abs(v)),desired.Z)
        else fly.hoverY=nil end
        local alpha=1-math.exp(-10*math.max(number("Fly Response Multiplier",1),.05)*dt)
        fly.velocity=fly.velocity:Lerp(desired,alpha)
        local velocity=fly.velocity
        if V["Fly Wall Check"] then velocity=slide(character,root,velocity,dt) end
        if float=="Platform" then
            if not fly.platform then
                local platform=Instance.new("Part");platform.Name="AngelFlyPlatform";platform.Size=Vector3.new(6,1,6)
                platform.Transparency=1;platform.Anchored=true;platform.CanCollide=true;platform.CanTouch=false;platform.CanQuery=false
                platform.Parent=workspace;fly.platform=platform
            end
            fly.platform.CFrame=CFrame.new(root.Position-Vector3.new(0,humanoid.HipHeight+root.Size.Y/2+.5,0))
        else removePlatform() end
        if V["Fly Face Camera"] and workspace.CurrentCamera then
            local look=workspace.CurrentCamera.CFrame.LookVector;local direction=Vector3.new(look.X,0,look.Z)
            if direction.Magnitude>.001 then root.CFrame=CFrame.lookAt(root.Position,root.Position+direction) end
        end
        if skipBurst(fly,dt,number("Fly Burst Interval",.2)) then return end
        if V["Fly Method"]=="CFrame" then
            applyVelocity(fly,root,Vector3.zero,false)
            root.CFrame=root.CFrame+velocity*dt
        else applyVelocity(fly,root,velocity,false) end
    end
    local function restoreWorldGravity()
        if gravity.applied~=nil then pcall(function()
            if workspace.Gravity==gravity.applied then workspace.Gravity=gravity.before end
        end) end
        gravity.applied=nil;gravity.before=nil
    end
    local function clearForce()
        if gravity.force then pcall(function()gravity.force:Destroy()end);gravity.force=nil end
        if gravity.attachment then pcall(function()gravity.attachment:Destroy()end);gravity.attachment=nil end
    end
    local function gravityStep()
        if not V["Enable Gravity"] then return end
        local value=math.clamp(number("Gravity Value",initialGravity),-500,500)
        if V["Gravity Method"]~="Force" then
            clearForce()
            if gravity.applied==nil or workspace.Gravity~=gravity.applied then gravity.before=workspace.Gravity end
            if workspace.Gravity~=value then workspace.Gravity=value end
            gravity.applied=value
        else
            restoreWorldGravity()
            local _,_,root=parts()
            if not root or root.Anchored then clearForce();return end
            if not gravity.attachment or gravity.attachment.Parent~=root then
                clearForce()
                local attachment=Instance.new("Attachment");attachment.Name="AngelGravityAtt";attachment.Parent=root
                gravity.attachment=attachment
                local force=Instance.new("VectorForce");force.Name="AngelGravityForce";force.Attachment0=attachment
                force.RelativeTo=Enum.ActuatorRelativeTo.World;force.ApplyAtCenterOfMass=true;force.Parent=root
                gravity.attachment,gravity.force=attachment,force
            end
            gravity.force.Force=Vector3.new(0,root.AssemblyMass*(workspace.Gravity-value),0)
        end
    end
    local oldFly=AngelHub.setters["Enable Fly"]
    AngelHub.setters["Enable Fly"]=function(value)
        local on=value==true and not stopped
        oldFly(on);releaseVelocity(fly);removePlatform();clearFlyInput();fly.hoverY=nil;fly.burst=0
        if on then makeButtons() else removeButtons() end
    end
    local oldSpeed=AngelHub.setters["Enable Speed"]
    AngelHub.setters["Enable Speed"]=function(value)
        oldSpeed(value==true and not stopped);restoreSpeed();releaseVelocity(speed);speed.burst=0
    end
    for _,name in ipairs({"Fly Up Key Value","Fly Down Key Value","Sprint Key Value"}) do
        local old=AngelHub.setters[name]
        if old then AngelHub.setters[name]=function(value)
            old(value)
            if name=="Sprint Key Value" then speed.sprintKey=nil;table.clear(speed.touch);stopSprintIfReleased()
            else clearFlyInput() end
        end end
    end
    local oldGravity=AngelHub.setters["Enable Gravity"]
    AngelHub.setters["Enable Gravity"]=function(value)
        oldGravity(value==true and not stopped);clearForce();restoreWorldGravity()
    end
    AngelHub.resetGravity=function()
        if stopped then return end
        AngelHub.setters["Gravity Value"](initialGravity)
        if not V["Enable Gravity"] then workspace.Gravity=initialGravity end
    end
    if mm2 and userInputService.TouchEnabled then
        local button=create("TextButton",{Name="AngelHubSprintButton",Text="SPRINT",Size=UDim2.fromOffset(72,48),
            Position=UDim2.new(1,-158,1,-96),TextSize=14,TextColor3=Color3.fromRGB(255,255,255),
            BackgroundColor3=Color3.fromRGB(25,25,25),BackgroundTransparency=.3,ZIndex=31},screenGui)
        create("UICorner",{CornerRadius=UDim.new(0,8)},button)
        button.InputBegan:Connect(function(input)
            if not stopped and (input.UserInputType==Enum.UserInputType.Touch or input.UserInputType==Enum.UserInputType.MouseButton1) then speed.touch[input]=true end
        end)
        button.InputEnded:Connect(function(input)speed.touch[input]=nil;stopSprintIfReleased() end)
    end
    connectInput(userInputService.InputBegan,function(input,processed)
        if stopped or processed or AngelHub.listening or userInputService:GetFocusedTextBox()
            or input.UserInputType~=Enum.UserInputType.Keyboard then return end
        local name=input.KeyCode and input.KeyCode.Name
        if not name then return end
        if V["Enable Fly"] and name==V["Fly Up Key Value"] then fly.up[input.KeyCode]=true end
        if V["Enable Fly"] and name==V["Fly Down Key Value"] then fly.down[input.KeyCode]=true end
        if mm2 and name==V["Sprint Key Value"] then speed.sprintKey=name end
    end)
    connectInput(userInputService.InputEnded,function(input)
        if input.KeyCode then
            fly.up[input.KeyCode]=nil;fly.down[input.KeyCode]=nil
            if input.KeyCode.Name==speed.sprintKey then speed.sprintKey=nil;stopSprintIfReleased() end
        end
        fly.up[input]=nil;fly.down[input]=nil
        if speed.touch[input] then speed.touch[input]=nil;stopSprintIfReleased() end
    end)
    local function clearKeys()clearFlyInput();speed.sprintKey=nil;table.clear(speed.touch);stopSprintIfReleased() end
    connectInput(userInputService.WindowFocusReleased,clearKeys)
    connectInput(userInputService.TextBoxFocused,clearKeys)
    local function runMovement(step,name,dt)
        if not V[name] then return end
        local ok,err=pcall(step,dt)
        if not ok then warn("Angel Hub "..name..": "..tostring(err));AngelHub.setters[name](false) end
    end
    connectInput(run.Heartbeat,function(dt)
        if stopped then return end
        dt=math.clamp(tonumber(dt) or 0,0,.1)
        if dt<=0 then return end
        if V["Enable Speed"] or sprintHeld() then
            local ok,err=pcall(speedStep,dt)
            if not ok then
                warn("Angel Hub Speed/Sprint: "..tostring(err));speed.sprintKey=nil;table.clear(speed.touch)
                clearSprintTrail();AngelHub.setters["Enable Speed"](false)
            end
        end
        runMovement(flyStep,"Enable Fly",dt)
        runMovement(gravityStep,"Enable Gravity",dt)
    end)
    screenGui.Destroying:Connect(function()
        stopped=true;clearKeys();removeButtons();removePlatform();releaseVelocity(fly);releaseVelocity(speed)
        clearSprintTrail();restoreSpeed();clearForce();restoreWorldGravity();AngelHub.resetGravity=nil
    end)
end

type Runtime = {
    framework: any,
    entity: any,
    host: any,
    services: any,
}

local Module = {
    Name = "AntiFling",
    PlaceId = 0,
    Events = {} :: {[string]: any},
    Initialized = false,
}

local activeCleanup: (() -> ())? = nil

function Module.init(context: Runtime): any
    local host: any = context.host
    local services: any = context.services
    local activity: any = services and services.activity
    local framework: any = context.framework
    local players: any = host.Players
    local localPlayer: any = host.LocalPlayer
    local world: any = host.workspace or workspace
    local featureConnections: any = host.featureConnections
    local disconnect: any = host.disconnectFeatureConnection

    local enabled: boolean = false
    local playerShield: boolean = true
    local impactShield: boolean = true
    local feature: any = nil
    local character: Model? = nil
    local links: {[BasePart]: {[BasePart]: any}} = {}
    local hazardSeen: {[BasePart]: number} = setmetatable({}, {__mode = "k"}) :: any
    local nextScan: number = 0
    local dangerUntil: number = 0
    local suspended: boolean = false
    local warnedOverlap: boolean = false

    -- Limits are deliberately conservative. The shield is local physics, not
    -- server authority; do not change other people's CanCollide or global groups.
    local SCAN_INTERVAL: number = 0.15
    local RADIUS: number = 30
    local HAZARD_HOLD: number = 0.75
    local LINEAR_LIMIT: number = 80
    local SURFACE_LIMIT: number = 70
    local MAX_LINKS: number = 2048
    local MAX_PARTS: number = 200

    local function isBusy(): boolean
        return activity ~= nil and type(activity.isActive) == "function"
            and (activity.isActive("fling") or activity.isActive("walkfling"))
    end

    local function destroyLinks(): ()
        for _, perLocal: {[BasePart]: any} in pairs(links) do
            for _, constraint: any in pairs(perLocal) do
                pcall(function(): () constraint:Destroy() end)
            end
        end
        table.clear(links)
        table.clear(hazardSeen)
        dangerUntil = 0
    end

    local function isToolPart(part: Instance): boolean
        local ok: boolean, result: any = pcall(function(): any
            return part:FindFirstAncestorOfClass("Tool")
        end)
        return ok and result ~= nil
    end

    local function listBodyParts(model: Model?, onlyCollidable: boolean): {BasePart}
        local parts: {BasePart} = {}
        if not model or not model.Parent then return parts end
        local ok: boolean, descendants: any = pcall(function(): any return model:GetDescendants() end)
        if not ok or type(descendants) ~= "table" then return parts end
        for _, object: Instance in ipairs(descendants) do
            if object:IsA("BasePart") and not isToolPart(object) then
                local part: BasePart = object :: BasePart
                if not onlyCollidable or part.CanCollide then table.insert(parts, part) end
            end
        end
        return parts
    end

    local function fastPart(part: BasePart, scale: number?): boolean
        local multiplier: number = scale or 1
        local radius: number = math.min(part.Size.Magnitude * 0.5, 20)
        return part.AssemblyLinearVelocity.Magnitude > LINEAR_LIMIT * multiplier
            or part.AssemblyAngularVelocity.Magnitude * radius > SURFACE_LIMIT * multiplier
    end

    local function supportAssembly(root: BasePart, humanoid: Humanoid, owner: Model): (BasePart?, boolean)
        local ok: boolean, result: any = pcall(function(): any
            local params: RaycastParams = RaycastParams.new()
            params.FilterType = Enum.RaycastFilterType.Exclude
            params.FilterDescendantsInstances = {owner}
            params.IgnoreWater = true
            params.RespectCanCollide = true
            local distance: number = math.max(4, (tonumber(humanoid.HipHeight) or 2) + root.Size.Y * 0.5 + 2)
            return world:Raycast(root.Position, Vector3.new(0, -distance, 0), params)
        end)
        if not ok then return nil, false end
        local ground: any = result and result.Instance
        if ground and ground:IsA("BasePart") then
            return ground.AssemblyRootPart or ground, true
        end
        return nil, true
    end

    local function synchronize(owner: Model, humanoid: Humanoid, root: BasePart, now: number): ()
        local ownParts: {BasePart} = listBodyParts(owner, true)
        local desired: {[BasePart]: boolean} = {}
        local candidates: {any} = {}
        local characterModels: {[Model]: boolean} = {}
        local roster: any = players:GetPlayers()

        -- Track every visible player body, not the shared world collision group.
        -- Nearest bodies get the bounded constraint budget first on large servers.
        for _, player: Player in ipairs(roster) do
            local other: Model? = player ~= localPlayer and player.Character or nil
            if other and other ~= owner and other.Parent then
                characterModels[other] = true
                local otherRoot: BasePart? = other:FindFirstChild("HumanoidRootPart") :: BasePart?
                local distance: number = otherRoot and (otherRoot.Position - root.Position).Magnitude or math.huge
                table.insert(candidates, {model = other, distance = distance,
                    parts = if playerShield then listBodyParts(other, false) else {}})
            end
        end
        table.sort(candidates, function(a: any, b: any): boolean return a.distance < b.distance end)
        if playerShield then
            for _, candidate: any in ipairs(candidates) do
                for _, part: BasePart in ipairs(candidate.parts) do
                    desired[part] = true
                    if candidate.distance <= RADIUS and fastPart(part, nil) then
                        dangerUntil = math.max(dangerUntil, now + HAZARD_HOLD)
                    end
                end
            end
        end

        -- The query is local and bounded. Anchored geometry, the supporting
        -- platform/assembly, seats and tools keep their normal interactions.
        -- A fast loose part is suppressed only briefly while it remains risky.
        if impactShield and not humanoid.Sit and not humanoid.SeatPart then
            local overlap: any = nil
            pcall(function(): ()
                overlap = OverlapParams.new()
                overlap.FilterType = Enum.RaycastFilterType.Exclude
                overlap.FilterDescendantsInstances = {owner}
                overlap.MaxParts = MAX_PARTS
                overlap.RespectCanCollide = true
            end)
            local ok: boolean, nearby: any = pcall(function(): any
                return world:GetPartBoundsInRadius(root.Position, RADIUS, overlap)
            end)
            if not ok or type(nearby) ~= "table" then
                if not warnedOverlap and type(host.notify) == "function" then
                    warnedOverlap = true
                    host.notify("Anti Fling: moving-object scan unavailable; player protection remains active.")
                end
            else
                local floorAssembly: BasePart?, floorChecked: boolean = supportAssembly(root, humanoid, owner)
                if not floorChecked then table.clear(hazardSeen) end
                for _, instance: Instance in ipairs(if floorChecked then nearby else {}) do
                    if instance:IsA("BasePart") then
                        local part: BasePart = instance :: BasePart
                        local isCharacterPart: boolean = false
                        local ancestor: Instance? = part.Parent
                        while ancestor and ancestor ~= world do
                            if characterModels[ancestor :: any] then isCharacterPart = true break end
                            ancestor = ancestor.Parent
                        end
                        local inOwner: boolean = part:IsDescendantOf(owner)
                        local isTool: boolean = isToolPart(part)
                        local assembly: BasePart? = part.AssemblyRootPart or part
                        -- Ordinary weapon/tools remain interactive. An extreme
                        -- spinning/flying handle is treated as a hazard, at 3x
                        -- the loose-object threshold, only while it is nearby.
                        if part.Parent and part.CanCollide and not part.Anchored and not inOwner
                            and (not isCharacterPart or isTool)
                            and (not floorAssembly or assembly ~= floorAssembly)
                            and fastPart(part, if isTool then 3 else nil) then
                            hazardSeen[part] = now
                            dangerUntil = math.max(dangerUntil, now + HAZARD_HOLD)
                        end
                    end
                end
            end
            for part: BasePart, seen: number in pairs(hazardSeen) do
                if not part.Parent or now - seen > HAZARD_HOLD then
                    hazardSeen[part] = nil
                else
                    desired[part] = true
                end
            end
        else
            table.clear(hazardSeen)
        end

        -- Allocate the finite budget afresh in priority order: newly dangerous
        -- objects first, then nearest players. Otherwise 2,048 older/farther
        -- links could permanently starve a new incoming threat.
        local ordered: {BasePart} = {}
        local included: {[BasePart]: boolean} = {}
        local function include(part: BasePart): ()
            if desired[part] and part.Parent and not included[part] then
                included[part] = true
                table.insert(ordered, part)
            end
        end
        local hazards: {BasePart} = {}
        for part: BasePart in pairs(hazardSeen) do table.insert(hazards, part) end
        table.sort(hazards, function(a: BasePart, b: BasePart): boolean
            return (a.Position - root.Position).Magnitude < (b.Position - root.Position).Magnitude
        end)
        for _, part: BasePart in ipairs(hazards) do include(part) end
        if playerShield then
            for _, candidate: any in ipairs(candidates) do
                for _, part: BasePart in ipairs(candidate.parts) do include(part) end
            end
        end
        local accepted: {[BasePart]: {[BasePart]: boolean}} = {}
        local allocated: number = 0
        for _, other: BasePart in ipairs(ordered) do
            if allocated >= MAX_LINKS then break end
            local perLocal: {[BasePart]: boolean} = {}
            for _, own: BasePart in ipairs(ownParts) do
                if allocated >= MAX_LINKS then break end
                if own ~= other then
                    perLocal[own] = true
                    allocated += 1
                end
            end
            if next(perLocal) then accepted[other] = perLocal end
        end
        -- Remove old or lower-priority pairs before filling the new budget.
        for other: BasePart, perLocal: {[BasePart]: any} in pairs(links) do
            local wanted: {[BasePart]: boolean}? = accepted[other]
            for own: BasePart, constraint: any in pairs(perLocal) do
                if not wanted or not wanted[own] or not own.Parent or not other.Parent then
                    pcall(function(): () constraint:Destroy() end)
                    perLocal[own] = nil
                end
            end
            if next(perLocal) == nil then links[other] = nil end
        end
        for other: BasePart, wanted: {[BasePart]: boolean} in pairs(accepted) do
            local perLocal: {[BasePart]: any} = links[other] or {}
            links[other] = perLocal
            for own: BasePart in pairs(wanted) do
                if not perLocal[own] then
                    local ok: boolean, constraint: any = pcall(function(): any
                        local link: NoCollisionConstraint = Instance.new("NoCollisionConstraint")
                        link.Part0, link.Part1 = own, other
                        link.Parent = own
                        return link
                    end)
                    if ok and constraint then perLocal[own] = constraint end
                end
            end
            if next(perLocal) == nil then links[other] = nil end
        end
    end

    local function step(): ()
        if not enabled then return end
        local owner: Model?, humanoid: Humanoid?, root: BasePart? = host.getCharacterParts()
        if owner ~= character then
            destroyLinks()
            character = owner
            nextScan = 0
        end
        if not owner or not humanoid or not root or not owner.Parent or humanoid.Health <= 0 then
            destroyLinks()
            return
        end
        if isBusy() then
            if not suspended then destroyLinks() end
            suspended = true
            return
        end
        if suspended then suspended = false nextScan = 0 end
        local now: number = os.clock()
        if now >= nextScan then
            nextScan = now + SCAN_INTERVAL
            local ok: boolean, err: any = pcall(synchronize, owner, humanoid, root, now)
            if not ok then
                -- A streaming object may disappear during the scan. Retry on
                -- the next interval, without leaving partial constraints stuck.
                destroyLinks()
                if type(host.warn) == "function" then host.warn("Anti Fling: " .. tostring(err)) end
            end
        end
        if now > dangerUntil or root.Anchored or humanoid.Sit or humanoid.SeatPart then return end
        -- Emergency damping is only armed by a nearby extreme mover. Normal
        -- sprinting, jumps, vehicles and game launch pads are not capped merely
        -- because this feature is on.
        local walkSpeed: number = math.max(16, tonumber(humanoid.WalkSpeed) or 16)
        local speed: any = framework and framework:GetModule("Speed Hack")
        if speed and speed.Enabled and speed.Options.Speed then
            walkSpeed = math.max(walkSpeed, tonumber(speed.Options.Speed.Value) or 0)
        end
        local jumpPower: number = math.max(50, tonumber(humanoid.JumpPower) or 50)
        local velocity: Vector3 = root.AssemblyLinearVelocity
        local planar: Vector3 = Vector3.new(velocity.X, 0, velocity.Z)
        local limit: number = math.max(120, walkSpeed * 4)
        local riseLimit: number = math.max(180, jumpPower * 4)
        local nextVelocity: Vector3 = velocity
        local changed: boolean = false
        if planar.Magnitude > limit then
            local move: Vector3 = humanoid.MoveDirection
            local direction: Vector3 = move.Magnitude > 0.05 and move.Unit * walkSpeed or Vector3.zero
            nextVelocity = Vector3.new(direction.X, nextVelocity.Y, direction.Z)
            changed = true
        end
        if nextVelocity.Y > riseLimit then
            nextVelocity = Vector3.new(nextVelocity.X, riseLimit, nextVelocity.Z)
            changed = true
        end
        pcall(function(): ()
            if root.AssemblyAngularVelocity.Magnitude > 25 then
                root.AssemblyAngularVelocity = Vector3.zero
            end
            if changed then root.AssemblyLinearVelocity = nextVelocity end
        end)
    end

    local function toggleAntiFling(value: boolean): ()
        enabled = value == true
        disconnect("AntiFling")
        destroyLinks()
        character = nil
        nextScan = 0
        suspended = false
        if enabled then
            featureConnections.AntiFling = host.TaskManager:Connect(step, nil, "Anti Fling")
        end
    end

    feature = host.createUniversalFeature(
        "Anti Fling",
        "Collision Shield: local player and extreme-object protection",
        2,
        toggleAntiFling,
        {categoryName = "Utilities"}
    )
    host.addToggleOption(feature, "Player Shield", true, function(value: boolean): ()
        playerShield = value == true
        nextScan = 0
        if not playerShield and not impactShield then destroyLinks() end
    end, "Prevent local physical collisions with other character bodies without changing their collision settings.")
    host.addToggleOption(feature, "Impact Shield", true, function(value: boolean): ()
        impactShield = value == true
        nextScan = 0
        if not playerShield and not impactShield then destroyLinks() end
    end, "Temporarily avoid nearby extreme-speed loose objects; support platforms, seated vehicles and ordinary tools are left alone.")
    host.addFeatureTooltip(feature,
        "Collision Shield only changes local collision pairs. Extremely fast/streamed objects and server-owned physics can still launch you.")

    activeCleanup = function(): ()
        toggleAntiFling(false)
    end
    Module.Initialized = true
    return feature
end

function Module.destroy(): ()
    if activeCleanup then pcall(activeCleanup) end
    activeCleanup = nil
    Module.Initialized = false
end

return Module

	-- Config files use Config Name; blank falls back to the default file.
	local function getConfigPath()
		local name = string.gsub(tostring(AngelHub.values["Config Name"] or ""), "[^%w_%- ]", "")
		if name == "" then
			return CONFIG_FILE
		end

		return CONFIG_FOLDER .. "/" .. name .. ".json"
	end

	-- Color3 is not JSON friendly, so it is stored as a tagged table.
	local function encodeValues()
		local data = {}

		for name, value in AngelHub.values do
			if typeof(value) == "Color3" then
				data[name] = {
					R = math.floor(value.R * 255 + 0.5),
					G = math.floor(value.G * 255 + 0.5),
					B = math.floor(value.B * 255 + 0.5),
					__color = true,
				}
			else
				data[name] = value
			end
		end

		return httpService:JSONEncode(data)
	end

	local function writeConfig(path)
		if not writefile then
			return false, "Saving is not supported here."
		end

		return pcall(function()
			if makefolder and isfolder and not isfolder(CONFIG_FOLDER) then
				makefolder(CONFIG_FOLDER)
			end

			writefile(path, encodeValues())
		end)
	end

	local function readConfig(path)
		if not (readfile and isfile and isfile(path)) then
			return nil, "No saved settings found."
		end

		local ok, data = pcall(function()
			return httpService:JSONDecode(readfile(path))
		end)

		if not ok or typeof(data) ~= "table" then
			return nil, "Settings file is corrupted."
		end

		return data
	end

	local function applyConfig(data)
        if data["WallCheck"] == nil then data["WallCheck"] = data["Aimbot WallCheck"] end
        if data["Macro"] == nil then data["Macro"] = data["Aimbot Macro"] end
        local legacyAim = {Bullet="Normal", ["Normal(Bullet)"]="Normal", Experimental="Normal",
            CFrame="Instakill", ["Silent AIM"]="Instakill", ["Silent(CFrame)"]="Instakill", ["Silent(Cframe)"]="Instakill"}
        data["Aim Method"] = legacyAim[data["Aim Method"]] or data["Aim Method"]
        data["Shoot Mode"] = legacyAim[data["Shoot Mode"]] or data["Shoot Mode"]
		for name, value in data do
			local setter = AngelHub.setters[name]
			if not setter then
				continue
			end

			if typeof(value) == "table" and value.__color then
				value = Color3.fromRGB(value.R, value.G, value.B)
			end

			pcall(setter, value)
		end
	end

	local function saveSettings()
		local ok, result = writeConfig(getConfigPath())
		setStatus(ok and "Settings saved." or ("Save failed: " .. tostring(result)))
	end

	local function loadSettings()
		local data, message = readConfig(getConfigPath())
		if not data then
			return setStatus(message)
		end

		applyConfig(data)
		setStatus("Settings loaded.")
	end

		for name, value in AngelHub.defaults do
			AngelHub.setters[name](value)
		end

		setStatus("Settings reset.")
