-- Only actions with verified local paths and call signatures are wired. The
-- script does not guess killer attack arguments or undocumented skill-check
-- result remotes. Auto Perfect uses the visible skill-check UI and configured input.
-- Session profiles remain in PlayerGui until the client exits.

local userInputService = game:GetService("UserInputService")
local tweenService = game:GetService("TweenService")
local httpService = game:GetService("HttpService")
local runService = game:GetService("RunService")
local replicatedStorage = game:GetService("ReplicatedStorage")

local ACCENT_A = Color3.fromRGB(235, 235, 238)
local ACCENT_B = Color3.fromRGB(110, 110, 116)
local BACKGROUND_TOP = Color3.fromRGB(25, 25, 27)
local BACKGROUND_BOTTOM = Color3.fromRGB(9, 9, 10)
local ROW_COLOR = Color3.fromRGB(34, 34, 36)
local ELEMENT_COLOR = Color3.fromRGB(17, 17, 19)
local TEXT_COLOR = Color3.fromRGB(240, 240, 242)
local OFF_TEXT_COLOR = Color3.fromRGB(192, 192, 196)
local DIM_TEXT_COLOR = Color3.fromRGB(122, 122, 128)

local TEXT_FONT = Enum.Font.SourceSans
local BOLD_FONT = Enum.Font.SourceSansBold

local DRAG_TWEEN = TweenInfo.new(0.06, Enum.EasingStyle.Quart, Enum.EasingDirection.Out)
local SLIDE_TWEEN = TweenInfo.new(0.10, Enum.EasingStyle.Quart, Enum.EasingDirection.Out)
local FAST_TWEEN = TweenInfo.new(0.18, Enum.EasingStyle.Quart, Enum.EasingDirection.Out)
local MEDIUM_TWEEN = TweenInfo.new(0.38, Enum.EasingStyle.Exponential, Enum.EasingDirection.Out)
local POP_TWEEN = TweenInfo.new(0.32, Enum.EasingStyle.Back, Enum.EasingDirection.Out)


local ZenithUI = {}

local function create(className, properties, parent)
	local instance = Instance.new(className)
	for key, value in properties do
		instance[key] = value
	end
	instance.Parent = parent
	return instance
end

local function tween(instance, info, goal)
	local newTween = tweenService:Create(instance, info, goal)
	newTween:Play()
	return newTween
end

local function isPress(input)
	return input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch
end

local function isMove(input)
	return input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch
end

local function addGradient(parent, rotation)
	return create("UIGradient", {
		Color = ColorSequence.new(ACCENT_A, ACCENT_B),
		Rotation = rotation,
	}, parent)
end

local function addStroke(parent, color, transparency, thickness)
	return create("UIStroke", {
		Color = color,
		Transparency = transparency,
		Thickness = thickness or 1,
		ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
	}, parent)
end

function ZenithUI.create(guiParent)

	local MAX_WIDTH, MAX_HEIGHT = 700, 1050

	local Hub = {
		values = {},
		pages = {},
		tabButtons = {},
		tabList = {},
		searchables = {},
		sections = {},
		boxes = {},
		setters = {},
		connections = {},
		cleanupCallbacks = {},
		listening = nil,
		cancelListening = nil,
		defaults = {},
		menuKey = Enum.KeyCode.RightShift,
	}

	function Hub:AddCleanup(callback)
		if type(callback) == "function" then
			table.insert(Hub.cleanupCallbacks, callback)
		end
	end

	local function connectInput(signal, callback)
		local connection = signal:Connect(callback)
		table.insert(Hub.connections, connection)
		return connection
	end

	local previous = guiParent:FindFirstChild("ZenithPreview")
	if previous and previous:IsA("ScreenGui") and previous:GetAttribute("ZenithOwned") == true then
		previous:Destroy()
	end

		Name = "ZenithPreview",
		ResetOnSpawn = false,
		DisplayOrder = 20,
		ZIndexBehavior = Enum.ZIndexBehavior.Sibling,
	}, guiParent)
	screenGui:SetAttribute("ZenithOwned", true)

	screenGui.Destroying:Connect(function()
		for _, connection in Hub.connections do
			if connection.Connected then connection:Disconnect() end
		end
		table.clear(Hub.connections)
		for _, callback in Hub.cleanupCallbacks do
			pcall(callback)
		end
		table.clear(Hub.cleanupCallbacks)
	end)

	-- Stacked Toast Notifications

		AnchorPoint = Vector2.new(1, 1),
		Position = UDim2.new(1, -16, 1, -16),
		Size = UDim2.fromOffset(230, 320),

		ZIndex = 20,

		Padding = UDim.new(0, 6),
		SortOrder = Enum.SortOrder.LayoutOrder,
		HorizontalAlignment = Enum.HorizontalAlignment.Right,
		VerticalAlignment = Enum.VerticalAlignment.Bottom,

			Size = UDim2.fromOffset(230, 34),
			BackgroundColor3 = Color3.fromRGB(18, 18, 20),
			BorderSizePixel = 0,

		create("UICorner", { CornerRadius = UDim.new(0, 7) }, toast)
		addStroke(toast, Color3.new(1, 1, 1), 0.8)

		local toastScale = create("UIScale", { Scale = 0.92 }, toast)
		local statusColor = (enabled == nil) and ACCENT_A or (enabled and Color3.fromRGB(120, 220, 140) or Color3.fromRGB(230, 100, 100))

		create("Frame", {
			Size = UDim2.new(0, 3, 1, -12),
			Position = UDim2.fromOffset(6, 6),
			BackgroundColor3 = statusColor,
			BorderSizePixel = 0,
		}, toast)

		create("TextLabel", {
			Size = UDim2.new(1, -80, 1, 0),
			Position = UDim2.fromOffset(18, 0),
			BackgroundTransparency = 1,
			Text = name,
			TextColor3 = TEXT_COLOR,
			TextSize = 13,
			Font = TEXT_FONT,
			TextXAlignment = Enum.TextXAlignment.Left,
			TextTruncate = Enum.TextTruncate.AtEnd,
		}, toast)

		if enabled ~= nil then
			create("TextLabel", {
				Size = UDim2.fromOffset(46, 34),
				Position = UDim2.new(1, -54, 0, 0),
				BackgroundTransparency = 1,
				Text = enabled and "ON" or "OFF",
				TextColor3 = statusColor,
				TextSize = 12,
				Font = BOLD_FONT,
				TextXAlignment = Enum.TextXAlignment.Right,
			}, toast)
		end

		tween(toastScale, POP_TWEEN, { Scale = 1 })

		task.delay(2, function()
			tween(toastScale, FAST_TWEEN, { Scale = 0.94 })
			tween(toast, MEDIUM_TWEEN, { GroupTransparency = 1 }).Completed:Wait()
			toast:Destroy()
		end)
	end

	-- Main Window
	local window = create("Frame", {
		Name = "Window",
		Size = UDim2.fromOffset(WINDOW_WIDTH, WINDOW_HEIGHT),
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.fromScale(0.5, 0.5),
		BackgroundColor3 = Color3.new(1, 1, 1),

		ClipsDescendants = true,

	create("UICorner", { CornerRadius = UDim.new(0, 8) }, window)
	create("UIGradient", { Color = ColorSequence.new(BACKGROUND_TOP, BACKGROUND_BOTTOM), Rotation = 90 }, window)

	window.Parent = viewport
	local function fit()
		local size = viewport.AbsoluteSize
		if size.X <= 12 or size.Y <= 12 then return end
		windowScale.Scale = math.min(requestedScale, (size.X - 12) / windowWidth, (size.Y - 12) / windowHeight)
	end
	viewport:GetPropertyChangedSignal("AbsoluteSize"):Connect(fit)
	function Hub:SetSize(value)
		requestedScale = value
		fit()
	end
	function Hub:SetTransparency(value)
		window.BackgroundTransparency = value / 100
	end
	function Hub:Close()
		screenGui:Destroy()
	end

		BorderSizePixel = 0,

		NumberSequenceKeypoint.new(0, 0.15),

		Size = UDim2.new(1, -260, 1, 0),

		TextSize = 15,
		Font = BOLD_FONT,

		BackgroundTransparency = 0.15,
		BorderSizePixel = 0,

		if not searchBox:IsFocused() then

		tween(searchStroke, FAST_TWEEN, { Color = ACCENT_A, Transparency = 0.05 })

			Position = UDim2.new(1, offset, 0.5, -11),
			BackgroundColor3 = ELEMENT_COLOR,
			BackgroundTransparency = 0.35,
			BorderSizePixel = 0,
			Text = text,
			TextColor3 = TEXT_COLOR,
			TextSize = 13,
			Font = BOLD_FONT,
			AutoButtonColor = false,
		}, titleBar)
		create("UICorner", { CornerRadius = UDim.new(0, 5) }, btn)
		local btnStroke = addStroke(btn, DIM_TEXT_COLOR, 0.75)

		btn.MouseEnter:Connect(function()
			tween(btn, FAST_TWEEN, { BackgroundColor3 = hoverColor, BackgroundTransparency = 0.08 })
			tween(btnStroke, FAST_TWEEN, { Color = ACCENT_A, Transparency = 0.35 })
		end)
		btn.MouseLeave:Connect(function()
			tween(btn, FAST_TWEEN, { BackgroundColor3 = ELEMENT_COLOR, BackgroundTransparency = 0.35 })
			tween(btnStroke, FAST_TWEEN, { Color = DIM_TEXT_COLOR, Transparency = 0.75 })
		end)
		btn.MouseButton1Down:Connect(function()
			tween(btn, FAST_TWEEN, { BackgroundTransparency = 0.45 })

	local closeButton = createTitleButton("X", -32, Color3.fromRGB(230, 60, 70))

	local dragStart, startPosition
	titleBar.InputBegan:Connect(function(input)
		if isPress(input) and input.Position.X < searchBox.AbsolutePosition.X then
			dragStart = input.Position
			startPosition = window.Position
		end
	end)
	connectInput(userInputService.InputChanged, function(input)
		if dragStart and isMove(input) then
			local delta = input.Position - dragStart
			tween(window, DRAG_TWEEN, {
				Position = UDim2.new(
					startPosition.X.Scale,
					startPosition.X.Offset + delta.X,
					startPosition.Y.Scale,
					startPosition.Y.Offset + delta.Y
				)
			})
		end
	end)
	connectInput(userInputService.InputEnded, function(input)
		if isPress(input) then dragStart = nil end
	end)

	local minimized = false

	local tabBar = create("Frame", {
		Size = UDim2.new(1, 0, 0, TAB_HEIGHT),
		Position = UDim2.fromOffset(0, TITLE_HEIGHT),
		BackgroundTransparency = 1,
	}, window)

		Size = UDim2.new(1, -12, 1, -6),
		Position = UDim2.new(0, 6, 0, 3),
		BackgroundColor3 = Color3.new(1, 1, 1),
		BackgroundTransparency = 0.9,
		BorderSizePixel = 0,

	create("UICorner", { CornerRadius = UDim.new(0, 6) }, indicator)
	addStroke(indicator, Color3.new(1, 1, 1), 0.8)

	local underline = create("Frame", {
		AnchorPoint = Vector2.new(0.5, 1),
		Size = UDim2.new(0.5, 0, 0, 2),
		Position = UDim2.fromScale(0.5, 1),
		BackgroundColor3 = Color3.new(1, 1, 1),
		BorderSizePixel = 0,
	}, indicator)
	create("UICorner", { CornerRadius = UDim.new(1, 0) }, underline)
	addGradient(underline, 0)

	local contentTop = TITLE_HEIGHT + TAB_HEIGHT + 8
	local content = create("Frame", {
		Size = UDim2.new(1, -20, 1, -(contentTop + 10)),
		Position = UDim2.fromOffset(10, contentTop),
		BackgroundTransparency = 1,
	}, window)

	-- Resize Grip in Bottom Right Corner
	local resizeHandle = create("TextButton", {
		AnchorPoint = Vector2.new(1, 1),
		Size = UDim2.fromOffset(18, 18),
		Position = UDim2.fromScale(1, 1),
		BackgroundTransparency = 1,
		Text = "",
		ZIndex = 10,
	}, window)

	for gripIndex = 1, 3 do
		create("Frame", {
			AnchorPoint = Vector2.new(1, 1),
			Size = UDim2.fromOffset(gripIndex * 4, 1),
			Position = UDim2.new(1, -3, 1, -3 - (3 - gripIndex) * 4 - 2),
			BackgroundColor3 = DIM_TEXT_COLOR,
			BorderSizePixel = 0,
		}, resizeHandle)
	end

	local resizing, resizeStart, startWidth, startHeight = false, nil, nil, nil

	resizeHandle.MouseEnter:Connect(function()
		for _, grip in resizeHandle:GetChildren() do
			tween(grip, FAST_TWEEN, { BackgroundColor3 = ACCENT_A })
		end
	end)

	resizeHandle.MouseLeave:Connect(function()
		if resizing then return end
		for _, grip in resizeHandle:GetChildren() do
			tween(grip, FAST_TWEEN, { BackgroundColor3 = DIM_TEXT_COLOR })
		end
	end)

	resizeHandle.InputBegan:Connect(function(input)
		if isPress(input) then
			resizing = true
			resizeStart = input.Position
			startWidth, startHeight = windowWidth, windowHeight
		end
	end)

	connectInput(userInputService.InputChanged, function(input)
		if resizing and isMove(input) then
			local delta = (input.Position - resizeStart) / windowScale.Scale
			windowWidth = math.clamp(startWidth + delta.X, MIN_WIDTH, MAX_WIDTH)
			windowHeight = math.clamp(startHeight + delta.Y, MIN_HEIGHT, MAX_HEIGHT)
			tween(window, DRAG_TWEEN, { Size = UDim2.fromOffset(windowWidth, windowHeight) })
			fit()
		end
	end)

	connectInput(userInputService.InputEnded, function(input)
		if resizing and isPress(input) then
			resizing = false
			for _, grip in resizeHandle:GetChildren() do
				tween(grip, FAST_TWEEN, { BackgroundColor3 = DIM_TEXT_COLOR })
			end
		end
	end)

	minimizeButton.Activated:Connect(function()
		minimized = not minimized
		resizeHandle.Visible = not minimized
		tween(window, MEDIUM_TWEEN, { Size = UDim2.fromOffset(windowWidth, minimized and TITLE_HEIGHT or windowHeight) })
	end)

	closeButton.Activated:Connect(function() Hub:Close() end)

	local function switchTab(name)
		if Hub.cancelListening then Hub.cancelListening() end
		local totalTabs = #Hub.tabList
		for idx, tabName in Hub.tabList do
			local page = Hub.pages[tabName]
			local selected = (tabName == name)
			page.Visible = selected
			tween(Hub.tabButtons[tabName], FAST_TWEEN, { TextColor3 = selected and TEXT_COLOR or DIM_TEXT_COLOR })
			if selected then
				page.GroupTransparency = 1
				tween(page, MEDIUM_TWEEN, { GroupTransparency = 0 })
				tween(indicator, MEDIUM_TWEEN, {
					Size = UDim2.new(1 / totalTabs, -12, 1, -6),
					Position = UDim2.new((idx - 1) / totalTabs, 6, 0, 3)
				})
			end
		end
		Hub.selected = name
	end

	-- Live Search Filter
	searchBox:GetPropertyChangedSignal("Text"):Connect(function()
		local query = string.lower(searchBox.Text)
		for _, element in Hub.searchables do
			element.Visible = (query == "" or string.find(string.lower(element.Name), query, 1, true) ~= nil)
		end
		for _, section in Hub.sections do
			section.Visible = query == ""
		end
		for _, box in Hub.boxes do
			local anyVisible = query == ""
			for _, child in box:GetChildren() do
				if child:IsA("GuiObject") and child.Visible and not table.find(Hub.sections, child) then
					anyVisible = true
					break
				end
			end
			box.Visible = anyVisible
		end
	end)

	-- Right Shift only shows/hides this UI. Draft binds never dispatch actions.
	connectInput(userInputService.InputBegan, function(input, gameProcessed)
		if input.UserInputType ~= Enum.UserInputType.Keyboard then return end
		if Hub.listening and not userInputService:GetFocusedTextBox() then
			return Hub.listening(input.KeyCode)
		end
		if not gameProcessed and input.KeyCode == Hub.menuKey then
			screenGui.Enabled = not screenGui.Enabled
		end
	end)

	-- Row Card Builder with Ripple Effect
	local function createCard(className, parent, text, height)
		local card = create(className, {
			Name = text,
			Size = UDim2.new(1, 0, 0, height),
			BackgroundColor3 = ROW_COLOR,
			BackgroundTransparency = 0.65,
			BorderSizePixel = 0,
		}, parent)

		create("UICorner", { CornerRadius = UDim.new(0, 5) }, card)
		local stroke = addStroke(card, ACCENT_A, 1)
		card.ClipsDescendants = true


		card.InputBegan:Connect(function(input)
			if not isPress(input) then return end
			local offset = Vector2.new(input.Position.X, input.Position.Y) - card.AbsolutePosition
			offset /= windowScale.Scale
			local ripple = create("Frame", {
				AnchorPoint = Vector2.new(0.5, 0.5),
				Position = UDim2.fromOffset(offset.X, offset.Y),
				Size = UDim2.fromOffset(0, 0),
				BackgroundColor3 = Color3.new(1, 1, 1),
				BackgroundTransparency = 0.8,
				BorderSizePixel = 0,
				ZIndex = 0,
			}, card)
			create("UICorner", { CornerRadius = UDim.new(1, 0) }, ripple)
			local diameter = math.max(card.AbsoluteSize.X, card.AbsoluteSize.Y) * 2 / windowScale.Scale
			tween(ripple, MEDIUM_TWEEN, {
				Size = UDim2.fromOffset(diameter, diameter),
				BackgroundTransparency = 1
			}).Completed:Connect(function()
				ripple:Destroy()
			end)
		end)

		card.MouseEnter:Connect(function()
			tween(card, FAST_TWEEN, { BackgroundTransparency = 0.35 })
			tween(stroke, FAST_TWEEN, { Transparency = 0.55 })
		end)
		card.MouseLeave:Connect(function()
			tween(card, FAST_TWEEN, { BackgroundTransparency = 0.65 })
			tween(stroke, FAST_TWEEN, { Transparency = 1 })
		end)

		return card, stroke
	end

	-- Create Tab Method
	function Hub:AddTab(name)
		table.insert(Hub.tabList, name)
		local totalTabs = #Hub.tabList

		local button = create("TextButton", {
			Name = name .. " Tab",
			Size = UDim2.fromScale(1 / totalTabs, 1),
			Position = UDim2.fromScale((totalTabs - 1) / totalTabs, 0),
			BackgroundTransparency = 1,
			Text = name,
			TextColor3 = DIM_TEXT_COLOR,
			TextSize = 13,
			Font = BOLD_FONT,
			AutoButtonColor = false,
		}, tabBar)

		for idx, tName in Hub.tabList do
			local btn = Hub.tabButtons[tName] or button
			btn.Size = UDim2.fromScale(1 / totalTabs, 1)

					if value ~= "" and not keys[value] then return end
					Hub.values[text] = value
					button.Text = value == "" and "..." or value
				end
				Hub.defaults[text], Hub.setters[text] = "", set
				set("")
				button.Activated:Connect(function()
					if Hub.cancelListening then Hub.cancelListening() end
					button.Text = "Press key"
					Hub.cancelListening = function()
						Hub.listening, Hub.cancelListening = nil, nil
						set(Hub.values[text])
					end
					Hub.listening = function(key)
						Hub.cancelListening()
						if key == Enum.KeyCode.Escape then set("")
						elseif key ~= Enum.KeyCode.Unknown and key ~= Hub.menuKey then set(key.Name) end

		skillCheckInputSent = false,

		skillCheckInputWarned = false,

	local playerGui = localPlayer:FindFirstChildOfClass("PlayerGui")
	local previousFolder = workspace:FindFirstChild("ZenithLocalVisuals")
	if previousFolder and previousFolder:GetAttribute("ZenithOwned") == true then
		previousFolder:Destroy()
	end
	local visualFolder = Instance.new("Folder")
	visualFolder.Name = "ZenithLocalVisuals"

	local savedWalkSpeed

	local function track(signal, callback, bucket)
		local connection = signal:Connect(callback)
		table.insert(bucket or runtime.connections, connection)
		return connection
	end

	local function clearConnections(bucket)
		for _, connection in bucket do
			if connection.Connected then connection:Disconnect() end
		end
		table.clear(bucket)
	end

	local function getCharacter()
		local character = localPlayer.Character
		if character and character.Parent then return character end
		return nil
	end

	local function getHumanoid(character)
		character = character or getCharacter()
		return character and character:FindFirstChildOfClass("Humanoid") or nil
	end

	local function getRoot(character)
		character = character or getCharacter()
		local root = character and character:FindFirstChild("HumanoidRootPart")
		return root and root:IsA("BasePart") and root or nil
	end

	local function getInteractable(character)
		return character and character:FindFirstChild("CheckInterractable") or nil
	end

	local function restoreCurrentMovement()
		clearConnections(runtime.characterConnections)
		if currentHumanoid and currentHumanoid.Parent then
			if runtime.walkEnabled and savedWalkSpeed ~= nil then
				currentHumanoid.WalkSpeed = savedWalkSpeed
			end
			if runtime.jumpEnabled and savedJumpPower ~= nil then
				currentHumanoid.JumpPower = savedJumpPower
			end
		end
		currentHumanoid = nil
		savedWalkSpeed, savedJumpPower = nil, nil
		appliedWalkSpeed, appliedJumpPower = nil, nil
	end

	local function attachCharacter(character)
		task.spawn(function()
			local humanoid = character:FindFirstChildOfClass("Humanoid") or character:WaitForChild("Humanoid", 10)
			if not runtime.alive or not humanoid or not humanoid:IsA("Humanoid") or localPlayer.Character ~= character then return end
			restoreCurrentMovement()
			currentHumanoid = humanoid
			savedWalkSpeed = humanoid.WalkSpeed
			savedJumpPower = humanoid.JumpPower
			if runtime.walkEnabled then
				appliedWalkSpeed = runtime.walkValue
				humanoid.WalkSpeed = runtime.walkValue
			end
			if runtime.jumpEnabled then
				appliedJumpPower = runtime.jumpValue
				humanoid.JumpPower = runtime.jumpValue
			end
			track(humanoid:GetPropertyChangedSignal("WalkSpeed"), function()
				if not runtime.alive or currentHumanoid ~= humanoid or not humanoid.Parent then return end
				if runtime.walkEnabled and humanoid.WalkSpeed ~= appliedWalkSpeed then
					savedWalkSpeed = humanoid.WalkSpeed
					appliedWalkSpeed = nil
				elseif not runtime.walkEnabled then
					savedWalkSpeed = humanoid.WalkSpeed
				end
			end, runtime.characterConnections)
			track(humanoid:GetPropertyChangedSignal("JumpPower"), function()
				if not runtime.alive or currentHumanoid ~= humanoid or not humanoid.Parent then return end
				if runtime.jumpEnabled and humanoid.JumpPower ~= appliedJumpPower then
					savedJumpPower = humanoid.JumpPower
					appliedJumpPower = nil
				elseif not runtime.jumpEnabled then
					savedJumpPower = humanoid.JumpPower
				end
			end, runtime.characterConnections)
		end)
	end

	local function canMoveCharacter()
		if not runtime.alive then return nil end
		local character = getCharacter()
		local humanoid = getHumanoid(character)
		local root = getRoot(character)
		if not character or not humanoid or not root or humanoid.Health <= 0 then return nil end
		if localPlayer:GetAttribute("IsDead") == true then return nil end
		if character:GetAttribute("IsCarried") == true or character:GetAttribute("IsHooked") == true then return nil end
		return character, humanoid, root
	end

	local function canRemainSurvivorInteraction(character, humanoid, root)
		local survivorTeam = teamsService:FindFirstChild("Survivors")
		if not survivorTeam or localPlayer.Team ~= survivorTeam then return false end
		if not character or not humanoid or not root or humanoid.Health <= 50 then return false end
		if localPlayer:GetAttribute("IsDead") == true or character:GetAttribute("Trapped") == true
			or character:GetAttribute("IsHooked") == true or character:GetAttribute("IsCarried") == true then
			return false
		end
		local ragdoll = character:FindFirstChild("RagdollTrigger")
		if ragdoll and ragdoll:IsA("BoolValue") and ragdoll.Value then return false end
		return true
	end

	local function canStartSurvivorInteraction(character, humanoid, root)
		if not canRemainSurvivorInteraction(character, humanoid, root)
			or collectionService:HasTag(root, "doing action")
			or runtime.currentRepairPoint or runtime.currentHealPoint or runtime.currentExitPoint or runtime.unhookCharacter then
			return false
		end
		local controller = getInteractable(character)
		if controller then
			for _, attribute in ipairs({ "isVaulting", "isSliding", "isDroppingPallet", "isRepairing", "isHealing", "isUnhooking", "isExiting" }) do
				if controller:GetAttribute(attribute) == true then return false end
			end
		end
		return true
	end

	local function nearestTaggedInteractionPoint(tag, character, root, maxDistance)
		local nearest, nearestDistance
		for _, point in ipairs(collectionService:GetTagged(tag)) do
			if point:IsA("BasePart") and point:IsDescendantOf(workspace)
				and not point:IsDescendantOf(character) and point.Parent
				and not collectionService:HasTag(point.Parent, "Blocked") then
				local distance = (point.Position - root.Position).Magnitude
				if distance <= maxDistance and (not nearestDistance or distance < nearestDistance) then
					nearest, nearestDistance = point, distance
				end
			end
		end
		if not nearest then return nil end
		if nearestDistance < 3.5 then return nearest end
		local rayParams = RaycastParams.new()
		rayParams.FilterType = Enum.RaycastFilterType.Exclude
		rayParams.FilterDescendantsInstances = { character }
		local result = workspace:Raycast(root.Position, nearest.Position - root.Position, rayParams)
		if result and result.Instance == nearest then return nearest end
		return nil
	end

	local function nearestGeneratorPoint(root)
		local map = workspace:FindFirstChild("Map")
		if not map then return nil end
		local nearest, nearestDistance
		for _, point in ipairs(collectionService:GetTagged("GeneratorPoint")) do
			if point:IsA("BasePart") and point:IsDescendantOf(map) then
				local generator = point.Parent
				local blocked = generator and collectionService:HasTag(generator, "Blocked")
				local doingAction = collectionService:HasTag(point, "doing action")
				local progress = generator and generator:GetAttribute("RepairProgress") or 0
				if type(progress) ~= "number" then progress = 0 end
				local distance = (point.Position - root.Position).Magnitude
				if not blocked and not doingAction and progress < 99.95
					and distance <= math.min(runtime.generatorRange, 4)
					and (not nearestDistance or distance < nearestDistance) then
					nearest, nearestDistance = point, distance
				end
			end
		end
		if nearest and nearestDistance >= 3.5 then
			local rayParams = RaycastParams.new()
			rayParams.FilterType = Enum.RaycastFilterType.Exclude
			rayParams.FilterDescendantsInstances = { root.Parent }
			local result = workspace:Raycast(root.Position, nearest.Position - root.Position, rayParams)
			if not result or result.Instance ~= nearest then return nil end
		end
		return nearest
	end

	local function updateAutoGenerator()
		if not runtime.autoGeneratorEnabled then return end
		local character, humanoid, root = canMoveCharacter()
		if not character or not canRemainSurvivorInteraction(character, humanoid, root)
			or collectionService:HasTag(root, "doing action") then
			stopAutoRepair()
			return
		end

		local point = nearestGeneratorPoint(root)
		if point == runtime.currentRepairPoint and runtime.repairCharacter == character then
			return
		end
		stopAutoRepair()
		if not point or not canStartSurvivorInteraction(character, humanoid, root) then return end

		local repairEvent = getRepairEvent()
		if not repairEvent then
			if not runtime.autoGeneratorRemoteWarned then
				runtime.autoGeneratorRemoteWarned = true
				hub:Notify("Auto Generator: RepairEvent is unavailable.")
			end
			return
		end
		local requestId = httpService:GenerateGUID(false)
		runtime.currentRepairPoint = point
		runtime.repairRequestId = requestId
		runtime.repairCharacter = character
		setRepairControllerState(character, requestId, true)
		local ok = pcall(function()
			repairEvent:FireServer(point, true, nil, requestId, character)
		end)
		if not ok then
			setRepairControllerState(character, nil, false)
			runtime.currentRepairPoint = nil
			runtime.repairRequestId = nil
			runtime.repairCharacter = nil
			if not runtime.autoGeneratorRemoteWarned then
				runtime.autoGeneratorRemoteWarned = true
				hub:Notify("Auto Generator: repair request failed.")
			end
		else
			runtime.autoGeneratorRemoteWarned = false
		end
	end

	local function nearestHealTarget(character, root)
		if not character or not root then return nil end
		local candidates, seen = {}, {}
		for _, player in ipairs(playersService:GetPlayers()) do
			if player ~= localPlayer and player.Character and not seen[player.Character] then
				seen[player.Character] = true
				table.insert(candidates, player.Character)
			end
		end
		for _, tag in ipairs({ "VDHealTestDummy", "SurvivorBot" }) do
			for _, model in ipairs(collectionService:GetTagged(tag)) do
				if model:IsA("Model") and model:IsDescendantOf(workspace) and not seen[model] then
					seen[model] = true
					table.insert(candidates, model)
				end
			end
		end

		local nearest, nearestDistance
		for _, targetCharacter in ipairs(candidates) do
			local targetRoot = targetCharacter:FindFirstChild("HumanoidRootPart")
			local targetHumanoid = targetCharacter:FindFirstChildOfClass("Humanoid")
			if targetCharacter ~= character and targetRoot and targetRoot:IsA("BasePart")
				and targetHumanoid and targetHumanoid.Health > 0
				and targetRoot:GetAttribute("CanGetHealed") == true
				and targetCharacter:GetAttribute("IsHooked") ~= true
				and targetCharacter:GetAttribute("IsCarried") ~= true
				and not collectionService:HasTag(targetRoot, "doing action") then
				local distance = (targetRoot.Position - root.Position).Magnitude
				if distance <= 5 and (not nearestDistance or distance < nearestDistance) then
					nearest, nearestDistance = targetRoot, distance
				end
			end
		end
		if not nearest then return nil end
		if nearestDistance < 3.5 then return nearest end

		local targetCharacter = nearest.Parent
		if not targetCharacter then return nil end
		local rayParams = RaycastParams.new()
		rayParams.FilterType = Enum.RaycastFilterType.Exclude
		rayParams.FilterDescendantsInstances = { character }
		local result = workspace:Raycast(root.Position, nearest.Position - root.Position, rayParams)
		if result and result.Instance:IsDescendantOf(targetCharacter) then return nearest end
		return nil
	end

	local function clearAutoUnhookState()
		runtime.unhookToken += 1

		runtime.unhookCharacter = nil

		runtime.unhookNextAt = 0

	local function updateAutoUnhook()
		if not runtime.autoUnhookEnabled then return end
		if os.clock() < runtime.unhookNextAt then return end
		local character, humanoid, root = canMoveCharacter()
		if not character or not canStartSurvivorInteraction(character, humanoid, root) then return end
		local point = nearestTaggedInteractionPoint("UnhookPoint", character, root, 4)
		if not point then return end

		local event = getUnhookEvent()
		if not event then
			if not runtime.autoUnhookRemoteWarned then
				runtime.autoUnhookRemoteWarned = true
				hub:Notify("Auto Rescue: UnHookEvent is unavailable.")
			end
			return
		end
		runtime.unhookToken += 1
		local token = runtime.unhookToken
		runtime.unhookCharacter = character
		runtime.unhookPoint = point
		runtime.unhookNextAt = os.clock() + 3
		setUnhookControllerState(character, true)
		local ok = pcall(function() event:FireServer(point, character) end)
		if not ok then
			setUnhookControllerState(character, false)
			runtime.unhookCharacter = nil
			runtime.unhookPoint = nil
			runtime.unhookToken += 1
			if not runtime.autoUnhookRemoteWarned then
				runtime.autoUnhookRemoteWarned = true
				hub:Notify("Auto Rescue: unhook request failed.")
			end
			return
		end
		runtime.autoUnhookRemoteWarned = false
		task.delay(2, function()
			if runtime.unhookToken ~= token then return end
			setUnhookControllerState(character, false)
			if runtime.unhookCharacter == character then
				runtime.unhookCharacter = nil
				runtime.unhookPoint = nil
			end
		end)
	end

	local function getSkillCheckInputName()
		local gamepadInput = false
		pcall(function()
			gamepadInput = string.find(userInputService:GetLastInputType().Name, "Gamepad", 1, true) == 1
		end)
		local attribute = gamepadInput and "pad_skillcheck" or "kb_skillcheck"
		local inputName = localPlayer:GetAttribute(attribute)
		if type(inputName) == "string" and inputName ~= "" then return inputName end
		return gamepadInput and "ButtonL1" or "Space"
	end

	local function sendSkillCheckInput()
		local virtualInput
		local serviceOk = pcall(function()
			virtualInput = game:GetService("VirtualInputManager")
		end)
		if not serviceOk or not virtualInput then return false end

		local inputName = getSkillCheckInputName()
		local mouseButton = inputName == "MouseButton1" and 0
			or inputName == "MouseButton2" and 1
			or inputName == "MouseButton3" and 2
		if mouseButton ~= nil then
			local camera = workspace.CurrentCamera
			local centre = camera and camera.ViewportSize / 2 or Vector2.new(400, 300)
			local downOk = pcall(function()
				virtualInput:SendMouseButtonEvent(centre.X, centre.Y, mouseButton, true, game, false)
			end)
			if not downOk then return false end
			pcall(function()
				virtualInput:SendMouseButtonEvent(centre.X, centre.Y, mouseButton, false, game, false)
			end)
			return true
		end

		local keyOk, keyCode = pcall(function() return Enum.KeyCode[inputName] end)
		if not keyOk or not keyCode then return false end
		local downOk = pcall(function()
			virtualInput:SendKeyEvent(true, keyCode, false, game)
		end)
		if not downOk then return false end
		pcall(function()
			virtualInput:SendKeyEvent(false, keyCode, false, game)
		end)
		return true
	end

	local function processAutoPerfect()
		if not runtime.autoPerfectEnabled then
			resetAutoPerfect()
			return
		end
		local character = getCharacter()
		local controller = character and character:FindFirstChild("CheckInterractable")
		if not controller or controller:GetAttribute("isRepairing") ~= true then
			resetAutoPerfect()
			return
		end
		local screenGui = playerGui and playerGui:FindFirstChild("SkillCheckPromptGui")
		if not screenGui or not screenGui:IsA("ScreenGui") or not screenGui.Enabled then
			resetAutoPerfect()
			return
		end
		local check = screenGui:FindFirstChild("Check")
		if not check or not check:IsA("GuiObject") or not check.Visible then
			resetAutoPerfect()
			return
		end
		local goal = check:FindFirstChild("Goal")
		local line = check:FindFirstChild("Line")
		if not goal or not goal:IsA("GuiObject") or not goal.Visible
			or not line or not line:IsA("GuiObject") or not line.Visible then
			resetAutoPerfect()
			return
		end

		if not runtime.skillCheckGuiVisible then
			runtime.skillCheckInputSent = false
		end
		runtime.skillCheckGuiVisible = true
		if runtime.skillCheckInputSent then return end

		local signedDifference = (line.Rotation - goal.Rotation + 180) % 360 - 180
		local difference = math.abs(signedDifference)
		local previousDifference = runtime.skillCheckPreviousAngleDifference
		runtime.skillCheckPreviousAngleDifference = signedDifference
		local crossedTarget = previousDifference ~= nil
			and previousDifference * signedDifference < 0
			and math.min(math.abs(previousDifference), difference) <= 20
		if difference > 4 and not crossedTarget then return end

		runtime.skillCheckInputSent = true
		if not sendSkillCheckInput() and not runtime.skillCheckInputWarned then
			runtime.skillCheckInputWarned = true
			hub:Notify("Auto Perfect could not send the configured skill-check input.")
		end
	end

	function runtime:SetAutoPerfectEnabled(enabled)
		if not runtime.alive then return end
		runtime.autoPerfectEnabled = enabled == true
		runtime.skillCheckInputWarned = false
		resetAutoPerfect()
	end

	function runtime:TeleportToTaggedPoint(tag, label)
		local character, _, root = canMoveCharacter()
		if not root then
			hub:Notify("Cannot teleport while character is unavailable.")
			return
		end
		local target, nearestDistance
		for _, instance in ipairs(collectionService:GetTagged(tag)) do
			if instance:IsA("BasePart") and instance:IsDescendantOf(workspace) and not instance:IsDescendantOf(character) then
				local distance = (instance.Position - root.Position).Magnitude
				if not nearestDistance or distance < nearestDistance then
					target, nearestDistance = instance, distance
				end
			end
		end
		if not target then
			hub:Notify(label .. " point not found.")
			return
		end
		local ok = pcall(function() root.CFrame = target.CFrame end)
		if ok then hub:Notify("Moved to nearest " .. label .. ".") else hub:Notify("Could not move to " .. label .. ".") end
	end

	function runtime:TeleportToExitGate()
		local _, _, root = canMoveCharacter()
		if not root then
			hub:Notify("Cannot teleport while character is unavailable.")
			return
		end
		local map = workspace:FindFirstChild("Map")
		local gate = map and map:FindFirstChild("Gate")
		local lever = gate and gate:FindFirstChild("ExitLever")
		local target = lever and lever:FindFirstChild("Tp")
		if not target or not target:IsA("BasePart") then
			hub:Notify("The verified ExitLever.Tp target is unavailable.")
			return
		end
		local ok = pcall(function() root.CFrame = target.CFrame end)
		if ok then hub:Notify("Moved to the exit lever marker.") else hub:Notify("Could not move to the exit lever marker.") end
	end

	function runtime:VaultNearestWindow()
		local character, humanoid, root = canMoveCharacter()
		if not root then
			hub:Notify("Character is unavailable.")
			return
		end
		if humanoid.Health <= 50 then
			hub:Notify("Vault interaction requires health above 50 in the source flow.")
			return
		end
		local ragdoll = character:FindFirstChild("RagdollTrigger")
		if ragdoll and ragdoll:IsA("BoolValue") and ragdoll.Value then
			hub:Notify("Cannot vault while ragdolled.")
			return
		end
		if collectionService:HasTag(root, "doing action") then
			hub:Notify("Finish the current action first.")
			return
		end
		local interactable = character:FindFirstChild("CheckInterractable")
		if interactable and interactable:GetAttribute("isVaulting") == true then
			hub:Notify("A vault is already in progress.")
			return
		end
		local nearest, nearestDistance
		for _, instance in ipairs(collectionService:GetTagged("VaultPoint")) do
			if instance:IsA("BasePart") and instance:IsDescendantOf(workspace) and not instance:IsDescendantOf(character)
				and instance.Parent and not collectionService:HasTag(instance.Parent, "Blocked") then
				local distance = (instance.Position - root.Position).Magnitude
				if distance <= 4 and (not nearestDistance or distance < nearestDistance) then
					nearest, nearestDistance = instance, distance
				end
			end
		end
		if not nearest then
			hub:Notify("No VaultPoint is within the verified 4-stud radius.")
			return
		end
		if nearestDistance >= 3.5 then
			local rayParams = RaycastParams.new()
			rayParams.FilterType = Enum.RaycastFilterType.Exclude
			rayParams.FilterDescendantsInstances = { character }
			local direction = nearest.Position - root.Position
			local result = workspace:Raycast(root.Position, direction, rayParams)
			if not result or result.Instance ~= nearest then
				hub:Notify("VaultPoint is not visible from the character.")
				return
			end
		end
		local remotes = replicatedStorage:FindFirstChild("Remotes")
		local window = remotes and remotes:FindFirstChild("Window")
		local vaultEvent = window and window:FindFirstChild("VaultEvent")
		local vaultBindable = window and window:FindFirstChild("Vaultbindable")
		if not vaultEvent or not vaultEvent:IsA("RemoteEvent") or not vaultBindable or not vaultBindable:IsA("BindableEvent") then
			hub:Notify("Verified vault event is unavailable.")
			return
		end
		local sprinting = character:GetAttribute("Sprinting") == true
		local ok, err = pcall(function()
			vaultEvent:FireServer(nearest, sprinting)
			vaultBindable:Fire(nearest, sprinting)
		end)
		if ok then
			hub:Notify("Vault request sent through the verified survivor flow.")
		else
			warn("Zenith: vault request failed:", err)
			hub:Notify("Vault request failed.")
		end
	end

	local function makeDrawing(adornee, anchor, color, labelText, showLabel)
		local highlight = Instance.new("Highlight")
		highlight.Name = "ZenithHighlight"
		highlight.Adornee = adornee
		highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
		highlight.FillColor = color
		highlight.FillTransparency = 1
		highlight.OutlineColor = color
		highlight.OutlineTransparency = 0.12
		highlight.Parent = visualFolder

		local billboard = Instance.new("BillboardGui")
		billboard.Name = "ZenithLabel"
		billboard.Adornee = anchor
		billboard.Size = UDim2.fromOffset(180, 22)
		billboard.StudsOffset = Vector3.new(0, 2.5, 0)
		billboard.AlwaysOnTop = true
		billboard.Enabled = showLabel
		billboard.Parent = visualFolder

		local label = Instance.new("TextLabel")
		label.Name = "Text"
		label.Size = UDim2.fromScale(1, 1)
		label.BackgroundTransparency = 1
		label.Text = labelText
		label.TextColor3 = color
		label.TextStrokeColor3 = Color3.new(0, 0, 0)
		label.TextStrokeTransparency = 0.25
		label.TextSize = 12
		label.Font = BOLD_FONT
		label.TextTruncate = Enum.TextTruncate.AtEnd
		label.Parent = billboard
		return { highlight = highlight, billboard = billboard, label = label }
	end

	local function destroyDrawing(key)
		local drawing = drawings[key]
		if drawing then
			if drawing.highlight then drawing.highlight:Destroy() end
			if drawing.billboard then drawing.billboard:Destroy() end
			drawings[key] = nil
		end
	end

	local function syncVisuals()
		if not runtime.alive or not visualFolder.Parent then return end
		local root = getRoot()
		if not root then
			for key in pairs(drawings) do destroyDrawing(key) end
			return
		end
		local origin = root.Position
		local maxDistance = runtime.visuals.maxDistance
		local desired = {}

		local function upsert(key, adornee, anchor, distance, color, text, showLabel)
			if not adornee or not adornee.Parent or not anchor or not anchor.Parent or distance > maxDistance then return end
			desired[key] = true
			local drawing = drawings[key]
			if not drawing or not drawing.highlight.Parent or not drawing.billboard.Parent then
				destroyDrawing(key)
				drawing = makeDrawing(adornee, anchor, color, text, showLabel)
				drawings[key] = drawing
			else
				drawing.highlight.Adornee = adornee
				drawing.highlight.OutlineColor = color
				drawing.highlight.FillColor = color
				drawing.billboard.Adornee = anchor
				drawing.billboard.Enabled = showLabel
				drawing.label.Text = text
				drawing.label.TextColor3 = color
			end
		end

		local function distanceTo(part)
			return (part.Position - origin).Magnitude
		end

		local survivorsTeam = teamsService:FindFirstChild("Survivors")
		local killerTeam = teamsService:FindFirstChild("Killer")
		for _, player in ipairs(playersService:GetPlayers()) do
			if player ~= localPlayer then
				local role, color, enabled
				if survivorsTeam and player.Team == survivorsTeam then
					role, color, enabled = "Survivor", runtime.visuals.survivorColor, runtime.visuals.survivor
			elseif killerTeam and player.Team == killerTeam then
					role, color, enabled = "Killer", runtime.visuals.killerColor, runtime.visuals.killer
				end
				local character = player.Character
				local humanoid = character and character:FindFirstChildOfClass("Humanoid")
				local playerRoot = character and character:FindFirstChild("HumanoidRootPart")
				if enabled and role and character and humanoid and humanoid.Health > 0 and player:GetAttribute("IsDead") ~= true
					and playerRoot and playerRoot:IsA("BasePart") then
					local distance = distanceTo(playerRoot)
					local labelParts = {}
					if runtime.visuals.names then table.insert(labelParts, player.DisplayName) end
					if runtime.visuals.distance then table.insert(labelParts, string.format("%d studs", math.floor(distance + 0.5))) end
					upsert(character, character, playerRoot, distance, color, table.concat(labelParts, " · "), #labelParts > 0)
				end
			end
		end

		local nearestHook, nearestHookDistance
		if runtime.visuals.hook and runtime.visuals.nearestHook then
			for _, instance in ipairs(collectionService:GetTagged("HookPoint")) do
				if instance:IsA("BasePart") and instance:IsDescendantOf(workspace) then
					local distance = distanceTo(instance)
					if distance <= maxDistance and (not nearestHookDistance or distance < nearestHookDistance) then
						nearestHook, nearestHookDistance = instance, distance
					end
				end
			end
		end

		local objectDefinitions = {
			{ tag = "GeneratorPoint", enabled = runtime.visuals.generator, name = "Generator", color = Color3.fromRGB(80, 190, 255) },
			{ tag = "HookPoint", enabled = runtime.visuals.hook, name = "Hook", color = Color3.fromRGB(255, 120, 120) },
			{ tag = "UnhookPoint", enabled = runtime.visuals.unhook, name = "Rescue point", color = Color3.fromRGB(110, 235, 150) },
			{ tag = "PalletPoint", enabled = runtime.visuals.pallet, name = "Pallet", color = Color3.fromRGB(245, 190, 80) },
			{ tag = "PalletPointSlide", enabled = runtime.visuals.palletSlide, name = "Pallet slide", color = Color3.fromRGB(255, 145, 70) },
			{ tag = "VaultPoint", enabled = runtime.visuals.window, name = "Vault / window", color = runtime.visuals.windowColor },
			{ tag = "ExitPoint", enabled = runtime.visuals.exitPoint, name = "Exit interaction", color = Color3.fromRGB(120, 230, 150) },
		}
		for _, definition in ipairs(objectDefinitions) do
			if definition.enabled then
				local candidates = {}
				for _, instance in ipairs(collectionService:GetTagged(definition.tag)) do
					if instance:IsA("BasePart") and instance:IsDescendantOf(workspace) then
						local nearestOnlyPass = definition.tag ~= "HookPoint" or not runtime.visuals.nearestHook or instance == nearestHook
					if nearestOnlyPass then
						local adornee = instance:FindFirstAncestorWhichIsA("Model") or instance
						local anchor = instance
						if definition.tag == "GeneratorPoint" and adornee:IsA("Model") then
							local generatorBody = adornee:FindFirstChild("GeneratorBody")
						if generatorBody and generatorBody:IsA("BasePart") then anchor = generatorBody end
					end
						local distance = distanceTo(anchor)
						local candidate = candidates[adornee]
						if not candidate or distance < candidate.distance then
							candidates[adornee] = { anchor = anchor, distance = distance }
						end
					end
				end
				end
				for adornee, candidate in pairs(candidates) do
					local distance = candidate.distance
					upsert(adornee, adornee, candidate.anchor, distance, definition.color,
						string.format("%s · %d studs", definition.name, math.floor(distance + 0.5)), true)
				end
			end
		end

		if runtime.visuals.exitGate then
			local map = workspace:FindFirstChild("Map")
			local gate = map and map:FindFirstChild("Gate")
			local lever = gate and gate:FindFirstChild("ExitLever")
			local anchor = lever and (lever:FindFirstChild("Tp") or lever:FindFirstChildWhichIsA("BasePart", true))
			if lever and anchor and anchor:IsA("BasePart") then
				local distance = distanceTo(anchor)
				upsert(lever, lever, anchor, distance, Color3.fromRGB(120, 230, 150),
					string.format("Exit lever · %d studs", math.floor(distance + 0.5)), true)
			end
		end

		for key in pairs(drawings) do
			if not desired[key] then destroyDrawing(key) end
		end
	end

	local visualElapsed = 0
	track(runService.Heartbeat, function(deltaTime)
		visualElapsed += deltaTime
		if visualElapsed >= 0.25 then
			visualElapsed = 0
			syncVisuals()
		end
	end)
	track(runService.Heartbeat, function(deltaTime)
		if runtime.autoGeneratorEnabled then
			runtime.generatorScanElapsed += deltaTime
			if runtime.generatorScanElapsed >= 0.2 then
				runtime.generatorScanElapsed = 0
				updateAutoGenerator()
			end
		end
		if runtime.autoHealEnabled then
			runtime.healScanElapsed += deltaTime
			if runtime.healScanElapsed >= 0.2 then
				runtime.healScanElapsed = 0
				updateAutoHeal()
			end
		end
		if runtime.autoExitEnabled then
			runtime.exitScanElapsed += deltaTime
			if runtime.exitScanElapsed >= 0.2 then
				runtime.exitScanElapsed = 0
				updateAutoExit()
			end
		end
		if runtime.autoUnhookEnabled then
			runtime.unhookScanElapsed += deltaTime
			if runtime.unhookScanElapsed >= 0.2 then
				runtime.unhookScanElapsed = 0
				updateAutoUnhook()
			end
		end
	end)
	track(runService.RenderStepped, processAutoPerfect)
	track(localPlayer.CharacterAdded, function(character)
		stopAutoRepair()
		stopAutoHeal()
		stopAutoExit()
		clearAutoUnhookState()
		attachCharacter(character)
	end)
	if localPlayer.Character then attachCharacter(localPlayer.Character) end

	function runtime:Cleanup()
		if not runtime.alive then return end
		stopAutoRepair()
		stopAutoHeal()
		stopAutoExit()
		clearAutoUnhookState()
		runtime.alive = false
		clearConnections(runtime.connections)
		restoreCurrentMovement()
		for key in pairs(drawings) do destroyDrawing(key) end
		if visualFolder.Parent then visualFolder:Destroy() end
	end
	hub:AddCleanup(function() runtime:Cleanup() end)
	return runtime
end

-- Profiles are session-local. Settings are stored in an owned PlayerGui folder.
local function loadProfileStore(parent)
	local root = parent:FindFirstChild("ZenithPreviewProfiles")
	if not root or not root:IsA("Folder") or root:GetAttribute("ZenithOwned") ~= true then
		root = create("Folder", { Name = "ZenithPreviewProfiles" }, parent)
		root:SetAttribute("ZenithOwned", true)
	end
	local profiles, autoload = {}, nil
	local raw = root:GetAttribute("ProfilesV1")
	if typeof(raw) == "string" then
		local ok, data = pcall(function() return httpService:JSONDecode(raw) end)
		if ok and typeof(data) == "table" and typeof(data.profiles) == "table" then
			for name, values in pairs(data.profiles) do
				if typeof(name) == "string" and typeof(values) == "table" then profiles[name] = values end
			end
			if typeof(data.autoload) == "string" and profiles[data.autoload] then autoload = data.autoload end
		end
	end
	return profiles, autoload, function(newAutoload)
		root:SetAttribute("ProfilesV1", httpService:JSONEncode({ profiles = profiles, autoload = newAutoload }))
	end
end

function Pages.configs(tab, hub, parent)
	local config = tab:AddSection("Config", "left")
	local interface = tab:AddSection("Interface", "right")
	local profiles, autoload, save = loadProfileStore(parent)
	local selected
	local nameFlag = "Config Name"
	local updateSelection
	local setAutoloadText
	local setStatus

	local function apply(values)
		for flag, default in pairs(hub.defaults) do
			local value = values[flag]
			if flag ~= nameFlag and typeof(value) == typeof(default) then hub.setters[flag](value) end
		end
	end
	local function snapshot()
		local values = table.clone(hub.values)
		values[nameFlag] = nil
		return values
	end
	local function selectProfile(name)
		selected = name
		updateSelection(name and ("Select Config: " .. name) or "Select Config...")
	end
	local function cycleProfile()
		local names = {}
		for name in pairs(profiles) do table.insert(names, name) end
		table.sort(names)
		if #names == 0 then return setStatus("No session profiles yet.") end
		local position = table.find(names, selected) or 0
		selectProfile(names[position % #names + 1])
	end

	config:AddTextbox(nameFlag, "Name", "")
	local selectionButton = config:AddButton("Select Config...", cycleProfile)
	updateSelection = function(value) selectionButton.Text = value end
	config:AddButton("Create Config", function()
		local name = (hub.values[nameFlag] or ""):match("^%s*(.-)%s*$")
		if not name:match("^[%w _%-]+$") or #name > 24 then
			return setStatus("Use 1–24 letters, digits, spaces, dashes or underscores.")
		end
		if profiles[name] then return setStatus("That profile exists. Select it to overwrite it.") end
		local count = 0
		for _ in pairs(profiles) do count += 1 end
		if count >= 8 then return setStatus("Session limit: eight profiles.") end
		profiles[name] = snapshot()
		save(autoload)
		selectProfile(name)
		setStatus("Created " .. name .. ".")
	end)
	config:AddButton("Load Config", function()
		if not selected then return setStatus("Select a profile first.") end
		apply(profiles[selected])
		setStatus("Loaded " .. selected .. ".")
	end)
	config:AddButton("Overwrite Config", function()
		if not selected then return setStatus("Select a profile first.") end
		profiles[selected] = snapshot()
		save(autoload)
		setStatus("Overwrote " .. selected .. ".")
	end)
	config:AddButton("Refresh Config List", function()
		profiles, autoload, save = loadProfileStore(parent)
		selectProfile(profiles[selected] and selected or nil)
		setAutoloadText("Autoload config: " .. (autoload or "None"))
		setStatus("Session profiles refreshed.")
	end)
	config:AddButton("Set as autoload", function()
		if not selected then return setStatus("Select a profile first.") end
		autoload = selected
		save(autoload)
		setAutoloadText("Autoload config: " .. autoload)
		setStatus("Will load the next time Zenith opens in this session.")
	end)
	config:AddButton("Remove autoload", function()
		autoload = nil
		save(nil)
		setAutoloadText("Autoload config: None")
		setStatus("Autoload cleared.")
	end)
	setAutoloadText = config:AddNote("Autoload config: " .. (autoload or "None"), 28)
	setStatus = config:AddNote("Profiles are stored for this client session only.", 36)
	interface:AddSlider("UI Size", 0.75, 1.5, 1.08, 2, function(value) hub:SetSize(value) end)
	interface:AddSlider("GUI Transparency", 0, 80, 5, 0, function(value) hub:SetTransparency(value) end, nil, "%")
	interface:AddButton("Unload", function() hub:Close() end)
	interface:AddNote("Source-verified controls only. Right Shift shows or hides Zenith; unsupported mechanics are documented on Main and Extra.", 48)
	hub:SetTransparency(5)
	if autoload then
		selectProfile(autoload)
		apply(profiles[autoload])
		setStatus("Loaded " .. autoload .. ".")
	end
end

if not game:IsLoaded() then game.Loaded:Wait() end
local client = playersService.LocalPlayer

if not parent or not parent:IsA("PlayerGui") then
	warn("Zenith: PlayerGui is unavailable.")
	return
end
local hub = ZenithUI.create(parent)
local features = createGameFeatures(hub, client)
Pages.main(hub:AddTab("Main"), features)
Pages.extra(hub:AddTab("Extra"), features)
Pages.visuals(hub:AddTab("Visuals"), features)
Pages.configs(hub:AddTab("Configs"), hub, parent)
