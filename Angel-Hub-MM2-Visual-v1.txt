local function make(class, props, parent)
    local o = Instance.new(class)
    for k, v in pairs(props or {}) do o[k] = v end
    if parent then o.Parent = parent end
    return o
end
local function corner(o, radius)
    make("UICorner", {CornerRadius = UDim.new(0, radius or 5)}, o)
end
local function stroke(o, color)
    make("UIStroke", {Color = color or C.border, Thickness = 1}, o)
end
local function text(o, size, color)
    o.Font = Enum.Font.Gotham
    o.TextSize = size or 11
    o.TextColor3 = color or C.text
    o.TextXAlignment = Enum.TextXAlignment.Left
    o.TextTruncate = Enum.TextTruncate.AtEnd
end
local function padding(o, left, right, top, bottom)
    make("UIPadding", {PaddingLeft = UDim.new(0, left or 0), PaddingRight = UDim.new(0, right or 0),
        PaddingTop = UDim.new(0, top or 0), PaddingBottom = UDim.new(0, bottom or 0)}, o)
end
local function listLayout(o, gap)
    return make("UIListLayout", {SortOrder = Enum.SortOrder.LayoutOrder, Padding = UDim.new(0, gap or 0)}, o)
end

local function fitWindow()
    local cam = workspace.CurrentCamera
    if not cam then return end
    local v = cam.ViewportSize
    sizeConstraint.MinSize = minimized and Vector2.new(1, 1) or Vector2.new(math.min(280, math.max(1, v.X - 16)), math.min(230, math.max(1, v.Y - 16)))
    window.Size = UDim2.fromOffset(math.min(690, math.max(1, v.X - 16)), minimized and 37 or math.min(590, math.max(1, v.Y - 16)))
end

    if cameraConn then cameraConn:Disconnect(); cameraConn = nil end
    local cam = workspace.CurrentCamera
    if cam then cameraConn = on(cam:GetPropertyChangedSignal("ViewportSize"), fitWindow) end
    fitWindow()
end

local function selectTab(name)
    activeTab = name
    for n, t in pairs(tabs) do
        t.page.Visible = n == name and not minimized
        t.button.BackgroundTransparency = n == name and 0 or 1
        t.button.TextColor3 = n == name and C.text or C.muted
    end
end

local function detach()
    if detached then return end
    detached = true
    draggingWindow, draggingSlider, captureKey = nil, nil, nil
    for _, c in ipairs(connections) do pcall(function() c:Disconnect() end) end
    table.clear(connections)
    gui:Destroy()
end
gui.Destroying:Connect(function()
    if not detached then
        detached = true
        for _, c in ipairs(connections) do pcall(function() c:Disconnect() end) end
        table.clear(connections)
    end
end)
on(close.MouseButton1Click, detach)

    minimized = not minimized

    minimize.Text = minimized and "+" or "−"
    if minimized then
        sizeConstraint.MinSize = Vector2.new(1, 1)
        window.Size = UDim2.fromOffset(window.AbsoluteSize.X, 37)
    else fitWindow(); selectTab(activeTab) end
end)

local function newCard(tab, side, titleText, opts)
    opts = opts or {}
    local holder = side == "left" and tabs[tab].left or tabs[tab].right
    local frame = make("Frame", {Name = titleText, AutomaticSize = Enum.AutomaticSize.Y,
        Size = UDim2.new(1, -2, 0, 0), BackgroundColor3 = C.panel, BorderSizePixel = 0}, holder)
    corner(frame, 8); stroke(frame); padding(frame, 7, 7, 7, 7); listLayout(frame, 4)
    local heading = make("Frame", {BackgroundTransparency = 1, Size = UDim2.new(1, 0, 0, 25), LayoutOrder = 0}, frame)
    local accent = make("Frame", {BackgroundColor3 = C.muted, BorderSizePixel = 0, Position = UDim2.fromOffset(0, 4), Size = UDim2.fromOffset(2, 14)}, heading)
    corner(accent, 2)
    local label = make("TextLabel", {Text = string.upper(titleText), BackgroundTransparency = 1,
        Position = UDim2.fromOffset(8, 0), Size = UDim2.new(1, opts.toggle and -48 or -8, 1, 0)}, heading)
    text(label, 10); label.Font = Enum.Font.GothamBold
    local card = {frame = frame, title = titleText, rows = {}, mm2 = opts.mm2 == true, toggle = nil}
    table.insert(cards, card)
    if opts.toggle then
        local enabled = false
        local b = make("TextButton", {Text = "", BackgroundColor3 = C.row, BorderSizePixel = 0,
            Position = UDim2.new(1, -27, 0, 4), Size = UDim2.fromOffset(25, 15)}, heading)
        corner(b, 8); stroke(b)
        local dot = make("Frame", {BackgroundColor3 = C.muted, BorderSizePixel = 0,
            Position = UDim2.fromOffset(2, 2), Size = UDim2.fromOffset(11, 11)}, b); corner(dot, 6)
        on(b.MouseButton1Click, function()
            enabled = not enabled
            b.BackgroundColor3 = enabled and Color3.fromRGB(94,94,101) or C.row
            dot.Position = UDim2.fromOffset(enabled and 12 or 2, 2)
            dot.BackgroundColor3 = enabled and C.text or C.muted
        end)
    end
    return card
end
local function newRow(card, name, height, condition)
    local row = make("Frame", {BackgroundColor3 = C.row, BorderSizePixel = 0,
        Size = UDim2.new(1, 0, 0, height or 29), LayoutOrder = #card.rows + 1}, card.frame)
    corner(row, 4)
    local entry = {frame = row, name = name, condition = condition}
    table.insert(card.rows, entry)
    return row
end

        Size = UDim2.new(width or 0.57, -8, 0, 28)}, row)
    text(t, 10); return t
end

    id = id or (card.title .. "." .. name); values[id] = default == true
    local row = newRow(card, name, 28, condition); labelIn(row, name, 0.77)

        Position = UDim2.new(1, -25, 0, 5), Size = UDim2.fromOffset(17, 17)}, row)

    id = id or card.title .. "." .. name; values[id] = default or ""

end

local function slider(card, name, id, min, max, default, step, condition)
    id = id or card.title .. "." .. name; step = step or 1; values[id] = default
    local row = newRow(card, name, 46, condition); labelIn(row, name, 0.65)
    local input = make("TextBox", {Text = tostring(default), ClearTextOnFocus = false, BackgroundTransparency = 1,
        Position = UDim2.new(0.65,0,0,2), Size = UDim2.new(0.35,-7,0,23)}, row)
    text(input, 10); input.TextXAlignment = Enum.TextXAlignment.Right
    local bar = make("TextButton", {Text = "", BackgroundColor3 = C.bg, BorderSizePixel = 0,
        Position = UDim2.new(0, 8, 0, 32), Size = UDim2.new(1, -16, 0, 7)}, row); corner(bar, 4); stroke(bar)
    local fill = make("Frame", {BackgroundColor3 = Color3.fromRGB(175,175,180), BorderSizePixel = 0, Size = UDim2.new()}, bar); corner(fill,4)
    local knob = make("Frame", {AnchorPoint = Vector2.new(0.5, 0.5), BackgroundColor3 = C.text,
        BorderSizePixel = 0, Position = UDim2.fromScale(0,0.5), Size = UDim2.fromOffset(12,12)}, bar); corner(knob, 6)
    local function apply(v)
        v = math.clamp(tonumber(v) or get(id), min, max)
        v = math.clamp(min + math.floor((v-min)/step + 0.5)*step, min, max)
        v = math.floor(v * 1000 + 0.5) / 1000
        values[id] = v; input.Text = tostring(v)
        local pct = (v-min)/(max-min); fill.Size = UDim2.fromScale(pct,1); knob.Position = UDim2.fromScale(pct,0.5)
    end
    local function move(x) if bar.AbsoluteSize.X > 0 then apply(min + math.clamp((x-bar.AbsolutePosition.X)/bar.AbsoluteSize.X,0,1)*(max-min)) end end
    on(bar.InputBegan, function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            draggingSlider = {input = i, move = move}; move(i.Position.X)
        end
    end)
    on(input.FocusLost, function() apply(input.Text) end); apply(default)
    return apply
end
local function dropdown(card, name, id, choices, default, condition, afterChange)
    id = id or card.title .. "." .. name; values[id] = default or choices[1]
    local row = newRow(card, name, 31, condition); row.AutomaticSize = Enum.AutomaticSize.Y
    labelIn(row, name, 0.48)
    local b = button(row, tostring(values[id]) .. "  ▾"); b.Position = UDim2.new(0.48,0,0,4); b.Size = UDim2.new(0.52,-6,0,22); b.TextSize = 10
    local options = make("Frame", {BackgroundTransparency = 1, Visible = false, Position = UDim2.fromOffset(5,31),
        Size = UDim2.new(1,-10,0,0), AutomaticSize = Enum.AutomaticSize.Y}, row); listLayout(options, 2)
    for _, choice in ipairs(choices) do
        local item = button(options, choice); item.Size = UDim2.new(1,0,0,23); item.TextSize = 10
        on(item.MouseButton1Click, function()
            set(id, choice); b.Text = tostring(choice) .. "  ▾"; options.Visible = false
            if afterChange then afterChange(choice) end
            if refresh then refresh() end
        end)
    end
    on(b.MouseButton1Click, function() options.Visible = not options.Visible end)
    return function(value) values[id] = value; b.Text = tostring(value) .. "  ▾"; options.Visible = false end
end

local function keybind(card, name, id, default, condition, changed)
    if MOBILE then return end
    id = id or card.title .. "." .. name; values[id] = default or Enum.KeyCode.Unknown
    local row = newRow(card, name, 29, condition); labelIn(row, name, 0.55)
    local b = button(row, values[id] == Enum.KeyCode.Unknown and "None" or values[id].Name)
    b.Position = UDim2.new(0.55,0,0,4); b.Size = UDim2.new(0.45,-6,0,21); b.TextSize = 10
    on(b.MouseButton1Click, function()
        if captureKey and captureKey.button then captureKey.button.Text = captureKey.oldText end
        captureKey = {button = b, oldText = b.Text, id = id, changed = changed}; b.Text = "Press key..."
    end)
end
local function colour(card, name, id, default, condition)
    id = id or card.title .. "." .. name; values[id] = default or "FFFFFF"
    local row = newRow(card, name, 29, condition); labelIn(row, name, 0.5)
    local box = make("TextBox", {Text = "#"..values[id], ClearTextOnFocus = false, BackgroundColor3 = C.bg,
        BorderSizePixel = 0, Position = UDim2.new(0.5,0,0,4), Size = UDim2.new(0.5,-6,0,21)}, row)
    text(box, 10); box.TextXAlignment = Enum.TextXAlignment.Center; corner(box,4); stroke(box)
    local swatch = make("Frame", {BackgroundColor3 = C.accent, BorderSizePixel = 0,
        Position = UDim2.new(0,3,0,3), Size = UDim2.fromOffset(15,15)}, box); corner(swatch,3)
    local function display()
        local h = values[id]
        box.Text = "#" .. h
        swatch.BackgroundColor3 = Color3.fromRGB(tonumber(string.sub(h,1,2),16), tonumber(string.sub(h,3,4),16), tonumber(string.sub(h,5,6),16))
    end
    on(box.FocusLost, function()
        local h = string.upper(string.gsub(box.Text, "#", ""))
        if string.match(h, "^%x%x%x%x%x%x$") then values[id] = h end
        display()
    end)
    display()
end

local function yes(id) return function() return get(id) == true end end
local function equals(id, ...)
    local opts = {...}
    return function() for _, v in ipairs(opts) do if get(id) == v then return true end end return false end
end
local function both(a,b) return function() return a() and b() end end

refresh = function()
    if detached then return end
    for _, card in ipairs(cards) do
        local allowed = not card.mm2 or IS_MM2
        local titleMatches = matchSearch(card.title)
        local any = false
        for _, row in ipairs(card.rows) do
            local conditional = not row.condition or row.condition()
            local visible = allowed and conditional and (searchText == "" or titleMatches or matchSearch(row.name))
            row.frame.Visible = visible
            if visible then any = true end
        end
        card.frame.Visible = allowed and (searchText == "" or titleMatches or any)
    end
end
on(search:GetPropertyChangedSignal("Text"),function()
    searchText = string.lower(search.Text)
    if menuSearchBox and menuSearchBox.Text ~= search.Text then menuSearchBox.Text = search.Text end
    refresh()
end)
on(UIS.InputBegan,function(input, processed)
    if captureKey and input.UserInputType == Enum.UserInputType.Keyboard then
        local c = captureKey; captureKey = nil
        local key = input.KeyCode == Enum.KeyCode.Escape and Enum.KeyCode.Unknown or input.KeyCode
        values[c.id] = key; c.button.Text = key == Enum.KeyCode.Unknown and "None" or key.Name
        if c.changed then c.changed(key) end
        return
    end
    if not MOBILE and not processed and not UIS:GetFocusedTextBox() and input.UserInputType == Enum.UserInputType.Keyboard and input.KeyCode == menuKey and menuKey ~= Enum.KeyCode.Unknown then
        window.Visible = not window.Visible
    end
end)
on(UIS.InputChanged,function(input)
    local mouseMove = input.UserInputType == Enum.UserInputType.MouseMovement
    if draggingSlider and (input == draggingSlider.input or (mouseMove and draggingSlider.input.UserInputType == Enum.UserInputType.MouseButton1)) then
        draggingSlider.move(input.Position.X)
    end
    if draggingWindow and (input == draggingWindow.input or (mouseMove and draggingWindow.input.UserInputType == Enum.UserInputType.MouseButton1)) then
        local delta = input.Position - draggingWindow.origin
        window.Position = draggingWindow.position + UDim2.fromOffset(delta.X,delta.Y)
    end
end)
on(UIS.InputEnded,function(input)
    if draggingSlider and (input == draggingSlider.input or input.UserInputType == Enum.UserInputType.MouseButton1) then draggingSlider = nil end
    if draggingWindow and (input == draggingWindow.input or input.UserInputType == Enum.UserInputType.MouseButton1) then draggingWindow = nil end
end)
refresh()
