-- Violent | standalone native Luau menu, with a transparent background.
-- Controls only edit menu state. No game scene, player overlays, or gameplay logic.

local MENU_WIDTH, MENU_HEIGHT = 308, 513
local EDITOR_WIDTH, EDITOR_GAP = 228, 12
local RGB = Color3.fromRGB
local WHITE = RGB(250, 250, 250)
local INK = RGB(9, 8, 9)
local PINK = RGB(255, 15, 143)
local SELECTED = RGB(255, 85, 176)
local BODY_FONT = Enum.Font.Gotham
local DISPLAY_FONT = Enum.Font.FredokaOne

local function create(class, parent, properties)
    local object = Instance.new(class)
    for key, value in pairs(properties) do object[key] = value end
    object.Parent = parent
    return object
end

local function rect(parent, name, x, y, w, h, color, alpha)
    return create("Frame", parent, {
        Name = name, Position = UDim2.fromOffset(x, y), Size = UDim2.fromOffset(w, h),
        BackgroundColor3 = color or INK, BackgroundTransparency = alpha or (color and 0 or 1),
        BorderSizePixel = 0,
    })
end

local function label(parent, name, text, x, y, w, h, size, options)
    options = options or {}
    return create("TextLabel", parent, {
        Name = name, Text = text, Position = UDim2.fromOffset(x, y), Size = UDim2.fromOffset(w, h),
        BackgroundTransparency = 1, BorderSizePixel = 0, Font = options.font or BODY_FONT,
        TextSize = size, TextColor3 = options.color or WHITE,
        TextStrokeTransparency = options.outline and 0 or 1,
        TextXAlignment = options.align or Enum.TextXAlignment.Left,
        TextYAlignment = Enum.TextYAlignment.Center,
    })
end

local function stroke(parent, color, thickness)
    return create("UIStroke", parent, { Color = color, Thickness = thickness, ApplyStrokeMode = Enum.ApplyStrokeMode.Border })
end

local function line(parent, name, x1, y1, x2, y2, color, width, alpha)
    local dx, dy = x2 - x1, y2 - y1
    local frame = rect(parent, name, (x1+x2)/2, (y1+y2)/2, math.sqrt(dx*dx + dy*dy), width, color, alpha)
    frame.AnchorPoint = Vector2.new(0.5, 0.5)
    frame.Rotation = math.deg(math.atan2(dy, dx))
    return frame
end

local function chain(parent, name, points, width, color)
    for i = 1, #points - 2, 2 do
        line(parent, name, points[i], points[i+1], points[i+2], points[i+3], color, width)
    end
end

local function target(parent, name, x, y, w, h, callback)
    local button = create("TextButton", parent, {
        Name = name, Position = UDim2.fromOffset(x, y), Size = UDim2.fromOffset(w, h),
        BackgroundTransparency = 1, Text = "", BorderSizePixel = 0,
        AutoButtonColor = false, Selectable = true, ZIndex = 2,
    })
    button.Activated:Connect(callback)
    return button
end

local function logo(parent, text, x, y, w, h, size, centered)
    return label(parent, text, text, x, y, w, h, size, {
        font = DISPLAY_FONT, outline = true, align = centered and Enum.TextXAlignment.Center or Enum.TextXAlignment.Left,
    })
end

local rows = {
    {id="skeleton",       text="Enable Skeleton",         y=106, kind="check", default=true},
    {id="boneWidth",      text="Skeleton Thickness",      y=132, kind="value", default=1, min=0.5, max=5, step=0.5},
    {id="boneColor",      text="Skeleton Color",          y=159, kind="color", default=RGB(255,255,255)},
    {id="highlight",      text="Hightlight ESP",          y=213, kind="check", default=true},
    {id="fill",           text="Hightlight Fill",         y=240, kind="check", default=true},
    {id="outline",        text="Hightlight Outline",      y=267, kind="check", default=true},
    {id="fillColor",      text="Hightlight Fill Color",   y=294, kind="color", default=RGB(255,0,234)},
    {id="outlineColor",   text="Hightlight Outline Color",y=321, kind="color", default=PINK},
    {id="health",         text="Show Health Bar",         y=375, kind="check", default=true},
    {id="healthWidth",    text="Healt Bar Size",            y=402, kind="value", default=2.5, min=1, max=8, step=0.5},

local function buildMenu(parent, actions)
    local panel = rect(parent, "Violent settings", 0, 0, MENU_WIDTH, MENU_HEIGHT, RGB(5,6,7), 0.27)
    local header = rect(panel, "Pink title banner", 0, 0, 308, 77, PINK)
    local title = logo(header, "VIOLENT", 0, 0, 308, 77, 44, true)
    create("UIStroke", title, {Color=INK, Thickness=2.4, ApplyStrokeMode=Enum.ApplyStrokeMode.Contextual})
    local category = rect(panel, "Category bar", 0, 77, 308, 29, RGB(3,3,3))
    label(category, "Category", "VISUALS / HUD", 8, 0, 200, 29, 14)
    local counter = label(category, "Position counter", "29/51", 238, 0, 62, 29, 14, {align=Enum.TextXAlignment.Right})
    local views = {}
    for index, spec in ipairs(rows) do
        local row = rect(panel, spec.text, 0, spec.y, 308, 25, SELECTED, 1)
        local caption = label(row, "Caption", spec.text, 8, 0, 260, 24, 14)
        local view = {row=row, caption=caption}
        if spec.kind == "check" then
            view.box = rect(row, "Checkbox", 282, 3, 18, 17)
            view.border = stroke(view.box, WHITE, 1.8)
            view.mark = rect(view.box, "Check mark", 0, 0, 18, 17)
            chain(view.mark, "Tick stroke", {3,9, 7,13, 14,5}, 1.9, WHITE)
        elseif spec.kind == "color" then
            view.swatch = rect(row, "Color swatch", 278, 3, 26, 18, spec.default)
            stroke(view.swatch, RGB(59,44,56), 0.8).Transparency = 0.44
        else
            view.value = label(row, "Value", tostring(spec.default), 244, 0, 57, 24, 14, {align=Enum.TextXAlignment.Right})
        end
        target(row, "Open " .. spec.text, 0, 0, 308, 25, function() actions.activate(index) end)
        views[index] = view
    end
    for _, section in ipairs({{"HIGHLIGHTS",214,90}, {"HEALTH BAR",376,94}, {"TEXT",457,41}}) do
        local name, y, width = table.unpack(section)
        rect(panel, name .. " underline", (308-width)/2, y-28+21, width, 2, RGB(206,44,137), 0.32)
        logo(panel, name, 0, y-28, 308, 26, 19, true)
    end
    local footer = rect(panel, "Release and selector", 0, 460, 308, 29, RGB(3,3,3), 0.22)
    label(footer, "Release", "Release, v1.4", 9, 0, 126, 27, 14, {font=Enum.Font.SourceSansItalic})
    chain(footer, "Up arrow", {147,11, 152,6, 157,11}, 2.1, WHITE)
    chain(footer, "Down arrow", {147,16, 152,21, 157,16}, 2.1, WHITE)
    target(footer, "Previous row", 134, 0, 37, 14, function() actions.move(-1) end)
    target(footer, "Next row", 134, 14, 37, 15, function() actions.move(1) end)
    rect(panel, "Text section right indicator", 281, 459, 20, 1, WHITE)
    rect(panel, "Status separator", 0, 490, 308, 3, RGB(184,53,126), 0.18)
    local status = rect(panel, "Status", 0, 493, 308, 20, RGB(3,3,3), 0.1)
    local statusText = label(status, "Status label", "", 8, 0, 294, 20, 11)
    return function(values, selected, message)
        counter.Text = tostring(21 + selected) .. "/51"
        statusText.Text = message
        for index, spec in ipairs(rows) do
            local view = views[index]
            local active = index == selected
            local textColor = active and INK or WHITE
            view.row.BackgroundTransparency = active and 0 or 1
            view.caption.TextColor3 = textColor
            if view.mark then
                view.mark.Visible = values[spec.id]
                view.border.Color = textColor
                for _, tick in ipairs(view.mark:GetChildren()) do tick.BackgroundColor3 = textColor end
            elseif view.swatch then
                view.swatch.BackgroundColor3 = values[spec.id]
            else
                view.value.Text = tostring(values[spec.id])
                view.value.TextColor3 = textColor
            end
        end
    end
end

local function button(parent, name, text, x, y, width, height, onClick)
    local tile = rect(parent, name .. " background", x, y, width, height, RGB(45,36,43))
    local caption = label(tile, "Label", text, 0, 0, width, height, 13, {align=Enum.TextXAlignment.Center})
    local hit = target(tile, name, 0, 0, width, height, onClick)
    return hit, caption
end

local function colorHex(color)
    return ("#%02X%02X%02X"):format(math.round(color.R*255), math.round(color.G*255), math.round(color.B*255))
end

local palette = {
    RGB(255,255,255), RGB(165,165,167), RGB(30,30,33), PINK, RGB(255,0,234), RGB(187,52,255),
    RGB(255,72,76), RGB(255,149,57), RGB(255,227,74), RGB(111,247,81), RGB(58,221,223), RGB(79,142,255),
}

local function buildEditor(parent, actions)
    local popup = rect(parent, "Value and color editor", 0, 0, EDITOR_WIDTH, 151, RGB(20,16,20))
    popup.ZIndex = 5
    popup.Active = true
    popup.Visible = false
    stroke(popup, RGB(110,47,86), 1)
    rect(popup, "Pink accent", 0, 0, 228, 3, PINK)
    local heading = label(popup, "Editor title", "", 10, 5, 169, 27, 13)
    button(popup, "Close editor", "×", 192, 7, 27, 22, actions.close)

    local colorPage = rect(popup, "Color editor", 0, 34, 228, 117)
    for i, color in ipairs(palette) do
        local x, y = 10 + ((i-1)%6)*35, math.floor((i-1)/6)*31
        local cell = rect(colorPage, "Palette " .. colorHex(color), x, y, 28, 23, color)
        stroke(cell, RGB(92,73,88), 1)
        target(cell, "Choose " .. colorHex(color), 0, 0, 28, 23, function() actions.setColor(color) end)
    end
    local hex = create("TextBox", colorPage, {
        Name="Hex color", Position=UDim2.fromOffset(10,66), Size=UDim2.fromOffset(126,25),
        BackgroundColor3=RGB(45,36,43), BorderSizePixel=0, Text="#FFFFFF", PlaceholderText="#RRGGBB",
        TextColor3=WHITE, TextSize=14, Font=BODY_FONT, ClearTextOnFocus=false,
    })
    local colorHint = label(colorPage, "Color hint", "Enter a hex color or pick a swatch", 10, 94, 210, 17, 10)
    local function applyHex()
        local digits = hex.Text:match("^%s*#?(%x%x%x%x%x%x)%s*$")
        if not digits then
            colorHint.Text = "Use six hex digits, e.g. #FF0F8F"
            return
        end
        actions.setColor(RGB(tonumber(digits:sub(1,2),16), tonumber(digits:sub(3,4),16), tonumber(digits:sub(5,6),16)))
    end
    button(colorPage, "Apply hex", "Apply", 144, 66, 73, 25, applyHex)
    hex.FocusLost:Connect(function(enterPressed) if enterPressed then applyHex() end end)

    local numeric = rect(popup, "Numeric editor", 0, 34, 228, 62)
    button(numeric, "Decrease", "−", 10, 4, 43, 28, function() actions.step(-1) end)
    local number = label(numeric, "Current number", "", 58, 4, 112, 28, 17, {align=Enum.TextXAlignment.Center})
    button(numeric, "Increase", "+", 175, 4, 43, 28, function() actions.step(1) end)
    local range = label(numeric, "Range", "", 10, 36, 208, 18, 11, {align=Enum.TextXAlignment.Center})
    local function positionEditor()
        local docked = parent.Size.X.Offset >= MENU_WIDTH + EDITOR_GAP + EDITOR_WIDTH
        local x = docked and (MENU_WIDTH + EDITOR_GAP) or (MENU_WIDTH - EDITOR_WIDTH) / 2
        popup.Position = UDim2.fromOffset(x, popup.Position.Y.Offset)
    end
    parent:GetPropertyChangedSignal("Size"):Connect(positionEditor)
    return function(spec, value)
        popup.Visible = spec ~= nil
        if not spec then return end
        local isColor = spec.kind == "color"
        heading.Text = spec.text
        colorPage.Visible, numeric.Visible = isColor, not isColor
        local height = isColor and 151 or 96
        popup.Size = UDim2.fromOffset(EDITOR_WIDTH, height)
        popup.Position = UDim2.fromOffset(0, math.min(spec.y, MENU_HEIGHT-height-12))
        positionEditor()
        if isColor then
            local formatted = colorHex(value)
            if hex.Text ~= formatted then hex.Text = formatted end
            colorHint.Text = "Enter a hex color or pick a swatch"
        else
            number.Text = tostring(value)
            range.Text = tostring(spec.min) .. " – " .. tostring(spec.max)
        end
    end
end

-- All state belongs to the menu; controls do not affect the experience.
local function runMenu(canvas, screen)
    local values = {}
    for _, spec in ipairs(rows) do values[spec.id] = spec.default end
    local selected, editing = 8, nil
    local actions = {}
    local drawMenu = buildMenu(canvas, actions)
    local drawEditor

    local function render(message)
        local spec = rows[selected]
        if not message then
            if spec.kind == "check" then message = spec.text .. (values[spec.id] and ": On" or ": Off")
            elseif spec.kind == "value" then message = spec.text .. ": " .. tostring(values[spec.id])
            else message = "Set color to: " .. spec.text end
        end
        drawMenu(values, selected, message)
        local editorSpec = editing and rows[editing]
        drawEditor(editorSpec, editorSpec and values[editorSpec.id])
    end

    function actions.activate(index)
        local sameEditor = editing == index
        selected, editing = index, nil
        local spec = rows[index]
        if spec.kind == "check" then values[spec.id] = not values[spec.id]
        elseif not sameEditor then editing = index end
        render()
    end

    function actions.move(direction)
        selected = (selected - 1 + direction) % #rows + 1
        editing = nil
        render()
    end

    function actions.close()
        editing = nil
        render()
    end

    function actions.step(direction)
        local spec = rows[selected]
        if spec.kind ~= "value" then return end
        values[spec.id] = math.clamp(values[spec.id] + direction*spec.step, spec.min, spec.max)
        render()
    end

    function actions.setColor(color)
        local spec = editing and rows[editing]
        if not spec or spec.kind ~= "color" then return end
        values[spec.id] = color
        render(spec.text .. ": " .. colorHex(color))
    end

    drawEditor = buildEditor(canvas, actions)
    render()

    local input = game:GetService("UserInputService")
    local key = Enum.KeyCode
    local listener = input.InputBegan:Connect(function(event, handled)
        if handled or input:GetFocusedTextBox() or event.UserInputType ~= Enum.UserInputType.Keyboard then return end
        if event.KeyCode == key.Up then actions.move(-1)
        elseif event.KeyCode == key.Down then actions.move(1)
        elseif event.KeyCode == key.Left then actions.step(-1)
        elseif event.KeyCode == key.Right then actions.step(1)
        elseif event.KeyCode == key.Return then actions.activate(selected)
        elseif event.KeyCode == key.Escape and editing then actions.close() end
    end)
    screen.Destroying:Connect(function() listener:Disconnect() end)
end

local function mount()
    local runService = game:GetService("RunService")
    if not runService:IsClient() then return end
    if not game:IsLoaded() then game.Loaded:Wait() end
    local player = game:GetService("Players").LocalPlayer
    local root = player and (player:FindFirstChildOfClass("PlayerGui") or player:WaitForChild("PlayerGui", 10))
    if not root or not root:IsA("PlayerGui") then
        warn("Violent replica: local PlayerGui is not available.")
        return
    end
    local screenName = "ViolentScreenshotReplica" -- reused to replace older copies
    local old = root:FindFirstChild(screenName)
    if old and old:IsA("ScreenGui") and old:GetAttribute("ViolentReplicaOwned") == true then old:Destroy() end
    local screen = create("ScreenGui", nil, {
        Name = screenName, ResetOnSpawn = false, DisplayOrder = 40,
        ZIndexBehavior = Enum.ZIndexBehavior.Sibling,
    })
    screen:SetAttribute("ViolentReplicaOwned", true)
    local viewport = rect(screen, "Transparent viewport", 0, 0, 0, 0)
    viewport.Size = UDim2.fromScale(1, 1)
    local canvas = rect(viewport, "Violent controls", 0, 0, MENU_WIDTH, MENU_HEIGHT)
    local scale = create("UIScale", canvas, {Scale = 1})
    local function fit()
        local size = viewport.AbsoluteSize
        if size.X < 24 or size.Y < 24 then return end
        local left = size.X < 380 and 8 or 27
        local top = 12
        local usableWidth = size.X - left - 8
        local factor = math.min(1, usableWidth/MENU_WIDTH, (size.Y-top-8)/MENU_HEIGHT)
        local dockedWidth = MENU_WIDTH + EDITOR_GAP + EDITOR_WIDTH
        local docked = usableWidth >= dockedWidth * factor
        canvas.Position = UDim2.fromOffset(left, top)
        canvas.Size = UDim2.fromOffset(docked and dockedWidth or MENU_WIDTH, MENU_HEIGHT)
        scale.Scale = factor
    end
    viewport:GetPropertyChangedSignal("AbsoluteSize"):Connect(fit)
    runMenu(canvas, screen)
    screen.Parent = root
    fit()
end

mount()
