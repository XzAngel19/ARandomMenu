local function create(className, properties, parent)
	local instance = Instance.new(className)
	for key, value in properties do
		instance[key] = value
	end
	instance.Parent = parent
	return instance
end

local function tween(instance, info, goal)
	local newTween = tweenService:Create(instance, info, goal)
	newTween:Play()
	return newTween
end

local function connectInput(signal, callback)
	local connection = signal:Connect(callback)
	table.insert(AngelHub.connections, connection)
	return connection
end

local function isPress(input)
	return input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch
end

local function isMove(input)
	return input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch
end

local function addGradient(parent, rotation)
	return create("UIGradient", { Color = ColorSequence.new(ACCENT_A, ACCENT_B), Rotation = rotation }, parent)
end

local function addStroke(parent, color, transparency, thickness)
	return create("UIStroke", {
		Color = color,
		Transparency = transparency,
		Thickness = thickness or 1,
		ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
	}, parent)
end

local function createCard(className, parent, text, height)
	local card = create(className, {
		Name = text,
		Size = UDim2.new(1, 0, 0, height),
		BackgroundColor3 = ROW_COLOR,
		BackgroundTransparency = 0.65,
		BorderSizePixel = 0,
	}, AngelHub.sectionBoxes[parent] or parent)

	create("UICorner", { CornerRadius = UDim.new(0, 5) }, card)
	local stroke = addStroke(card, ACCENT_A, 1)
	card.ClipsDescendants = true

	local group = AngelHub.currentGroup
	if group then
		table.insert(group.cards, card)
		card:SetAttribute("GroupHidden", group.collapsed)
		card.Visible = not group.collapsed
	end

	card.InputBegan:Connect(function(input)
		if not isPress(input) then
			return
		end

		local offset = Vector2.new(input.Position.X, input.Position.Y) - card.AbsolutePosition
		local ripple = create("Frame", {
			AnchorPoint = Vector2.new(0.5, 0.5),
			Position = UDim2.fromOffset(offset.X, offset.Y),
			Size = UDim2.fromOffset(0, 0),
			BackgroundColor3 = Color3.new(1, 1, 1),
			BackgroundTransparency = 0.8,
			BorderSizePixel = 0,
			ZIndex = 0,
		}, card)

		create("UICorner", { CornerRadius = UDim.new(1, 0) }, ripple)
		local diameter = math.max(card.AbsoluteSize.X, card.AbsoluteSize.Y) * 2

		tween(ripple, MEDIUM_TWEEN, { Size = UDim2.fromOffset(diameter, diameter), BackgroundTransparency = 1 }).Completed:Connect(function()
			ripple:Destroy()
		end)
	end)

	card.MouseEnter:Connect(function()
		tween(card, FAST_TWEEN, { BackgroundTransparency = 0.35 })
		tween(stroke, FAST_TWEEN, { Transparency = 0.55 })
	end)

	card.MouseLeave:Connect(function()
		tween(card, FAST_TWEEN, { BackgroundTransparency = 0.65 })
		tween(stroke, FAST_TWEEN, { Transparency = 1 })
	end)

	return card, stroke
end

local function addToggle(parent, text, keybind, default)
	local startState = default == true and not UNIMPLEMENTED_TOGGLES[text]
	local holder, stroke = createCard("TextButton", parent, text, 28)
	holder.Text = ""
	holder.AutoButtonColor = false

	local box = create("Frame", {
		Size = UDim2.fromOffset(16, 16),
		Position = UDim2.new(0, 8, 0.5, -8),
		BackgroundColor3 = ELEMENT_COLOR,
		BorderSizePixel = 0,
	}, holder)

	create("UICorner", { CornerRadius = UDim.new(0, 4) }, box)
	local boxStroke = addStroke(box, startState and ACCENT_A or DIM_TEXT_COLOR, startState and 0 or 0.5)

	local check = create("Frame", {
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.fromScale(0.5, 0.5),
		Size = startState and UDim2.fromOffset(10, 10) or UDim2.fromOffset(0, 0),
		BackgroundColor3 = Color3.new(1, 1, 1),
		BorderSizePixel = 0,
	}, box)

	create("UICorner", { CornerRadius = UDim.new(0, 3) }, check)
	addGradient(check, 45)

	local label = create("TextLabel", {
		Size = UDim2.new(1, keybind and -100 or -44, 1, 0),
		Position = UDim2.fromOffset(32, 0),
		BackgroundTransparency = 1,
		Text = text,
		TextColor3 = startState and TEXT_COLOR or OFF_TEXT_COLOR,
		TextSize = 13,
		Font = TEXT_FONT,
		TextXAlignment = Enum.TextXAlignment.Left,
		TextTruncate = Enum.TextTruncate.AtEnd,
	}, holder)

	local accentBar = create("Frame", {
		Size = UDim2.new(0, 3, 1, -10),
		Position = UDim2.fromOffset(0, 5),
		BackgroundColor3 = Color3.new(1, 1, 1),
		BackgroundTransparency = startState and 0 or 1,
		BorderSizePixel = 0,
	}, holder)

	create("UICorner", { CornerRadius = UDim.new(1, 0) }, accentBar)
	addGradient(accentBar, 90)

	local state = startState
	AngelHub.values[text] = state
	AngelHub.defaults[text] = state

	AngelHub.setters[text] = function(value)
		state = value == true
		AngelHub.values[text] = state

		tween(check, POP_TWEEN, { Size = state and UDim2.fromOffset(10, 10) or UDim2.fromOffset(0, 0) })
		tween(boxStroke, FAST_TWEEN, { Color = state and ACCENT_A or DIM_TEXT_COLOR, Transparency = state and 0 or 0.5 })
		tween(label, FAST_TWEEN, { TextColor3 = state and TEXT_COLOR or OFF_TEXT_COLOR })
		tween(accentBar, FAST_TWEEN, { BackgroundTransparency = state and 0 or 1 })
	end

	if keybind then
		local keybindName = text .. " Keybind"
		local keybindButton = create("TextButton", {
			Size = UDim2.fromOffset(52, 22),
			Position = UDim2.new(1, -58, 0.5, -11),
			BackgroundTransparency = 1,
			Text = "[" .. keybind .. "]",
			TextColor3 = DIM_TEXT_COLOR,
			TextSize = 12,
			Font = TEXT_FONT,
			TextXAlignment = Enum.TextXAlignment.Right,
			AutoButtonColor = false,
		}, holder)

		local function applyKey(keyName)
			local keyCode = nil
			for _, item in Enum.KeyCode:GetEnumItems() do
				if item.Name == keyName then
					keyCode = item
					break
				end
			end

			AngelHub.keybinds[text] = keyCode
			AngelHub.values[keybindName] = keyCode and keyName or ""
			keybindButton.Text = keyCode and ("[" .. keyName .. "]") or "[None]"
			keybindButton.Visible = AngelHub.values["Keybind Hints"] ~= false
		end

		AngelHub.defaults[keybindName] = keybind
		AngelHub.setters[keybindName] = applyKey
		AngelHub.keyLabels[text] = keybindButton
		applyKey(keybind)

		keybindButton.MouseEnter:Connect(function()
			tween(keybindButton, FAST_TWEEN, { TextColor3 = ACCENT_A })
		end)

		keybindButton.MouseLeave:Connect(function()
			if not AngelHub.listening then
				tween(keybindButton, FAST_TWEEN, { TextColor3 = DIM_TEXT_COLOR })
			end
		end)

		keybindButton.Activated:Connect(function()
			if AngelHub.cancelListening then
				AngelHub.cancelListening()
			end

			local previousKey = AngelHub.values[keybindName]
			keybindButton.Text = "[...]"
			tween(keybindButton, FAST_TWEEN, { TextColor3 = ACCENT_A })

			AngelHub.cancelListening = function()
				AngelHub.listening = nil
				AngelHub.cancelListening = nil
				tween(keybindButton, FAST_TWEEN, { TextColor3 = DIM_TEXT_COLOR })
				applyKey(previousKey)
			end

			AngelHub.listening = function(keyCode)
				AngelHub.listening = nil
				AngelHub.cancelListening = nil
				tween(keybindButton, FAST_TWEEN, { TextColor3 = DIM_TEXT_COLOR })
				applyKey(keyCode == Enum.KeyCode.Escape and "" or keyCode.Name)
			end
		end)
	end

	holder.MouseButton1Down:Connect(function()
		tween(holder, FAST_TWEEN, { BackgroundTransparency = 0 })
		tween(stroke, FAST_TWEEN, { Transparency = 0 })
	end)

	holder.MouseButton1Up:Connect(function()
		tween(holder, FAST_TWEEN, { BackgroundTransparency = 0.35 })
		tween(stroke, FAST_TWEEN, { Transparency = 0.55 })
	end)

	holder.Activated:Connect(function()
		local accepted = AngelHub.setters[text](not state)
		if accepted ~= false and AngelHub.notify then
			AngelHub.notify(text, state)
		end
	end)

	holder.MouseEnter:Connect(function()
		tween(label, FAST_TWEEN, { Position = UDim2.fromOffset(36, 0) })
	end)

	holder.MouseLeave:Connect(function()
		tween(label, FAST_TWEEN, { Position = UDim2.fromOffset(32, 0) })
	end)

	table.insert(AngelHub.searchables, holder)
	return holder
end

    AngelHub.moduleNames = AngelHub.moduleNames or {}
    AngelHub.moduleNames[text] = true

local function addSlider(parent, text, minValue, maxValue, default, decimals)
    local manualMovement = ({
        ["Speed Value"] = true, ["Jump Power"] = true, ["High Jump Velocity"] = true,
        ["Long Jump Speed"] = true, ["Infinite Jump Power"] = true,
        ["Fly Horizontal Speed"] = true, ["Fly Vertical Speed"] = true,
        ["Wall Crawl Speed"] = true,
    })[text] == true
	local holder = createCard("Frame", parent, text, 44)
	local startAlpha = (default - minValue) / (maxValue - minValue)

	create("TextLabel", {
		Size = UDim2.new(1, -70, 0, 20),
		Position = UDim2.fromOffset(10, 3),
		BackgroundTransparency = 1,
		Text = text,
		TextColor3 = TEXT_COLOR,
		TextSize = 13,
		Font = TEXT_FONT,
		TextXAlignment = Enum.TextXAlignment.Left,
	}, holder)

	local valueLabel = create(manualMovement and "TextBox" or "TextLabel", {
		Size = UDim2.fromOffset(56, 20),
		Position = UDim2.new(1, -66, 0, 3),
		BackgroundTransparency = 1,
		Text = tostring(default),
		TextColor3 = OFF_TEXT_COLOR,
		TextSize = 13,
		Font = BOLD_FONT,
		TextXAlignment = Enum.TextXAlignment.Right,
	}, holder)

	local bar = create("Frame", {
		Size = UDim2.new(1, -20, 0, 6),
		Position = UDim2.fromOffset(10, 29),
		BackgroundColor3 = ELEMENT_COLOR,
		BorderSizePixel = 0,
	}, holder)

	create("UICorner", { CornerRadius = UDim.new(1, 0) }, bar)
	addStroke(bar, Color3.new(1, 1, 1), 0.88)

	local fill = create("Frame", { Size = UDim2.fromScale(startAlpha, 1), BackgroundColor3 = Color3.new(1, 1, 1), BorderSizePixel = 0 }, bar)
	create("UICorner", { CornerRadius = UDim.new(1, 0) }, fill)
	addGradient(fill, 0)

	local knob = create("Frame", {
		AnchorPoint = Vector2.new(0.5, 0.5),
		Size = UDim2.fromOffset(12, 12),
		Position = UDim2.new(startAlpha, 0, 0.5, 0),
		BackgroundColor3 = Color3.new(1, 1, 1),
		BorderSizePixel = 0,
	}, bar)

	create("UICorner", { CornerRadius = UDim.new(1, 0) }, knob)
	addStroke(knob, ACCENT_A, 0, 2)
	local knobScale = create("UIScale", {}, knob)

	local hitbox = create("Frame", { Size = UDim2.new(1, -20, 0, 22), Position = UDim2.fromOffset(10, 21), BackgroundTransparency = 1 }, holder)

	AngelHub.values[text] = default
	AngelHub.defaults[text] = default

	local dragging = false

	local function setValue(value)
		local factor = 10 ^ decimals
        if type(value) ~= "number" or value ~= value or value == math.huge or value == -math.huge then return end
        if manualMovement then value = math.clamp(value, -10000, maxValue) else value = math.clamp(value, minValue, maxValue) end
		value = math.floor(value * factor + 0.5) / factor
		local snapped = math.clamp((value - minValue) / (maxValue - minValue), 0, 1)

		tween(fill, SLIDE_TWEEN, { Size = UDim2.fromScale(snapped, 1) })
		tween(knob, SLIDE_TWEEN, { Position = UDim2.new(snapped, 0, 0.5, 0) })
		valueLabel.Text = tostring(value)
		AngelHub.values[text] = value
	end

	AngelHub.setters[text] = setValue
    if manualMovement then
        valueLabel.ClearTextOnFocus = false
        valueLabel.FocusLost:Connect(function()
            local number = tonumber(valueLabel.Text)
            if number then AngelHub.setters[text](number) else valueLabel.Text = tostring(AngelHub.values[text]) end
        end)
    end

	local function updateFromX(positionX)
		local alpha = math.clamp((positionX - bar.AbsolutePosition.X) / bar.AbsoluteSize.X, 0, 1)
		AngelHub.setters[text](minValue + (maxValue - minValue) * alpha)
	end

	hitbox.InputBegan:Connect(function(input)
		if not isPress(input) then
			return
		end

		dragging = true
		tween(knobScale, POP_TWEEN, { Scale = 1.35 })
		tween(valueLabel, FAST_TWEEN, { TextColor3 = ACCENT_A })
		updateFromX(input.Position.X)
	end)

	connectInput(userInputService.InputChanged, function(input)
		if dragging and isMove(input) then
			updateFromX(input.Position.X)
		end
	end)

	connectInput(userInputService.InputEnded, function(input)
		if not dragging or not isPress(input) then
			return
		end

		dragging = false
		tween(knobScale, POP_TWEEN, { Scale = 1 })
		tween(valueLabel, FAST_TWEEN, { TextColor3 = OFF_TEXT_COLOR })
	end)

	table.insert(AngelHub.searchables, holder)
	return holder
end

local function addCycle(parent, text, options, onChange)
	local holder = createCard("Frame", parent, text, 56)

	create("TextLabel", {
		Size = UDim2.new(1, -20, 0, 20),
		Position = UDim2.fromOffset(10, 3),
		BackgroundTransparency = 1,
		Text = text,
		TextColor3 = TEXT_COLOR,
		TextSize = 13,
		Font = TEXT_FONT,
		TextXAlignment = Enum.TextXAlignment.Left,
	}, holder)

	local button = create("TextButton", {
		Size = UDim2.new(1, -20, 0, 24),
		Position = UDim2.fromOffset(10, 26),
		BackgroundColor3 = ELEMENT_COLOR,
		BorderSizePixel = 0,
		Text = options[1],
		TextColor3 = TEXT_COLOR,
		TextSize = 13,
		Font = TEXT_FONT,
		AutoButtonColor = false,
	}, holder)

	create("UICorner", { CornerRadius = UDim.new(0, 4) }, button)
	local buttonStroke = addStroke(button, DIM_TEXT_COLOR, 0.6)

	local index = 1
	AngelHub.values[text] = options[1]
	AngelHub.defaults[text] = options[1]

	AngelHub.setters[text] = function(value)
		local found = table.find(options, value)
		if not found then
			return
		end

		index = found
		AngelHub.values[text] = options[index]
		button.Text = options[index]

		if onChange then
			onChange(options[index])
		end
	end

    AngelHub.cycleOptions[text] = function(items)
        if type(items) ~= "table" then return end
        local nextOptions = {}
        for _, item in ipairs(items) do
            if type(item) == "string" and item ~= "" and not table.find(nextOptions, item) then table.insert(nextOptions, item) end
        end
        if #nextOptions == 0 then nextOptions = {"Waiting for maps"} end
        local selected = AngelHub.values[text]
        options = nextOptions
        index = table.find(options, selected) or 1
        AngelHub.setters[text](options[index])
    end

	button.MouseEnter:Connect(function()
		tween(buttonStroke, FAST_TWEEN, { Color = ACCENT_A, Transparency = 0 })
	end)

	button.MouseLeave:Connect(function()
		tween(buttonStroke, FAST_TWEEN, { Color = DIM_TEXT_COLOR, Transparency = 0.6 })
	end)

	button.Activated:Connect(function()
        local nextIndex = index % #options + 1
        AngelHub.setters[text](options[nextIndex])
        button.TextTransparency = 1
        tween(button, FAST_TWEEN, { TextTransparency = 0 })
	end)

	table.insert(AngelHub.searchables, holder)
	return holder
end

local function addButton(parent, text, callback)
	local holder = createCard("Frame", parent, text, 34)

	local button = create("TextButton", {
		Size = UDim2.new(1, -16, 0, 24),
		Position = UDim2.fromOffset(8, 5),
		BackgroundColor3 = ELEMENT_COLOR,
		BorderSizePixel = 0,
		Text = text,
		TextColor3 = TEXT_COLOR,
		TextSize = 13,
		Font = TEXT_FONT,
		AutoButtonColor = false,
	}, holder)

	create("UICorner", { CornerRadius = UDim.new(0, 4) }, button)
	local buttonStroke = addStroke(button, DIM_TEXT_COLOR, 0.6)

	button.MouseEnter:Connect(function()
		tween(buttonStroke, FAST_TWEEN, { Color = ACCENT_A, Transparency = 0 })
		tween(button, FAST_TWEEN, { BackgroundColor3 = Color3.fromRGB(28, 28, 30) })
	end)

	button.MouseLeave:Connect(function()
		tween(buttonStroke, FAST_TWEEN, { Color = DIM_TEXT_COLOR, Transparency = 0.6 })
		tween(button, FAST_TWEEN, { BackgroundColor3 = ELEMENT_COLOR })
	end)

	button.MouseButton1Down:Connect(function()
		tween(button, FAST_TWEEN, { BackgroundColor3 = Color3.fromRGB(46, 46, 48) })
	end)

	button.MouseButton1Up:Connect(function()
		tween(button, FAST_TWEEN, { BackgroundColor3 = Color3.fromRGB(28, 28, 30) })
	end)

	button.Activated:Connect(function()
        if callback then callback(); return end
        local action = AngelHub.actions and AngelHub.actions[text]
        if action then
            local ok, err = pcall(action)
            if not ok then warn("Angel Hub action failed: " .. tostring(err)) end
        elseif AngelHub.notify then
            AngelHub.notify("Not implemented: " .. text, false, true)
        end
	end)

	table.insert(AngelHub.searchables, holder)

local function makeDraggable(handle, target)
	local dragStart, startPosition

	handle.InputBegan:Connect(function(input)
		if not isPress(input) then
			return
		end

		dragStart = input.Position
		startPosition = target.Position
	end)

	connectInput(userInputService.InputChanged, function(input)
		if not dragStart or not isMove(input) then
			return
		end

		local delta = input.Position - dragStart
		target.Position = UDim2.new(startPosition.X.Scale, startPosition.X.Offset + delta.X, startPosition.Y.Scale, startPosition.Y.Offset + delta.Y)
	end)

	connectInput(userInputService.InputEnded, function(input)
		if isPress(input) then
			dragStart = nil
		end
	end)
end

-- FIRST FUNCTIONAL PASS. Independent implementations; no ARandomMenu runtime.
-- Only the features listed in supportedModules/actions below are connected.
-- Everything else is deliberately prevented from pretending to be active.
local function installGameplay(screenGui)
    local run = game:GetService("RunService")
    local lighting = game:GetService("Lighting")
    local V = AngelHub.values
    local enabled = {}
    local groups = {}
    local noclipParts = setmetatable({}, {__mode = "k"})
    local noclipCharacter
    local lastJump = -math.huge
    local jumpHeld, jumpTouch, jumpKey = false, nil, nil
    local afkElapsed = 0
    local xrayStep = function() end
    local updateItemEsp = function() end
    local mm2FarmStep = function() end
    local mm2KnifeStep = function() end
    local mm2CoinAuraStep = function() end
    local mm2VoteStep = function() end
    local mm2ShootStep = function() end
    local mm2UnboxStep = function() end
    local mm2WinStep = function() end
    local mm2GunAssistStep = function() end
    local mm2CoinEspStep = function() end
    local mm2NotificationsStep = function() end
    local airWalkStep = function() end
    local flyStep = function() end
    local chamsStep = function() end
    local nametagsStep = function() end
    local playerHealthStep = function() end
    local pointerStep = function() end
    local markersStep = function() end
    local mm2Cleanup = function() end
    local cleanupItemEsp = function() end
    local stopped = false
    local runtimeConnections = {}
    local supportedModules = {
        ["Noclip"] = true, ["Enable Gravity"] = true, ["Enable Speed"] = true,
        ["Custom Jump Power"] = true, ["Infinite Jump"] = true,
        ["Fullbright"] = true, ["Time Changer"] = true, ["Fast Time"] = true,
        ["Custom FOV"] = true, ["Custom Zoom"] = true,
        ["Instant Prompts"] = true, ["Proximity Prompts"] = true, ["Click Detectors"] = true, ["Air Walk"] = true, ["Enable Fly"] = true, ["Walk Fling"] = true,
        ["X-Ray"] = true, ["Anti AFK"] = true, ["Chams"] = true, ["Nametags"] = true, ["HP Players"] = true, ["Player Pointers"] = true,
        ["Gun ESP"] = true, ["Trap ESP"] = true,
        ["Auto Farm"] = true, ["Knife Aura"] = true, ["Coins Aura"] = true,
        ["Auto Vote"] = true, ["Auto Shoot"] = true, ["Auto Unbox"] = true, ["Auto Win"] = true,
        ["Insta Kill As Sheriff"] = true, ["Insta Kill As Murder"] = true, ["Gun Aura"] = true, ["Auto Get Gun"] = true, ["Coin ESP"] = true,
    }
    local function message(text)
        if AngelHub.notify then AngelHub.notify(text, false, true) end
    end
    local function listen(signal, callback)
        local connection = signal:Connect(callback)
        table.insert(runtimeConnections, connection)
        return connection
    end
    local function restore(name)
        local entry = groups[name]
        if not entry then return end
        for property, previous in pairs(entry.original) do
            pcall(function()
                -- Do not overwrite a value another script changed afterwards.
                if entry.object[property] == entry.applied[property] then
                    entry.object[property] = previous
                end
            end)
        end
        groups[name] = nil
    end
    local function apply(name, object, properties)
        local group = groups[name]
        if not object then restore(name); return end
        if group and group.object ~= object then restore(name); group = nil end
        if not group then
            group = {object = object, original = {}, applied = {}}
            groups[name] = group
        end
        for property, value in pairs(properties) do
            pcall(function()
                if group.original[property] == nil then group.original[property] = object[property] end
                object[property] = value
                group.applied[property] = value
            end)
        end
    end
    local function characterParts()
        local character = localPlayer.Character
        local humanoid = character and character:FindFirstChildOfClass("Humanoid")
        local root = character and character:FindFirstChild("HumanoidRootPart")
        if not humanoid or humanoid.Health <= 0 or not root or not root:IsA("BasePart") then return character, nil, nil end
        return character, humanoid, root
    end
    local function restoreNoclip()
        for part, original in pairs(noclipParts) do
            pcall(function() if part.CanCollide == false then part.CanCollide = original end end)
            noclipParts[part] = nil
        end
        noclipCharacter = nil
    end
    local function updateNoclip(character)
        if not enabled["Noclip"] or not character then restoreNoclip(); return end
        if character ~= noclipCharacter then restoreNoclip(); noclipCharacter = character end
        for _, object in ipairs(character:GetDescendants()) do
            if object:IsA("BasePart") then
                if noclipParts[object] == nil then noclipParts[object] = object.CanCollide end
                if object.CanCollide then object.CanCollide = false end
            end
        end
    end
    local fastTime
    local function update(dt)
        if stopped then return end
        local character, humanoid, root = characterParts()
        updateNoclip(character)
        if enabled["Enable Gravity"] then apply("gravity", workspace, {Gravity = V["Gravity Value"]}) else restore("gravity") end
        if enabled["Enable Speed"] and V["Speed Mode"] == "WalkSpeed" and V["Speed Value"] >= 0 and humanoid then
            apply("speed", humanoid, {WalkSpeed = V["Speed Value"]})
        else restore("speed") end
        if dt > 0 and enabled["Enable Speed"] and (V["Speed Mode"] == "Velocity" or (V["Speed Mode"] == "WalkSpeed" and V["Speed Value"] < 0)) and humanoid and root and not root.Anchored and not humanoid.Sit then
            local move = humanoid.MoveDirection
            local vel = root.AssemblyLinearVelocity
            local desiredX, desiredZ = move.X * V["Speed Value"], move.Z * V["Speed Value"]
            local alpha = math.clamp(V["Speed Response"] * dt * 12, 0, 1)
            if V["Speed Response"] >= 1 then alpha = 1 end
            local x = vel.X + (desiredX - vel.X) * alpha
            local z = vel.Z + (desiredZ - vel.Z) * alpha
            if move.Magnitude < 0.01 then
                local brake = math.clamp((1 - V["Speed Braking"]) * dt * 10, 0, 1)
                x, z = x * (1 - brake), z * (1 - brake)
            end
            if V["Speed Wall Check"] then
                local ok, hit = pcall(function()
                    local params = RaycastParams.new()
                    params.FilterType = Enum.RaycastFilterType.Exclude
                    params.FilterDescendantsInstances = {character}
                    params.RespectCanCollide = true
                    params.IgnoreWater = true
                    local step = Vector3.new(x * math.min(dt, 0.1), 0, z * math.min(dt, 0.1))
                    if step.Magnitude < 0.001 then return nil end
                    if workspace.Blockcast then
                        return workspace:Blockcast(root.CFrame, root.Size * 0.9, step, params)
                    end
                    return workspace:Raycast(root.Position, step, params)
                end)
                if not ok or hit then x, z = 0, 0 end
            end
            root.AssemblyLinearVelocity = Vector3.new(x, V["Speed Keep Vertical"] and vel.Y or 0, z)
        end
        if enabled["Infinite Jump"] and V["Infinite Jump Mode"] == "Rise" and jumpHeld and humanoid and root and not root.Anchored then
            local vel = root.AssemblyLinearVelocity
            root.AssemblyLinearVelocity = Vector3.new(vel.X, V["Infinite Jump Power"], vel.Z)
        end
        xrayStep(dt, root)
        local farmOk, farmError = pcall(mm2FarmStep, dt, character, humanoid, root)
        if not farmOk then
            warn("Angel Hub Farm error: " .. tostring(farmError))
            if enabled["Auto Farm"] then AngelHub.setters["Auto Farm"](false); message("Auto Farm stopped: runtime error") end
        end
        local shotOk, shotError = pcall(mm2ShootStep, dt, character, humanoid, root)
        if not shotOk then
            warn("Angel Hub Auto Shoot error: " .. tostring(shotError))
            if enabled["Auto Shoot"] then AngelHub.setters["Auto Shoot"](false) end
            if enabled["Insta Kill As Sheriff"] then AngelHub.setters["Insta Kill As Sheriff"](false) end
            message("Shooting stopped: runtime error")
        end
        mm2UnboxStep()
        local gunOk, gunError = pcall(mm2GunAssistStep, dt, character, humanoid, root)
        if not gunOk then
            warn("Angel Hub Gun assist error: " .. tostring(gunError))
            if enabled["Gun Aura"] then AngelHub.setters["Gun Aura"](false) end
            if enabled["Auto Get Gun"] then AngelHub.setters["Auto Get Gun"](false) end
            message("Gun assist stopped: runtime error")
        end
        local markersOk, markersError = pcall(markersStep, dt, root)
        if not markersOk and not AngelHub.markersError then
            AngelHub.markersError = true; warn("Angel Hub markers error: " .. tostring(markersError))
        end
        local notificationOk, notificationError = pcall(mm2NotificationsStep, dt, root)
        if not notificationOk and not AngelHub.notificationError then
            AngelHub.notificationError = true; warn("Angel Hub notifications error: " .. tostring(notificationError))
        end
        local pointerOk, pointerError = pcall(pointerStep, dt, root)
        if not pointerOk then

            if enabled["Player Pointers"] then AngelHub.setters["Player Pointers"](false); message("Player Pointers stopped: runtime error") end
        end
        local hpOk, hpError = pcall(playerHealthStep, dt, root)

        end
        local tagOk, tagError = pcall(nametagsStep, dt, root)

        local chamsOk, chamsError = pcall(chamsStep, dt)

        end
        local airOk, airError = pcall(airWalkStep, dt, character, humanoid, root)
        if not airOk then

        end

    local promptState = setmetatable({}, {__mode = "k"})

    local function setInteraction(object, property, value)
        pcall(function()
            local byProperty = promptState[object]
            if not byProperty then byProperty = {}; promptState[object] = byProperty end
            local entry = byProperty[property]
            if not entry then entry = {original = object[property]}; byProperty[property] = entry end
            object[property] = value
            entry.applied = value
        end)
    end
    local function restoreInteraction(kind)
        for object, fields in pairs(promptState) do
            for property, entry in pairs(fields) do
                local matching = kind == nil
                    or (kind == "Instant Prompts" and property == "HoldDuration")
                    or (kind == "Proximity Prompts" and property == "MaxActivationDistance" and object:IsA("ProximityPrompt"))
                    or (kind == "Click Detectors" and property == "MaxActivationDistance" and object:IsA("ClickDetector"))
                if matching then
                    pcall(function() if object[property] == entry.applied then object[property] = entry.original end end)
                    fields[property] = nil
                end
            end
            if next(fields) == nil then promptState[object] = nil end
        end
    end

    local function parseNames(value)
        local found = {}
        for name in string.gmatch(string.lower(value or ""), "[^,]+") do
            name = string.match(name, "^%s*(.-)%s*$")
            if name ~= "" then found[name] = true end
        end

    xrayStep = function(dt, root)
        if not enabled["X-Ray"] then return end
        xrayPause = math.max(0, xrayPause - dt)
        if xrayPause > 0 then return end
        local named, kept = parseNames(V["XRay Part Names"]), parseNames(V["XRay Keep Visible"])
        local camera = workspace.CurrentCamera
        local origin = camera and camera.CFrame.Position or root and root.Position
        for _ = 1, 200 do
            local part = xrayParts[xrayCursor]
            if not part then xrayCursor = 1; xrayPause = 0.2; break end
            if not part.Parent or not part:IsDescendantOf(workspace) then
                xrayRestore(part); xrayIndexed[part] = nil
                xrayParts[xrayCursor] = xrayParts[#xrayParts]; xrayParts[#xrayParts] = nil
            else
                local name = string.lower(part.Name)
                local floor = part.Size.Y <= math.min(part.Size.X, part.Size.Z) * 0.35
                local valid = part.Transparency < 0.98 and not kept[name]
                    and (V["XRay Include Unanchored"] or part.Anchored)
                    and (not V["XRay Collidable Only"] or part.CanCollide)
                    and not (V["XRay Keep Floors"] and floor and root and part.Position.Y <= root.Position.Y)
                if valid and V["XRay Limit Range"] and origin then
                    valid = (part.Position - origin).Magnitude <= V["XRay Range"] + part.Size.Magnitude * 0.5
                end
                local ancestor = part.Parent
                while valid and ancestor and ancestor ~= workspace do
                    if ancestor:IsA("Tool") or (ancestor:IsA("Model") and ancestor:FindFirstChildOfClass("Humanoid")) then valid = false end
                    ancestor = ancestor.Parent
                end
                local scope = V["XRay Scope"]
                if scope == "Named Parts" then valid = valid and named[name] == true end
                if scope == "Walls" then valid = valid and not floor and part.Size.Y > math.min(part.Size.X, part.Size.Z) * 1.5 end
                if valid then
                    local current = part.LocalTransparencyModifier
                    local record = xrayRecords[part]
                    if not record then record = {base = current}; xrayRecords[part] = record
                    elseif current ~= record.applied then record.base = current end
                    record.applied = math.max(record.base, V["XRay Transparency"])
                    if current ~= record.applied then part.LocalTransparencyModifier = record.applied end
                else xrayRestore(part) end
                xrayCursor += 1
            end
        end
    end

    local function gunCandidate(object)
        if not object or not object:IsDescendantOf(workspace) then return nil end
        local ancestor = object.Parent
        while ancestor and ancestor ~= workspace do
            if ancestor:IsA("Model") and ancestor:FindFirstChildOfClass("Humanoid") then return nil end
            ancestor = ancestor.Parent
        end
        if object:IsA("BasePart") or object:IsA("Model") then return object end
        if object:IsA("Tool") then
            local parent = object.Parent
            if parent and parent:FindFirstChildOfClass("Humanoid") then return nil end
            local part = object:FindFirstChild("Handle") or object:FindFirstChildWhichIsA("BasePart", true)
            if part and part:IsA("BasePart") then return part end
        end
        return nil
    end
    local nextGunSweep, cachedGun = 0, nil
    local function findDroppedGun()
        if cachedGun and gunCandidate(cachedGun) then return cachedGun end
        cachedGun = nil
        local map = mm2Map()
        local candidate = (map and map:FindFirstChild("GunDrop")) or workspace:FindFirstChild("GunDrop")
        cachedGun = gunCandidate(candidate)
        if cachedGun then return cachedGun end
        local ok, collection = pcall(function() return game:GetService("CollectionService") end)
        if ok and collection then
            for _, tag in ipairs({"GunDrop", "Weapon_Gun"}) do
                for _, tagged in ipairs(collection:GetTagged(tag)) do
                    if tag ~= "Weapon_Gun" or tagged:IsA("Tool") then
                        cachedGun = gunCandidate(tagged)
                        if cachedGun then return cachedGun end
                    end
                end
            end
        end
        for _, object in ipairs(workspace:GetChildren()) do
            if object:IsA("Model") or object:IsA("Folder") then
                cachedGun = gunCandidate(object:FindFirstChild("GunDrop"))
                if cachedGun then return cachedGun end
            end
        end
        local now = os.clock()
        if now >= nextGunSweep then
            nextGunSweep = now + 1.5
            cachedGun = gunCandidate(workspace:FindFirstChild("GunDrop", true))
        end
        return cachedGun
    end

        local reference = spawnPart(spawns) or spawnPart(container)
            or map:FindFirstChildWhichIsA("BasePart", true)
        -- Fail closed when either location is unknown; never move while in lobby.
        if not lobbyRef or not reference then return false end
        return (root.Position - lobbyRef.Position).Magnitude > 350
            and (root.Position - reference.Position).Magnitude <= 1500
    end
    local farmPark
    leaveFarmPark = function()
        local parked = farmPark
        if not parked then return true end
        local character, humanoid, root = characterParts()
        local nearPlatform = character == parked.character and humanoid and root
            and (root.Position - parked.position).Magnitude <= 120
        if nearPlatform then
            if root.Anchored or not parked.map or parked.map.Parent ~= workspace then
                message("Safe Zone: map return unavailable; platform kept")
                return false
            end
            local ok = pcall(function() character:PivotTo(parked.returnFrame) end)
            if not ok then message("Safe Zone: return failed; platform kept"); return false end
            pcall(function()
                root.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
                root.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
            end)
        end
        farmPark = nil
        pcall(function() parked.part:Destroy() end)
        return true
    end
    local function parkFarmCharacter(character, root, map, returnFrame)
        if not returnFrame or not map or map.Parent ~= workspace or not farmOnMap(root, map) then return false end
        if farmPark and not leaveFarmPark() then return false end
        local original = returnFrame.Position
        local destroyY = workspace.FallenPartsDestroyHeight
        if type(destroyY) ~= "number" or destroyY ~= destroyY or math.abs(destroyY) > 1000000 then destroyY = -500 end
        local safeY = math.max(original.Y + 280, 380, destroyY + 80)
        local platformPosition = Vector3.new(original.X + 650, safeY, original.Z + 650)
        local lobbyRef = farmLobbyRef()
        if lobbyRef and (platformPosition - lobbyRef.Position).Magnitude < 400 then
            platformPosition = Vector3.new(original.X - 650, safeY, original.Z - 650)
        end
        if lobbyRef and (platformPosition - lobbyRef.Position).Magnitude < 400 then return false end
        local part
        local ok = pcall(function()
            part = Instance.new("Part")
            part.Name = "AngelHub_FarmSafeZone"
            part.Anchored = true; part.CanCollide = true; part.CanQuery = false; part.CanTouch = false
            part.CastShadow = false; part.Size = Vector3.new(64, 4, 64)
            part.Material = Enum.Material.SmoothPlastic; part.Transparency = 0.15
            part.CFrame = CFrame.new(platformPosition)
            part.Parent = workspace
        end)
        if not ok or not part or not part.Parent then
            if part then pcall(function() part:Destroy() end) end
            message("Safe Zone: could not create platform")
            return false
        end
        local moved = pcall(function() character:PivotTo(CFrame.new(platformPosition + Vector3.new(0, 5, 0))) end)
        if not moved then pcall(function() part:Destroy() end); message("Safe Zone: move failed"); return false end
        farmPark = {character = character, map = map, part = part, returnFrame = returnFrame, position = platformPosition}
        pcall(function()
            root.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
            root.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
        end)
        return true
    end
    local function farmUsable(coin, map)
        if not coin or not coin.Parent or not coin:IsA("BasePart") or not coin:IsDescendantOf(map) then return false end
        if coin:GetAttribute("Collected") == true or coin:GetAttribute("Delete") == true then return false end
        local visual = coin:FindFirstChild("CoinVisual")
        if visual and (visual:GetAttribute("Collected") == true or visual:GetAttribute("Delete") == true) then return false end
        return true
    end

    end

    local function farmStart(gameplay, collected, ended)
        farmStop()
        farmAmount, farmLimit = 0, 50
        farmBagSampleKnown = false
        farmTickElapsed, farmScanElapsed, farmDone = 0, 0.5, false
        farmCounter("Coins Collected", farmTotal)
        local whole = math.floor(farmElapsed)
        farmCounter("Farm Time", string.format("%02d:%02d", math.floor(whole / 60), whole % 60))
        table.insert(farmConnections, collected.OnClientEvent:Connect(function(_, amount, limit)
            if not enabled["Auto Farm"] then return end
            local character, humanoid, root = characterParts()
            local map = mm2Map()
            if not humanoid or not farmOnMap(root, map) or (farmWaitingMap and farmWaitingMap == map) then return end
            if farmRoundMap ~= map then farmRoundMap, farmAmount, farmLimit, farmDone, farmBagSampleKnown = map, 0, 50, false, false end
            if farmMap ~= map or character ~= farmCharacter then
                farmStopMovement()
                farmMap, farmCharacter = map, character
                farmTarget, farmScanElapsed = nil, 0.5
                table.clear(farmCoins); table.clear(farmBlacklist)
            end
            if type(limit) == "number" and limit >= 50 then farmLimit = limit end
            if type(amount) == "number" and amount >= 0 then
                if amount > farmAmount then
                    farmTotal += farmBagSampleKnown and (amount - farmAmount) or 1
                    farmCounter("Coins Collected", farmTotal)
                    farmTarget, farmScanElapsed = nil, 0.5
                end
                farmAmount, farmBagSampleKnown = amount, true
                if amount < math.max(50, farmLimit) then farmDone = false end
            end
        end))
        table.insert(farmConnections, ended.OnClientEvent:Connect(function()
            if not enabled["Auto Farm"] then return end
            farmWaitingMap = farmMap or mm2Map()
            farmStopMovement(); leaveFarmPark(); farmMap, farmCharacter, farmRoundMap, farmTarget = nil, nil, nil, nil
            table.clear(farmCoins)
        end))
        for _, name in ipairs({"RoundStart", "RoleSelect", "ShowRoleSelect", "ShowRoleSelectNew", "CoinsStarted"}) do
            local remote = gameplay:FindFirstChild(name)
            if remote and remote:IsA("RemoteEvent") then
                table.insert(farmConnections, remote.OnClientEvent:Connect(function(duration)
                    if name ~= "RoundStart" or (tonumber(duration) and tonumber(duration) > 0) then farmWaitingMap = nil end
                end))
            end
        end
    end
    mm2FarmStep = function(dt, character, humanoid, root)
        if not enabled["Auto Farm"] then return end
        farmElapsed += math.max(dt, 0)
        local seconds = math.floor(farmElapsed)
        farmCounter("Farm Time", string.format("%02d:%02d", math.floor(seconds / 60), seconds % 60))
        farmTickElapsed += dt
        if farmTickElapsed < 0.1 then return end
        local elapsed = farmTickElapsed
        farmTickElapsed = 0
        local map = mm2Map()
        if farmPark then
            if V["Farm Safe Zone"] and character == farmPark.character and humanoid and root
                and farmPark.map == map and (root.Position - farmPark.position).Magnitude <= 120 then return end
            if not leaveFarmPark() then return end
        end
        if not character or not humanoid or not root or root.Anchored or not farmOnMap(root, map) or (farmWaitingMap and farmWaitingMap == map) then
            farmStopMovement(); farmMap, farmCharacter, farmTarget = nil, nil, nil
            return
        end
        if farmRoundMap ~= map then farmRoundMap, farmAmount, farmLimit, farmDone, farmBagSampleKnown = map, 0, 50, false, false end
        if map ~= farmMap or character ~= farmCharacter then
            farmStopMovement()
            farmMap, farmCharacter = map, character
            farmTarget, farmScanElapsed = nil, 0.5
            table.clear(farmCoins); table.clear(farmBlacklist)
        end
        if farmAmount >= math.max(50, farmLimit) then
            local wasBelow, belowHome = farmBelowEngaged, farmHomeCFrame
            farmStopMovement()
            if not farmDone then
                farmDone = true
                local spawns = map:FindFirstChild("Spawns") or map:FindFirstChild("Spawns", true)
                local part = spawnPart(spawns)
                if V["Farm Safe Zone"] then
                    local returnFrame = belowHome or (part and (part.CFrame + Vector3.new(0, 3, 0)))
                    if not parkFarmCharacter(character, root, map, returnFrame) then message("Safe Zone unavailable") end
                elseif not wasBelow and part and farmOnMap(root, map) then
                    characterTeleport(part)
                end
            end
            return
        end
        farmScanElapsed += elapsed
        if farmScanElapsed >= 0.5 then farmScanElapsed = 0; farmScan(map) end
        local best, bestDistance
        local now = os.clock()
        for _, coin in ipairs(farmCoins) do
            if farmUsable(coin, map) and (not farmBlacklist[coin] or farmBlacklist[coin] <= now) then
                local distance = (coin.Position - root.Position).Magnitude
                if not bestDistance or distance < bestDistance then best, bestDistance = coin, distance end
            end
        end
        if not best then
            farmTarget = nil
            if farmBelowEngaged then
                local y = farmHoverTarget and farmHoverTarget.Y or root.Position.Y
                farmHoverTarget = Vector3.new(root.Position.X, y, root.Position.Z)
            else farmStopMovement() end
            return
        end
        if best ~= farmTarget then
            stopFarmCarry()
            farmTarget, farmTargetAt, farmBestDistance, farmLastJump = best, now, bestDistance, now
        elseif bestDistance < farmBestDistance - 0.35 then
            stopFarmCarry()
            farmBestDistance, farmTargetAt = bestDistance, now
        end
        if V["Farm Mode"] == "Walk" and now - farmTargetAt >= 1.5 and now - farmLastJump >= 1.5 then
            farmLastJump, farmWalkCarryUntil = now, now + 1
            pcall(function() humanoid.Jump = true end)
        end
        if now - farmTargetAt > 6 then
            farmBlacklist[best] = now + 6; farmTarget = nil
            if farmBelowEngaged then
                local y = farmHoverTarget and farmHoverTarget.Y or root.Position.Y
                farmHoverTarget = Vector3.new(root.Position.X, y, root.Position.Z)
            else farmStopMovement() end
            return
        end
        if V["Farm Mode"] == "Below" then
            if not farmBelowEngaged then
                if humanoid.FloorMaterial == Enum.Material.Air then return end
                farmHomeCFrame, farmBelowEngaged = root.CFrame, true
            end
            for _, object in ipairs(character:GetDescendants()) do
                if object:IsA("BasePart") then
                    if farmCollision[object] == nil then farmCollision[object] = object.CanCollide end
                    object.CanCollide = false
                end
            end
            apply("farmBody", humanoid, {PlatformStand = true, AutoRotate = false})
            apply("farmSize", root, {Size = Vector3.new(4, 8, 4)})
            local destroyY = workspace.FallenPartsDestroyHeight
            if type(destroyY) ~= "number" or destroyY ~= destroyY or math.abs(destroyY) > 1000000 then destroyY = -500 end
            local low = math.max(destroyY + 40, farmHomeCFrame.Position.Y - 90)
            local high = math.max(low, farmHomeCFrame.Position.Y + 80)
            farmHoverTarget = Vector3.new(best.Position.X, math.clamp(best.Position.Y - 3, low, high), best.Position.Z)
            if bestDistance <= 5 and type(firetouchinterest) == "function" then
                pcall(firetouchinterest, root, best, 0); pcall(firetouchinterest, root, best, 1)
            end
            return
        end
        local delta = best.Position - root.Position
        local flat = Vector3.new(delta.X, 0, delta.Z)
        if flat.Magnitude < 1.4 then
            farmStopMovement()
            if bestDistance <= 3 and type(firetouchinterest) == "function" then
                pcall(firetouchinterest, root, best, 0); pcall(firetouchinterest, root, best, 1)
            end
            return
        end
        apply("farmWalk", humanoid, {WalkSpeed = V["Farm Speed"]})
        if now < farmWalkCarryUntil then
            local direction, speed = flat.Unit, math.clamp(V["Farm Speed"], 5, 60)
            local velocity = root.AssemblyLinearVelocity
            farmCarryRoot, farmCarryX, farmCarryZ = root, direction.X * speed, direction.Z * speed
            root.AssemblyLinearVelocity = Vector3.new(farmCarryX, velocity.Y, farmCarryZ)
        elseif farmCarryRoot then stopFarmCarry() end
        humanoid:Move(flat.Unit, false)
        farmMoved = true
    end
    local physicsOk, farmPhysicsSignal = pcall(function() return run.PreSimulation end)
    if not physicsOk or not farmPhysicsSignal then
        local ok, signal = pcall(function() return run.Stepped end)

            root.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
            end)
            if not ok then

            end

    local function restoreCoinAura()
        local coin, saved = auraCurrent, auraState
        auraCurrent, auraState = nil, nil
        if not coin or not saved then return end
        pcall(function() if coin.CFrame == saved.appliedCFrame then coin.CFrame = saved.cframe end end)
        pcall(function() if coin.CanCollide == saved.appliedCollide then coin.CanCollide = saved.collide end end)
        if saved.visual then
            pcall(function() if saved.visual.Anchored == saved.appliedAnchored then saved.visual.Anchored = saved.anchored end end)
        end
        if saved.weld then
            pcall(function() if saved.weld.Enabled == saved.appliedWeld then saved.weld.Enabled = saved.weldEnabled end end)
        end
    end

        end

            if farmUsable(coin, map) then
                local position = coin == auraCurrent and auraState.cframe.Position or coin.Position
                local distance = (position - root.Position).Magnitude
                if distance <= 15 and (not nearest or distance < nearest) then best, nearest = coin, distance end
            end
        end
        if best ~= auraCurrent then
            restoreCoinAura()

                    if weld then auraState.weld, auraState.weldEnabled = weld, weld.Enabled end
                end

        best.CFrame = root.CFrame; saved.appliedCFrame = root.CFrame
        pcall(firetouchinterest, root, best, 0); pcall(firetouchinterest, root, best, 1)
    end

    local function cleanVoteTitle(value)
        local text = string.match(tostring(value or ""), "^%s*(.-)%s*$")
        local upper = string.upper(text)
        if upper == "" or upper == "MAP NAME" or upper == "LOADING" or upper == "NONE"
            or upper == "MAP 1" or upper == "MAP 2" or upper == "MAP 3" then return nil end
        return text
    end

    local function votingSlots(lobby)
        local slots, names, boardActive = {}, {}, false
        if not lobby then return slots, names, false end
        local pads = lobby:FindFirstChild("VotePads", true)
        local icons = lobby:FindFirstChild("VoteIcons", true)
        for slot = 1, 3 do
            local model = lobby:FindFirstChild("VotePad" .. slot, true)
            if model and model.Parent and model.Parent.Name == "VoteIcons" then model = nil end
            local icon = icons and icons:FindFirstChild("VotePad" .. slot)
            local label = model and model:FindFirstChild("MapName", true)
            local name = voteTitle(label) or voteTitle(icon and icon:FindFirstChild("MapName", true))
            if not name and icon then
                local map = icon:FindFirstChild("Map")
                if map and map:IsA("ObjectValue") and map.Value then name = cleanVoteTitle(map.Value.Name) end
            end
            local detector = pads and pads:FindFirstChild("Detector" .. slot, true)
            local pad = model and model:FindFirstChild("Pad", true)
            if pad and not pad:IsA("BasePart") then pad = nil end
            if detector and not detector:IsA("BasePart") then detector = nil end
            local surface = model and (model:FindFirstChild("VoteInfoGui") or model:FindFirstChild("MapInfoGui"))
            local iconSurface = icon and icon:FindFirstChild("Surface")
            local iconFlag = icon and icon:FindFirstChild("Voting")
            if (surface and surface:IsA("SurfaceGui") and surface.Enabled)
                or (iconSurface and iconSurface:IsA("SurfaceGui") and iconSurface.Enabled)
                or (iconFlag and iconFlag:IsA("BoolValue") and iconFlag.Value) then boardActive = true end
            if name then
                table.insert(names, name)
                slots[slot] = {name = name, pad = pad, detector = detector}
            end
        end
        return slots, names, boardActive
    end

            end

        if not target or not handle or not handle:IsA("BasePart") then return end
        local targetCharacter, generation = target.Character, knifeGeneration
        if not pcall(function() knife:Activate() end) then return end
        knifeLastSwing = now

    local function shootMurderer(root, ignoreProximity)
        local nearest, distance
        local mayUseRoles = shootRoleMap ~= nil and farmOnMap(root, shootRoleMap)
        for _, player in ipairs(playersService:GetPlayers()) do

                local hasGun = getTool(character, "Gun") or getTool(backpack, "Gun")
                local role = mayUseRoles and readShootRole(player) or nil
                local okDead, dead = pcall(function() return character:GetAttribute("Dead") end)
                local okState, deadState = pcall(function()
                    local state = humanoid:GetState()
                    return state == Enum.HumanoidStateType.Dead or state == Enum.HumanoidStateType.Physics
                end)
                if humanoid and humanoid.Health > 0 and targetRoot and targetRoot:IsA("BasePart")
                    and not (okDead and dead == true) and not (okState and deadState) and role ~= "Dead"
                    and not hasGun and (hasKnife or role == "Murderer") then
                    local d = (targetRoot.Position - root.Position).Magnitude
                    if (ignoreProximity or not V["Shoot Proximity"] or d <= V["Shoot Proximity Aura"]) and (not distance or d < distance) then
                        nearest, distance = player, d
                    end
                end
            end
        end
        return nearest
    end
    local function gunOrigin(character, root, gun)

    local function bulletPathClear(origin, endpoint, character, target)
        local direction = endpoint - origin
        if direction.Magnitude < 0.05 then return false end
        local ignored = {character}
        local params = RaycastParams.new()
        params.FilterType = Enum.RaycastFilterType.Exclude
        params.IgnoreWater = true
        for _ = 1, 32 do
            params.FilterDescendantsInstances = ignored
            local hit = workspace:Raycast(origin, direction, params)
            if not hit then return true end
            local object = hit.Instance
            if object and object:IsDescendantOf(target) then return true end
            if not object or not object:IsA("BasePart") or object.Transparency < 1 then return false end
            table.insert(ignored, object)
        end
        return false
    end

    local function tryAutoShoot()
        if stopped or not shootingActive() or shootRoundBlocked or shootBusy then return end
        local forceWin = winSheriffActive()
        local mode = forceWin and "CFrame" or V["Shoot Mode"]
        if mode ~= "Bullet" and mode ~= "CFrame" then return end
        local character, humanoid, root = characterParts()
        if not humanoid or not root then return end
        if forceWin and not farmOnMap(root, mm2Map()) then return end
        local backpack = localPlayer:FindFirstChildOfClass("Backpack")
        if getTool(character, "Knife") or getTool(backpack, "Knife") then return end
        local gun = getTool(character, "Gun") or getTool(backpack, "Gun")
        if not gun then return end
        local target = shootMurderer(root, forceWin)
        if not target or not target.Character then return end
        if gun.Parent ~= character then
            pcall(function() humanoid:EquipTool(gun) end)
            if gun.Parent ~= character then return end
        end
        if gun.Enabled == false or gun:GetAttribute("Disabled") == true or gun:GetAttribute("Reloading") == true or gun:GetAttribute("IsReloading") == true then return end
        local remote = gun:FindFirstChild("Shoot")
        if not remote or not remote:IsA("RemoteEvent") then return end
        local targetRoot = target.Character:FindFirstChild("HumanoidRootPart")
        if not targetRoot or not targetRoot:IsA("BasePart") then return end
        local now = os.clock()
        if now - lastGunShot < 0.22 then return end
        local origin = gunOrigin(character, root, gun)
        if not origin then return end
        local velocity = targetRoot.AssemblyLinearVelocity
        local endpoint = targetRoot.Position + velocity * 0.08
        for _, number in ipairs({origin.Position.X, origin.Position.Y, origin.Position.Z, endpoint.X, endpoint.Y, endpoint.Z}) do
            if number ~= number or math.abs(number) > 1000000 then return end
        end
        if (endpoint - origin.Position).Magnitude < 0.05 then return end
        local wallCheck = forceWin and V["Win Sheriff WallCheck"] == true or (not forceWin and V["Shoot WallCheck"] == true)
        if wallCheck then
            local ok, clear = pcall(bulletPathClear, origin.Position, endpoint, character, target.Character)
            if not ok or not clear then return end
        end
        local startPoint, endPoint = origin.Position, endpoint
        if mode == "CFrame" then
            -- Approximation of the reference's immediate Silent fallback, not the full predictor.
            local chord = Vector3.new(endpoint.X - targetRoot.Position.X, 0, endpoint.Z - targetRoot.Position.Z)
            local line = Vector3.new(endpoint.X - origin.Position.X, 0, endpoint.Z - origin.Position.Z)
            local axis = chord.Magnitude > 0.15 and chord.Unit or (line.Magnitude > 0.05 and line.Unit or Vector3.new(0, 0, -1))
            startPoint, endPoint = targetRoot.Position - axis * 1.5, endpoint + axis * 1.5
            if (endPoint - startPoint).Magnitude < 0.2 then startPoint, endPoint = endpoint + Vector3.new(0, 2, -1.5), endpoint end
        end
        shootBusy = true
        lastGunShot = now
        local ok, err = pcall(function() remote:FireServer(CFrame.lookAt(startPoint, endPoint), CFrame.new(endPoint)) end)
        shootBusy = false
        if not ok then warn("Angel Hub Auto Shoot dispatch uncertain: " .. tostring(err)) end
    end

        for _, section in ipairs({"Materials", "Weapons"}) do

    local function refreshUnboxCatalog()
        if stopped or not isMM2 or unboxRefreshing then return end
        unboxRefreshing = true
        local ok, err = pcall(function()
            local sync = shopSync()

            local list, ids = {}, {}
            local function add(id, data, category)

                for currency, amount in pairs(price) do
                    if type(currency) == "string" and currency ~= "Robux" and type(amount) == "number" and amount > 0 and amount < 1000000000 then
                        accepted[currency] = amount
                    end
                end
                if not next(accepted) then return end

                    or (type(data.ItemData) == "table" and data.ItemData.Name)) or nil
                table.insert(list, {id = id, name = type(name) == "string" and name or id, category = category, prices = accepted})
                ids[id] = true
            end
            for _, category in ipairs({"MysteryBox", "Eggs"}) do
                if type(sync[category]) == "table" then
                    for id, data in pairs(sync[category]) do add(id, data, category) end
                end

                table.insert(labels, label); index[label] = entry
                for currency in pairs(entry.prices) do currencies[currency] = true end
            end
            local choices = {"Auto"}
            for currency in pairs(currencies) do if currency ~= "Auto" then table.insert(choices, currency) end end

    local function chooseUnboxCurrency(entry, profile)
        local selected = V["Pay With"]
        if selected ~= "Auto" then
            local price = entry.prices[selected]
            return price and unboxBalance(profile, selected) and selected or nil
        end
        -- Never select Gems or keys implicitly: those may have been purchased.
        local candidates = {"Coins", "Candies2025", "SnowTokens2024", "BeachBalls2025", "Candies2024", "Candies2022", "SnowTokens2023", "SnowTokens2022", "SnowTokens2020", "Eggs2025", "RareEggs2025", "Hearts2023"}
        for _, currency in ipairs(candidates) do
            local balance, price = unboxBalance(profile, currency), entry.prices[currency]
            if balance and price and balance >= price then return currency end
        end
        return nil
    end

    local function liveUnboxEntry(sync, entry)

    local function tryAutoUnbox(generation)
        if stopped or not enabled["Auto Unbox"] then return end
        local label, selection = V["Box To Open"], V["Pay With"]
        local entry = unboxCatalog[label]
        if not entry then return end
        local sync = shopSync()
        if not sync or not liveUnboxEntry(sync, entry) then return end
        local livePrices = unboxPrice(sync, entry.id)
        if not livePrices then return end
        local profile = getUnboxProfile()
        if not profile then return end
        if unboxReceipt then
            local observed = unboxBalance(profile, unboxReceipt.currency)
            if not observed or observed >= unboxReceipt.balance then return end
            unboxReceipt = nil
        end
        local currency = chooseUnboxCurrency(entry, profile)
        if not currency then return end
        local balance, price = unboxBalance(profile, currency), entry.prices[currency]
        if not balance or not price or balance < price or livePrices[currency] ~= price then return end
        local shop = storageChild("Remotes", "Shop")
        local remote = shop and shop:FindFirstChild("OpenCrate")
        local complete = shop and shop:FindFirstChild("CrateComplete")
        if not remote or not remote:IsA("RemoteFunction") or not complete or not complete:IsA("RemoteEvent") then return end
        if stopped or not enabled["Auto Unbox"] or generation ~= unboxGeneration
            or V["Box To Open"] ~= label or V["Pay With"] ~= selection then return end
        -- Keep a receipt across OFF/ON until the balance actually updates.
        unboxReceipt = {currency = currency, balance = balance}
        local ok, reward = pcall(function() return remote:InvokeServer(entry.id, entry.category, currency) end)
        if not ok or type(reward) ~= "string" or reward == "" or reward == "false" or reward == "nil" then
            if not stopped then stopAutoUnbox("Auto Unbox: uncertain result; no retry until balance changes") end
            return
        end
        unboxCount += 1
        local animated = false
        if not stopped and enabled["Auto Unbox"] and generation == unboxGeneration and V["Reveal Animation"] == "On" then
            local sync = shopSync()
            if not stopped and enabled["Auto Unbox"] and generation == unboxGeneration
                and entry.category == "MysteryBox" and sync and type(sync.MysteryBox) == "table" and type(sync.MysteryBox[entry.id]) == "table"
                and type(sync.MysteryBox[entry.id].Contents) == "table" and type(sync.Weapons) == "table" and type(sync.Weapons[reward]) == "table" then
                local controller = shop:FindFirstChild("BoxController")
                if controller and controller:IsA("BindableEvent") then
                    unboxRevealUntil = os.clock() + 5
                    animated = pcall(function() controller:Fire({{MysteryBoxId = entry.id, RewardedItemId = reward}}) end)
                end
            end
        end
        if not animated then pcall(function() complete:FireServer(reward) end) end
        if not stopped then
            message("Auto Unbox: " .. tostring(reward) .. " (" .. tostring(unboxCount) .. ")")
            if enabled["Auto Unbox"] and unboxCount >= math.max(1, tonumber(V["Max Boxes Per Run"]) or 10) then stopAutoUnbox("Auto Unbox: run limit reached") end
        end
    end
    mm2UnboxStep = function()
        if not isMM2 then return end
        if os.clock() >= unboxRefreshAt then
            unboxRefreshAt = os.clock() + 15
            task.spawn(refreshUnboxCatalog)

        if not enabled["Auto Unbox"] or unboxBusy or os.clock() < unboxNext or os.clock() < unboxRevealUntil then return end
        unboxNext = os.clock() + math.max(0.8, tonumber(V["Unbox Delay"]) or 1.5)
        local generation = unboxGeneration
        unboxBusy = true
        task.spawn(function()
            local ok, err = pcall(tryAutoUnbox, generation)
            unboxBusy = false

                root.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
                root.AssemblyAngularVelocity = Vector3.new(0, 0, 0)

            if not character or not character.Parent or not humanoid or not root or root.Anchored or humanoid.Sit or humanoid.SeatPart then stopVelocity(); clearInput(); return end
            if root ~= lastRoot then stopVelocity() end
            local move = humanoid.MoveDirection
            local horizontal = Vector3.new(move.X, 0, move.Z)
            if horizontal.Magnitude > 1 then horizontal = horizontal.Unit end
            local vertical = (next(heldUp) and 1 or 0) - (next(heldDown) and 1 or 0)
            local speed = tonumber(V["Fly Horizontal Speed"]) or 60
            local lift = tonumber(V["Fly Vertical Speed"]) or 60
            if speed ~= speed or lift ~= lift or math.abs(speed) > 10000 or math.abs(lift) > 10000 then error("invalid fly speed") end
            local target = horizontal * speed + Vector3.new(0, vertical * lift, 0)
            local current = root.AssemblyLinearVelocity
            local response = math.clamp(tonumber(V["Fly Response Multiplier"]) or 1, 0.5, 2)
            local alpha = math.clamp(dt * 12 * response, 0, 1)
            local velocity = current + (target - current) * alpha
            if V["Fly Wall Check"] and velocity.Magnitude > 0.001 then
                local ok, hit = pcall(function()
                    local params = RaycastParams.new(); params.FilterType = Enum.RaycastFilterType.Exclude
                    params.FilterDescendantsInstances = {character}; params.RespectCanCollide = true; params.IgnoreWater = true
                    local step = velocity * math.min(dt, 0.1)
                    if workspace.Blockcast then return workspace:Blockcast(root.CFrame, root.Size * 0.9, step, params) end
                    return workspace:Raycast(root.Position, step, params)
                end)
                if not ok or hit then velocity = Vector3.new(0, 0, 0) end
            end
            root.AssemblyLinearVelocity = velocity; lastRoot, lastVelocity = root, velocity
            if V["Fly Face Camera"] and workspace.CurrentCamera then
                local look = workspace.CurrentCamera.CFrame.LookVector
                local flat = Vector3.new(look.X, 0, look.Z)

            end

    do
        local collisions = setmetatable({}, {__mode = "k"})
        local trackedCharacter, pulse = nil, nil
        local oscillation = 0.1
        local function restorePulse()
            local state = pulse; pulse = nil
            if not state then return end
            pcall(function()
                local value = state.root.AssemblyLinearVelocity
                if value == state.spike or value == state.oscillated then
                    state.root.AssemblyLinearVelocity = state.velocity
                    state.root.Velocity = state.velocity
                end
            end)
        end
        local function restoreCollisions()
            for part, original in pairs(collisions) do
                pcall(function() if part.CanCollide == false then part.CanCollide = original end end)
                collisions[part] = nil
            end
            trackedCharacter = nil
        end
        function walkFling.cleanup()
            restorePulse(); restoreCollisions(); oscillation = 0.1
        end
        function walkFling.heartbeat(character, humanoid, root)
            if not enabled["Walk Fling"] then return end
            restorePulse()
            if not character or not character.Parent or not humanoid or not root or root.Anchored or humanoid.Sit or humanoid.SeatPart then
                restoreCollisions(); return
            end
            if trackedCharacter ~= character then restoreCollisions(); trackedCharacter = character end
            for _, part in ipairs(character:GetChildren()) do
                if part:IsA("BasePart") then
                    if collisions[part] == nil then collisions[part] = part.CanCollide end
                    part.CanCollide = false
                end
            end
            local velocity = root.AssemblyLinearVelocity
            local spike = velocity * 30000 + Vector3.new(0, 30000, 0)
            pulse = {character=character, root=root, velocity=velocity, spike=spike}
            root.Velocity = spike; root.AssemblyLinearVelocity = spike
        end
        function walkFling.render()
            local state = pulse
            if not enabled["Walk Fling"] or not state or state.character ~= localPlayer.Character then return end
            pcall(function()
                if state.root.AssemblyLinearVelocity == state.spike then
                    state.root.Velocity = state.velocity; state.root.AssemblyLinearVelocity = state.velocity
                    state.restored = true
                end
            end)
        end
        function walkFling.stepped()
            local state = pulse
            if not enabled["Walk Fling"] or not state or not state.restored or state.character ~= localPlayer.Character then return end
            pcall(function()
                if state.root.AssemblyLinearVelocity == state.velocity then
                    local motion = state.velocity + Vector3.new(0, oscillation, 0)
                    state.oscillated = motion
                    state.root.Velocity = motion; state.root.AssemblyLinearVelocity = motion
                    oscillation = -oscillation
                end
            end)
        end

        local function travel(position, surface)
            local character, human, root = characterParts()
            if not character or not human or not root or root.Anchored then message("Teleport: character unavailable or anchored"); return end
            if enabled["Auto Win"] or enabled["Auto Farm"] or enabled["Air Walk"] or enabled["Enable Fly"] or enabled["Walk Fling"] then message("Teleport: disable Auto Win, Auto Farm, Air Walk, Fly and Walk Fling first"); return end
            if V["Teleport Movement"] ~= "Instant" and V["Teleport Movement"] ~= "Glide" then message("Teleport: unknown movement"); return end
            local x, y, z = validNumber(position and position.X), validNumber(position and position.Y), validNumber(position and position.Z)
            if not x or not y or not z then message("Teleport: invalid destination"); return end
            local before = root.Position
            local height = surface and math.clamp(tonumber(V["Teleport Height Offset"]) or 3, 0, 12) or 0
            local destination = Vector3.new(x, y + height, z)
            if V["Teleport Movement"] == "Glide" then
                local speed = tonumber(V["Teleport Glide Speed"]) or 100
                if speed ~= speed or speed <= 0 then message("Teleport: Glide speed must be positive"); return end
                glide = {character=character, humanoid=human, root=root, target=destination,
                    speed=math.clamp(speed, 1, 500), ticket=travelGeneration}
                lastPosition = before
                return
            end
            local ok = pcall(function()
                character:PivotTo(root.CFrame + (destination - root.Position))
                if not V["Teleport Keep Momentum"] then
                    root.AssemblyLinearVelocity = Vector3.new(0, 0, 0); root.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
                end
            end)
            if ok then lastPosition = before else message("Teleport failed") end
        end
        local function startPick(mode)

        local function searchDestination(mode, ticket)
            local character, human, root = characterParts()
            if not character or not human or not root then message("Teleport: character unavailable"); return end
            local origin = root.Position
            local range = math.clamp(tonumber(V["Teleport Search Range"]) or 500, 50, 5000)
            local needle = string.lower(string.match(tostring(V[mode == "Item" and "Teleport Item" or "Teleport Player"] or ""), "^%s*(.-)%s*$"))
            if mode ~= "Nearest Player" and needle == "" then message("Teleport: enter a name"); return end
            local best, bestDistance, exact, ambiguous = nil, nil, nil, false
            if mode == "Item" then
                local visited = 0
                for _, object in ipairs(workspace:GetDescendants()) do
                    if stopped or ticket ~= travelGeneration then return end
                    visited += 1
                    if visited > 20000 then break end
                    if visited % 500 == 0 then task.wait(); if stopped or ticket ~= travelGeneration then return end end
                    if object:IsA("BasePart") and object:IsDescendantOf(workspace) then
                        local parent = object.Parent
                        local text = string.lower(object.Name)
                        local parentText = parent and (parent:IsA("Model") or parent:IsA("Tool")) and string.lower(parent.Name) or ""
                        local match = string.find(text, needle, 1, true) or string.find(parentText, needle, 1, true)
                        local ancestor, insideCharacter = parent, false
                        while match and ancestor and ancestor ~= workspace do
                            if ancestor:FindFirstChildOfClass("Humanoid") then insideCharacter = true; break end
                            ancestor = ancestor.Parent
                        end
                        if match and not insideCharacter then
                            local distance = (object.Position - origin).Magnitude
                            if distance <= range and (not bestDistance or distance < bestDistance) then best, bestDistance = object, distance end
                        end
                    end
                end
            else
                for _, player in ipairs(playersService:GetPlayers()) do
                    if stopped or ticket ~= travelGeneration then return end
                    if player ~= localPlayer then
                        local model = player.Character
                        local targetHuman = model and model:FindFirstChildOfClass("Humanoid")
                        local target = model and model:FindFirstChild("HumanoidRootPart")
                        local name, display = string.lower(player.Name), string.lower(player.DisplayName)
                        local isExact = name == needle
                        local match = mode == "Nearest Player" or isExact or string.sub(name, 1, #needle) == needle or string.sub(display, 1, #needle) == needle
                        local allowed = match and model and model.Parent and model:IsDescendantOf(workspace)
                            and targetHuman and targetHuman.Health > 0 and target and target:IsA("BasePart")
                        if allowed and V["Teleport Skip Friends"] then
                            local ok, friend = pcall(function() return player:IsFriendsWith(localPlayer.UserId) end)
                            if stopped or ticket ~= travelGeneration then return end
                            allowed = ok and friend == false
                        end
                        if allowed then
                            local distance = (target.Position - origin).Magnitude
                            if distance <= range then
                                if mode == "Nearest Player" then
                                    if not bestDistance or distance < bestDistance then best, bestDistance = player, distance end
                                elseif isExact then exact = player
                                elseif best then ambiguous = true else best = player end
                            end
                        end
                    end
                end
                if mode == "Named Player" then
                    if exact then best = exact elseif ambiguous then message("Teleport: player name is ambiguous"); return end
                end
            end
            if stopped or ticket ~= travelGeneration or localPlayer.Character ~= character then return end
            if not best then message("Teleport: destination not found in range"); return end
            if mode == "Item" then
                if not best:IsDescendantOf(workspace) or (best.Position - root.Position).Magnitude > range then return end
                travel(best.Position, true)
            else
                local model = best.Character
                local targetHuman = model and model:FindFirstChildOfClass("Humanoid")
                local target = model and model:FindFirstChild("HumanoidRootPart")
                if not model or not model.Parent or not model:IsDescendantOf(workspace)
                    or not targetHuman or targetHuman.Health <= 0 or not target or not target:IsA("BasePart")
                    or (target.Position - root.Position).Magnitude > range then return end
                local look = target.CFrame.LookVector
                local flat = Vector3.new(look.X, 0, look.Z)
                local offset = math.clamp(tonumber(V["Teleport Player Offset"]) or 4, 0, 20)
                local position = target.Position - (flat.Magnitude > 0.01 and flat.Unit * offset or Vector3.new(0, 0, 0))
                travel(position, true)
            end
        end
        function markers.teleport()
            travelGeneration += 1; glide = nil
            local ticket, destination = travelGeneration, V["Teleport Destination"]
            if destination == "Tap Point" then startPick("teleport")
            elseif destination == "Waypoint" then
                if pick then restoreMenu() end
                local point = store[placeKey] and store[placeKey][V["Teleport Point"]]
                if not point then message("Teleport: select a saved point"); return end
                travel(Vector3.new(point[1], point[2], point[3]), point[4] == "surface")
            elseif destination == "Last Position" then
                if pick then restoreMenu() end
                if not lastPosition then message("Teleport: no previous point teleport"); return end
                travel(lastPosition, false)
            elseif destination == "Nearest Player" or destination == "Named Player" or destination == "Item" then
                if pick then restoreMenu() end
                task.spawn(function()
                    local ok, err = pcall(searchDestination, destination, ticket)
                    if not ok then warn("Angel Hub Teleport search error: " .. tostring(err)); if not stopped and ticket == travelGeneration then message("Teleport: search failed") end end
                end)
            else message("Teleport: destination not implemented: " .. tostring(destination)) end
        end
        function markers.input(input, processed)
            if not pick then return end
            if input.KeyCode == Enum.KeyCode.Escape then restoreMenu(); message("Point selection cancelled"); return end
            if processed or userInputService:GetFocusedTextBox() or not input.Position
                or (input.UserInputType ~= Enum.UserInputType.Touch and input.UserInputType ~= Enum.UserInputType.MouseButton1) then return end
            local camera = workspace.CurrentCamera
            if not camera then return end
            local ok, hit = pcall(function()

                local params = RaycastParams.new(); params.FilterType = Enum.RaycastFilterType.Exclude
                params.FilterDescendantsInstances = localPlayer.Character and {localPlayer.Character} or {}
                params.IgnoreWater = true; params.RespectCanCollide = true
                return workspace:Raycast(ray.Origin, ray.Direction * 5000, params)
            end)
            if not ok or not hit or not hit.Position then message("No surface found; try again"); return end
            local current = pick
            restoreMenu()
            if current.mode == "marker" then save(hit.Position, true, current.name) else travel(hit.Position, true) end
        end
        function markers.step(dt, root)
            if glide then
                local trip = glide
                if stopped or trip.ticket ~= travelGeneration or V["Teleport Movement"] ~= "Glide"
                    or enabled["Auto Win"] or enabled["Auto Farm"] or enabled["Air Walk"] or enabled["Enable Fly"] or enabled["Walk Fling"]
                    or localPlayer.Character ~= trip.character or trip.humanoid.Health <= 0
                    or not root or root ~= trip.root or root.Anchored then
                    glide = nil
                elseif dt > 0 then
                    local delta = trip.target - root.Position
                    local distance = delta.Magnitude
                    local step = math.min(trip.speed * math.clamp(dt, 0, 0.1), distance)
                    local ok = pcall(function()
                        if distance > 0.0001 then trip.character:PivotTo(root.CFrame + delta.Unit * step) end
                        if not V["Teleport Keep Momentum"] then
                            root.AssemblyLinearVelocity = Vector3.new(0, 0, 0); root.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
                        end
                    end)
                    if not ok then glide = nil; message("Teleport: Glide stopped")
                    elseif distance <= step + 0.001 then glide = nil end
                end
            end
            if pick and os.clock() >= pick.deadline then restoreMenu(); message("Point selection timed out") end
            elapsed += math.max(dt, 0)
            if elapsed < 0.25 then return end
            elapsed = 0
            local points = store[placeKey] or {}
            for name, coords in pairs(points) do
                local visual = visuals[name]
                if visual and (not visual.part.Parent or not visual.gui.Parent) then removeVisual(name); visual = nil end
                if not visual then
                    local part = Instance.new("Part"); part.Name = "AngelHubMarker"
                    part.Anchored = true; part.CanCollide = false; part.CanQuery = false; part.CanTouch = false
                    part.Transparency = 1; part.Size = Vector3.new(0.1, 0.1, 0.1)
                    part.CFrame = CFrame.new(Vector3.new(coords[1], coords[2], coords[3])); part.Parent = workspace
                    local gui = Instance.new("BillboardGui"); gui.Name = "AngelHubMarkerLabel"; gui.Adornee = part
                    gui.Size = UDim2.fromOffset(200, 30); gui.AlwaysOnTop = true; gui.Parent = screenGui
                    local label = Instance.new("TextLabel"); label.Size = UDim2.fromScale(1, 1); label.BackgroundTransparency = 1
                    label.Font = Enum.Font.SourceSansBold; label.TextSize = 14; label.TextStrokeTransparency = 0.2; label.Parent = gui
                    visual = {part=part, gui=gui, label=label}; visuals[name] = visual
                end
                local distance = root and (root.Position - Vector3.new(coords[1], coords[2], coords[3])).Magnitude
                visual.label.Text = name .. (V["Marker Distance"] and distance and (" [" .. tostring(math.floor(distance + 0.5)) .. "]") or "")
                visual.label.TextColor3 = V["Marker Colour"]
            end
            for name in pairs(visuals) do if not points[name] then removeVisual(name) end end
        end
        for _, name in ipairs({"Teleport Destination", "Teleport Movement", "Teleport Player", "Teleport Item", "Teleport Point",
            "Teleport Search Range", "Teleport Skip Friends", "Teleport Player Offset", "Teleport Height Offset"}) do
            local old = AngelHub.setters[name]
            if old then
                AngelHub.setters[name] = function(value)
                    local previous = V[name]
                    local result = old(value)
                    if V[name] ~= previous then
                        travelGeneration += 1; glide = nil
                        if name == "Teleport Destination" or name == "Teleport Movement" then restoreMenu() end
                    end
                    return result
                end
            end
        end
        function markers.cancelTravel() travelGeneration += 1; glide = nil; restoreMenu() end
        function markers.cleanup() markers.cancelTravel(); clearVisuals() end
    end

                                end

                                local teamColour = not isMM2 and V["Pointer Team Colours"] and player.Team and player.Team.TeamColor.Color or nil

                if name == "Air Walk" then airWalk.toggle(value == true) end
                if name == "Enable Fly" then fly.toggle(value == true) end

        end

        AngelHub.setters["Farm Safe Zone"] = function(value)
            oldFarmSafeZone(value)
            if value ~= true then
                leaveFarmPark()
            elseif enabled["Auto Farm"] and farmAmount >= math.max(50, farmLimit) then
                farmDone = false
            end
        end
    end
    local oldFarmMode = AngelHub.setters["Farm Mode"]
    if oldFarmMode then
        AngelHub.setters["Farm Mode"] = function(value)
            local previous = V["Farm Mode"]
            oldFarmMode(value)
            if enabled["Auto Farm"] and V["Farm Mode"] ~= previous then
                AngelHub.setters["Auto Farm"](false); message("Auto Farm stopped: mode changed")
            end
        end
    end
    local oldDistance = AngelHub.setters["Interaction Distance"]
    if oldDistance then
        AngelHub.setters["Interaction Distance"] = function(value)
            oldDistance(value)
            scanInteractions()
        end

                AngelHub.setters["Enable Fly"](false); message("Fly stopped: unsupported " .. name)

        end end

        end

            end

            if window and window.Visible then

        local velocity = root.AssemblyLinearVelocity
        root.AssemblyLinearVelocity = Vector3.new(velocity.X, V["Infinite Jump Power"], velocity.Z)

    if run.RenderStepped and run.Stepped then

        if enabled["Walk Fling"] then

            local flingOk, flingError = pcall(walkFling.heartbeat, character, humanoid, root)
            if not flingOk then
                warn("Angel Hub Walk Fling error: " .. tostring(flingError))
                AngelHub.setters["Walk Fling"](false); message("Walk Fling stopped: runtime error")

	-- Config files use Config Name; blank falls back to the default file.
	local function getConfigPath()
		local name = string.gsub(tostring(AngelHub.values["Config Name"] or ""), "[^%w_%- ]", "")
		if name == "" then
			return CONFIG_FILE
		end

		return CONFIG_FOLDER .. "/" .. name .. ".json"
	end

	-- Color3 is not JSON friendly, so it is stored as a tagged table.
	local function encodeValues()
		local data = {}

		for name, value in AngelHub.values do
			if typeof(value) == "Color3" then
				data[name] = {
					R = math.floor(value.R * 255 + 0.5),
					G = math.floor(value.G * 255 + 0.5),
					B = math.floor(value.B * 255 + 0.5),
					__color = true,
				}
			else
				data[name] = value
			end
		end

		return httpService:JSONEncode(data)
	end

	local function writeConfig(path)
		if not writefile then
			return false, "Saving is not supported here."
		end

		return pcall(function()
			if makefolder and isfolder and not isfolder(CONFIG_FOLDER) then
				makefolder(CONFIG_FOLDER)
			end

			writefile(path, encodeValues())
		end)
	end

	local function readConfig(path)
		if not (readfile and isfile and isfile(path)) then
			return nil, "No saved settings found."
		end

		local ok, data = pcall(function()
			return httpService:JSONDecode(readfile(path))
		end)

		if not ok or typeof(data) ~= "table" then
			return nil, "Settings file is corrupted."
		end

		return data
	end

	local function applyConfig(data)
        -- Restore point lists before the saved list selections, regardless of JSON key order.
        if type(data["Marker Points by Place"]) == "table" and AngelHub.setters["Marker Points by Place"] then
            pcall(AngelHub.setters["Marker Points by Place"], data["Marker Points by Place"])
        end
		for pass = 1, 2 do
		for name, value in data do
            if name == "Marker Points by Place" or (name == "Walk Fling" and value == true) then continue end
			if (AngelHub.moduleNames[name] == true) ~= (pass == 2) then continue end
			local setter = AngelHub.setters[name]
			if not setter then
				continue
			end

			if typeof(value) == "table" and value.__color then
				value = Color3.fromRGB(value.R, value.G, value.B)
			end

			pcall(setter, value)
		end
		end
	end

	local function saveSettings()
		local ok, result = writeConfig(getConfigPath())
		setStatus(ok and "Settings saved." or ("Save failed: " .. tostring(result)))
	end

	local function loadSettings()
		local data, message = readConfig(getConfigPath())
		if not data then
			return setStatus(message)
		end

		applyConfig(data)
		setStatus("Settings loaded.")
	end

		for name, value in AngelHub.defaults do

		end

		setStatus("Settings reset.")

		if input.UserInputType ~= Enum.UserInputType.Keyboard then
			return
		end

		if AngelHub.listening then
			return AngelHub.listening(input.KeyCode)
		end

		if gameProcessed then
			return
		end

		if input.KeyCode.Name == AngelHub.values["Menu Keybind"] then
			window.Visible = not window.Visible
			return
		end

		for featureName, keyCode in AngelHub.keybinds do
			if keyCode == input.KeyCode then
				local accepted = AngelHub.setters[featureName](not AngelHub.values[featureName])
				if accepted ~= false then AngelHub.notify(featureName, AngelHub.values[featureName]) end
			end
		end
	end)
