local function create(className, properties, parent)
	local instance = Instance.new(className)
	for key, value in properties do
		instance[key] = value
	end
	instance.Parent = parent
	return instance
end

local function tween(instance, info, goal)
	local newTween = tweenService:Create(instance, info, goal)
	newTween:Play()
	return newTween
end

local function connectInput(signal, callback)
	local connection = signal:Connect(callback)
	table.insert(AngelHub.connections, connection)
	return connection
end

local function isPress(input)
	return input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch
end

local function isMove(input)
	return input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch
end

local function addGradient(parent, rotation)
	return create("UIGradient", { Color = ColorSequence.new(ACCENT_A, ACCENT_B), Rotation = rotation }, parent)
end

local function addStroke(parent, color, transparency, thickness)
	return create("UIStroke", {
		Color = color,
		Transparency = transparency,
		Thickness = thickness or 1,
		ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
	}, parent)
end

local function createCard(className, parent, text, height)
	local card = create(className, {
		Name = text,
		Size = UDim2.new(1, 0, 0, height),
		BackgroundColor3 = ROW_COLOR,
		BackgroundTransparency = 0.65,
		BorderSizePixel = 0,
	}, AngelHub.sectionBoxes[parent] or parent)

	create("UICorner", { CornerRadius = UDim.new(0, 5) }, card)
	local stroke = addStroke(card, ACCENT_A, 1)
	card.ClipsDescendants = true

	local group = AngelHub.currentGroup
	if group then
		table.insert(group.cards, card)
		card:SetAttribute("GroupHidden", group.collapsed)
		card.Visible = not group.collapsed
	end

	card.InputBegan:Connect(function(input)
		if not isPress(input) then
			return
		end

		local offset = Vector2.new(input.Position.X, input.Position.Y) - card.AbsolutePosition
		local ripple = create("Frame", {
			AnchorPoint = Vector2.new(0.5, 0.5),
			Position = UDim2.fromOffset(offset.X, offset.Y),
			Size = UDim2.fromOffset(0, 0),
			BackgroundColor3 = Color3.new(1, 1, 1),
			BackgroundTransparency = 0.8,
			BorderSizePixel = 0,
			ZIndex = 0,
		}, card)

		create("UICorner", { CornerRadius = UDim.new(1, 0) }, ripple)
		local diameter = math.max(card.AbsoluteSize.X, card.AbsoluteSize.Y) * 2

		tween(ripple, MEDIUM_TWEEN, { Size = UDim2.fromOffset(diameter, diameter), BackgroundTransparency = 1 }).Completed:Connect(function()
			ripple:Destroy()
		end)
	end)

	card.MouseEnter:Connect(function()
		tween(card, FAST_TWEEN, { BackgroundTransparency = 0.35 })
		tween(stroke, FAST_TWEEN, { Transparency = 0.55 })
	end)

	card.MouseLeave:Connect(function()
		tween(card, FAST_TWEEN, { BackgroundTransparency = 0.65 })
		tween(stroke, FAST_TWEEN, { Transparency = 1 })
	end)

	return card, stroke
end

local function addToggle(parent, text, keybind, default)
	local startState = default == true and not UNIMPLEMENTED_TOGGLES[text]
	local holder, stroke = createCard("TextButton", parent, text, 28)
	holder.Text = ""
	holder.AutoButtonColor = false

	local box = create("Frame", {
		Size = UDim2.fromOffset(16, 16),
		Position = UDim2.new(0, 8, 0.5, -8),
		BackgroundColor3 = ELEMENT_COLOR,
		BorderSizePixel = 0,
	}, holder)

	create("UICorner", { CornerRadius = UDim.new(0, 4) }, box)
	local boxStroke = addStroke(box, startState and ACCENT_A or DIM_TEXT_COLOR, startState and 0 or 0.5)

	local check = create("Frame", {
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.fromScale(0.5, 0.5),
		Size = startState and UDim2.fromOffset(10, 10) or UDim2.fromOffset(0, 0),
		BackgroundColor3 = Color3.new(1, 1, 1),
		BorderSizePixel = 0,
	}, box)

	create("UICorner", { CornerRadius = UDim.new(0, 3) }, check)
	addGradient(check, 45)

	local label = create("TextLabel", {
		Size = UDim2.new(1, keybind and -100 or -44, 1, 0),
		Position = UDim2.fromOffset(32, 0),
		BackgroundTransparency = 1,
		Text = text,
		TextColor3 = startState and TEXT_COLOR or OFF_TEXT_COLOR,
		TextSize = 13,
		Font = TEXT_FONT,
		TextXAlignment = Enum.TextXAlignment.Left,
		TextTruncate = Enum.TextTruncate.AtEnd,
	}, holder)

	local accentBar = create("Frame", {
		Size = UDim2.new(0, 3, 1, -10),
		Position = UDim2.fromOffset(0, 5),
		BackgroundColor3 = Color3.new(1, 1, 1),
		BackgroundTransparency = startState and 0 or 1,
		BorderSizePixel = 0,
	}, holder)

	create("UICorner", { CornerRadius = UDim.new(1, 0) }, accentBar)
	addGradient(accentBar, 90)

	local state = startState
	AngelHub.values[text] = state
	AngelHub.defaults[text] = state

	AngelHub.setters[text] = function(value)
		state = value == true
		AngelHub.values[text] = state

		tween(check, POP_TWEEN, { Size = state and UDim2.fromOffset(10, 10) or UDim2.fromOffset(0, 0) })
		tween(boxStroke, FAST_TWEEN, { Color = state and ACCENT_A or DIM_TEXT_COLOR, Transparency = state and 0 or 0.5 })
		tween(label, FAST_TWEEN, { TextColor3 = state and TEXT_COLOR or OFF_TEXT_COLOR })
		tween(accentBar, FAST_TWEEN, { BackgroundTransparency = state and 0 or 1 })
	end

	if keybind then
		local keybindName = text .. " Keybind"
		local keybindButton = create("TextButton", {
			Size = UDim2.fromOffset(52, 22),
			Position = UDim2.new(1, -58, 0.5, -11),
			BackgroundTransparency = 1,
			Text = "[" .. keybind .. "]",
			TextColor3 = DIM_TEXT_COLOR,
			TextSize = 12,
			Font = TEXT_FONT,
			TextXAlignment = Enum.TextXAlignment.Right,
			AutoButtonColor = false,
		}, holder)

		local function applyKey(keyName)
			local keyCode = nil
			for _, item in Enum.KeyCode:GetEnumItems() do
				if item.Name == keyName then
					keyCode = item
					break
				end
			end

			AngelHub.keybinds[text] = keyCode
			AngelHub.values[keybindName] = keyCode and keyName or ""
			keybindButton.Text = keyCode and ("[" .. keyName .. "]") or "[None]"
			keybindButton.Visible = AngelHub.values["Keybind Hints"] ~= false
		end

		AngelHub.defaults[keybindName] = keybind
		AngelHub.setters[keybindName] = applyKey
		AngelHub.keyLabels[text] = keybindButton
		applyKey(keybind)

		keybindButton.MouseEnter:Connect(function()
			tween(keybindButton, FAST_TWEEN, { TextColor3 = ACCENT_A })
		end)

		keybindButton.MouseLeave:Connect(function()
			if not AngelHub.listening then
				tween(keybindButton, FAST_TWEEN, { TextColor3 = DIM_TEXT_COLOR })
			end
		end)

		keybindButton.Activated:Connect(function()
			if AngelHub.cancelListening then
				AngelHub.cancelListening()
			end

			local previousKey = AngelHub.values[keybindName]
			keybindButton.Text = "[...]"
			tween(keybindButton, FAST_TWEEN, { TextColor3 = ACCENT_A })

			AngelHub.cancelListening = function()
				AngelHub.listening = nil
				AngelHub.cancelListening = nil
				tween(keybindButton, FAST_TWEEN, { TextColor3 = DIM_TEXT_COLOR })
				applyKey(previousKey)
			end

			AngelHub.listening = function(keyCode)
				AngelHub.listening = nil
				AngelHub.cancelListening = nil
				tween(keybindButton, FAST_TWEEN, { TextColor3 = DIM_TEXT_COLOR })
				applyKey(keyCode == Enum.KeyCode.Escape and "" or keyCode.Name)
			end
		end)
	end

	holder.MouseButton1Down:Connect(function()
		tween(holder, FAST_TWEEN, { BackgroundTransparency = 0 })
		tween(stroke, FAST_TWEEN, { Transparency = 0 })
	end)

	holder.MouseButton1Up:Connect(function()
		tween(holder, FAST_TWEEN, { BackgroundTransparency = 0.35 })
		tween(stroke, FAST_TWEEN, { Transparency = 0.55 })
	end)

	holder.Activated:Connect(function()
		local accepted = AngelHub.setters[text](not state)
		if accepted ~= false and AngelHub.notify then
			AngelHub.notify(text, state)
		end
	end)

	holder.MouseEnter:Connect(function()
		tween(label, FAST_TWEEN, { Position = UDim2.fromOffset(36, 0) })
	end)

	holder.MouseLeave:Connect(function()
		tween(label, FAST_TWEEN, { Position = UDim2.fromOffset(32, 0) })
	end)

	table.insert(AngelHub.searchables, holder)
	return holder
end

local function addSlider(parent, text, minValue, maxValue, default, decimals)
    local manualMovement = ({
        ["Speed Value"] = true, ["Jump Power"] = true, ["High Jump Velocity"] = true,
        ["Long Jump Speed"] = true, ["Infinite Jump Power"] = true,
        ["Fly Horizontal Speed"] = true, ["Fly Vertical Speed"] = true,
        ["Wall Crawl Speed"] = true,
    })[text] == true
	local holder = createCard("Frame", parent, text, 44)
	local startAlpha = (default - minValue) / (maxValue - minValue)

	create("TextLabel", {
		Size = UDim2.new(1, -70, 0, 20),
		Position = UDim2.fromOffset(10, 3),
		BackgroundTransparency = 1,
		Text = text,
		TextColor3 = TEXT_COLOR,
		TextSize = 13,
		Font = TEXT_FONT,
		TextXAlignment = Enum.TextXAlignment.Left,
	}, holder)

	local valueLabel = create(manualMovement and "TextBox" or "TextLabel", {
		Size = UDim2.fromOffset(56, 20),
		Position = UDim2.new(1, -66, 0, 3),
		BackgroundTransparency = 1,
		Text = tostring(default),
		TextColor3 = OFF_TEXT_COLOR,
		TextSize = 13,
		Font = BOLD_FONT,
		TextXAlignment = Enum.TextXAlignment.Right,
	}, holder)

	local bar = create("Frame", {
		Size = UDim2.new(1, -20, 0, 6),
		Position = UDim2.fromOffset(10, 29),
		BackgroundColor3 = ELEMENT_COLOR,
		BorderSizePixel = 0,
	}, holder)

	create("UICorner", { CornerRadius = UDim.new(1, 0) }, bar)
	addStroke(bar, Color3.new(1, 1, 1), 0.88)

	local fill = create("Frame", { Size = UDim2.fromScale(startAlpha, 1), BackgroundColor3 = Color3.new(1, 1, 1), BorderSizePixel = 0 }, bar)
	create("UICorner", { CornerRadius = UDim.new(1, 0) }, fill)
	addGradient(fill, 0)

	local knob = create("Frame", {
		AnchorPoint = Vector2.new(0.5, 0.5),
		Size = UDim2.fromOffset(12, 12),
		Position = UDim2.new(startAlpha, 0, 0.5, 0),
		BackgroundColor3 = Color3.new(1, 1, 1),
		BorderSizePixel = 0,
	}, bar)

	create("UICorner", { CornerRadius = UDim.new(1, 0) }, knob)
	addStroke(knob, ACCENT_A, 0, 2)
	local knobScale = create("UIScale", {}, knob)

	local hitbox = create("Frame", { Size = UDim2.new(1, -20, 0, 22), Position = UDim2.fromOffset(10, 21), BackgroundTransparency = 1 }, holder)

	AngelHub.values[text] = default
	AngelHub.defaults[text] = default

	local dragging = false

	local function setValue(value)
		local factor = 10 ^ decimals
        if type(value) ~= "number" or value ~= value or value == math.huge or value == -math.huge then return end
        if manualMovement then value = math.clamp(value, -10000, maxValue) else value = math.clamp(value, minValue, maxValue) end
		value = math.floor(value * factor + 0.5) / factor
		local snapped = math.clamp((value - minValue) / (maxValue - minValue), 0, 1)

		tween(fill, SLIDE_TWEEN, { Size = UDim2.fromScale(snapped, 1) })
		tween(knob, SLIDE_TWEEN, { Position = UDim2.new(snapped, 0, 0.5, 0) })
		valueLabel.Text = tostring(value)
		AngelHub.values[text] = value
	end

	AngelHub.setters[text] = setValue
    if manualMovement then
        valueLabel.ClearTextOnFocus = false
        valueLabel.FocusLost:Connect(function()
            local number = tonumber(valueLabel.Text)
            if number then AngelHub.setters[text](number) else valueLabel.Text = tostring(AngelHub.values[text]) end
        end)
    end

	local function updateFromX(positionX)
		local alpha = math.clamp((positionX - bar.AbsolutePosition.X) / bar.AbsoluteSize.X, 0, 1)
		AngelHub.setters[text](minValue + (maxValue - minValue) * alpha)
	end

	hitbox.InputBegan:Connect(function(input)
		if not isPress(input) then
			return
		end

		dragging = true
		tween(knobScale, POP_TWEEN, { Scale = 1.35 })
		tween(valueLabel, FAST_TWEEN, { TextColor3 = ACCENT_A })
		updateFromX(input.Position.X)
	end)

	connectInput(userInputService.InputChanged, function(input)
		if dragging and isMove(input) then
			updateFromX(input.Position.X)
		end
	end)

	connectInput(userInputService.InputEnded, function(input)
		if not dragging or not isPress(input) then
			return
		end

		dragging = false
		tween(knobScale, POP_TWEEN, { Scale = 1 })
		tween(valueLabel, FAST_TWEEN, { TextColor3 = OFF_TEXT_COLOR })
	end)

	table.insert(AngelHub.searchables, holder)
	return holder
end

local function addCycle(parent, text, options, onChange)
	local holder = createCard("Frame", parent, text, 56)

	create("TextLabel", {
		Size = UDim2.new(1, -20, 0, 20),
		Position = UDim2.fromOffset(10, 3),
		BackgroundTransparency = 1,
		Text = text,
		TextColor3 = TEXT_COLOR,
		TextSize = 13,
		Font = TEXT_FONT,
		TextXAlignment = Enum.TextXAlignment.Left,
	}, holder)

	local button = create("TextButton", {
		Size = UDim2.new(1, -20, 0, 24),
		Position = UDim2.fromOffset(10, 26),
		BackgroundColor3 = ELEMENT_COLOR,
		BorderSizePixel = 0,
		Text = options[1],
		TextColor3 = TEXT_COLOR,
		TextSize = 13,
		Font = TEXT_FONT,
		AutoButtonColor = false,
	}, holder)

	create("UICorner", { CornerRadius = UDim.new(0, 4) }, button)
	local buttonStroke = addStroke(button, DIM_TEXT_COLOR, 0.6)

	local index = 1
	AngelHub.values[text] = options[1]
	AngelHub.defaults[text] = options[1]

	AngelHub.setters[text] = function(value)
		local found = table.find(options, value)
		if not found then
			return
		end

		index = found
		AngelHub.values[text] = options[index]
		button.Text = options[index]

		if onChange then
			onChange(options[index])
		end
	end

	button.MouseEnter:Connect(function()
		tween(buttonStroke, FAST_TWEEN, { Color = ACCENT_A, Transparency = 0 })
	end)

	button.MouseLeave:Connect(function()
		tween(buttonStroke, FAST_TWEEN, { Color = DIM_TEXT_COLOR, Transparency = 0.6 })
	end)

	button.Activated:Connect(function()
        local nextIndex = index % #options + 1
        AngelHub.setters[text](options[nextIndex])
        button.TextTransparency = 1
        tween(button, FAST_TWEEN, { TextTransparency = 0 })
	end)

	table.insert(AngelHub.searchables, holder)
	return holder
end

	local holder = createCard("Frame", parent, text, 34)

		Size = UDim2.new(1, -16, 0, 24),
		Position = UDim2.fromOffset(8, 5),
		BackgroundColor3 = ELEMENT_COLOR,
		BorderSizePixel = 0,
		Text = text,
		TextColor3 = TEXT_COLOR,
		TextSize = 13,
		Font = TEXT_FONT,
		AutoButtonColor = false,
	}, holder)

	create("UICorner", { CornerRadius = UDim.new(0, 4) }, button)
	local buttonStroke = addStroke(button, DIM_TEXT_COLOR, 0.6)

	button.MouseEnter:Connect(function()
		tween(buttonStroke, FAST_TWEEN, { Color = ACCENT_A, Transparency = 0 })
		tween(button, FAST_TWEEN, { BackgroundColor3 = Color3.fromRGB(28, 28, 30) })
	end)

	button.MouseLeave:Connect(function()
		tween(buttonStroke, FAST_TWEEN, { Color = DIM_TEXT_COLOR, Transparency = 0.6 })
		tween(button, FAST_TWEEN, { BackgroundColor3 = ELEMENT_COLOR })
	end)

	button.MouseButton1Down:Connect(function()
		tween(button, FAST_TWEEN, { BackgroundColor3 = Color3.fromRGB(46, 46, 48) })
	end)

	button.MouseButton1Up:Connect(function()
		tween(button, FAST_TWEEN, { BackgroundColor3 = Color3.fromRGB(28, 28, 30) })
	end)

	button.Activated:Connect(function()
        if callback then callback(); return end
        local action = AngelHub.actions and AngelHub.actions[text]
        if action then
            local ok, err = pcall(action)
            if not ok then warn("Angel Hub action failed: " .. tostring(err)) end
        elseif AngelHub.notify then
            AngelHub.notify("Not implemented: " .. text, false, true)
        end
	end)

	table.insert(AngelHub.searchables, holder)
	return holder
end

local function makeDraggable(handle, target)
	local dragStart, startPosition

	handle.InputBegan:Connect(function(input)
		if not isPress(input) then
			return
		end

		dragStart = input.Position
		startPosition = target.Position
	end)

	connectInput(userInputService.InputChanged, function(input)
		if not dragStart or not isMove(input) then
			return
		end

		local delta = input.Position - dragStart
		target.Position = UDim2.new(startPosition.X.Scale, startPosition.X.Offset + delta.X, startPosition.Y.Scale, startPosition.Y.Offset + delta.Y)
	end)

	connectInput(userInputService.InputEnded, function(input)
		if isPress(input) then
			dragStart = nil
		end
	end)
end

-- FIRST FUNCTIONAL PASS. Independent implementations; no ARandomMenu runtime.
-- Only the features listed in supportedModules/actions below are connected.
-- Everything else is deliberately prevented from pretending to be active.
local function installGameplay(screenGui)
    local run = game:GetService("RunService")
    local lighting = game:GetService("Lighting")
    local V = AngelHub.values
    local enabled = {}
    local groups = {}
    local noclipParts = setmetatable({}, {__mode = "k"})
    local noclipCharacter
    local lastJump = -math.huge
    local jumpHeld, jumpTouch, jumpKey = false, nil, nil
    local afkElapsed = 0
    local xrayStep = function() end
    local stopped = false
    local runtimeConnections = {}
    local supportedModules = {
        ["Noclip"] = true, ["Enable Gravity"] = true, ["Enable Speed"] = true,
        ["Custom Jump Power"] = true, ["Infinite Jump"] = true,
        ["Fullbright"] = true, ["Time Changer"] = true, ["Fast Time"] = true,
        ["Custom FOV"] = true, ["Custom Zoom"] = true,
        ["Instant Prompts"] = true, ["Proximity Prompts"] = true, ["Click Detectors"] = true,
        ["X-Ray"] = true, ["Anti AFK"] = true,
    }
    local function message(text)
        if AngelHub.notify then AngelHub.notify(text, false, true) end
    end
    local function listen(signal, callback)
        local connection = signal:Connect(callback)
        table.insert(runtimeConnections, connection)
        return connection
    end
    local function restore(name)
        local entry = groups[name]
        if not entry then return end
        for property, previous in pairs(entry.original) do
            pcall(function()
                -- Do not overwrite a value another script changed afterwards.
                if entry.object[property] == entry.applied[property] then
                    entry.object[property] = previous
                end
            end)
        end
        groups[name] = nil
    end
    local function apply(name, object, properties)
        local group = groups[name]
        if not object then restore(name); return end
        if group and group.object ~= object then restore(name); group = nil end
        if not group then
            group = {object = object, original = {}, applied = {}}
            groups[name] = group
        end
        for property, value in pairs(properties) do
            pcall(function()
                if group.original[property] == nil then group.original[property] = object[property] end
                object[property] = value
                group.applied[property] = value
            end)
        end
    end
    local function characterParts()
        local character = localPlayer.Character
        local humanoid = character and character:FindFirstChildOfClass("Humanoid")
        local root = character and character:FindFirstChild("HumanoidRootPart")
        if not humanoid or humanoid.Health <= 0 or not root or not root:IsA("BasePart") then return character, nil, nil end
        return character, humanoid, root
    end
    local function restoreNoclip()
        for part, original in pairs(noclipParts) do
            pcall(function() if part.CanCollide == false then part.CanCollide = original end end)
            noclipParts[part] = nil
        end
        noclipCharacter = nil
    end
    local function updateNoclip(character)
        if not enabled["Noclip"] or not character then restoreNoclip(); return end
        if character ~= noclipCharacter then restoreNoclip(); noclipCharacter = character end
        for _, object in ipairs(character:GetDescendants()) do
            if object:IsA("BasePart") then
                if noclipParts[object] == nil then noclipParts[object] = object.CanCollide end
                if object.CanCollide then object.CanCollide = false end
            end
        end
    end
    local fastTime
    local function update(dt)
        if stopped then return end
        local character, humanoid, root = characterParts()
        updateNoclip(character)
        if enabled["Enable Gravity"] then apply("gravity", workspace, {Gravity = V["Gravity Value"]}) else restore("gravity") end
        if enabled["Enable Speed"] and V["Speed Mode"] == "WalkSpeed" and V["Speed Value"] >= 0 and humanoid then
            apply("speed", humanoid, {WalkSpeed = V["Speed Value"]})
        else restore("speed") end
        if dt > 0 and enabled["Enable Speed"] and (V["Speed Mode"] == "Velocity" or (V["Speed Mode"] == "WalkSpeed" and V["Speed Value"] < 0)) and humanoid and root and not root.Anchored and not humanoid.Sit then
            local move = humanoid.MoveDirection
            local vel = root.AssemblyLinearVelocity
            local desiredX, desiredZ = move.X * V["Speed Value"], move.Z * V["Speed Value"]
            local alpha = math.clamp(V["Speed Response"] * dt * 12, 0, 1)
            if V["Speed Response"] >= 1 then alpha = 1 end
            local x = vel.X + (desiredX - vel.X) * alpha
            local z = vel.Z + (desiredZ - vel.Z) * alpha
            if move.Magnitude < 0.01 then
                local brake = math.clamp((1 - V["Speed Braking"]) * dt * 10, 0, 1)
                x, z = x * (1 - brake), z * (1 - brake)
            end
            if V["Speed Wall Check"] then
                local ok, hit = pcall(function()
                    local params = RaycastParams.new()
                    params.FilterType = Enum.RaycastFilterType.Exclude
                    params.FilterDescendantsInstances = {character}
                    params.RespectCanCollide = true
                    params.IgnoreWater = true
                    local step = Vector3.new(x * math.min(dt, 0.1), 0, z * math.min(dt, 0.1))
                    if step.Magnitude < 0.001 then return nil end
                    if workspace.Blockcast then
                        return workspace:Blockcast(root.CFrame, root.Size * 0.9, step, params)
                    end
                    return workspace:Raycast(root.Position, step, params)
                end)
                if not ok or hit then x, z = 0, 0 end
            end
            root.AssemblyLinearVelocity = Vector3.new(x, V["Speed Keep Vertical"] and vel.Y or 0, z)
        end
        if enabled["Infinite Jump"] and V["Infinite Jump Mode"] == "Rise" and jumpHeld and humanoid and root and not root.Anchored then
            local vel = root.AssemblyLinearVelocity
            root.AssemblyLinearVelocity = Vector3.new(vel.X, V["Infinite Jump Power"], vel.Z)
        end
        xrayStep(dt, root)
        if enabled["Anti AFK"] and dt > 0 then
            afkElapsed += dt
            if afkElapsed >= 150 then afkElapsed = 0; task.spawn(AngelHub.pokeAfk) end
        end
        if enabled["Custom Jump Power"] and humanoid then
            apply("jump", humanoid, {UseJumpPower = true, JumpPower = V["Jump Power"]})
        else restore("jump") end
        if enabled["Fullbright"] then apply("bright", lighting, {Brightness = V["Fullbright Brightness"]}) else restore("bright") end
        if enabled["Time Changer"] then
            if enabled["Fast Time"] then
                fastTime = (fastTime or lighting.ClockTime) + math.min(math.max(dt, 0), 0.25) * 6
                fastTime = fastTime % 24
                apply("clock", lighting, {ClockTime = fastTime})
            else
                fastTime = nil
                apply("clock", lighting, {ClockTime = V["Time Value"]})
            end
        else fastTime = nil; restore("clock") end
        if enabled["Custom FOV"] then apply("fov", workspace.CurrentCamera, {FieldOfView = V["FOV Value"]}) else restore("fov") end
        if enabled["Custom Zoom"] then
            local max = V["Zoom Max Distance"]
            local min = math.min(V["Zoom Min Distance"], max)
            apply("zoom", localPlayer, {CameraMinZoomDistance = min, CameraMaxZoomDistance = max})
            if V["Unlock Forced First Person"] then apply("cameraMode", localPlayer, {CameraMode = Enum.CameraMode.Classic}) else restore("cameraMode") end
        else restore("zoom"); restore("cameraMode") end
    end
    local function pulse()
        local ok, err = pcall(update, 0)
        if not ok then warn("Angel Hub gameplay update failed: " .. tostring(err)) end
    end
    local promptState = setmetatable({}, {__mode = "k"})
    local scanGeneration = 0
    local function setInteraction(object, property, value)
        pcall(function()
            local byProperty = promptState[object]
            if not byProperty then byProperty = {}; promptState[object] = byProperty end
            local entry = byProperty[property]
            if not entry then entry = {original = object[property]}; byProperty[property] = entry end
            object[property] = value
            entry.applied = value
        end)
    end
    local function restoreInteraction(kind)
        for object, fields in pairs(promptState) do
            for property, entry in pairs(fields) do
                local matching = kind == nil
                    or (kind == "Instant Prompts" and property == "HoldDuration")
                    or (kind == "Proximity Prompts" and property == "MaxActivationDistance" and object:IsA("ProximityPrompt"))
                    or (kind == "Click Detectors" and property == "MaxActivationDistance" and object:IsA("ClickDetector"))
                if matching then
                    pcall(function() if object[property] == entry.applied then object[property] = entry.original end end)
                    fields[property] = nil
                end
            end
            if next(fields) == nil then promptState[object] = nil end
        end
    end
    local function updateInteraction(object)
        if object:IsA("ProximityPrompt") then
            if enabled["Instant Prompts"] then setInteraction(object, "HoldDuration", 0) end
            if enabled["Proximity Prompts"] then setInteraction(object, "MaxActivationDistance", V["Interaction Distance"]) end
        elseif object:IsA("ClickDetector") and enabled["Click Detectors"] then
            setInteraction(object, "MaxActivationDistance", V["Interaction Distance"])
        end
    end
    local function scanInteractions()
        scanGeneration += 1
        local generation = scanGeneration
        if not (enabled["Instant Prompts"] or enabled["Proximity Prompts"] or enabled["Click Detectors"]) then return end
        task.spawn(function()
            local objects = workspace:GetDescendants()
            for i, object in ipairs(objects) do
                if stopped or generation ~= scanGeneration then return end
                updateInteraction(object)
                if i % 200 == 0 then task.wait() end
            end
        end)
    end
    listen(workspace.DescendantAdded, function(object)
        if not stopped then updateInteraction(object) end
    end)
    -- Local-only map fading. Characters and held tools are excluded.
    local xrayRecords = setmetatable({}, {__mode = "k"})
    local xrayIndexed = setmetatable({}, {__mode = "k"})
    local xrayParts, xrayCursor, xrayPause, xrayGeneration = {}, 1, 0, 0
    local function xrayRestore(part)
        local record = xrayRecords[part]
        if not record then return end
        pcall(function()
            if part.LocalTransparencyModifier == record.applied then part.LocalTransparencyModifier = record.base end
        end)
        xrayRecords[part] = nil
    end
    local function xrayClear()
        for part in pairs(xrayRecords) do xrayRestore(part) end
    end
    local function xrayAdd(part)
        if part:IsA("BasePart") and not xrayIndexed[part] then
            xrayIndexed[part] = true
            table.insert(xrayParts, part)
        end
    end
    local function xrayToggle(on)
        xrayGeneration += 1
        xrayClear()
        table.clear(xrayParts); table.clear(xrayIndexed)
        xrayCursor, xrayPause = 1, 0
        if not on then return end
        local generation = xrayGeneration
        task.spawn(function()
            for index, part in ipairs(workspace:GetDescendants()) do
                if stopped or generation ~= xrayGeneration then return end
                xrayAdd(part)
                if index % 200 == 0 then task.wait() end
            end
        end)
    end
    listen(workspace.DescendantAdded, function(part)
        if enabled["X-Ray"] then xrayAdd(part) end
    end)
    local function parseNames(value)
        local found = {}
        for name in string.gmatch(string.lower(value or ""), "[^,]+") do
            name = string.match(name, "^%s*(.-)%s*$")
            if name ~= "" then found[name] = true end
        end
        return found
    end
    xrayStep = function(dt, root)
        if not enabled["X-Ray"] then return end
        xrayPause = math.max(0, xrayPause - dt)
        if xrayPause > 0 then return end
        local named, kept = parseNames(V["XRay Part Names"]), parseNames(V["XRay Keep Visible"])
        local camera = workspace.CurrentCamera
        local origin = camera and camera.CFrame.Position or root and root.Position
        for _ = 1, 200 do
            local part = xrayParts[xrayCursor]
            if not part then xrayCursor = 1; xrayPause = 0.2; break end
            if not part.Parent or not part:IsDescendantOf(workspace) then
                xrayRestore(part); xrayIndexed[part] = nil
                xrayParts[xrayCursor] = xrayParts[#xrayParts]; xrayParts[#xrayParts] = nil
            else
                local name = string.lower(part.Name)
                local floor = part.Size.Y <= math.min(part.Size.X, part.Size.Z) * 0.35
                local valid = part.Transparency < 0.98 and not kept[name]
                    and (V["XRay Include Unanchored"] or part.Anchored)
                    and (not V["XRay Collidable Only"] or part.CanCollide)
                    and not (V["XRay Keep Floors"] and floor and root and part.Position.Y <= root.Position.Y)
                if valid and V["XRay Limit Range"] and origin then
                    valid = (part.Position - origin).Magnitude <= V["XRay Range"] + part.Size.Magnitude * 0.5
                end
                local ancestor = part.Parent
                while valid and ancestor and ancestor ~= workspace do
                    if ancestor:IsA("Tool") or (ancestor:IsA("Model") and ancestor:FindFirstChildOfClass("Humanoid")) then valid = false end
                    ancestor = ancestor.Parent
                end
                local scope = V["XRay Scope"]
                if scope == "Named Parts" then valid = valid and named[name] == true end
                if scope == "Walls" then valid = valid and not floor and part.Size.Y > math.min(part.Size.X, part.Size.Z) * 1.5 end
                if valid then
                    local current = part.LocalTransparencyModifier
                    local record = xrayRecords[part]
                    if not record then record = {base = current}; xrayRecords[part] = record
                    elseif current ~= record.applied then record.base = current end
                    record.applied = math.max(record.base, V["XRay Transparency"])
                    if current ~= record.applied then part.LocalTransparencyModifier = record.applied end
                else xrayRestore(part) end
                xrayCursor += 1
            end
        end
    end
    AngelHub.pokeAfk = function()
        if stopped or not enabled["Anti AFK"] then return end
        local camera = workspace.CurrentCamera
        local position = camera and camera.ViewportSize and camera.ViewportSize / 2 or Vector2.new(400, 300)
        local ok = pcall(function()
            local input = game:GetService("VirtualUser")
            input:CaptureController()
            input:Button2Down(position, camera and camera.CFrame)
            input:Button2Up(position, camera and camera.CFrame)
        end)
        if not ok then
            ok = pcall(function()
                local input = game:GetService("VirtualInputManager")
                input:SendMouseButtonEvent(position.X, position.Y, 0, true, game, false)
                input:SendMouseButtonEvent(position.X, position.Y, 0, false, game, false)
            end)
        end
        if not ok then message("Anti AFK: input unavailable") end
    end

    local function restoreAll()
        scanGeneration += 1
        xrayToggle(false)
        restoreNoclip()
        restoreInteraction(nil)
        for name in pairs(groups) do restore(name) end
    end
    local moduleNames = AngelHub.moduleNames or {}
    for name in pairs(moduleNames) do
        local old = AngelHub.setters[name]
        if old then
            AngelHub.setters[name] = function(value)
                if value == true and not supportedModules[name] then
                    old(false); enabled[name] = nil
                    message("Not implemented: " .. name)
                    return false
                end
                if value == true and name == "Enable Speed" and V["Speed Mode"] ~= "WalkSpeed" and V["Speed Mode"] ~= "Velocity" then
                    old(false); enabled[name] = nil; message("Speed: WalkSpeed or Velocity only"); return false
                end
                old(value == true); enabled[name] = value == true
                if name == "Infinite Jump" and not value then jumpHeld, jumpTouch, jumpKey = false, nil, nil end
                if name == "Anti AFK" then afkElapsed = 0 end
                if name == "X-Ray" then xrayToggle(value == true) end
                if name == "Instant Prompts" or name == "Proximity Prompts" or name == "Click Detectors" then
                    if not enabled[name] then restoreInteraction(name) end
                    scanInteractions()
                end
                pulse()
            end
        end
    end
    local oldDistance = AngelHub.setters["Interaction Distance"]
    if oldDistance then
        AngelHub.setters["Interaction Distance"] = function(value)
            oldDistance(value)
            scanInteractions()
        end
    end

        if old then
            AngelHub.setters[name] = function(value)
                old(false)
                if value then message("Not implemented: " .. name); return false end
            end
        end
    end

        if old then AngelHub.setters[name] = function(value)
            if value then old(false); message("Not implemented: " .. name); return false else old(false) end
        end end
    end

        local setting, feature, allowed = item[1], item[2], item[3]
        local old = AngelHub.setters[setting]
        AngelHub.setters[setting] = function(value)
            old(value)
            if enabled[feature] and V[setting] ~= allowed and V[setting] ~= "Velocity" then

            if V["Long Jump Mode"] ~= "Velocity" then message("Long Jump: Velocity only in this build"); return end
            local _, humanoid, root = characterParts()
            if not humanoid or not root then return end
            local move = humanoid.MoveDirection
            if move.Magnitude < 0.01 then
                local camera = workspace.CurrentCamera
                if not camera then return end
                local look = camera.CFrame.LookVector
                move = Vector3.new(look.X, 0, look.Z)
            end
            if move.Magnitude < 0.01 then return end
            local velocity = root.AssemblyLinearVelocity
            local direction = move.Unit * V["Long Jump Speed"]
            root.AssemblyLinearVelocity = Vector3.new(direction.X, velocity.Y, direction.Z)
        end,
    }
    local function typing()
        return userInputService:GetFocusedTextBox() ~= nil
    end
    local function jumpButtonTouch(input)
        local gui = localPlayer:FindFirstChildOfClass("PlayerGui")
        local touchGui = gui and gui:FindFirstChild("TouchGui")
        local button = touchGui and touchGui:FindFirstChild("JumpButton", true)
        if not button or not input.Position or not button.Visible then return false end
        local p, at, size = input.Position, button.AbsolutePosition, button.AbsoluteSize
        return p.X >= at.X and p.X <= at.X + size.X and p.Y >= at.Y and p.Y <= at.Y + size.Y
    end
    listen(userInputService.InputBegan, function(input, processed)
        if not enabled["Infinite Jump"] or typing() then return end
        if (input.KeyCode == Enum.KeyCode.Space or input.KeyCode == Enum.KeyCode.ButtonA) and not processed then
            jumpHeld, jumpKey = true, input.KeyCode
        elseif input.UserInputType == Enum.UserInputType.Touch and jumpButtonTouch(input) then
            jumpHeld, jumpTouch = true, input
        end
    end)
    listen(userInputService.InputEnded, function(input)
        if (jumpKey and input.KeyCode == jumpKey) or (jumpTouch and input == jumpTouch) then
            jumpHeld, jumpTouch, jumpKey = false, nil, nil
        end
    end)
    listen(userInputService.WindowFocusReleased, function() jumpHeld, jumpTouch, jumpKey = false, nil, nil end)
    listen(userInputService.TextBoxFocused, function() jumpHeld, jumpTouch, jumpKey = false, nil, nil end)

        AngelHub.actions["Get Gun"] = function()
            if type(firetouchinterest) ~= "function" then message("Get Gun: touch API unavailable"); return end
            local _, humanoid, root = characterParts()
            if not humanoid or not root then message("Get Gun: character unavailable"); return end
            local drop = workspace:FindFirstChild("GunDrop", true)
            if not drop then message("Get Gun: dropped gun not found"); return end
            local part = drop:IsA("BasePart") and drop or drop:FindFirstChild("Handle", true) or drop:FindFirstChildWhichIsA("BasePart", true)
            if not part or not part:IsA("BasePart") or not part:IsDescendantOf(workspace) then
                message("Get Gun: dropped gun has no touch part"); return
            end
            local ok1 = pcall(firetouchinterest, root, part, 0)
            local ok2 = pcall(firetouchinterest, root, part, 1)
            if not ok1 or not ok2 then message("Get Gun: touch failed") end
        end
        listen(userInputService.InputBegan, function(input, processed)
            if processed or AngelHub.listening or typing() or input.UserInputType ~= Enum.UserInputType.Keyboard then return end
            if V["Get Gun Key Value"] and input.KeyCode.Name == V["Get Gun Key Value"] then
                AngelHub.actions["Get Gun"]()
            end
        end)
    end
    listen(userInputService.JumpRequest, function()
        if not enabled["Infinite Jump"] or V["Infinite Jump Mode"] ~= "Normal" or typing() then return end
        local _, humanoid, root = characterParts()
        if not humanoid or not root or root.Anchored or humanoid.FloorMaterial ~= Enum.Material.Air then return end
        local now = os.clock()
        if now - lastJump < V["Infinite Jump Interval"] then return end
        lastJump = now
        local velocity = root.AssemblyLinearVelocity
        root.AssemblyLinearVelocity = Vector3.new(velocity.X, V["Infinite Jump Power"], velocity.Z)
    end)
    listen(run.Heartbeat, function(dt)
        local ok, err = pcall(update, dt)
        if not ok and not AngelHub.runtimeError then AngelHub.runtimeError = true; warn("Angel Hub gameplay error: " .. tostring(err)) end
    end)
    screenGui.Destroying:Connect(function()
        stopped = true
        for _, connection in ipairs(runtimeConnections) do connection:Disconnect() end
        table.clear(runtimeConnections)
        restoreAll()
        AngelHub.actions = nil
        AngelHub.pokeAfk = nil
    end)
end

	-- Config files use Config Name; blank falls back to the default file.
	local function getConfigPath()
		local name = string.gsub(tostring(AngelHub.values["Config Name"] or ""), "[^%w_%- ]", "")
		if name == "" then
			return CONFIG_FILE
		end

		return CONFIG_FOLDER .. "/" .. name .. ".json"
	end

	-- Color3 is not JSON friendly, so it is stored as a tagged table.
	local function encodeValues()
		local data = {}

		for name, value in AngelHub.values do
			if typeof(value) == "Color3" then
				data[name] = {
					R = math.floor(value.R * 255 + 0.5),
					G = math.floor(value.G * 255 + 0.5),
					B = math.floor(value.B * 255 + 0.5),
					__color = true,
				}
			else
				data[name] = value
			end
		end

		return httpService:JSONEncode(data)
	end

	local function writeConfig(path)
		if not writefile then
			return false, "Saving is not supported here."
		end

		return pcall(function()
			if makefolder and isfolder and not isfolder(CONFIG_FOLDER) then
				makefolder(CONFIG_FOLDER)
			end

			writefile(path, encodeValues())
		end)
	end

	local function readConfig(path)
		if not (readfile and isfile and isfile(path)) then
			return nil, "No saved settings found."
		end

		local ok, data = pcall(function()
			return httpService:JSONDecode(readfile(path))
		end)

		if not ok or typeof(data) ~= "table" then
			return nil, "Settings file is corrupted."
		end

		return data
	end

	local function applyConfig(data)
		for pass = 1, 2 do
		for name, value in data do
			if (AngelHub.moduleNames[name] == true) ~= (pass == 2) then continue end
			local setter = AngelHub.setters[name]
			if not setter then
				continue
			end

			if typeof(value) == "table" and value.__color then
				value = Color3.fromRGB(value.R, value.G, value.B)
			end

			pcall(setter, value)
		end
		end
	end

	local function saveSettings()
		local ok, result = writeConfig(getConfigPath())
		setStatus(ok and "Settings saved." or ("Save failed: " .. tostring(result)))
	end

	local function loadSettings()
		local data, message = readConfig(getConfigPath())
		if not data then
			return setStatus(message)
		end

		applyConfig(data)
		setStatus("Settings loaded.")
	end

	local function resetSettings()
		for name, value in AngelHub.defaults do
			AngelHub.setters[name](value)
		end

		setStatus("Settings reset.")
	end

		end

		for featureName, keyCode in AngelHub.keybinds do
			if keyCode == input.KeyCode then
				local accepted = AngelHub.setters[featureName](not AngelHub.values[featureName])
				if accepted ~= false then AngelHub.notify(featureName, AngelHub.values[featureName]) end
			end
		end
	end)

	task.spawn(function()
		while screenGui.Parent do
			local visible = AngelHub.values["Keybind Hints"] ~= false

			for _, keybindButton in AngelHub.keyLabels do
				keybindButton.Visible = visible
			end

			task.wait(0.5)
		end
	end)

	switchTab("Main")
end

local function onInitializeError(error)
	warn("Failed to initialize.")
	warn(error)
	warn(debug.traceback())
end

xpcall(initializeScript, onInitializeError)
